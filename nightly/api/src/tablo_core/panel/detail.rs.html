1//! The detail page: `GET {prefix}/{slug}/{id}`, read-only.
2//!
3//! One record, rendered through [`Resource::view`] — the same `Schema` a form
4//! uses, read the other way round. It lives beside the form handlers rather
5//! than in `list.rs` because it is a record page: it loads through the same
6//! tenant-scoped query (`find_by_key`), checks the same `can_view`
7//! policy,
8//! and answers the same 404 for an unknown or out-of-scope id.
9
10use topcoat::{
11    context::Cx,
12    router::{Body, error::not_found, path_param_segment},
13    view::{BoxView, HoistView, ViewExt, internal::ThenView, view},
14};
15
16use super::{
17    actions::load_detail,
18    gate::{gate, list_url},
19};
20use crate::{db::db, form::RecordForm, resource::Resource};
21
22/// Detail page GET.
23///
24/// A resource that declares no [`view`](Resource::view) has no detail page:
25/// the handler 404s rather than rendering an empty shell, which keeps
26/// "declares nothing" and "no such page" the same answer for a hand-typed URL
27/// and makes the row link's absence honest.
28///
29/// The record loads through the tenant-scoped query (`find_by_key` — the
30/// tenancy half derived by the framework, — and the resource's own
31/// soft-delete scope, ADR-0002), so an unknown id and an id outside the
32/// request's scope get one answer, as everywhere else in the panel. `can_view`
33/// on the loaded record is a 403 rather than a 404: the record exists and this
34/// caller may not see it.
35pub(crate) fn resource_view<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
36    Box::pin(HoistView::new(ThenView::new(async move {
37        gate::<R>(cx)?;
38        if !R::viewed(cx) {
39            return Err(not_found().into());
40        }
41        let id = path_param_segment(cx, "id").to_string();
42        let mut db = db(cx);
43        let record = load_detail::<R>(cx, &mut db).await?;
44        // The form's projection over `view_values`, so the page and the form
45        // agree about what a field holds (ADR-0016); `NoForm` projects nothing.
46        let mut values = R::view_values(cx, &record);
47        values.extend(<R::Form as RecordForm>::hydrate(cx, &record));
48        let body = R::view(cx)
49            .render(cx, crate::schema::Source::view(&values))
50            .await?;
51        // Relations render from the record itself: the `Schema`
52        // above carries only its string projection, and the related rows are
53        // already loaded by `query`'s `include`, so this adds no query.
54        let relations = R::view_relations(cx, &record);
55        // The record's own label titles the page when the resource declares
56        // one. The fallback is the page's name plus the URL's record
57        // key, which is what the route carries (the display key drives the list
58        // and the record key the action URLs).
59        let title = detail_title::<R>(cx, &record, &id);
60        let back = list_url(cx, &R::slug());
61        let public = R::public_url(cx, &record);
62        Ok(view! {
63            cx =>
64            tablo_ui::page(
65                tablo_ui::page_header(
66                    tablo_ui::page_title((title))
67                    <a
68                        (crate::resource::runtime_link(cx, &back))
69                        class="text-sm text-muted-foreground underline"
70                    >
71                        "Back to list"
72                    </a>
73                    if let Some(public) = public {
74                        <a
75                            href=(public)
76                            class="text-sm text-muted-foreground underline"
77                        >
78                            "View public post"
79                        </a>
80                    }
81                )
82                tablo_ui::page_content(
83                    <div class="flex flex-col gap-4">
84                        (body)
85                        if let Some(relations) = relations {
86                            (relations)
87                        }
88                    </div>
89                )
90            )
91        }
92        .boxed())
93    })))
94}

96/// The detail page's title: the record's label when the resource
97/// declares one ([`Resource::record_label`]), else the page's name and the
98/// URL's record key.
99fn detail_title<R: Resource>(cx: &Cx, record: &R::Model, id: &str) -> String {
100    R::record_label(cx, record).unwrap_or_else(|| format!("{} {id}", R::label()))
101}
