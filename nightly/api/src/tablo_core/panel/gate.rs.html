8/// The mount prefix of the [`Panel`] that built this Router (e.g. `/admin`).
9/// Installed by [`Panel::build`](super::Panel::build) so generic handlers can derive every
10/// resource URL as `{prefix}/{slug}` — correct by construction even when a
11/// table renders away from its own list route — instead of sniffing the
12/// request path (item 6).
13#[derive(Debug, Clone)]
14pub(crate) struct PanelPrefix(pub(crate) String);

16/// Demo/deployment hint rendered under the login form (auth feature).

21/// Defense-in-depth companion to the auth gate (ADR-0013): every
22/// panel handler and the live-search shard re-check the resolved user, so a
23/// missing or mis-mounted gate cannot silently open a handler. A no-op when
24/// the panel explicitly disabled auth.
25#[cfg(feature = "auth")]
26pub(crate) fn enforce_auth(cx: &Cx) -> Result<(), topcoat::Error> {
27    if crate::auth::enforced(cx) {
28        crate::auth::require_authenticated(cx)?;
29    }
30    Ok(())
31}
32
33/// Auth compiled out: the gate does not exist either, so nothing to enforce.
34#[cfg(not(feature = "auth"))]
35pub(crate) fn enforce_auth(_cx: &Cx) -> Result<(), topcoat::Error> {
36    Ok(())
37}

39/// Enforce tenancy gating for resources that require it.
40///
41/// Wired into every resource handler; a no-op unless the resource overrides
42/// `Resource::requires_tenant`. Fails closed (403) when no tenant is present
43/// instead of serving unscoped rows.
44pub(crate) fn enforce_tenant<R: Resource>(cx: &Cx) -> Result<(), topcoat::Error> {
45    if R::requires_tenant() {
46        crate::tenancy::require_tenant(cx)?;
47    }
48    Ok(())
49}
50
51/// The gate every resource handler runs first: the authenticated user, then
52/// the resource's tenant. A no-op when auth is compiled out and the resource
53/// declares no tenant.
54pub(crate) fn gate<R: Resource>(cx: &Cx) -> Result<(), topcoat::Error> {
55    enforce_auth(cx)?;
56    enforce_tenant::<R>(cx)
57}

59/// The panel's URL prefix: the [`PanelPrefix`] app context installed by
60/// [`Panel::build`](super::Panel::build), else the request path's first segment, else `/admin`.
61///
62/// A bare `CxTestBuilder` installs no prefix, so a test rendering under
63/// `/admin/...` still derives `/admin`.
64pub(crate) fn panel_prefix(cx: &Cx) -> String {
65    topcoat::context::try_app_context::<PanelPrefix>(cx)
66        .map(|p| p.0.clone())
67        .unwrap_or_else(|| {
68            let path = topcoat::router::request::uri(cx).path().to_string();
69            path.split('/')
70                .nth(1)
71                .filter(|s| !s.is_empty())
72                .map(|s| format!("/{s}"))
73                .unwrap_or_else(|| "/admin".to_string())
74        })
75}
76
77/// The list URL for a resource: `{panel prefix}/{slug}`.
78pub(crate) fn list_url(cx: &Cx, slug: &str) -> String {
79    format!("{}/{slug}", panel_prefix(cx))
80}

95    #[test]
96    fn list_url_prefers_panel_prefix_over_request_path() {
97        use topcoat::context::CxTestBuilder;
98
99        // With the panel prefix installed, the resource URL is derived from
100        // the declaration — even on a path that is not the list route.
101        let (parts, ()) = http::Request::builder()
102            .uri("/admin/users/42/edit")
103            .body(())
104            .unwrap()
105            .into_parts();
106        let cx = CxTestBuilder::new()
107            .request_context(parts)
108            .app_context(PanelPrefix("/admin".to_string()))
109            .build();
110        assert_eq!(list_url(&cx, "users"), "/admin/users");
111
112        let (parts, ()) = http::Request::builder()
113            .uri("/backoffice/users")
114            .body(())
115            .unwrap()
116            .into_parts();
117        let cx = CxTestBuilder::new()
118            .request_context(parts)
119            .app_context(PanelPrefix("/backoffice".to_string()))
120            .build();
121        assert_eq!(list_url(&cx, "users"), "/backoffice/users");
122
123        // Without a panel prefix (bare test builder), fall back to the
124        // request path's first segment.
125        let (parts, ()) = http::Request::builder()
126            .uri("/admin/users/42/edit")
127            .body(())
128            .unwrap()
129            .into_parts();
130        let cx = CxTestBuilder::new().request_context(parts).build();
131        assert_eq!(list_url(&cx, "users"), "/admin/users");
132    }

134    /// The named runtime endpoints answer the gate: a request
135    /// without a session to a shard's fixed path is refused with 401, not a
136    /// login redirect and not the shard's content. The path is stable, so the
137    /// refusal is the only thing that keeps it from being probed.
138    #[cfg(feature = "auth")]
139    #[tokio::test]
140    async fn named_shard_endpoints_answer_401_without_a_session() {
141        let db = Db::builder()
142            .models(toasty::models!(
143                crate::auth::AdminUser,
144                crate::auth::AuthSession
145            ))
146            .connect("sqlite::memory:")
147            .await
148            .unwrap();
149        db.push_schema().await.unwrap();
150        let router = Panel::new("admin")
151            .app_context(db)
152            .auth(crate::Auth::password())
153            .build()
154            .expect("panel builds");
155
156        for path in [TABLE_SEARCH_PATH, crate::notification::LIVE_TOASTER_PATH] {
157            let request = http::Request::builder()
158                .method(http::Method::POST)
159                .uri(path)
160                .header(http::header::CONTENT_TYPE, "application/json")
161                .body(Body::from("{}".to_owned()))
162                .unwrap();
163            let response = router.handle(request).await;
164            assert_eq!(
165                response.status(),
166                http::StatusCode::UNAUTHORIZED,
167                "an unauthenticated shard request must answer 401: {path}"
168            );
169        }
170    }
