8/// The mount prefix of the [`Panel`] that built this Router (e.g. `/admin`).
9/// Installed by [`Panel::build`](super::Panel::build) so generic handlers can derive every
10/// resource URL as `{prefix}/{slug}` — correct by construction even when a
11/// table renders away from its own list route — instead of sniffing the
12/// request path (item 6).
13#[derive(Debug, Clone)]
14pub(crate) struct PanelPrefix(pub(crate) String);
15
16/// Demo/deployment hint rendered under the login form.
17#[derive(Debug, Clone)]
18pub(crate) struct LoginHint(pub(crate) String);

20/// Defense-in-depth companion to the auth gate (ADR-0013): every
21/// panel handler and the live-search shard re-check the resolved user, so a
22/// missing or mis-mounted gate cannot silently open a handler. A no-op when
23/// the panel explicitly disabled auth.
24pub(crate) fn enforce_auth(cx: &Cx) -> Result<(), topcoat::Error> {
25    if crate::auth::enforced(cx) {
26        crate::auth::require_authenticated(cx)?;
27    }
28    Ok(())
29}
30
31/// Enforce tenancy gating for resources that require it.
32///
33/// Wired into every resource handler; a no-op unless the resource overrides
34/// `Resource::requires_tenant`. Fails closed (403) when no tenant is present
35/// instead of serving unscoped rows.
36pub(crate) fn enforce_tenant<R: Resource>(cx: &Cx) -> Result<(), topcoat::Error> {
37    if R::requires_tenant() {
38        crate::tenancy::require_tenant(cx)?;
39    }
40    Ok(())
41}
42
43/// The gate every resource handler runs first: the authenticated user, then
44/// the resource's tenant. A no-op when auth is compiled out and the resource
45/// declares no tenant.
46pub(crate) fn gate<R: Resource>(cx: &Cx) -> Result<(), topcoat::Error> {
47    enforce_auth(cx)?;
48    enforce_tenant::<R>(cx)
49}

51/// The panel's URL prefix: the [`PanelPrefix`] app context installed by
52/// [`Panel::build`](super::Panel::build), else the request path's first segment, else `/admin`.
53///
54/// A bare `CxTestBuilder` installs no prefix, so a test rendering under
55/// `/admin/...` still derives `/admin`.
56pub(crate) fn panel_prefix(cx: &Cx) -> String {
57    topcoat::context::try_app_context::<PanelPrefix>(cx)
58        .map(|p| p.0.clone())
59        .unwrap_or_else(|| {
60            let path = topcoat::router::request::uri(cx).path().to_string();
61            path.split('/')
62                .nth(1)
63                .filter(|s| !s.is_empty())
64                .map(|s| format!("/{s}"))
65                .unwrap_or_else(|| "/admin".to_string())
66        })
67}
68
69/// The list URL for a resource: `{panel prefix}/{slug}`.
70pub(crate) fn list_url(cx: &Cx, slug: &str) -> String {
71    format!("{}/{slug}", panel_prefix(cx))
72}
