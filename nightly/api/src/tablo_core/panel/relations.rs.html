23/// The relation tables of `R`'s record `owner`, for the page at the request's
24/// path: one section per [`Resource::relations`] entry, in declaration order.
25/// A `read_only` page — the detail page — shows the rows without write
26/// actions, as Filament's view page does; the edit page carries them.
27pub(crate) fn render_relations<'a, R: Resource>(
28    cx: &'a Cx,
29    owner: &R::Model,
30    read_only: bool,
31) -> Vec<BoxView<'a>> {
32    let page = topcoat::router::request::uri(cx).path().to_string();
33    R::relations()
34        .iter()
35        .map(|relation| relation.render(cx, owner, &page, read_only))
36        .collect()
37}
38
39/// One relation's section: `C`'s list table over the rows the owner holds,
40/// its URL parameters prefixed with the relation's key, and — unless the
41/// page is read-only — its write actions returning to this page and a create
42/// link that opens `C`'s form with the owner chosen.
43///
44/// `C`'s own gates apply: a request that lacks a tenant `C` requires, or that
45/// `C::can_view_any` refuses, gets no section, and the row actions keep their
46/// per-row policy. A live-search table renders its server-side form here; the
47/// live shard serves `C`'s list.
48pub(crate) fn relation_table<C: Resource>(cx: &Cx, relation: BoundRelation) -> BoxView<'_> {
49    Box::pin(ThenView::new(async move {
50        if enforce_tenant::<C>(cx).is_err() || !C::can_view_any(cx) {
51            return Ok(().boxed());
52        }
53        let BoundRelation {
54            key,
55            label,
56            scope,
57            seed,
58            page,
59            read_only,
60        } = relation;
61        let declared = declared_chrome::<C>(cx);
62        let chrome = if read_only {
63            TableChrome {
64                view: declared.view,
65                ..TableChrome::default()
66            }
67        } else {
68            declared
69        };
70        let table = wire_table::<C>(cx, false, chrome);
71        let state = table.normalize_state(&TableState::from_cx_prefixed(cx, &key));
72        // A write returns to the page as the table shows it, without a
73        // dialog left open on a row the write removed.
74        let table = table.returning_to(state.list_url(&page));
75        let create_url = (!read_only && <C::Form as RecordForm>::HAS_FORM && C::can_create(cx))
76            .then(|| {
77                let query = form_urlencoded::Serializer::new(String::new())
78                    .append_pair(&seed.0, &seed.1)
79                    .append_pair(RETURN_PARAM, &state.list_url(&page))
80                    .finish();
81                format!("{}?{query}", create_page_url(&list_url(cx, &C::slug())))
82            });
83        let body = match load_scoped_page::<C>(cx, &table, &state, scope).await {
84            Ok(rows) => table.render_with_state(cx, rows, &state, &page).await?,
85            Err(error) => table_error_view::<C>(cx, &state, &error, &page, None),
86        };
87        let create_label = format!("New {}", C::label());
88        Ok(view! {
89            cx =>
90            <section class="flex flex-col gap-3" data-relation=(key)>
91                <div class="flex items-center justify-between gap-4">
92                    <h2 class="text-lg font-semibold tracking-tight text-foreground">
93                        (label)
94                    </h2>
95                    if let Some(url) = create_url {
96                        <a
97                            (runtime_link(cx, &url))
98                            class=(tablo_ui::button_variants(
99                                tablo_ui::ButtonVariant::Outline,
100                                tablo_ui::ButtonSize::Sm,
101                            ))
102                        >
103                            icon(data: tablo_ui::icons::PLUS)
104                            (create_label)
105                        </a>
106                    }
107                </div>
108                (body)
109            </section>
110        }
111        .boxed())
112    }))
113}
