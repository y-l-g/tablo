25/// Renders the relation tables of `R`'s record `owner`.
26pub(crate) fn render_relations<'a, R: Resource>(
27    cx: &'a Cx,
28    owner: &R::Model,
29    read_only: bool,
30) -> Vec<BoxView<'a>> {
31    let page = topcoat::router::request::uri(cx).path().to_string();
32    declared::<R>(cx)
33        .relations
34        .iter()
35        .map(|relation| relation.render(cx, owner, &page, read_only, &R::slug()))
36        .collect()
37}

39/// Derives one relation table's action chrome, view-only on a read-only page.
40pub(crate) fn relation_chrome<C: Resource>(cx: &Cx, read_only: bool) -> TableChrome {
41    let declared = declared_chrome::<C>(cx);
42    if read_only {
43        TableChrome {
44            view: declared.view,
45            ..TableChrome::default()
46        }
47    } else {
48        declared
49    }
50}

52/// Renders one relation's section as `C`'s list table over the rows the owner holds.
53pub(crate) fn relation_table<C: Resource>(cx: &Cx, relation: BoundRelation) -> BoxView<'_> {
54    Box::pin(HoistView::new(ThenView::new(async move {
55        if enforce_tenant::<C>(cx).is_err() || !can::<C>(cx, Ability::ViewAny) {
56            return Ok(().boxed());
57        }
58        let table = wire_table::<C>(cx, false, relation_chrome::<C>(cx, relation.read_only));
59        let state = table.normalize_state(&TableState::from_cx_prefixed(cx, &relation.key));
60        let BoundRelation {
61            ref seed,
62            ref page,
63            read_only,
64            ..
65        } = relation;
66        let table = table.returning_to(state.list_url(page));
67        let create_url =
68            (!read_only && <C::Form as RecordForm>::HAS_FORM && can::<C>(cx, Ability::Create))
69                .then(|| {
70                    let query = form_urlencoded::Serializer::new(String::new())
71                        .append_pair(&seed.0, &seed.1)
72                        .append_pair(RETURN_PARAM, &state.list_url(page))
73                        .finish();
74                    format!("{}?{query}", create_page_url(&list_url(cx, &C::slug())))
75                });
76        if table.is_live_search() {
77            return relation_table_live::<C>(cx, table, state, relation, create_url).await;
78        }
79        let BoundRelation {
80            key,
81            label,
82            scope,
83            page,
84            ..
85        } = relation;
86        let body = match load_scoped_page::<C>(cx, &table, &state, scope).await {
87            Ok(rows) => table.render_with_state(cx, rows, &state, &page).await?,
88            Err(error) => table_error_view::<C>(cx, &state, &error, &page, None),
89        };
90        let header = relation_header::<C>(cx, label, create_url);
91        Ok(view! {
92            cx =>
93            <section class="flex flex-col gap-3" data-relation=(key)>
94                (header)
95                (body)
96            </section>
97        }
98        .boxed())
99    })))
100}

102/// Renders a live-search relation's section with its bars hoisted above the streamed region.
103async fn relation_table_live<C: Resource>(
104    cx: &Cx,
105    table: Table<C::Model>,
106    state: TableState,
107    relation: BoundRelation,
108    create_url: Option<String>,
109) -> Result<BoxView<'_>>
110where
111    C::Model: toasty::schema::Model + Send + Sync + 'static,
112{
113    let BoundRelation {
114        parent,
115        key,
116        label,
117        seed,
118        page,
119        read_only,
120        scope: _,
121    } = relation;
122    // Keyed by page and relation key so one relation's search never filters the next.
123    let signals = TableState::signals_for(
124        &cx.keyed(format!("{page}#{key}").as_str()),
125        &request_query(cx),
126    );
127    let host = if table.search_enabled() {
128        Some(
129            table
130                .render_live_search_bar(cx, &state, &page, &signals)
131                .await?,
132        )
133    } else {
134        None
135    };
136    let filter_bar = if table.filter_bar_enabled() {
137        Some(
138            table
139                .render_live_filter_bar(cx, &state, &page, &signals)
140                .await?,
141        )
142    } else {
143        None
144    };
145    let table = table.hide_search().hide_filter_bar().unframed();
146    let skeleton = table.render_skeleton(cx, &state).await?;
147    let delete_dialog = table.render_delete_dialog(cx, &state).await?;
148    let header = relation_header::<C>(cx, label, create_url);
149    let invocation_key = key.clone();
150    let lazy_rows = ThenView::new(async move {
151        let retry_signals = signals.clone();
152        let rendered = table
153            .render_live_relation_invocation(
154                cx,
155                &parent,
156                &invocation_key,
157                RelationRequest {
158                    seed: seed.1.clone(),
159                    page: page.clone(),
160                    read_only,
161                },
162                signals,
163            )
164            .await;
165        match rendered {
166            Ok(view) => Ok(view),
167            Err(error) => Ok(table_error_view::<C>(
168                cx,
169                &state,
170                &error,
171                &page,
172                Some(&retry_signals),
173            )),
174        }
175    });
176
177    Ok(view! {
178        cx =>
179        <section class="flex flex-col gap-3" data-relation=(key)>
180            (header)
181            <div class=(TABLE_CARD_CLASS)>
182                if let Some(host) = host {
183                    (host)
184                }
185                if let Some(bar) = filter_bar {
186                    (bar)
187                }
188                suspense(fallback: skeleton, (lazy_rows.boxed()))
189            </div>
190            if let Some(dialog) = delete_dialog {
191                (dialog)
192            }
193        </section>
194    }
195    .boxed())
196}

198/// Renders a relation section's heading row with its create-child link.
199fn relation_header<C: Resource>(cx: &Cx, label: String, create_url: Option<String>) -> BoxView<'_> {
200    let create_label = format!("New {}", C::label());
201    view! {
202        cx =>
203        <div class="flex items-center justify-between gap-4">
204            <h2 class="text-lg font-semibold tracking-tight text-foreground">
205                (label)
206            </h2>
207            if let Some(url) = create_url {
208                <a
209                    (runtime_link(cx, &url))
210                    class=(tablo_ui::button_variants(
211                        tablo_ui::ButtonVariant::Outline,
212                        tablo_ui::ButtonSize::Sm,
213                    ))
214                >
215                    icon(data: tablo_ui::icons::PLUS)
216                    (create_label)
217                </a>
218            }
219        </div>
220    }
221    .boxed()
