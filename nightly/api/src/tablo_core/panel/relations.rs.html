26/// The relation tables of `R`'s record `owner`, for the page at the request's
27/// path: one section per [`Resource::relations`] entry, in declaration order.
28/// A `read_only` page — the detail page — shows the rows without write
29/// actions, as Filament's view page does; the edit page carries them.
30pub(crate) fn render_relations<'a, R: Resource>(
31    cx: &'a Cx,
32    owner: &R::Model,
33    read_only: bool,
34) -> Vec<BoxView<'a>> {
35    let page = topcoat::router::request::uri(cx).path().to_string();
36    declared::<R>(cx)
37        .relations
38        .iter()
39        .map(|relation| relation.render(cx, owner, &page, read_only, &R::slug()))
40        .collect()
41}

43/// The action chrome of one relation's table: the resource's declared chrome,
44/// or view-only on a read-only page — the detail page shows the rows without
45/// write actions, as Filament's view page does; the edit page carries them.
46pub(crate) fn relation_chrome<C: Resource>(cx: &Cx, read_only: bool) -> TableChrome {
47    let declared = declared_chrome::<C>(cx);
48    if read_only {
49        TableChrome {
50            view: declared.view,
51            ..TableChrome::default()
52        }
53    } else {
54        declared
55    }
56}

58/// One relation's section: `C`'s list table over the rows the owner holds,
59/// its URL parameters prefixed with the relation's key, and — unless the
60/// page is read-only — its write actions returning to this page and a create
61/// link that opens `C`'s form with the owner chosen.
62///
63/// `C`'s own gates apply: a request that lacks a tenant `C` requires, or that
64/// `ViewAny` refuses, gets no section, and the row actions keep their
65/// per-row policy. A live-search table renders its search and filter bars
66/// eagerly above the streamed region while the relation shard invocation
67/// fills the table below, so sort, search, filters and pagination re-render
68/// in place with focus and scroll surviving; the GET form stays inside
69/// `<noscript>` as the no-JS fallback.
70pub(crate) fn relation_table<C: Resource>(cx: &Cx, relation: BoundRelation) -> BoxView<'_> {
71    Box::pin(HoistView::new(ThenView::new(async move {
72        if enforce_tenant::<C>(cx).is_err() || !can::<C>(cx, Ability::ViewAny) {
73            return Ok(().boxed());
74        }
75        let table = wire_table::<C>(cx, false, relation_chrome::<C>(cx, relation.read_only));
76        let state = table.normalize_state(&TableState::from_cx_prefixed(cx, &relation.key));
77        let BoundRelation {
78            ref seed,
79            ref page,
80            read_only,
81            ..
82        } = relation;
83        // A write returns to the page as the table shows it, without a
84        // dialog left open on a row the write removed.
85        let table = table.returning_to(state.list_url(page));
86        let create_url =
87            (!read_only && <C::Form as RecordForm>::HAS_FORM && can::<C>(cx, Ability::Create))
88                .then(|| {
89                    let query = form_urlencoded::Serializer::new(String::new())
90                        .append_pair(&seed.0, &seed.1)
91                        .append_pair(RETURN_PARAM, &state.list_url(page))
92                        .finish();
93                    format!("{}?{query}", create_page_url(&list_url(cx, &C::slug())))
94                });
95        if table.is_live_search() {
96            return relation_table_live::<C>(cx, table, state, relation, create_url).await;
97        }
98        let BoundRelation {
99            key,
100            label,
101            scope,
102            page,
103            ..
104        } = relation;
105        let body = match load_scoped_page::<C>(cx, &table, &state, scope).await {
106            Ok(rows) => table.render_with_state(cx, rows, &state, &page).await?,
107            Err(error) => table_error_view::<C>(cx, &state, &error, &page, None),
108        };
109        let header = relation_header::<C>(cx, label, create_url);
110        Ok(view! {
111            cx =>
112            <section class="flex flex-col gap-3" data-relation=(key)>
113                (header)
114                (body)
115            </section>
116        }
117        .boxed())
118    })))
119}

121/// A live-search relation's section: the same header and create link as the
122/// static section, the search and filter bars hoisted above the streamed
123/// region, and the relation shard invocation below.
124///
125/// The bars sit outside the swapped region for the same focus rationale as
126/// the live list page: a `<select>` change re-renders the table, and a
127/// control inside the swapped region would lose focus and collapse its popup
128/// mid-change. The signals are keyed by page and relation key, so several
129/// relations on one page never share a query or a selection.
130async fn relation_table_live<C: Resource>(
131    cx: &Cx,
132    table: Table<C::Model>,
133    state: TableState,
134    relation: BoundRelation,
135    create_url: Option<String>,
136) -> Result<BoxView<'_>>
137where
138    C::Model: toasty::schema::Model + Send + Sync + 'static,
139{
140    let BoundRelation {
141        parent,
142        key,
143        label,
144        seed,
145        page,
146        read_only,
147        scope: _,
148    } = relation;
149    // Seeded with the request's query as written: the shard parses its own
150    // prefixed parameters out of it and normalizes them as this page did.
151    //
152    // Keyed by page and relation key: runtime navigation carries every signal
153    // the next page shares with this one, and one call site would otherwise
154    // give every relation on the page the same ids — one relation's search
155    // would filter the next.
156    let signals = TableState::signals_for(
157        &cx.keyed(format!("{page}#{key}").as_str()),
158        &request_query(cx),
159    );
160    let host = if table.search_enabled() {
161        Some(
162            table
163                .render_live_search_bar(cx, &state, &page, &signals)
164                .await?,
165        )
166    } else {
167        None
168    };
169    // The filter bar is hoisted next to the search host: a
170    // `<select>` change re-renders the table, and a control inside the
171    // swapped region would lose focus and collapse its popup mid-change.
172    let filter_bar = if table.filter_bar_enabled() {
173        Some(
174            table
175                .render_live_filter_bar(cx, &state, &page, &signals)
176                .await?,
177        )
178    } else {
179        None
180    };
181    // The shard's table renders neither bar, so neither does the placeholder
182    // it replaces.
183    // Nor does it draw a card: the section's card holds the bars and the
184    // table together.
185    let table = table.hide_search().hide_filter_bar().unframed();
186    let skeleton = table.render_skeleton(cx, &state).await?;
187    // The delete confirmation dialog is not part of the swapped table
188    // region: a keystroke starts a new result set and must never carry
189    // (or re-open) a dialog, so the live section renders it eagerly once.
190    let delete_dialog = table.render_delete_dialog(cx, &state).await?;
191    let header = relation_header::<C>(cx, label, create_url);
192    // The view below moves `key` into the section marker; the streamed rows
193    // only borrow it, so they take a clone.
194    let invocation_key = key.clone();
195    let lazy_rows = ThenView::new(async move {
196        // The retry link inside the table writes the same signals the
197        // toolbar does, so a bad cursor recovers in place.
198        let retry_signals = signals.clone();
199        let rendered = table
200            .render_live_relation_invocation(
201                cx,
202                &parent,
203                &invocation_key,
204                RelationRequest {
205                    seed: seed.1.clone(),
206                    page: page.clone(),
207                    read_only,
208                },
209                signals,
210            )
211            .await;
212        match rendered {
213            Ok(view) => Ok(view),
214            Err(error) => Ok(table_error_view::<C>(
215                cx,
216                &state,
217                &error,
218                &page,
219                Some(&retry_signals),
220            )),
221        }
222    });
223
224    Ok(view! {
225        cx =>
226        <section class="flex flex-col gap-3" data-relation=(key)>
227            (header)
228            // The section draws the table's card (`TABLE_CARD_CLASS`) so the
229            // hoisted bars share it with the table they drive, as on the live
230            // list page.
231            <div class=(TABLE_CARD_CLASS)>
232                if let Some(host) = host {
233                    (host)
234                }
235                if let Some(bar) = filter_bar {
236                    (bar)
237                }
238                suspense(fallback: skeleton, (lazy_rows.boxed()))
239            </div>
240            if let Some(dialog) = delete_dialog {
241                (dialog)
242            }
243        </section>
244    }
245    .boxed())
246}

248/// A relation section's heading row: the relation's label and, when the
249/// page allows it, the link that creates a child seeded with this owner.
250fn relation_header<C: Resource>(cx: &Cx, label: String, create_url: Option<String>) -> BoxView<'_> {
251    let create_label = format!("New {}", C::label());
252    view! {
253        cx =>
254        <div class="flex items-center justify-between gap-4">
255            <h2 class="text-lg font-semibold tracking-tight text-foreground">
256                (label)
257            </h2>
258            if let Some(url) = create_url {
259                <a
260                    (runtime_link(cx, &url))
261                    class=(tablo_ui::button_variants(
262                        tablo_ui::ButtonVariant::Outline,
263                        tablo_ui::ButtonSize::Sm,
264                    ))
265                >
266                    icon(data: tablo_ui::icons::PLUS)
267                    (create_label)
268                </a>
269            }
270        </div>
271    }
272    .boxed()
