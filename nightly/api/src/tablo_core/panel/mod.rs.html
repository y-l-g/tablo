1//! `Panel` — an admin panel an app mounts into its router.
2//!
3//! A panel registers each declared [`Resource`]'s routes and each [`Page`] at
4//! `{prefix}/{slug}`, and the home page at the prefix (Filament-style routes —
5//! ADR-0008), on a [`RouterBuilder`](topcoat::router::RouterBuilder) the app
6//! owns. A router mounts any number of panels at distinct prefixes. See
7//! `CONTEXT.md`.
8//!
9//! Layout: the [`Panel`] builder and its navigation seam live here; mounting
10//! (declaration checks, route paths) in `build`; each mounted panel's state
11//! and the request's panel in `state`; the URL helpers in `url`; the
12//! auth/tenant gate and prefix URLs in `gate`; shell rendering in `shell`;
13//! list + live shard support in `list`; form decoding and create/edit in
14//! `forms`; the record detail page in `detail`; delete/bulk/export/options in
15//! `actions`; the registered-page handler in `pages`; the live-search registry
16//! + shard dispatch in `search`; and response hardening headers in `headers`.

78/// The table the panel's list page serves for `R`, for a page that owns its
79/// table instead of mounting the panel's list route.
80///
81/// The page-owned seam (GH #154 §2) pairs this with
82/// [`TablePage::load`](crate::resource::TablePage::load) and
83/// [`Table::render_with_state`](crate::resource::Table::render_with_state).
84/// The table carries `R::table`'s columns, key, page size, search toolbar and
85/// filter bar, plus the action chrome `R`'s declarations imply — the row
86/// Delete link and bulk column from
87/// [`DeleteAny`](crate::policy::Ability::DeleteAny), the Edit link
88/// from a record form ([`RecordForm::HAS_FORM`])
89/// and the View link from a non-empty [`view`](crate::resource::Resource::view) — each gated
90/// per row by `View`/`Update`/`Delete`, the wiring the panel's own
91/// list applies. The chrome has no other entry point: a page-owned table that
92/// must agree with the resource's routes takes its wiring from here.
93pub fn wired_table<R: Resource>(cx: &topcoat::context::Cx) -> crate::resource::Table<R::Model> {
94    self::list::wire_table_actions::<R>(cx, false)

97/// An admin panel: resources and pages under one prefix, framed by one shell
98/// and gated by one [`Auth`](crate::auth::Auth).
99///
100/// The app owns the router and mounts the panel into it with
101/// [`RouterBuilderPanelExt::panel`]:
102///
103/// ```ignore
104/// let router = Router::builder()
105///     .discover()
106///     .app_context(db)
107///     .panel(Panel::new("admin").resource::<UserResource>())?
108///     .build();
109/// ```
110pub struct Panel {
111    prefix: String,
112    shell_assets: Option<ShellAssets>,
113    brand: Option<Brand>,
114    dark_mode: Option<bool>,
115    nav_items: Vec<NavigationItem>,
116    pages: Vec<PageFn>,
117    routes: Vec<RouteFn>,
118    root: Option<Root>,
119    /// The layout framing the panel's pages; `None` is the shipped shell.
120    layout: Option<LayoutRenderFn>,
121    /// The URL each registered resource and page is served at, by type: what
122    /// the [`url`] helpers answer.
123    urls: HashMap<TypeId, String>,
124    /// Every slug a resource or a page mounts at: one namespace, since both
125    /// mount at `{prefix}/{slug}`.
126    slugs: Vec<String>,
127    /// The slugs of the registered resources, a subset of `slugs`.
128    resource_slugs: Vec<String>,
129    /// Each registered resource's relation keys, by resource type name:
130    /// mounting checks that each names a registered resource, once.
131    relations: Vec<(&'static str, Vec<String>)>,
132    search_handlers: HashMap<String, SearchFn>,
133    /// Each registered resource's relations' live-search loaders, by
134    /// (parent slug, child slug): the shard behind a record page's
135    /// relation tables.
136    relation_handlers: HashMap<(String, String), RelationSearchFn>,
137    /// `Content-Security-Policy: frame-ancestors …` for every response under
138    /// the prefix; `None` opts out. Defaults to `'self'`.
139    frame_ancestors: Option<String>,
140    /// Per-resource declaration checks, monomorphized at
141    /// `resource::<R>()` and run at mount, before anything is served.
142    resource_checks: Vec<ResourceCheck>,
143    /// Registration failures collected by the declarative builders
144    /// `Panel::resource` cannot return `Result`, so a bad `slug`
145    /// or a duplicate is recorded here and reported when the panel is mounted.
146    registration_errors: Vec<String>,
147    /// Where file field bytes go; `None` stores the sanitized basename.
148    uploads: Option<crate::upload::InstalledUploader>,
149    /// App-owned filesystem directories served with the hardening headers
150    /// `(route pattern, directory)`.
151    served_dirs: Vec<(String, PathBuf)>,
152    login_hint: Option<String>,
153    auth: crate::auth::Auth,
154}

157    /// Create a `Panel` mounted at `prefix` (e.g. `"admin"` → `"/admin"`).
158    /// An empty prefix mounts at `"/admin"`.
159    pub fn new(prefix: impl Into<String>) -> Self {
160        let raw = prefix.into();
161        let trimmed = raw.trim().trim_matches('/').to_string();
162        let prefix = if trimmed.is_empty() {
163            "/admin".to_string()
164        } else {
165            format!("/{trimmed}")
166        };
167        // The prefix is free-form too, and every route path is built from it
168        // validate it once here rather than panicking at the first
169        // `route_path` call.
170        let registration_errors: Vec<String> = prefix
171            .trim_matches('/')
172            .split('/')
173            .filter_map(|segment| validate_route_segment("panel prefix", segment).err())
174            .collect();
175        Self {
176            prefix,
177            shell_assets: None,
178            brand: None,
179            dark_mode: None,
180            nav_items: Vec::new(),
181            pages: Vec::new(),
182            routes: Vec::new(),
183            root: None,
184            layout: None,
185            urls: HashMap::new(),
186            slugs: Vec::new(),
187            resource_slugs: Vec::new(),
188            relations: Vec::new(),
189            search_handlers: HashMap::new(),
190            relation_handlers: HashMap::new(),
191            frame_ancestors: Some(headers::DEFAULT_FRAME_ANCESTORS.to_string()),
192            registration_errors,
193            resource_checks: Vec::new(),
194            uploads: None,
195            served_dirs: Vec::new(),
196            login_hint: None,
197            auth: crate::auth::Auth::default(),
198        }
199    }

217    /// Serve a directory of files, with the headers that keep an uploaded file
218    /// inert on the panel's origin.
219    ///
220    /// `path` is a route pattern ending in a catch-all (e.g.
221    /// `"/uploads/{*file}"`), and `dir` the directory those URLs read from —
222    /// see Topcoat's `DirectoryRoute` for the resolution rules. The path is
223    /// **not** panel-relative: a served directory holds files a record points
224    /// at (an upload store's output), which are not a page of the panel and
225    /// must not move when the panel is mounted elsewhere. This is **public**:
226    /// the auth gate covers only the panel prefix and `/_topcoat/runtime`
227    /// (ADR-0013), so a served directory sits outside it and its URLs answer
228    /// whoever asks, with no session — an app that needs protected files owns
229    /// that route itself (ADR-0017).
230    ///
231    /// Every file the directory route serves carries `nosniff`, a sandboxing
232    /// `Content-Security-Policy`, and `Content-Disposition: attachment` for
233    /// anything but common raster images, audio/video and plain text, so an
234    /// uploaded document cannot run script on the panel's origin. A
235    /// 404 keeps Topcoat's `text/plain` error page; a 405 carries only `Allow`
236    /// and an empty body. Neither carries user content.
237    pub fn serve_dir(mut self, path: impl Into<String>, dir: impl Into<PathBuf>) -> Self {
238        let path = path.into();
239        if !is_directory_pattern(&path) {
240            self.registration_errors.push(format!(
241                "serve_dir path '{path}': must end in a catch-all like '/uploads/{{*file}}'"
242            ));
243        }
244        self.served_dirs.push((path, dir.into()));
245        self
246    }

259    /// Declare a `Resource` for this panel (the declarative seam, ADR-0008).
260    ///
261    /// Registers the resource's **list page** at `{prefix}/{slug}` (e.g.
262    /// `Panel::new("admin").resource::<AuditResource>()` serves `/admin/audits`),
263    /// its detail page, delete, bulk delete, and CSV export, and derives its
264    /// [`NavigationItem`] from the same slug, so the sidebar and the router can
265    /// never disagree. Without a [`home`](Self::home) page, the panel root
266    /// redirects to the first declared resource's list. Multiple calls
267    /// compose. Sidebar order comes from the resource's
268    /// [`Resource::navigation`] override, defaulting to declaration order
269    /// (#165).
270    ///
271    /// A resource whose [`Form`](Resource::Form) is a record form
272    /// ([`RecordForm::HAS_FORM`]) also gets the create page, the edit page, and
273    /// the relationship-options endpoint; one that names
274    /// [`NoForm`](crate::NoForm) gets none of them. The route set follows the
275    /// resource type, so one type serves the same routes in every panel.
276    ///
277    /// [`RouterBuilderPanelExt::panel`] checks that the resource's [`form`](Resource::form)
278    /// agrees with its `Form`, that a resource with no form does not allow
279    /// `Create`, and, for a resource with a form, that the
280    /// form's struct and its `Schema` agree: every control is bound by exactly
281    /// one field and every field's key is a declared control; an optional
282    /// control, or one inside a `Repeater` or a variant group, binds a field
283    /// with a blank answer; a [`Tenancy::column`](crate::Tenancy::column)
284    /// resource's form does not claim its tenant column; and, where `Create` allows it, every
285    /// non-nullable column is set by the form, by toasty, by the tenant stamp, or by an
286    /// override that names it in [`Resource::CREATE_COLUMNS`]. It also checks that each of the
287    /// resource's [`relations`](Resource::relations) names a resource this
288    /// panel registers, once: the relation's actions go to that resource's
289    /// routes.
290    ///
291    /// A slug another resource or [page](Self::page) holds, one the panel
292    /// routes itself (`login`, `logout`), or one that is not one URL segment
293    /// is recorded here and reported by [`RouterBuilderPanelExt::panel`], which returns `Err`
294    /// instead of panicking: two routes over one slug would shadow each other,
295    /// and a hostile `slug()` must not reach a route path or a response header.
296    pub fn resource<R: Resource>(mut self) -> Self {
297        let Some(url) = self.register_common::<R>() else {
298            return self;
299        };
300        if <R::Form as RecordForm>::HAS_FORM {
301            self.register_form_routes::<R>(&url);
302        }
303        self.finish_registration::<R>(url);
304        let declared = R::relations();
305        let keys = declared
306            .iter()
307            .map(|relation| relation.key().to_string())
308            .collect::<Vec<_>>();
309        for relation in &declared {
310            if let Some(error) = relation.misdeclared() {
311                self.registration_errors.push(format!(
312                    "resource `{}`'s relation `{}`: {error}",
313                    std::any::type_name::<R>(),
314                    relation.key()
315                ));
316            }
317            self.relation_handlers.insert(
318                (R::slug(), relation.key().to_string()),
319                relation.search_handler(),
320            );
321        }
322        if !keys.is_empty() {
323            self.relations.push((std::any::type_name::<R>(), keys));
324        }
325        self
326    }

330    fn register_form_routes<R: Resource>(&mut self, url: &str) {
331        // Create page — GET renders form, POST handles submission.
332        let create_url = format!("{url}/{CREATE_ROUTE_SEGMENT}");
333        self.pages.push(PageFn::new(
334            http::Method::GET,
335            route_path(&create_url),
336            resource_create::<R>,
337        ));
338        self.pages.push(PageFn::new(
339            http::Method::POST,
340            route_path(&create_url),
341            resource_create_post::<R>,
342        ));
343        // Edit page — GET renders hydrated form, POST handles update.
344        let edit_url = format!("{url}/{RECORD_ROUTE_PARAM}/{EDIT_ROUTE_SEGMENT}");
345        self.pages.push(PageFn::new(
346            http::Method::GET,
347            route_path(&edit_url),
348            resource_edit::<R>,
349        ));
350        self.pages.push(PageFn::new(
351            http::Method::POST,
352            route_path(&edit_url),
353            resource_edit_post::<R>,
354        ));
355        // Relationship option search — GET for searchable selects past the cap
356        // `{list_url}/options?field=&q=` reusing the related
357        // table's searchable columns, bounded, policy-checked.
358        let options_url = format!("{url}/options");
359        self.routes.push(RouteFn::new(
360            http::Method::GET,
361            route_path(&options_url),
362            resource_options::<R>,
363        ));
364    }

366    /// The routes every resource registers: the list, the detail page,
367    /// delete, bulk delete, and export. Returns the list URL,
368    /// or `None` when the slug was refused.
369    fn register_common<R: Resource>(&mut self) -> Option<String> {
370        let url = self.claim_slug::<R>("Resource::slug", R::slug())?;
371        self.resource_slugs.push(R::slug());
372        self.resource_checks.push(check_resource::<R>);
373        self.pages.push(PageFn::new(
374            http::Method::GET,
375            route_path(&url),
376            resource_list::<R>,
377        ));
378        // Detail page — GET renders the record read-only. Registered
379        // unconditionally, unlike the row link: registration runs before the
380        // mount has the `Db` the view declaration resolves through.
381        // The handler 404s a resource that declares no view, which is the same
382        // answer as an unknown id and costs one comparison.
383        //
384        // `RECORD_ROUTE_PARAM` shares its position with the literal `create`
385        // segment: topcoat routes through `matchit`, which prefers a static
386        // segment over a parameter one, so the create page keeps being reached
387        // regardless of registration order.
388        let detail_url = format!("{url}/{RECORD_ROUTE_PARAM}");
389        self.pages.push(PageFn::new(
390            http::Method::GET,
391            route_path(&detail_url),
392            resource_view::<R>,
393        ));
394        // Delete action — POST via row button (requires confirmation).
395        let delete_url = format!("{url}/{RECORD_ROUTE_PARAM}/{DELETE_ROUTE_SEGMENT}");
396        self.pages.push(PageFn::new(
397            http::Method::POST,
398            route_path(&delete_url),
399            resource_delete::<R>,
400        ));
401        // Bulk delete — POST with `ids` form field (comma-separated).
402        let bulk_delete_url = format!("{url}/{BULK_DELETE_ROUTE_SEGMENT}");
403        self.pages.push(PageFn::new(
404            http::Method::POST,
405            route_path(&bulk_delete_url),
406            resource_bulk_delete::<R>,
407        ));
408        // Custom actions — POST per row and for the bulk selection, the action
409        // named by the last segment. Registered only for a resource that
410        // declares some: the list-level route's static `actions` segment
411        // would otherwise shadow the edit and delete routes of a record whose
412        // key is `actions`, for nothing.
413        if !R::actions().entries().is_empty() {
414            let row_action_url =
415                format!("{url}/{RECORD_ROUTE_PARAM}/{ACTIONS_ROUTE_SEGMENT}/{ACTION_ROUTE_PARAM}");
416            self.pages.push(PageFn::new(
417                http::Method::POST,
418                route_path(&row_action_url),
419                resource_row_action::<R>,
420            ));
421            let bulk_action_url = format!("{url}/{ACTIONS_ROUTE_SEGMENT}/{ACTION_ROUTE_PARAM}");
422            self.pages.push(PageFn::new(
423                http::Method::POST,
424                route_path(&bulk_action_url),
425                resource_bulk_action::<R>,
426            ));
427        }
428        // CSV export — GET over the tenant-scoped export query + Table
429        // filters/sort (ADR-0012).
430        let export_url = format!("{url}/export");
431        self.routes.push(RouteFn::new(
432            http::Method::GET,
433            route_path(&export_url),
434            resource_export::<R>,
435        ));
436        Some(url)
437    }

439    /// The registration tail every resource shares: its live-search handler,
440    /// the panel root, and its sidebar entry.
441    fn finish_registration<R: Resource>(&mut self, url: String) {
442        // Live-search handler: the slug-dispatched `#[shard]` cannot be generic
443        // (inventory only discovers concrete fns), so each resource
444        // monomorphizes its table loader here, keyed by list path.
445        self.search_handlers
446            .insert(url.clone(), search_handler_for::<R>());
447        if self.root.is_none() {
448            self.root = Some(Root::Redirect(url));
449        }
450        let nav_item = self.nav_item::<R>();
451        self.nav_items.push(nav_item);
452    }

454    /// Claim `{prefix}/{slug}` for the resource or page `T`, returning that
455    /// URL, or record why the slug is refused: it is not one URL segment, the
456    /// panel routes it itself, or another resource or page holds it.
457    fn claim_slug<T: 'static>(&mut self, kind: &str, slug: String) -> Option<String> {
458        let owner = std::any::type_name::<T>();
459        let refused = if let Err(error) = validate_route_segment(kind, &slug) {
460            Some(error)
461        } else if RESERVED_SLUGS.contains(&slug.as_str()) {
462            Some(format!(
463                "slug '{slug}' of `{owner}`: the panel routes `{}/{slug}` itself",
464                self.prefix
465            ))
466        } else if self.slugs.contains(&slug) {
467            Some(format!(
468                "duplicate slug '{slug}': `{owner}` mounts where another resource or page does \
469                 — each needs a distinct `slug()`"
470            ))
471        } else {
472            None
473        };
474        if let Some(error) = refused {
475            self.registration_errors.push(error);
476            return None;
477        }
478        let url = format!("{}/{slug}", self.prefix);
479        self.slugs.push(slug);
480        self.urls.insert(TypeId::of::<T>(), url.clone());
481        Some(url)
482    }

484    /// Declare a [`Page`] for this panel: its `GET` at `{prefix}/{slug}`
485    /// and its sidebar entry at the same URL.
486    ///
487    /// Pages and resources share one slug namespace, and the slug is checked
488    /// as a resource's is ([`Self::resource`]): a refused one is reported by
489    /// [`RouterBuilderPanelExt::panel`].
490    pub fn page<P: Page>(mut self) -> Self {
491        if let Some(url) = self.claim_slug::<P>("Page::slug", P::slug()) {
492            let item = self.mount_page::<P>(&url);
493            self.nav_items.push(item);
494        }
495        self
496    }

498    /// Declare the panel's home page (Filament's dashboard): the [`Page`]
499    /// served at the panel prefix itself, with a sidebar entry there.
500    ///
501    /// It replaces the redirect to the first resource's list, and its entry
502    /// leads the sidebar among entries of the same `order`. `P::slug()` is
503    /// not read. A second call is recorded and reported by [`RouterBuilderPanelExt::panel`].
504    pub fn home<P: Page>(mut self) -> Self {
505        if matches!(self.root, Some(Root::Home)) {
506            self.registration_errors.push(format!(
507                "Panel::home: a home page is already registered; `{}` would shadow it",
508                std::any::type_name::<P>()
509            ));
510            return self;
511        }
512        self.root = Some(Root::Home);
513        self.urls.insert(TypeId::of::<P>(), self.prefix.clone());
514        let item = self.mount_page::<P>(&self.prefix.clone());
515        // First among equal `order`s whatever the call order, as Filament's
516        // dashboard leads its sidebar.
517        self.nav_items.insert(0, item);
518        self
519    }

550    /// Set the `frame-ancestors` directive the panel sends on every response
551    /// under its prefix.
552    ///
553    /// Defaults to `'self'`: the admin only frames itself, so a hostile page
554    /// cannot clickjack it. Pass what your deployment needs — `"'self'
555    /// https://intranet.example"` to allow an internal portal, or `"'none'"`
556    /// to forbid framing outright.
557    ///
558    /// The layer only fills the gap, so an app that sets its own
559    /// `Content-Security-Policy` (its own layer or route) keeps it. Use
560    /// [`Self::without_frame_ancestors`] to send nothing at all and leave the
561    /// decision to a proxy.
562    pub fn frame_ancestors(mut self, ancestors: impl Into<String>) -> Self {
563        self.frame_ancestors = Some(ancestors.into());
564        self
565    }

567    /// Send no `frame-ancestors` directive: the escape hatch for
568    /// deployments whose proxy owns the whole CSP.
569    ///
570    /// Off by default in the sense that nothing is *added* — the panel's
571    /// `'self'` default is what this opts out of.
572    pub fn without_frame_ancestors(mut self) -> Self {
573        self.frame_ancestors = None;
574        self
575    }

577    /// Set the theme a visitor who has not chosen one sees: dark when `true`.
578    ///
579    /// The shell always renders the theme toggle, and a visitor's stored
580    /// choice wins over this default in both directions. Without this call
581    /// the panel starts light.
582    ///
583    /// The choice persists through `theme.js` (`localStorage` and a `theme`
584    /// cookie). The server renders `<html class>` from the cookie when it
585    /// carries `dark` or `light`, and from this default otherwise; the
586    /// blocking `theme_init_script` then applies `localStorage`, then the
587    /// cookie, with this default as its fallback. Rendering the cookie matters
588    /// for runtime navigation, which copies `<html>`'s attributes from the next
589    /// page and runs no script.
590    pub fn dark_mode(mut self, enabled: bool) -> Self {
591        self.dark_mode = Some(enabled);
592        self
593    }
594
595    /// Configure authentication (ADR-0013).
596    ///
597    /// The default is the shipped [`Auth::password`](crate::auth::Auth::password)
598    /// over [`AdminUser`](crate::auth::AdminUser); swap in an app-owned
599    /// authenticator with `Panel::auth(Auth::custom(..))`, or opt a public
600    /// demo out explicitly with `Panel::auth(Auth::disabled())`.
601    ///
602    /// Each panel has its own auth and its own login page, and a session
603    /// belongs to the panel that signed it in: a user signed in to one panel
604    /// is not signed in to another, and signing in to another ends the first
605    /// session.
606    pub fn auth(mut self, auth: crate::auth::Auth) -> Self {
607        self.auth = auth;
608        self
609    }

611    /// A muted line rendered under the login form, for demo credentials or
612    /// deployment hints (e.g. `"Demo: admin@example.com / password"`).
613    pub fn login_hint(mut self, hint: impl Into<String>) -> Self {
614        self.login_hint = Some(hint.into());
615        self
616    }

631impl Panel {
632    /// Derive a [`NavigationItem`] for `R` using this panel's mount prefix.
633    ///
634    /// The one panel-aware navigation seam for a [`Resource`]:
635    /// [`Panel::resource`](Self::resource) calls it, so a resource's
636    /// [`Resource::navigation`] override reaches the sidebar instead of being
637    /// dead API. The override owns the **label, ordering and
638    /// grouping**; the panel owns the **URL**, because it is the only party
639    /// that knows where the resource is mounted. Concretely: an explicit
640    /// [`NavTarget::Url`] in `R::navigation()` is taken as returned, and only
641    /// a [`NavTarget::Derived`] target — the default, which names no URL
642    /// because [`Resource::navigation`] takes no prefix — is resolved to
643    /// `{prefix}/{slug}`.
644    ///
645    /// So `Panel::new("backoffice")` yields `"/backoffice/{slug}"` — never a
646    /// hard-coded `"/admin"` — for default and overridden items alike, a URL an
647    /// override spelled out stays exactly as written, and an override's `order`
648    /// decides sidebar order.
649    pub(crate) fn nav_item<R: Resource>(&self) -> NavigationItem {
650        R::navigation().resolved(&format!("{}/{}", self.prefix, R::slug()))
651    }
652}
