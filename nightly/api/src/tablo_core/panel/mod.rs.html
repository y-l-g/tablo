1//! `Panel` — the admin application shell.
2//!
3//! Owns the [`Router`](topcoat::router::Router) and the `Db` in `app_context`, and registers each
4//! declared [`Resource`]'s list page at `{prefix}/{slug}` (Filament-style
5//! routes — ADR-0008). See `CONTEXT.md`.
6//!
7//! Layout: the [`Panel`] builder and its navigation seam live here; assembly
8//! (`build`, declaration checks, route paths) in `build`; the auth/tenant
9//! gate and prefix URLs in `gate`; shell rendering in `shell`; list + live
10//! shard support in `list`; form decoding and create/edit in `forms`; the
11//! record detail page in `detail`; delete/bulk/export/options in `actions`;
12//! the live-search registry + shard dispatch in `search`; and response
13//! hardening headers in `headers`.

62/// The table the panel's list page serves for `R`, for a page that owns its
63/// table instead of mounting the panel's list route.
64///
65/// The page-owned seam (GH #154 §2) pairs this with
66/// [`Table::load`](crate::resource::Table::load) and
67/// [`Table::render_with_state`](crate::resource::Table::render_with_state).
68/// The table carries `R::table`'s columns, key, page size, search toolbar and
69/// filter bar, plus the action chrome `R`'s declarations imply — the row
70/// Delete link and bulk column from
71/// [`can_delete_any`](crate::resource::Resource::can_delete_any), the Edit link
72/// from a record form ([`RecordForm::HAS_FORM`])
73/// and the View link from [`viewed`](crate::resource::Resource::viewed) — each gated
74/// per row by `can_view`/`can_update`/`can_delete`, the wiring the panel's own
75/// list applies. The chrome has no other entry point: a page-owned table that
76/// must agree with the resource's routes takes its wiring from here.
77pub fn wired_table<R: Resource>(cx: &topcoat::context::Cx) -> crate::resource::Table<R::Model> {
78    self::list::wire_table_actions::<R>(cx, false)
79}

81/// The admin application.
82///
83/// ```ignore
84/// Panel::new("admin")
85///     .app_context(db)
86///     .resource::<UserResource>()
87///     .build().expect("panel builds")
88/// ```
89pub struct Panel {
90    prefix: String,
91    db: Option<Db>,
92    assets: Option<AssetConfig>,
93    shell_assets: Option<ShellAssets>,
94    brand: Option<Brand>,
95    dark_mode: Option<bool>,
96    nav_items: Vec<NavigationItem>,
97    pages: Vec<PageFn>,
98    routes: Vec<RouteFn>,
99    root_target: Option<String>,
100    slugs: Vec<String>,
101    search_handlers: HashMap<String, SearchFn>,
102    /// `Content-Security-Policy: frame-ancestors …` for every response
103    /// `None` opts out. Defaults to `'self'`.
104    frame_ancestors: Option<String>,
105    /// Per-resource declaration checks, monomorphized at
106    /// `resource::<R>()` and run by `build` before anything is served.
107    resource_checks: Vec<ResourceCheck>,
108    /// Registration failures collected by the declarative builders
109    /// `Panel::resource` cannot return `Result`, so a bad `slug`
110    /// or a duplicate is recorded here and reported by `build`.
111    registration_errors: Vec<String>,
112    /// Where `FileUpload` bytes go; `None` stores the sanitized basename.
113    uploads: Option<crate::upload::InstalledUploader>,
114    /// App-owned filesystem directories served from this panel's router
115    /// `(route pattern, directory)`.
116    served_dirs: Vec<(String, PathBuf)>,
117    login_hint: Option<String>,
118    auth: crate::auth::Auth,
119}

121impl Panel {
122    /// Create a `Panel` mounted at `prefix` (e.g. `"admin"` → `"/admin"`).
123    pub fn new(prefix: impl Into<String>) -> Self {
124        let raw = prefix.into();
125        let trimmed = raw.trim_matches('/').trim().to_string();
126        let prefix = if trimmed.is_empty() {
127            "/admin".to_string()
128        } else {
129            format!("/{trimmed}")
130        };
131        // The prefix is free-form too, and every route path is built from it
132        // validate it once here rather than panicking at the first
133        // `route_path` call.
134        let registration_errors: Vec<String> = prefix
135            .trim_matches('/')
136            .split('/')
137            .filter_map(|segment| validate_route_segment("panel prefix", segment).err())
138            .collect();
139        Self {
140            prefix,
141            db: None,
142            assets: None,
143            shell_assets: None,
144            brand: None,
145            dark_mode: None,
146            nav_items: Vec::new(),
147            pages: Vec::new(),
148            routes: Vec::new(),
149            root_target: None,
150            slugs: Vec::new(),
151            search_handlers: HashMap::new(),
152            frame_ancestors: Some(headers::DEFAULT_FRAME_ANCESTORS.to_string()),
153            registration_errors,
154            resource_checks: Vec::new(),
155            uploads: None,
156            served_dirs: Vec::new(),
157            login_hint: None,
158            auth: crate::auth::Auth::default(),
159        }
160    }

185    /// Serve a directory of files from this panel's router.
186    ///
187    /// `path` is a route pattern ending in a catch-all (e.g.
188    /// `"/uploads/{*file}"`), and `dir` the directory those URLs read from —
189    /// see Topcoat's `DirectoryRoute` for the resolution rules. The path is
190    /// **not** panel-relative: a served directory holds files a record points
191    /// at (an upload store's output), which are not a page of the panel and
192    /// must not move when the panel is mounted elsewhere. This is **public**:
193    /// the auth gate covers only the panel prefix and `/_topcoat/runtime`
194    /// (ADR-0013), so a served directory sits outside it and its URLs answer
195    /// whoever asks, with no session — an app that needs protected files owns
196    /// that route itself (ADR-0017).
197    ///
198    /// Every file the directory route serves carries `nosniff`, a sandboxing
199    /// `Content-Security-Policy`, and `Content-Disposition: attachment` for
200    /// anything but common raster images, audio/video and plain text, so an
201    /// uploaded document cannot run script on the panel's origin. A
202    /// 404 keeps Topcoat's `text/plain` error page; a 405 carries only `Allow`
203    /// and an empty body. Neither carries user content.
204    ///
205    /// The Panel owns the [`Router`](topcoat::router::Router), so this is the app's only way to
206    /// mount a route the framework does not own.
207    pub fn serve_dir(mut self, path: impl Into<String>, dir: impl Into<PathBuf>) -> Self {
208        let path = path.into();
209        if !is_directory_pattern(&path) {
210            self.registration_errors.push(format!(
211                "serve_dir path '{path}': must end in a catch-all like '/uploads/{{*file}}'"
212            ));
213        }
214        self.served_dirs.push((path, dir.into()));
215        self
216    }

237    /// Declare a `Resource` for this panel (the declarative seam, ADR-0008).
238    ///
239    /// Registers the resource's **list page** at `{prefix}/{slug}` (e.g.
240    /// `Panel::new("admin").resource::<AuditResource>()` serves `/admin/audits`),
241    /// its detail page, delete, bulk delete, and CSV export, and derives its
242    /// [`NavigationItem`] from the same slug, so the sidebar and the router can
243    /// never disagree. The panel root redirects to the first declared
244    /// resource's list. Multiple calls compose. Sidebar order comes from the
245    /// resource's [`Resource::navigation`] override, defaulting to declaration
246    /// order (#165).
247    ///
248    /// A resource whose [`Form`](Resource::Form) is a record form
249    /// ([`RecordForm::HAS_FORM`]) also gets the create page, the edit page, and
250    /// the relationship-options endpoint; one that names
251    /// [`NoForm`](crate::NoForm) gets none of them. The route set follows the
252    /// resource type, so one type serves the same routes in every panel.
253    ///
254    /// [`Panel::build`] checks that the resource's [`form`](Resource::form)
255    /// agrees with its `Form`, that a resource with no form does not allow
256    /// `can_create`, and, for a resource with a form, that the
257    /// form's struct and its `Schema` agree: every control is bound by exactly
258    /// one field and every field's key is a declared control; an optional
259    /// control, or one inside a `Repeater` or a variant group, binds a field
260    /// with a blank answer; a gated resource's form does not claim its tenant
261    /// column; and, where `can_create` allows it, every non-nullable column is
262    /// set by the form, by toasty, by the tenant stamp, or by an override that
263    /// names it in [`Resource::CREATE_COLUMNS`].
264    ///
265    /// A duplicate slug or a slug that is not one URL segment
266    /// is recorded here and reported by [`Panel::build`], which
267    /// returns `Err` instead of panicking: two resources over one slug would
268    /// shadow each other's routes, and a hostile `slug()` must not reach a
269    /// route path or a response header.
270    pub fn resource<R: Resource>(mut self) -> Self {
271        let Some(url) = self.register_common::<R>() else {
272            return self;
273        };
274        if <R::Form as RecordForm>::HAS_FORM {
275            self.register_form_routes::<R>(&url);
276        }
277        self.finish_registration::<R>(url);
278        self
279    }

281    /// The create page, the edit page, and the relationship-options endpoint
282    /// of a resource with a record form.
283    fn register_form_routes<R: Resource>(&mut self, url: &str) {
284        // Create page — GET renders form, POST handles submission.
285        let create_url = format!("{url}/{CREATE_ROUTE_SEGMENT}");
286        self.pages.push(PageFn::new(
287            http::Method::GET,
288            route_path(&create_url),
289            resource_create::<R>,
290        ));
291        self.pages.push(PageFn::new(
292            http::Method::POST,
293            route_path(&create_url),
294            resource_create_post::<R>,
295        ));
296        // Edit page — GET renders hydrated form, POST handles update.
297        let edit_url = format!("{url}/{RECORD_ROUTE_PARAM}/{EDIT_ROUTE_SEGMENT}");
298        self.pages.push(PageFn::new(
299            http::Method::GET,
300            route_path(&edit_url),
301            resource_edit::<R>,
302        ));
303        self.pages.push(PageFn::new(
304            http::Method::POST,
305            route_path(&edit_url),
306            resource_edit_post::<R>,
307        ));
308        // Relationship option search — GET for searchable selects past the cap
309        // `{list_url}/options?field=&q=` reusing the related
310        // table's searchable columns, bounded, policy-checked.
311        let options_url = format!("{url}/options");
312        self.routes.push(RouteFn::new(
313            http::Method::GET,
314            route_path(&options_url),
315            resource_options::<R>,
316        ));
317    }

319    /// The routes every resource registers: the list, the detail page,
320    /// delete, bulk delete, and export. Returns the list URL,
321    /// or `None` when the slug was refused.
322    fn register_common<R: Resource>(&mut self) -> Option<String> {
323        let slug = R::slug();
324        if let Err(error) = validate_route_segment("Resource::slug", &slug) {
325            self.registration_errors.push(error);
326            return None;
327        }
328        if self.slugs.iter().any(|s| s == &slug) {
329            self.registration_errors.push(format!(
330                "duplicate resource slug '{slug}': each Resource needs a distinct slug (see Resource::slug)"
331            ));
332            return None;
333        }
334        self.slugs.push(slug);
335        self.resource_checks.push(check_resource::<R>);
336        let url = format!("{}/{}", self.prefix, R::slug());
337        self.pages.push(PageFn::new(
338            http::Method::GET,
339            route_path(&url),
340            resource_list::<R>,
341        ));
342        // Detail page — GET renders the record read-only. Registered
343        // unconditionally, unlike the row link: registration runs before a
344        // request exists, so `R::view(cx)` is not declarable here. The handler
345        // 404s a resource that declares no view, which is the same answer as an
346        // unknown id and costs one comparison.
347        //
348        // `RECORD_ROUTE_PARAM` shares its position with the literal `create`
349        // segment: topcoat routes through `matchit`, which prefers a static
350        // segment over a parameter one, so the create page keeps being reached
351        // regardless of registration order.
352        let detail_url = format!("{url}/{RECORD_ROUTE_PARAM}");
353        self.pages.push(PageFn::new(
354            http::Method::GET,
355            route_path(&detail_url),
356            resource_view::<R>,
357        ));
358        // Delete action — POST via row button (requires confirmation).
359        let delete_url = format!("{url}/{RECORD_ROUTE_PARAM}/{DELETE_ROUTE_SEGMENT}");
360        self.pages.push(PageFn::new(
361            http::Method::POST,
362            route_path(&delete_url),
363            resource_delete::<R>,
364        ));
365        // Bulk delete — POST with `ids` form field (comma-separated).
366        let bulk_delete_url = format!("{url}/{BULK_DELETE_ROUTE_SEGMENT}");
367        self.pages.push(PageFn::new(
368            http::Method::POST,
369            route_path(&bulk_delete_url),
370            resource_bulk_delete::<R>,
371        ));
372        // CSV export — GET over the tenant-scoped export query + Table
373        // filters/sort (ADR-0012).
374        let export_url = format!("{url}/export");
375        self.routes.push(RouteFn::new(
376            http::Method::GET,
377            route_path(&export_url),
378            resource_export::<R>,
379        ));
380        Some(url)
381    }

383    /// The registration tail every resource shares: its live-search handler,
384    /// the panel root, and its sidebar entry.
385    fn finish_registration<R: Resource>(&mut self, url: String) {
386        // Live-search handler: the slug-dispatched `#[shard]` cannot be generic
387        // (inventory only discovers concrete fns), so each resource
388        // monomorphizes its table loader here, keyed by list path.
389        self.search_handlers
390            .insert(url.clone(), search_handler_for::<R>());
391        if self.root_target.is_none() {
392            self.root_target = Some(url);
393        }
394        let nav_item = self.nav_item::<R>();
395        self.nav_items.push(nav_item);
396    }

404    /// Set the `frame-ancestors` directive the panel sends.
405    ///
406    /// Defaults to `'self'`: the admin only frames itself, so a hostile page
407    /// cannot clickjack it. Pass what your deployment needs — `"'self'
408    /// https://intranet.example"` to allow an internal portal, or `"'none'"`
409    /// to forbid framing outright.
410    ///
411    /// The layer only fills the gap, so an app that sets its own
412    /// `Content-Security-Policy` (its own layer or route) keeps it. Use
413    /// [`Self::without_frame_ancestors`] to send nothing at all and leave the
414    /// decision to a proxy.
415    pub fn frame_ancestors(mut self, ancestors: impl Into<String>) -> Self {
416        self.frame_ancestors = Some(ancestors.into());
417        self
418    }

455impl Panel {
456    /// Derive a [`NavigationItem`] for `R` using this panel's mount prefix.
457    ///
458    /// The one panel-aware navigation seam for a [`Resource`]:
459    /// [`Panel::resource`](Self::resource) calls it, so a resource's
460    /// [`Resource::navigation`] override reaches the sidebar instead of being
461    /// dead API. The override owns the **label, ordering and
462    /// grouping**; the panel owns the **URL**, because it is the only party
463    /// that knows where the resource is mounted. Concretely: an explicit
464    /// [`NavTarget::Url`] in `R::navigation()` is taken as returned, and only
465    /// a [`NavTarget::Derived`] target — the default, which names no URL
466    /// because [`Resource::navigation`] takes no prefix — is resolved to
467    /// `{prefix}/{slug}`.
468    ///
469    /// So `Panel::new("backoffice")` yields `"/backoffice/{slug}"` — never a
470    /// hard-coded `"/admin"` — for default and overridden items alike, a URL an
471    /// override spelled out stays exactly as written, and an override's `order`
472    /// decides sidebar order.
473    pub(crate) fn nav_item<R: Resource>(&self) -> NavigationItem {
474        R::navigation().resolved(&self.prefix, &R::slug())
475    }
