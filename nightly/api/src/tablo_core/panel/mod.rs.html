1//! `Panel` — the admin application shell.
2//!
3//! Owns the [`Router`](topcoat::router::Router) and the `Db` in `app_context`, and registers each
4//! declared [`Resource`]'s list page and each [`Page`] at `{prefix}/{slug}`,
5//! and the home page at the prefix (Filament-style routes — ADR-0008). See
6//! `CONTEXT.md`.
7//!
8//! Layout: the [`Panel`] builder and its navigation seam live here; assembly
9//! (`build`, declaration checks, route paths) in `build`; the auth/tenant
10//! gate and prefix URLs in `gate`; shell rendering in `shell`; list + live
11//! shard support in `list`; form decoding and create/edit in `forms`; the
12//! record detail page in `detail`; delete/bulk/export/options in `actions`;
13//! the registered-page handler in `pages`; the live-search registry + shard
14//! dispatch in `search`; and response hardening headers in `headers`.

75/// The table the panel's list page serves for `R`, for a page that owns its
76/// table instead of mounting the panel's list route.
77///
78/// The page-owned seam (GH #154 §2) pairs this with
79/// [`TablePage::load`](crate::resource::TablePage::load) and
80/// [`Table::render_with_state`](crate::resource::Table::render_with_state).
81/// The table carries `R::table`'s columns, key, page size, search toolbar and
82/// filter bar, plus the action chrome `R`'s declarations imply — the row
83/// Delete link and bulk column from
84/// [`can_delete_any`](crate::resource::Resource::can_delete_any), the Edit link
85/// from a record form ([`RecordForm::HAS_FORM`])
86/// and the View link from a non-empty [`view`](crate::resource::Resource::view) — each gated
87/// per row by `can_view`/`can_update`/`can_delete`, the wiring the panel's own
88/// list applies. The chrome has no other entry point: a page-owned table that
89/// must agree with the resource's routes takes its wiring from here.
90pub fn wired_table<R: Resource>(cx: &topcoat::context::Cx) -> crate::resource::Table<R::Model> {
91    self::list::wire_table_actions::<R>(cx, false)
92}

94/// The admin application.

146    /// Create a `Panel` mounted at `prefix` (e.g. `"admin"` → `"/admin"`).
147    pub fn new(prefix: impl Into<String>) -> Self {
148        let raw = prefix.into();
149        let trimmed = raw.trim_matches('/').trim().to_string();
150        let prefix = if trimmed.is_empty() {
151            "/admin".to_string()
152        } else {
153            format!("/{trimmed}")
154        };
155        // The prefix is free-form too, and every route path is built from it
156        // validate it once here rather than panicking at the first
157        // `route_path` call.
158        let registration_errors: Vec<String> = prefix
159            .trim_matches('/')
160            .split('/')
161            .filter_map(|segment| validate_route_segment("panel prefix", segment).err())
162            .collect();
163        Self {
164            prefix,
165            db: None,
166            assets: None,
167            shell_assets: None,
168            brand: None,
169            dark_mode: None,
170            nav_items: Vec::new(),
171            pages: Vec::new(),
172            routes: Vec::new(),
173            root: None,
174            slugs: Vec::new(),
175            resource_slugs: Vec::new(),
176            relations: Vec::new(),
177            search_handlers: HashMap::new(),
178            relation_handlers: HashMap::new(),
179            frame_ancestors: Some(headers::DEFAULT_FRAME_ANCESTORS.to_string()),
180            registration_errors,
181            resource_checks: Vec::new(),
182            uploads: None,
183            served_dirs: Vec::new(),
184            login_hint: None,
185            auth: crate::auth::Auth::default(),
186        }
187    }

212    /// Serve a directory of files from this panel's router.
213    ///
214    /// `path` is a route pattern ending in a catch-all (e.g.
215    /// `"/uploads/{*file}"`), and `dir` the directory those URLs read from —
216    /// see Topcoat's `DirectoryRoute` for the resolution rules. The path is
217    /// **not** panel-relative: a served directory holds files a record points
218    /// at (an upload store's output), which are not a page of the panel and
219    /// must not move when the panel is mounted elsewhere. This is **public**:
220    /// the auth gate covers only the panel prefix and `/_topcoat/runtime`
221    /// (ADR-0013), so a served directory sits outside it and its URLs answer
222    /// whoever asks, with no session — an app that needs protected files owns
223    /// that route itself (ADR-0017).
224    ///
225    /// Every file the directory route serves carries `nosniff`, a sandboxing
226    /// `Content-Security-Policy`, and `Content-Disposition: attachment` for
227    /// anything but common raster images, audio/video and plain text, so an
228    /// uploaded document cannot run script on the panel's origin. A
229    /// 404 keeps Topcoat's `text/plain` error page; a 405 carries only `Allow`
230    /// and an empty body. Neither carries user content.
231    ///
232    /// The Panel owns the [`Router`](topcoat::router::Router), so this is the app's only way to
233    /// mount a route the framework does not own.
234    pub fn serve_dir(mut self, path: impl Into<String>, dir: impl Into<PathBuf>) -> Self {
235        let path = path.into();
236        if !is_directory_pattern(&path) {
237            self.registration_errors.push(format!(
238                "serve_dir path '{path}': must end in a catch-all like '/uploads/{{*file}}'"
239            ));
240        }
241        self.served_dirs.push((path, dir.into()));
242        self
243    }

264    /// Declare a `Resource` for this panel (the declarative seam, ADR-0008).
265    ///
266    /// Registers the resource's **list page** at `{prefix}/{slug}` (e.g.
267    /// `Panel::new("admin").resource::<AuditResource>()` serves `/admin/audits`),
268    /// its detail page, delete, bulk delete, and CSV export, and derives its
269    /// [`NavigationItem`] from the same slug, so the sidebar and the router can
270    /// never disagree. Without a [`home`](Self::home) page, the panel root
271    /// redirects to the first declared resource's list. Multiple calls
272    /// compose. Sidebar order comes from the resource's
273    /// [`Resource::navigation`] override, defaulting to declaration order
274    /// (#165).
275    ///
276    /// A resource whose [`Form`](Resource::Form) is a record form
277    /// ([`RecordForm::HAS_FORM`]) also gets the create page, the edit page, and
278    /// the relationship-options endpoint; one that names
279    /// [`NoForm`](crate::NoForm) gets none of them. The route set follows the
280    /// resource type, so one type serves the same routes in every panel.
281    ///
282    /// [`Panel::build`] checks that the resource's [`form`](Resource::form)
283    /// agrees with its `Form`, that a resource with no form does not allow
284    /// `can_create`, and, for a resource with a form, that the
285    /// form's struct and its `Schema` agree: every control is bound by exactly
286    /// one field and every field's key is a declared control; an optional
287    /// control, or one inside a `Repeater` or a variant group, binds a field
288    /// with a blank answer; a gated resource's form does not claim its tenant
289    /// column; and, where `can_create` allows it, every non-nullable column is
290    /// set by the form, by toasty, by the tenant stamp, or by an override that
291    /// names it in [`Resource::CREATE_COLUMNS`]. It also checks that each of the
292    /// resource's [`relations`](Resource::relations) names a resource this
293    /// panel registers, once: the relation's actions go to that resource's
294    /// routes.
295    ///
296    /// A slug another resource or [page](Self::page) holds, one the panel
297    /// routes itself (`login`, `logout`), or one that is not one URL segment
298    /// is recorded here and reported by [`Panel::build`], which returns `Err`
299    /// instead of panicking: two routes over one slug would shadow each other,
300    /// and a hostile `slug()` must not reach a route path or a response header.
301    pub fn resource<R: Resource>(mut self) -> Self {
302        let Some(url) = self.register_common::<R>() else {
303            return self;
304        };
305        if <R::Form as RecordForm>::HAS_FORM {
306            self.register_form_routes::<R>(&url);
307        }
308        self.finish_registration::<R>(url);
309        let declared = R::relations();
310        let keys = declared
311            .iter()
312            .map(|relation| relation.key().to_string())
313            .collect::<Vec<_>>();
314        for relation in &declared {
315            if let Some(error) = relation.misdeclared() {
316                self.registration_errors.push(format!(
317                    "resource `{}`'s relation `{}`: {error}",
318                    std::any::type_name::<R>(),
319                    relation.key()
320                ));
321            }
322            self.relation_handlers.insert(
323                (R::slug(), relation.key().to_string()),
324                relation.search_handler(),
325            );
326        }
327        if !keys.is_empty() {
328            self.relations.push((std::any::type_name::<R>(), keys));
329        }
330        self
331    }

333    /// The create page, the edit page, and the relationship-options endpoint
334    /// of a resource with a record form.
335    fn register_form_routes<R: Resource>(&mut self, url: &str) {
336        // Create page — GET renders form, POST handles submission.
337        let create_url = format!("{url}/{CREATE_ROUTE_SEGMENT}");
338        self.pages.push(PageFn::new(
339            http::Method::GET,
340            route_path(&create_url),
341            resource_create::<R>,
342        ));
343        self.pages.push(PageFn::new(
344            http::Method::POST,
345            route_path(&create_url),
346            resource_create_post::<R>,
347        ));
348        // Edit page — GET renders hydrated form, POST handles update.
349        let edit_url = format!("{url}/{RECORD_ROUTE_PARAM}/{EDIT_ROUTE_SEGMENT}");
350        self.pages.push(PageFn::new(
351            http::Method::GET,
352            route_path(&edit_url),
353            resource_edit::<R>,
354        ));
355        self.pages.push(PageFn::new(
356            http::Method::POST,
357            route_path(&edit_url),
358            resource_edit_post::<R>,
359        ));
360        // Relationship option search — GET for searchable selects past the cap
361        // `{list_url}/options?field=&q=` reusing the related
362        // table's searchable columns, bounded, policy-checked.
363        let options_url = format!("{url}/options");
364        self.routes.push(RouteFn::new(
365            http::Method::GET,
366            route_path(&options_url),
367            resource_options::<R>,
368        ));
369    }

371    /// The routes every resource registers: the list, the detail page,
372    /// delete, bulk delete, and export. Returns the list URL,
373    /// or `None` when the slug was refused.
374    fn register_common<R: Resource>(&mut self) -> Option<String> {
375        let url = self.claim_slug::<R>("Resource::slug", R::slug())?;
376        self.resource_slugs.push(R::slug());
377        self.resource_checks.push(check_resource::<R>);
378        self.pages.push(PageFn::new(
379            http::Method::GET,
380            route_path(&url),
381            resource_list::<R>,
382        ));
383        // Detail page — GET renders the record read-only. Registered
384        // unconditionally, unlike the row link: registration runs before
385        // `Panel::build` has the `Db` the view declaration resolves through.
386        // The handler 404s a resource that declares no view, which is the same
387        // answer as an unknown id and costs one comparison.
388        //
389        // `RECORD_ROUTE_PARAM` shares its position with the literal `create`
390        // segment: topcoat routes through `matchit`, which prefers a static
391        // segment over a parameter one, so the create page keeps being reached
392        // regardless of registration order.
393        let detail_url = format!("{url}/{RECORD_ROUTE_PARAM}");
394        self.pages.push(PageFn::new(
395            http::Method::GET,
396            route_path(&detail_url),
397            resource_view::<R>,
398        ));
399        // Delete action — POST via row button (requires confirmation).
400        let delete_url = format!("{url}/{RECORD_ROUTE_PARAM}/{DELETE_ROUTE_SEGMENT}");
401        self.pages.push(PageFn::new(
402            http::Method::POST,
403            route_path(&delete_url),
404            resource_delete::<R>,
405        ));
406        // Bulk delete — POST with `ids` form field (comma-separated).
407        let bulk_delete_url = format!("{url}/{BULK_DELETE_ROUTE_SEGMENT}");
408        self.pages.push(PageFn::new(
409            http::Method::POST,
410            route_path(&bulk_delete_url),
411            resource_bulk_delete::<R>,
412        ));
413        // Custom actions — POST per row and for the bulk selection, the action
414        // named by the last segment. Registered only for a resource that
415        // declares some: the list-level route's static `actions` segment
416        // would otherwise shadow the edit and delete routes of a record whose
417        // key is `actions`, for nothing.
418        if !R::actions().entries().is_empty() {
419            let row_action_url =
420                format!("{url}/{RECORD_ROUTE_PARAM}/{ACTIONS_ROUTE_SEGMENT}/{ACTION_ROUTE_PARAM}");
421            self.pages.push(PageFn::new(
422                http::Method::POST,
423                route_path(&row_action_url),
424                resource_row_action::<R>,
425            ));
426            let bulk_action_url = format!("{url}/{ACTIONS_ROUTE_SEGMENT}/{ACTION_ROUTE_PARAM}");
427            self.pages.push(PageFn::new(
428                http::Method::POST,
429                route_path(&bulk_action_url),
430                resource_bulk_action::<R>,
431            ));
432        }
433        // CSV export — GET over the tenant-scoped export query + Table
434        // filters/sort (ADR-0012).
435        let export_url = format!("{url}/export");
436        self.routes.push(RouteFn::new(
437            http::Method::GET,
438            route_path(&export_url),
439            resource_export::<R>,
440        ));
441        Some(url)
442    }

444    /// The registration tail every resource shares: its live-search handler,
445    /// the panel root, and its sidebar entry.
446    fn finish_registration<R: Resource>(&mut self, url: String) {
447        // Live-search handler: the slug-dispatched `#[shard]` cannot be generic
448        // (inventory only discovers concrete fns), so each resource
449        // monomorphizes its table loader here, keyed by list path.
450        self.search_handlers
451            .insert(url.clone(), search_handler_for::<R>());
452        if self.root.is_none() {
453            self.root = Some(Root::Redirect(url));
454        }
455        let nav_item = self.nav_item::<R>();
456        self.nav_items.push(nav_item);
457    }

459    /// Claim `{prefix}/{slug}` for the resource or page `T`, returning that
460    /// URL, or record why the slug is refused: it is not one URL segment, the
461    /// panel routes it itself, or another resource or page holds it.
462    fn claim_slug<T>(&mut self, kind: &str, slug: String) -> Option<String> {
463        let owner = std::any::type_name::<T>();
464        let refused = if let Err(error) = validate_route_segment(kind, &slug) {
465            Some(error)
466        } else if RESERVED_SLUGS.contains(&slug.as_str()) {
467            Some(format!(
468                "slug '{slug}' of `{owner}`: the panel routes `{}/{slug}` itself",
469                self.prefix
470            ))
471        } else if self.slugs.contains(&slug) {
472            Some(format!(
473                "duplicate slug '{slug}': `{owner}` mounts where another resource or page does \
474                 — each needs a distinct `slug()`"
475            ))
476        } else {
477            None
478        };
479        if let Some(error) = refused {
480            self.registration_errors.push(error);
481            return None;
482        }
483        let url = format!("{}/{slug}", self.prefix);
484        self.slugs.push(slug);
485        Some(url)
486    }

488    /// Declare a [`Page`] for this panel: its `GET` at `{prefix}/{slug}`
489    /// and its sidebar entry at the same URL.
490    ///
491    /// Pages and resources share one slug namespace, and the slug is checked
492    /// as a resource's is ([`Self::resource`]): a refused one is reported by
493    /// [`Panel::build`].
494    pub fn page<P: Page>(mut self) -> Self {
495        if let Some(url) = self.claim_slug::<P>("Page::slug", P::slug()) {
496            let item = self.mount_page::<P>(&url);
497            self.nav_items.push(item);
498        }
499        self
500    }

502    /// Declare the panel's home page (Filament's dashboard): the [`Page`]
503    /// served at the panel prefix itself, with a sidebar entry there.
504    ///
505    /// It replaces the redirect to the first resource's list, and its entry
506    /// leads the sidebar among entries of the same `order`. `P::slug()` is
507    /// not read. A second call is recorded and reported by [`Panel::build`].
508    pub fn home<P: Page>(mut self) -> Self {
509        if matches!(self.root, Some(Root::Home)) {
510            self.registration_errors.push(format!(
511                "Panel::home: a home page is already registered; `{}` would shadow it",
512                std::any::type_name::<P>()
513            ));
514            return self;
515        }
516        self.root = Some(Root::Home);
517        let item = self.mount_page::<P>(&self.prefix.clone());
518        // First among equal `order`s whatever the call order, as Filament's
519        // dashboard leads its sidebar.
520        self.nav_items.insert(0, item);
521        self
522    }

582    /// authenticator with `Panel::auth(Auth::custom(..))`, or opt a public
583    /// demo out explicitly with `Panel::auth(Auth::disabled())`.
584    pub fn auth(mut self, auth: crate::auth::Auth) -> Self {
585        self.auth = auth;
586        self
587    }

589    /// A muted line rendered under the login form, for demo credentials or
590    /// deployment hints (e.g. `"Demo: admin@example.com / password"`).
591    pub fn login_hint(mut self, hint: impl Into<String>) -> Self {
592        self.login_hint = Some(hint.into());
593        self
594    }

609impl Panel {
610    /// Derive a [`NavigationItem`] for `R` using this panel's mount prefix.
611    ///
612    /// The one panel-aware navigation seam for a [`Resource`]:
613    /// [`Panel::resource`](Self::resource) calls it, so a resource's
614    /// [`Resource::navigation`] override reaches the sidebar instead of being
615    /// dead API. The override owns the **label, ordering and
616    /// grouping**; the panel owns the **URL**, because it is the only party
617    /// that knows where the resource is mounted. Concretely: an explicit
618    /// [`NavTarget::Url`] in `R::navigation()` is taken as returned, and only
619    /// a [`NavTarget::Derived`] target — the default, which names no URL
620    /// because [`Resource::navigation`] takes no prefix — is resolved to
621    /// `{prefix}/{slug}`.
622    ///
623    /// So `Panel::new("backoffice")` yields `"/backoffice/{slug}"` — never a
624    /// hard-coded `"/admin"` — for default and overridden items alike, a URL an
625    /// override spelled out stays exactly as written, and an override's `order`
626    /// decides sidebar order.
627    pub(crate) fn nav_item<R: Resource>(&self) -> NavigationItem {
628        R::navigation().resolved(&format!("{}/{}", self.prefix, R::slug()))
629    }
630}
