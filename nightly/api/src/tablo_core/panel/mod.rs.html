1//! `Panel` — the admin application shell.
2//!
3//! Owns the [`Router`](topcoat::router::Router) and the `Db` in `app_context`, and registers each
4//! declared [`Resource`]'s list page and each [`Page`] at `{prefix}/{slug}`,
5//! and the home page at the prefix (Filament-style routes — ADR-0008). See
6//! `CONTEXT.md`.
7//!
8//! Layout: the [`Panel`] builder and its navigation seam live here; assembly
9//! (`build`, declaration checks, route paths) in `build`; the auth/tenant
10//! gate and prefix URLs in `gate`; shell rendering in `shell`; list + live
11//! shard support in `list`; form decoding and create/edit in `forms`; the
12//! record detail page in `detail`; delete/bulk/export/options in `actions`;
13//! the registered-page handler in `pages`; the live-search registry + shard
14//! dispatch in `search`; and response hardening headers in `headers`.

72/// The table the panel's list page serves for `R`, for a page that owns its
73/// table instead of mounting the panel's list route.
74///
75/// The page-owned seam (GH #154 §2) pairs this with
76/// [`TablePage::load`](crate::resource::TablePage::load) and
77/// [`Table::render_with_state`](crate::resource::Table::render_with_state).
78/// The table carries `R::table`'s columns, key, page size, search toolbar and
79/// filter bar, plus the action chrome `R`'s declarations imply — the row
80/// Delete link and bulk column from
81/// [`can_delete_any`](crate::resource::Resource::can_delete_any), the Edit link
82/// from a record form ([`RecordForm::HAS_FORM`])
83/// and the View link from [`viewed`](crate::resource::Resource::viewed) — each gated
84/// per row by `can_view`/`can_update`/`can_delete`, the wiring the panel's own
85/// list applies. The chrome has no other entry point: a page-owned table that
86/// must agree with the resource's routes takes its wiring from here.
87pub fn wired_table<R: Resource>(cx: &topcoat::context::Cx) -> crate::resource::Table<R::Model> {
88    self::list::wire_table_actions::<R>(cx, false)
89}

91/// The admin application.
92///
93/// ```ignore
94/// Panel::new("admin")

99pub struct Panel {
100    prefix: String,

143    /// Create a `Panel` mounted at `prefix` (e.g. `"admin"` → `"/admin"`).
144    pub fn new(prefix: impl Into<String>) -> Self {
145        let raw = prefix.into();
146        let trimmed = raw.trim_matches('/').trim().to_string();
147        let prefix = if trimmed.is_empty() {
148            "/admin".to_string()
149        } else {
150            format!("/{trimmed}")
151        };
152        // The prefix is free-form too, and every route path is built from it
153        // validate it once here rather than panicking at the first
154        // `route_path` call.
155        let registration_errors: Vec<String> = prefix
156            .trim_matches('/')
157            .split('/')
158            .filter_map(|segment| validate_route_segment("panel prefix", segment).err())
159            .collect();
160        Self {
161            prefix,
162            db: None,
163            assets: None,
164            shell_assets: None,
165            brand: None,
166            dark_mode: None,
167            nav_items: Vec::new(),
168            pages: Vec::new(),
169            routes: Vec::new(),
170            root: None,
171            slugs: Vec::new(),
172            resource_slugs: Vec::new(),
173            relations: Vec::new(),
174            search_handlers: HashMap::new(),
175            relation_handlers: HashMap::new(),
176            frame_ancestors: Some(headers::DEFAULT_FRAME_ANCESTORS.to_string()),
177            registration_errors,
178            resource_checks: Vec::new(),
179            uploads: None,
180            served_dirs: Vec::new(),
181            login_hint: None,
182            auth: crate::auth::Auth::default(),
183        }
184    }

209    /// Serve a directory of files from this panel's router.
210    ///
211    /// `path` is a route pattern ending in a catch-all (e.g.
212    /// `"/uploads/{*file}"`), and `dir` the directory those URLs read from —
213    /// see Topcoat's `DirectoryRoute` for the resolution rules. The path is
214    /// **not** panel-relative: a served directory holds files a record points
215    /// at (an upload store's output), which are not a page of the panel and
216    /// must not move when the panel is mounted elsewhere. This is **public**:
217    /// the auth gate covers only the panel prefix and `/_topcoat/runtime`
218    /// (ADR-0013), so a served directory sits outside it and its URLs answer
219    /// whoever asks, with no session — an app that needs protected files owns
220    /// that route itself (ADR-0017).
221    ///
222    /// Every file the directory route serves carries `nosniff`, a sandboxing
223    /// `Content-Security-Policy`, and `Content-Disposition: attachment` for
224    /// anything but common raster images, audio/video and plain text, so an
225    /// uploaded document cannot run script on the panel's origin. A
226    /// 404 keeps Topcoat's `text/plain` error page; a 405 carries only `Allow`
227    /// and an empty body. Neither carries user content.
228    ///
229    /// The Panel owns the [`Router`](topcoat::router::Router), so this is the app's only way to
230    /// mount a route the framework does not own.
231    pub fn serve_dir(mut self, path: impl Into<String>, dir: impl Into<PathBuf>) -> Self {
232        let path = path.into();
233        if !is_directory_pattern(&path) {
234            self.registration_errors.push(format!(
235                "serve_dir path '{path}': must end in a catch-all like '/uploads/{{*file}}'"
236            ));
237        }
238        self.served_dirs.push((path, dir.into()));
239        self
240    }

261    /// Declare a `Resource` for this panel (the declarative seam, ADR-0008).
262    ///
263    /// Registers the resource's **list page** at `{prefix}/{slug}` (e.g.
264    /// `Panel::new("admin").resource::<AuditResource>()` serves `/admin/audits`),
265    /// its detail page, delete, bulk delete, and CSV export, and derives its
266    /// [`NavigationItem`] from the same slug, so the sidebar and the router can
267    /// never disagree. Without a [`home`](Self::home) page, the panel root
268    /// redirects to the first declared resource's list. Multiple calls
269    /// compose. Sidebar order comes from the resource's
270    /// [`Resource::navigation`] override, defaulting to declaration order
271    /// (#165).
272    ///
273    /// A resource whose [`Form`](Resource::Form) is a record form
274    /// ([`RecordForm::HAS_FORM`]) also gets the create page, the edit page, and
275    /// the relationship-options endpoint; one that names
276    /// [`NoForm`](crate::NoForm) gets none of them. The route set follows the
277    /// resource type, so one type serves the same routes in every panel.
278    ///
279    /// [`Panel::build`] checks that the resource's [`form`](Resource::form)
280    /// agrees with its `Form`, that a resource with no form does not allow
281    /// `can_create`, and, for a resource with a form, that the
282    /// form's struct and its `Schema` agree: every control is bound by exactly
283    /// one field and every field's key is a declared control; an optional
284    /// control, or one inside a `Repeater` or a variant group, binds a field
285    /// with a blank answer; a gated resource's form does not claim its tenant
286    /// column; and, where `can_create` allows it, every non-nullable column is
287    /// set by the form, by toasty, by the tenant stamp, or by an override that
288    /// names it in [`Resource::CREATE_COLUMNS`]. It also checks that each of the
289    /// resource's [`relations`](Resource::relations) names a resource this
290    /// panel registers, once: the relation's actions go to that resource's
291    /// routes.
292    ///
293    /// A slug another resource or [page](Self::page) holds, one the panel
294    /// routes itself (`login`, `logout`), or one that is not one URL segment
295    /// is recorded here and reported by [`Panel::build`], which returns `Err`
296    /// instead of panicking: two routes over one slug would shadow each other,
297    /// and a hostile `slug()` must not reach a route path or a response header.
298    pub fn resource<R: Resource>(mut self) -> Self {
299        let Some(url) = self.register_common::<R>() else {
300            return self;
301        };
302        if <R::Form as RecordForm>::HAS_FORM {
303            self.register_form_routes::<R>(&url);
304        }
305        self.finish_registration::<R>(url);
306        let declared = R::relations();
307        let keys = declared
308            .iter()
309            .map(|relation| relation.key().to_string())
310            .collect::<Vec<_>>();
311        for relation in &declared {
312            self.relation_handlers.insert(
313                (R::slug(), relation.key().to_string()),
314                relation.search_handler(),
315            );
316        }
317        if !keys.is_empty() {
318            self.relations.push((std::any::type_name::<R>(), keys));
319        }
320        self
321    }

323    /// The create page, the edit page, and the relationship-options endpoint
324    /// of a resource with a record form.
325    fn register_form_routes<R: Resource>(&mut self, url: &str) {
326        // Create page — GET renders form, POST handles submission.
327        let create_url = format!("{url}/{CREATE_ROUTE_SEGMENT}");
328        self.pages.push(PageFn::new(
329            http::Method::GET,
330            route_path(&create_url),
331            resource_create::<R>,
332        ));
333        self.pages.push(PageFn::new(
334            http::Method::POST,
335            route_path(&create_url),
336            resource_create_post::<R>,
337        ));
338        // Edit page — GET renders hydrated form, POST handles update.
339        let edit_url = format!("{url}/{RECORD_ROUTE_PARAM}/{EDIT_ROUTE_SEGMENT}");
340        self.pages.push(PageFn::new(
341            http::Method::GET,
342            route_path(&edit_url),
343            resource_edit::<R>,
344        ));
345        self.pages.push(PageFn::new(
346            http::Method::POST,
347            route_path(&edit_url),
348            resource_edit_post::<R>,
349        ));
350        // Relationship option search — GET for searchable selects past the cap
351        // `{list_url}/options?field=&q=` reusing the related
352        // table's searchable columns, bounded, policy-checked.
353        let options_url = format!("{url}/options");
354        self.routes.push(RouteFn::new(
355            http::Method::GET,
356            route_path(&options_url),
357            resource_options::<R>,
358        ));
359    }

361    /// The routes every resource registers: the list, the detail page,
362    /// delete, bulk delete, and export. Returns the list URL,
363    /// or `None` when the slug was refused.
364    fn register_common<R: Resource>(&mut self) -> Option<String> {
365        let url = self.claim_slug::<R>("Resource::slug", R::slug())?;
366        self.resource_slugs.push(R::slug());
367        self.resource_checks.push(check_resource::<R>);
368        self.pages.push(PageFn::new(
369            http::Method::GET,
370            route_path(&url),
371            resource_list::<R>,
372        ));
373        // Detail page — GET renders the record read-only. Registered
374        // unconditionally, unlike the row link: registration runs before a
375        // request exists, so `R::view(cx)` is not declarable here. The handler
376        // 404s a resource that declares no view, which is the same answer as an
377        // unknown id and costs one comparison.
378        //
379        // `RECORD_ROUTE_PARAM` shares its position with the literal `create`
380        // segment: topcoat routes through `matchit`, which prefers a static
381        // segment over a parameter one, so the create page keeps being reached
382        // regardless of registration order.
383        let detail_url = format!("{url}/{RECORD_ROUTE_PARAM}");
384        self.pages.push(PageFn::new(
385            http::Method::GET,
386            route_path(&detail_url),
387            resource_view::<R>,
388        ));
389        // Delete action — POST via row button (requires confirmation).
390        let delete_url = format!("{url}/{RECORD_ROUTE_PARAM}/{DELETE_ROUTE_SEGMENT}");
391        self.pages.push(PageFn::new(
392            http::Method::POST,
393            route_path(&delete_url),
394            resource_delete::<R>,
395        ));
396        // Bulk delete — POST with `ids` form field (comma-separated).
397        let bulk_delete_url = format!("{url}/{BULK_DELETE_ROUTE_SEGMENT}");
398        self.pages.push(PageFn::new(
399            http::Method::POST,
400            route_path(&bulk_delete_url),
401            resource_bulk_delete::<R>,
402        ));
403        // CSV export — GET over the tenant-scoped export query + Table
404        // filters/sort (ADR-0012).
405        let export_url = format!("{url}/export");
406        self.routes.push(RouteFn::new(
407            http::Method::GET,
408            route_path(&export_url),
409            resource_export::<R>,
410        ));
411        Some(url)
412    }

414    /// The registration tail every resource shares: its live-search handler,
415    /// the panel root, and its sidebar entry.
416    fn finish_registration<R: Resource>(&mut self, url: String) {
417        // Live-search handler: the slug-dispatched `#[shard]` cannot be generic
418        // (inventory only discovers concrete fns), so each resource
419        // monomorphizes its table loader here, keyed by list path.
420        self.search_handlers
421            .insert(url.clone(), search_handler_for::<R>());
422        if self.root.is_none() {
423            self.root = Some(Root::Redirect(url));
424        }
425        let nav_item = self.nav_item::<R>();
426        self.nav_items.push(nav_item);
427    }

429    /// Claim `{prefix}/{slug}` for the resource or page `T`, returning that
430    /// URL, or record why the slug is refused: it is not one URL segment, the
431    /// panel routes it itself, or another resource or page holds it.
432    fn claim_slug<T>(&mut self, kind: &str, slug: String) -> Option<String> {
433        let owner = std::any::type_name::<T>();
434        let refused = if let Err(error) = validate_route_segment(kind, &slug) {
435            Some(error)
436        } else if RESERVED_SLUGS.contains(&slug.as_str()) {
437            Some(format!(
438                "slug '{slug}' of `{owner}`: the panel routes `{}/{slug}` itself",
439                self.prefix
440            ))
441        } else if self.slugs.contains(&slug) {
442            Some(format!(
443                "duplicate slug '{slug}': `{owner}` mounts where another resource or page does \
444                 — each needs a distinct `slug()`"
445            ))
446        } else {
447            None
448        };
449        if let Some(error) = refused {
450            self.registration_errors.push(error);
451            return None;
452        }
453        let url = format!("{}/{slug}", self.prefix);
454        self.slugs.push(slug);
455        Some(url)
456    }

458    /// Declare a [`Page`] for this panel: its `GET` at `{prefix}/{slug}`
459    /// and its sidebar entry at the same URL.
460    ///
461    /// Pages and resources share one slug namespace, and the slug is checked
462    /// as a resource's is ([`Self::resource`]): a refused one is reported by
463    /// [`Panel::build`].
464    pub fn page<P: Page>(mut self) -> Self {
465        if let Some(url) = self.claim_slug::<P>("Page::slug", P::slug()) {
466            let item = self.mount_page::<P>(&url);
467            self.nav_items.push(item);
468        }
469        self
470    }

472    /// Declare the panel's home page (Filament's dashboard): the [`Page`]
473    /// served at the panel prefix itself, with a sidebar entry there.
474    ///
475    /// It replaces the redirect to the first resource's list, and its entry
476    /// leads the sidebar among entries of the same `order`. `P::slug()` is
477    /// not read. A second call is recorded and reported by [`Panel::build`].
478    pub fn home<P: Page>(mut self) -> Self {
479        if matches!(self.root, Some(Root::Home)) {
480            self.registration_errors.push(format!(
481                "Panel::home: a home page is already registered; `{}` would shadow it",
482                std::any::type_name::<P>()
483            ));
484            return self;
485        }
486        self.root = Some(Root::Home);
487        let item = self.mount_page::<P>(&self.prefix.clone());
488        // First among equal `order`s whatever the call order, as Filament's
489        // dashboard leads its sidebar.
490        self.nav_items.insert(0, item);
491        self
492    }

575impl Panel {
576    /// Derive a [`NavigationItem`] for `R` using this panel's mount prefix.
577    ///
578    /// The one panel-aware navigation seam for a [`Resource`]:
579    /// [`Panel::resource`](Self::resource) calls it, so a resource's
580    /// [`Resource::navigation`] override reaches the sidebar instead of being
581    /// dead API. The override owns the **label, ordering and
582    /// grouping**; the panel owns the **URL**, because it is the only party
583    /// that knows where the resource is mounted. Concretely: an explicit
584    /// [`NavTarget::Url`] in `R::navigation()` is taken as returned, and only
585    /// a [`NavTarget::Derived`] target — the default, which names no URL
586    /// because [`Resource::navigation`] takes no prefix — is resolved to
587    /// `{prefix}/{slug}`.
588    ///
589    /// So `Panel::new("backoffice")` yields `"/backoffice/{slug}"` — never a
590    /// hard-coded `"/admin"` — for default and overridden items alike, a URL an
591    /// override spelled out stays exactly as written, and an override's `order`
592    /// decides sidebar order.
593    pub(crate) fn nav_item<R: Resource>(&self) -> NavigationItem {
594        R::navigation().resolved(&format!("{}/{}", self.prefix, R::slug()))
595    }
596}
