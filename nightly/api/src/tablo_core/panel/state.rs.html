1//! What one mounted panel serves from, and how a request finds its panel.
2//!
3//! [`RouterBuilderPanelExt::panel`](super::RouterBuilderPanelExt::panel) turns
4//! each [`Panel`](super::Panel) into a [`PanelState`] and records it in the
5//! router's [`Panels`]. The panel's layer puts its state on every request under
6//! its prefix as [`CurrentPanel`], and every handler, the shell and the auth
7//! gate read the panel through [`current`]: a router holds any number of
8//! panels, and each request sees exactly one.
9
10use std::{any::TypeId, collections::HashMap, sync::Arc};
11
12use topcoat::context::{Cx, try_app_context, try_request_context};
13
14use super::{
15    search::{RelationSearchFn, SearchFn},
16    shell::{Brand, ShellAssets},
17};
18use crate::{auth::Auth, resource::NavigationItem, upload::InstalledUploader};

20/// One mounted panel: its prefix, its shell, its auth, and the registries its
21/// generic handlers dispatch through.
22pub(crate) struct PanelState {
23    /// The mount prefix, e.g. `/admin`: every URL the panel serves starts
24    /// with it, and no other panel's prefix overlaps it.
25    pub(crate) prefix: String,
26    pub(crate) nav_items: Vec<NavigationItem>,
27    pub(crate) brand: Option<Brand>,
28    pub(crate) dark_mode: bool,
29    pub(crate) shell_assets: Option<ShellAssets>,
30    /// Live-search loaders by list path.
31    pub(crate) search: HashMap<String, SearchFn>,
32    /// Relation live-search loaders by (parent slug, child slug).
33    pub(crate) relations: HashMap<(String, String), RelationSearchFn>,
34    /// Where the prefix redirects when the panel has no home page.
35    pub(crate) root_redirect: Option<String>,
36    pub(crate) auth: Auth,
37    pub(crate) login_hint: Option<String>,
38    pub(crate) uploads: Option<InstalledUploader>,
39    /// The URL each registered resource and page is served at, by type.
40    pub(crate) urls: HashMap<TypeId, String>,
41}
42
43impl PanelState {
44    /// Whether the panel requires a signed-in user.
45    pub(crate) fn gates(&self) -> bool {
46        !self.auth.is_disabled()
47    }
48
49    /// Whether `path` is the prefix itself or a path below it.
50    pub(crate) fn serves(&self, path: &str) -> bool {
51        under_prefix(&self.prefix, path)
52    }
53}
54
55/// Whether `path` is `prefix` or sits below it, segment by segment: `/admin`
56/// serves `/admin`, `/admin/users` and `/admin?q=`, never `/administer`.
57pub(crate) fn under_prefix(prefix: &str, path: &str) -> bool {
58    path.strip_prefix(prefix)
59        .is_some_and(|rest| rest.is_empty() || rest.starts_with(['/', '?']))
60}
61
62/// Every panel mounted on the router, in mount order: the app-context value
63/// the auth gates and the shard dispatch read.
64#[derive(Default)]
65pub(crate) struct Panels(pub(crate) Vec<Arc<PanelState>>);
66
67impl Panels {
68    /// The panel mounted at exactly `prefix`.
69    pub(crate) fn by_prefix(&self, prefix: &str) -> Option<&Arc<PanelState>> {
70        self.0.iter().find(|panel| panel.prefix == prefix)
71    }
72
73    /// The panel serving `path`. Prefixes never overlap, so at most one does.
74    pub(crate) fn by_path(&self, path: &str) -> Option<&Arc<PanelState>> {
75        self.0.iter().find(|panel| panel.serves(path))
76    }
77
78    /// Whether every mounted panel requires a signed-in user.
79    pub(crate) fn all_gate(&self) -> bool {
80        self.0.iter().all(|panel| panel.gates())
81    }
82
83    /// Whether some mounted panel requires a signed-in user.
84    pub(crate) fn any_gates(&self) -> bool {
85        self.0.iter().any(|panel| panel.gates())
86    }
87}
88
89/// The request's panel, as the request context carries it.
90#[derive(Clone)]
91pub(crate) struct CurrentPanel(pub(crate) Arc<PanelState>);
92
93/// The request's panel: the one whose prefix the request is under, the one a
94/// live table's shard re-renders for, or the panel whose session signed in a
95/// runtime request. A router with a single panel answers it for any request.
96pub(crate) fn current(cx: &Cx) -> Option<&Arc<PanelState>> {
97    if let Some(CurrentPanel(panel)) = try_request_context::<CurrentPanel>(cx) {
98        return Some(panel);
99    }
100    match try_app_context::<Panels>(cx).map(|panels| panels.0.as_slice()) {
101        Some([only]) => Some(only),
102        _ => None,
103    }
104}
105
106/// The mounted panels, if the router mounts any.
107pub(crate) fn panels(cx: &Cx) -> Option<&Panels> {
108    try_app_context::<Panels>(cx)
109}
