27impl Panel {
28    /// Build the [`Router`], discovering all `#[page]` / `#[layout]` / `#[shard]`
29    /// items linked into the binary, mounting the browser-runtime layer
30    /// (`RouterBuilderRuntimeExt::runtime`, required by `runtime::script`),
31    /// installing the `Db` and the panel navigation on the `app_context`,
32    /// registering each declared resource's list page, and pointing the
33    /// panel root at the first resource.
34    ///
35    /// # Errors
36    ///
37    /// Reports what the declarative builders could only record:
38    /// a missing [`Db`], a duplicate or malformed resource slug, a malformed
39    /// panel prefix, or `shell_assets` declared without `assets`. Configuring
40    /// a panel wrong is a boot failure, not a request-time panic, so it comes
41    /// back as an error the caller can log or exit on.
42    pub fn build(self) -> topcoat::Result<Router> {
43        if !self.registration_errors.is_empty() {
44            return Err(std::io::Error::other(format!(
45                "Panel::build: {}",
46                self.registration_errors.join("; ")
47            ))
48            .into());
49        }
50        if self.shell_assets.is_some() && self.assets.is_none() {
51            return Err(std::io::Error::other(
52                "Panel::build requires assets when shell_assets are configured",
53            )
54            .into());
55        }
56        let Panel {
57            prefix,
58            db,
59            assets,
60            shell_assets,
61            brand,
62            dark_mode,
63            nav_items,
64            pages,
65            routes,
66            root_target,
67            slugs: _,
68            search_handlers,
69            frame_ancestors,
70            registration_errors: _,
71            resource_checks,
72            uploads,
73            served_dirs,
74            login_hint,
75            auth,
76        } = self;
77        let db = db.ok_or_else(|| {
78            topcoat::Error::from(std::io::Error::other(
79                "Panel::build requires a Db via app_context",
80            ))
81        })?;
82        // Declaration checks: a resource whose table or form could
83        // never render is a configuration error, and the declaration is
84        // knowable here — waiting for the first request only moves the failure
85        // somewhere less useful. `table`, `form` and `can_create` are pure
86        // declarations, so they must not need request-scoped context.
87        if !resource_checks.is_empty() {
88            let cx = validation_cx(&db);
89            let failures: Vec<String> = resource_checks
90                .iter()
91                .filter_map(|check| check(&cx).err())
92                .collect();
93            if !failures.is_empty() {
94                return Err(std::io::Error::other(format!(
95                    "Panel::build: {}",
96                    failures.join("; ")
97                ))
98                .into());
99            }
100        }
101        crate::auth::assert_models_registered(&db, &auth);
102        let mut builder = Router::builder()
103            .discover()
104            .cookies()
105            // Form bodies (urlencoded buffered, multipart streamed) share one
106            // cap: without this layer Topcoat's 2 MiB default would
107            // 413 uploads the framework otherwise accepts.
108            .layer(topcoat::router::BodyLimit::max(MAX_FORM_BYTES))
109            .app_context(db);
110        // Clickjacking hardening: a response anyone can frame is a
111        // threat on every deployment, so the panel ships the directive itself
112        // and apps that need framing opt out (or supply their own policy,
113        // which wins — the layer only fills the gap).
114        if let Some(directive) = frame_ancestors {
115            builder = builder.layer(headers::FrameAncestors::new(directive));
116        }
117        // Auth (ADR-0013): sessions plus the resolving gate under the panel
118        // and runtime prefixes, and the login/logout routes. Disabled skips
119        // all three but still installs the `Auth` value for the shell.
120        if !auth.is_disabled() {
121            builder = crate::auth::install(builder, &prefix);
122            let login_path = route_path(&format!("{prefix}/login"));
123            let logout_path = route_path(&format!("{prefix}/logout"));
124            // A credential POST carries no upload: the login route
125            // gets its own cap, scoped by path so it wins over the panel's
126            // 10 MiB form cap.
127            builder = builder.layer(
128                topcoat::router::BodyLimit::max(crate::auth::MAX_LOGIN_BYTES)
129                    .at(login_path.clone()),
130            );
131            builder = builder
132                .route(RouteFn::new(
133                    http::Method::GET,
134                    login_path.clone(),
135                    crate::auth::login_page,
136                ))
137                .route(RouteFn::new(
138                    http::Method::POST,
139                    login_path,
140                    crate::auth::login_post,
141                ))
142                .route(RouteFn::new(
143                    http::Method::POST,
144                    logout_path,
145                    crate::auth::logout_post,
146                ));
147        }
148        if !search_handlers.is_empty() {
149            builder = builder.app_context(SearchRegistry(search_handlers));
150        }
151        // The mount prefix travels with the Router so generic handlers derive
152        // resource URLs from the declaration instead of sniffing the request
153        // path (item 6 / B4).
154        builder = builder.app_context(PanelPrefix(prefix.clone()));
155        if !nav_items.is_empty() {
156            builder = builder.app_context(nav_items);
157        }
158        if let Some(assets) = assets {
159            builder = builder.assets(assets);
160        }
161        if let Some(shell_assets) = shell_assets {
162            builder = builder.app_context(shell_assets);
163        }
164        if let Some(brand) = brand {
165            builder = builder.app_context(brand);
166        }
167        if let Some(enabled) = dark_mode {
168            builder = builder.app_context(DarkMode(enabled));
169        }
170        // Where uploaded bytes go: installed once, found by the form
171        // handlers and the multipart parser through the app context.
172        if let Some(uploads) = uploads {
173            builder = builder.app_context(uploads);
174        }
175        for (path, dir) in served_dirs {
176            // Files the panel serves share its origin, so each directory route
177            // is wrapped in the hardening layer that makes them inert
178            // The same path scopes the layer to that route only.
179            builder = builder
180                .layer(headers::ServedFileHeaders::new(&path))
181                .serve_dir(route_path(&path), dir);
182        }
183        for page in pages {
184            builder = builder.page(page);
185        }
186        for route in routes {
187            builder = builder.route(route);
188        }
189        // The panel root has no home page of its own; until custom pages exist,
190        // the prefix serves a redirect to the first resource's
191        // list so the mount point is never a dead URL.
192        if let Some(target) = root_target {
193            builder = builder
194                .app_context(RootRedirect(target))
195                .route(RouteFn::new(
196                    http::Method::GET,
197                    route_path(&prefix),
198                    panel_root_redirect,
199                ));
200        }
201        builder = builder.app_context(auth);
202        if let Some(hint) = login_hint {
203            builder = builder.app_context(LoginHint(hint));
204        }
205        // The runtime layer registers last, outside every other pathless
206        // layer: a page re-run is a marked POST the layer rewrites into a
207        // GET for the page's own URL, and the layers it wraps must receive
208        // the rewritten GET rather than the discarded POST. Panel links
209        // navigate through it without prefetching: a prefetch renders the
210        // destination, list queries included, for a page the user may never
211        // open.
212        Ok(builder
213            .runtime()
214            .prefetch(topcoat::runtime::PrefetchMode::Never)
215            .build())
216    }
217}

219/// Where the panel root redirects (the first declared resource's list).
220/// Lives on the `app_context` because page handlers are plain `fn` pointers
221/// and cannot capture.
222#[derive(Debug, Clone)]
223struct RootRedirect(String);
224
225/// The panel root: a temporary redirect to the first declared resource's
226/// list, so the mount point is never a dead URL (custom pages remain future
227/// work; see `docs/guide/src/panel-and-routing.md`). Filament registers its
228/// home page here.
229pub(crate) fn panel_root_redirect(cx: &Cx, _body: Body) -> RouteFuture<'_> {
230    Box::pin(async move {
231        // Defense in depth: every panel handler re-checks the
232        // resolved user, so a missing or mis-mounted gate cannot leak the
233        // first resource's slug via the redirect target.
234        enforce_auth(cx)?;
235        let RootRedirect(target) = app_context::<RootRedirect>(cx);
236        Err(redirect(target.clone()).into())
237    })
238}

240/// Whether a path is a route pattern ending in a catch-all, which is the only
241/// shape [`DirectoryRoute`](topcoat::router::DirectoryRoute) accepts.
242///
243/// Checked where the path is declared rather than where it is used: upstream
244/// `serve_dir` panics on anything else, and `Panel::build` reports instead of
245/// panicking — but the path comes from the app, and it would panic
246/// first in [`route_path`] (which refuses to spell a route it cannot parse) and
247/// then inside `DirectoryRoute::new` (which needs the catch-all last). Asking
248/// both conditions here turns a typo into a build error instead of a panic
249/// during the build.
250pub(super) fn is_directory_pattern(path: &str) -> bool {
251    Path::from_str(path)
252        .ok()
253        .and_then(|parsed| parsed.segments().next_back())
254        .is_some_and(|segment| segment.as_catch_all().is_some())
255}

257/// Validate one path segment a panel derives routes from: a
258/// `Resource::slug()` override, or a segment of the panel prefix.
259///
260/// Both reach a route path and, through the panel, a response body. A hostile
261/// value — quote, backslash, CR/LF, `..`, slash, URL punctuation, a route
262/// pattern character — must fail at registration rather than at request time,
263/// so this is the export filename sanitizer's rule tightened to what a URL
264/// segment can be: the export drops the offending characters because it must
265/// still produce a download, while a route has no meaningful fallback.
266///
267/// The route pattern characters that a literal segment cannot carry (`{`, `}`,
268/// `(`, `)`) are rejected rather than escaped: `Path::from_str` treats
269/// `{`/`(` as the start of a parameter or group segment, so a balanced pair
270/// silently becomes a pattern and an unbalanced one panics [`route_path`]. `*`
271/// stays accepted — it is a literal in a static segment — and the catch-all
272/// spelling `{*name}` needs the `{` this rule already refuses.
273pub(super) fn validate_route_segment(kind: &str, segment: &str) -> Result<(), String> {
274    if segment.is_empty() {
275        return Err(format!("{kind}: path segment must not be empty"));
276    }
277    if segment == "." || segment == ".." {
278        return Err(format!(
279            "{kind} '{segment}': a path segment may not be '.' or '..'"
280        ));
281    }
282    if let Some(bad) = segment.chars().find(|c| {
283        c.is_control()
284            || c.is_whitespace()
285            || matches!(
286                c,
287                '"' | '\\' | '/' | '?' | '#' | '%' | '&' | '=' | '{' | '}' | '(' | ')'
288            )
289    }) {
290        return Err(format!(
291            "{kind} '{segment}': a path segment may not contain {bad:?} (quotes, backslashes, control characters, whitespace, URL punctuation and the route pattern characters '{{', '}}', '(' and ')' are rejected)"
292        ));
293    }
294    Ok(())
295}

297/// A resource's build-time declaration check: monomorphized once per
298/// declared resource by [`Panel::resource`], run by [`Panel::build`] with the
299/// app's values and no request.
300pub(super) type ResourceCheck = fn(&Cx) -> Result<(), String>;
301
302/// What a declared resource must be able to promise before the panel serves it.
303///
304/// The trait defaults every method but `table`, so a resource that overrides
305/// nothing else compiles and only fails when a user reaches the page that needs
306/// the missing piece. The essentials that are *declarations* — a tenant
307/// predicate for a gated resource and a page size the list can serve — are
308/// checked here, at build, and reported with the
309/// resource's type name, together with the agreement between the resource's
310/// `Form` and its `form()` schema ([`check_form_declaration`]).
311/// Runtime essentials (the record fns) keep their loud failure.
312///
313/// A declaration that panics is a boot failure too: `Resource::table` and
314/// `Resource::form` run code that panics on a mis-declaration, and this check's
315/// contract is a registration error the caller can log or exit on. The whole
316/// body is caught, because `R::Model::schema()` and the policy predicates are
317/// part of the same declaration, and the panic's own message is carried into
318/// the error. `AssertUnwindSafe` is sound because nothing observes the captured
319/// state after an unwind: `cx` is the build-time `validation_cx`, and the panic
320/// fails the whole `build`.
321pub(super) fn check_resource<R: Resource>(cx: &Cx) -> Result<(), String> {
322    caught::<R>(|| check_resource_inner::<R>(cx))
323}
324
325/// The message out of a caught panic payload.
326///
327/// The declaration panics this catches are `assert!`/`panic!("…")` with a
328/// formatted string, so `&str` and `String` cover every one of them; anything
329/// else is reported by shape rather than silently dropped.
330fn panic_message(payload: &(dyn std::any::Any + Send)) -> String {
331    if let Some(message) = payload.downcast_ref::<&str>() {
332        (*message).to_string()
333    } else if let Some(message) = payload.downcast_ref::<String>() {
334        message.clone()
335    } else {
336        "a non-string panic payload".to_string()
337    }
338}

340/// The body of [`check_resource`], unwound through `catch_unwind` so a
341/// mis-declared resource is a registration error rather than a boot panic.
342fn check_resource_inner<R: Resource>(cx: &Cx) -> Result<(), String> {
343    // A gated resource that supplies no tenant predicate is misdeclared, and
344    // the declaration is checkable without a request:
345    // `R::tenant_scope` is pure, and the default derivation answers by the
346    // model's *shape* — a `tenant_id` UUID field, found by name and type — not
347    // by the tenant value, so the nil UUID is enough to ask whether a predicate
348    // exists at all. Refusing here is what `build`'s contract promises a
349    // declaration error gets; the request-time error in `apply_tenant_scope`
350    // stays as the backstop for a resource whose predicate is only `None` for
351    // some tenants, and for app code that calls `scoped_query` outside a panel.
352    //
353    // Checked before the declarations below because the gate and the scope
354    // govern every handler this resource registers, not just the list and
355    // create pages those checks are about.
356    if R::requires_tenant() && R::tenant_scope(uuid::Uuid::nil()).is_none() {
357        return Err(format!(
358            "resource `{}` requires a tenant, but the framework cannot scope it: `{}` declares no \
359             `tenant_id` UUID column to derive the filter from, and the resource does not override \
360             `tenant_scope` — declare the column, override `tenant_scope`, or drop \
361             `requires_tenant` and scope in `query` (GH #231)",
362            std::any::type_name::<R>(),
363            std::any::type_name::<R::Model>(),
364        ));
365    }
366    // Declaring the table and the view runs their own misdeclaration checks
367    // (a duplicate column or field name, a zero page size, a lens that is not
368    // a single field, a modifier on the wrong control), which panic; the
369    // `catch_unwind` around this body turns them into this resource's
370    // registration error instead of a failure on the first list or detail
371    // request.
372    let _ = R::table(cx);
373    let _ = R::view(cx);
374    check_form_declaration::<R>(cx)
375}

377/// A resource with a record form declares its schema and runs
378/// [`check_form_inner`]. A resource whose form serves no pages
379/// ([`RecordForm::HAS_FORM`] false) declares no schema, and a policy that
380/// allows create would link to a page that does not exist.
381fn check_form_declaration<R: Resource>(cx: &Cx) -> Result<(), String> {
382    let resource = std::any::type_name::<R>();
383    let form = std::any::type_name::<R::Form>();
384    if <R::Form as RecordForm>::HAS_FORM {
385        // A form with fields and the empty `form()` default is a missing
386        // override; the key check below would only name the first field.
387        if R::form(cx).is_empty() && !<R::Form as RecordForm>::fields(cx).is_empty() {
388            return Err(format!(
389                "resource `{resource}` names record form `{form}` but does not override `form()`, \
390                 whose default declares no controls"
391            ));
392        }
393        return check_form_inner::<R>(cx);
394    }
395    if !R::form(cx).is_empty() {
396        return Err(format!(
397            "resource `{resource}` declares a form schema but its `Form`, `{form}`, serves no \
398             form — name the record form in `type Form`"
399        ));
400    }
401    if R::can_create(cx) {
402        return Err(format!(
403            "resource `{resource}` allows create but has no form — name its record form in `type \
404             Form` and declare `form()`"
405        ));
406    }
407    Ok(())
408}

410fn check_form_inner<R: Resource>(cx: &Cx) -> Result<(), String> {
411    let form = R::form(cx);
412    let resource = std::any::type_name::<R>();
413    let fields = <R::Form as RecordForm>::fields(cx);
414    let controls = form.controls();
415    // Key agreement, reported control-first: a control no field binds is the
416    // direction that drops what the user typed.
417    for control in &controls {
418        let claims = fields
419            .iter()
420            .filter(|field| field.keys.contains(&control.name))
421            .count();
422        if claims == 0 {
423            return Err(format!(
424                "resource `{resource}` renders form control `{}` but no field of its record form \
425                 binds it, so what the user types there is never written",
426                control.name
427            ));
428        }
429        if claims > 1 {
430            return Err(format!(
431                "resource `{resource}` binds form control `{}` from more than one record-form field",
432                control.name
433            ));
434        }
435    }
436    for field in &fields {
437        for key in &field.keys {
438            if !controls.iter().any(|control| &control.name == key) {
439                return Err(format!(
440                    "resource `{resource}`'s record form field `{}` binds key `{key}`, but the form \
441                     declares no control for it",
442                    field.name
443                ));
444            }
445        }
446        // Blank agreement: an empty submission must resolve wherever the
447        // schema lets one through.
448        if field.answers_blank {
449            continue;
450        }
451        if let Some(control) = controls.iter().find(|control| {
452            field.keys.contains(&control.name) && (!control.required || control.in_repeater)
453        }) {
454            let place = if control.in_repeater {
455                "sits inside a `Repeater`, so it may be posted empty"
456            } else {
457                "is optional"
458            };
459            return Err(format!(
460                "resource `{resource}`'s form control `{}` {place}, but record form field `{}` has \
461                 no blank answer — declare `#[form(blank = ..)]`, make the field an \
462                 `Option`, or make the control required",
463                control.name, field.name
464            ));
465        }
466    }
467    // Tenant ownership: the framework stamps a gated resource's tenant column
468    // on create; a form that claimed it would let the client choose.
469    if R::requires_tenant()
470        && let Some(column) = crate::tenancy::tenant_field_name::<R::Model>()
471        && let Some(field) = fields.iter().find(|field| field.keys.contains(&column))
472    {
473        return Err(format!(
474            "resource `{resource}` requires a tenant, but record form field `{}` claims its tenant \
475             column `{column}` — the framework stamps it on create; drop it from the form",
476            field.name
477        ));
478    }
479    // Column coverage: a create that leaves a non-nullable column unset fails
480    // at the driver on every submit, with no field to point the user at.
481    if R::can_create(cx) {
482        check_create_columns::<R>(&fields)?;
483    }
484    // `.unique()` is a promise the panel makes and the database has to keep
485    // (item 3): the marker turns the app-side pre-check on, so a field
486    // whose column carries no unique index makes the panel enforce a rule
487    // nothing else does — a duplicate the check lets through, or a rule the
488    // database never asked for. The declaration checks are the only place both
489    // halves are reachable without a request, so the pair is refused here
490    // rather than discovered by a user. It is checked whatever the policies say:
491    // a `unique()` marker is wrong on a form the panel would not even serve.
492    // `lens_field_unique` recognizes composite indexes too, which is what makes
493    // `#[unique(tenant_id, email)]` — the tenant-scoped arrangement the panel
494    // documents — pass.
495    let model = R::Model::schema();
496    let root = model.as_root_unwrap();
497    for field in form.fields().filter(|field| field.is_unique()) {
498        let name = field.name();
499        // A bound lens always resolves, so a name with no field at all is a
500        // mis-declared schema — but it is not worth a second error string: it
501        // fails the same way, one message below.
502        let backed = root
503            .fields
504            .iter()
505            .filter(|field| field.name.app_unwrap() == name)
506            .any(|field| crate::schema::lens_field_unique(field, root));
507        if !backed {
508            return Err(format!(
509                "resource `{}` marks form field `{name}` unique, but `{}::{name}` carries no unique index — add `#[unique]` (or `#[unique(..)]`) to the column or drop `.unique()`, which would otherwise check a rule the database does not enforce",
510                std::any::type_name::<R>(),
511                std::any::type_name::<R::Model>()
512            ));
513        }
514    }
515    Ok(())
516}

518/// Every non-nullable column a create must set is set by something: the
519/// record form, toasty (`#[auto]`, `#[default(..)]`), the tenant stamp, or the
520/// resource's own `CREATE_COLUMNS`.
521fn check_create_columns<R: Resource>(
522    fields: &[crate::form::FormField<<R::Form as RecordForm>::Field>],
523) -> Result<(), String> {
524    let resource = std::any::type_name::<R>();
525    let prefilled = crate::form::prefilled_fields::<R::Model>();
526    let tenant = R::requires_tenant()
527        .then(crate::tenancy::tenant_field_name::<R::Model>)
528        .flatten();
529    let model = R::Model::schema();
530    let root = model.as_root_unwrap();
531    for name in R::CREATE_COLUMNS {
532        if !root
533            .fields
534            .iter()
535            .any(|field| field.name.app.as_deref() == Some(*name))
536        {
537            return Err(format!(
538                "resource `{resource}` names `{name}` in `CREATE_COLUMNS`, but `{}` has no such \
539                 field",
540                std::any::type_name::<R::Model>()
541            ));
542        }
543    }
544    for (index, field) in root.fields.iter().enumerate() {
545        let Some(name) = field.name.app.as_deref() else {
546            continue;
547        };
548        let filled = field.nullable()
549            || field.is_relation()
550            || prefilled.get(index).copied().unwrap_or(false)
551            || tenant.as_deref() == Some(name)
552            || fields.iter().any(|claim| claim.name == name)
553            || R::CREATE_COLUMNS.contains(&name);
554        if !filled {
555            return Err(format!(
556                "resource `{resource}` allows create, but nothing writes the non-nullable column \
557                 `{name}`: its record form has no such field, toasty fills no `#[default(..)]` for \
558                 it, and `CREATE_COLUMNS` does not name it — every create would fail at the driver"
559            ));
560        }
561    }
562    Ok(())
563}

565/// Run a declaration check, turning a panic in the app's declarations into a
566/// registration error naming the resource.
567fn caught<R: Resource>(check: impl FnOnce() -> Result<(), String>) -> Result<(), String> {
568    match std::panic::catch_unwind(std::panic::AssertUnwindSafe(check)) {
569        Ok(result) => result,
570        Err(payload) => Err(format!(
571            "resource `{}` panicked while declaring itself: {}",
572            std::any::type_name::<R>(),
573            panic_message(payload.as_ref())
574        )),
575    }
576}

578/// A context for the build-time declaration checks: the app's own values, no
579/// request. Resources must be able to describe their table and form from this
580/// — that they cannot read a request here is the contract, not a limitation.
581fn validation_cx(db: &Db) -> Cx {
582    let mut app_context = topcoat::context::AppContext::new();
583    app_context.insert(db.clone());
584    Cx::new(std::sync::Arc::new(app_context))
585}

587/// Parse a panel route path, panicking on malformed input — the paths are
588/// built from the panel prefix and the resource slug, both validated at
589/// registration ([`validate_route_segment`]), so a malformed path here is a
590/// framework bug rather than user input.
591pub(crate) fn route_path(path: &str) -> topcoat::router::PathBuf {
592    Path::from_str(path)
593        .expect("panel route paths are well-formed")
594        .to_owned()
595}
