29impl Panel {
30    /// Build the [`Router`], discovering all `#[page]` / `#[layout]` / `#[shard]`
31    /// items linked into the binary, mounting the browser-runtime layer
32    /// (`RouterBuilderRuntimeExt::runtime`, required by `runtime::script`),
33    /// installing the `Db` and the panel navigation on the `app_context`,
34    /// registering each declared resource's list page, and pointing the
35    /// panel root at the first resource.
36    ///
37    /// # Errors
38    ///
39    /// Reports what the declarative builders could only record:
40    /// a missing [`Db`], a duplicate or malformed resource slug, a malformed
41    /// panel prefix, or `shell_assets` declared without `assets`. Configuring
42    /// a panel wrong is a boot failure, not a request-time panic, so it comes
43    /// back as an error the caller can log or exit on.
44    ///
45    /// With the `auth` feature off nothing authenticates requests, so a panel
46    /// that has not acknowledged that with
47    /// [`Panel::auth(Auth::disabled())`](Self::auth) is also an error
48    /// (ADR-0013).
49    pub fn build(self) -> topcoat::Result<Router> {
50        if !self.registration_errors.is_empty() {
51            return Err(std::io::Error::other(format!(
52                "Panel::build: {}",
53                self.registration_errors.join("; ")
54            ))
55            .into());
56        }
57        if self.shell_assets.is_some() && self.assets.is_none() {
58            return Err(std::io::Error::other(
59                "Panel::build requires assets when shell_assets are configured",
60            )
61            .into());
62        }
63        let Panel {
64            prefix,
65            db,
66            assets,
67            shell_assets,
68            brand,
69            dark_mode,
70            nav_items,
71            pages,
72            routes,
73            root_target,
74            slugs: _,
75            search_handlers,
76            frame_ancestors,
77            registration_errors: _,
78            resource_checks,
79            uploads,
80            served_dirs,
81            #[cfg(feature = "auth")]
82            login_hint,
83            #[cfg(feature = "auth")]
84            auth,
85            #[cfg(not(feature = "auth"))]
86            auth_disabled,
87        } = self;
88        let db = db.ok_or_else(|| {
89            topcoat::Error::from(std::io::Error::other(
90                "Panel::build requires a Db via app_context",
91            ))
92        })?;
93        // Declaration checks: a resource whose table or form could
94        // never render is a configuration error, and the declaration is
95        // knowable here — waiting for the first request only moves the failure
96        // somewhere less useful. `table`, `form` and `can_create` are pure
97        // declarations, so they must not need request-scoped context.
98        if !resource_checks.is_empty() {
99            let cx = validation_cx(&db);
100            let failures: Vec<String> = resource_checks
101                .iter()
102                .filter_map(|check| check(&cx).err())
103                .collect();
104            if !failures.is_empty() {
105                return Err(std::io::Error::other(format!(
106                    "Panel::build: {}",
107                    failures.join("; ")
108                ))
109                .into());
110            }
111        }
112        // Auth compiled out (ADR-0013): `enforce_auth` is a no-op and no gate
113        // is installed, so a panel that reaches here would serve every page and
114        // mutation to anyone. The opt-out stays a line of app code.
115        #[cfg(not(feature = "auth"))]
116        if !auth_disabled {
117            return Err(std::io::Error::other(
118                "Panel::build: tablo-core is built without the `auth` feature, so nothing \
119                 authenticates requests; call `.auth(Auth::disabled())` to serve the panel \
120                 ungated, or enable the feature",
121            )
122            .into());
123        }
124        #[cfg(feature = "auth")]
125        crate::auth::assert_models_registered(&db, &auth);
126        let mut builder = Router::builder()
127            .discover()
128            .cookies()
129            // Form bodies (urlencoded buffered, multipart streamed) share one
130            // cap: without this layer Topcoat's 2 MiB default would
131            // 413 uploads the framework otherwise accepts.
132            .layer(topcoat::router::BodyLimit::max(MAX_FORM_BYTES))
133            .app_context(db);
134        // Clickjacking hardening: a response anyone can frame is a
135        // threat on every deployment, so the panel ships the directive itself
136        // and apps that need framing opt out (or supply their own policy,
137        // which wins — the layer only fills the gap).
138        if let Some(directive) = frame_ancestors {
139            builder = builder.layer(headers::FrameAncestors::new(directive));
140        }
141        // Auth (ADR-0013): sessions plus the resolving gate under the panel
142        // and runtime prefixes, and the login/logout routes. Disabled skips
143        // all three but still installs the `Auth` value for the shell.
144        #[cfg(feature = "auth")]
145        {
146            if !auth.is_disabled() {
147                builder = crate::auth::install(builder, &prefix);
148                let login_path = route_path(&format!("{prefix}/login"));
149                let logout_path = route_path(&format!("{prefix}/logout"));
150                // A credential POST carries no upload: the login route
151                // gets its own cap, scoped by path so it wins over the panel's
152                // 10 MiB form cap.
153                builder = builder.layer(
154                    topcoat::router::BodyLimit::max(crate::auth::MAX_LOGIN_BYTES)
155                        .at(login_path.clone()),
156                );
157                builder = builder
158                    .route(RouteFn::new(
159                        http::Method::GET,
160                        login_path.clone(),
161                        crate::auth::login_page,
162                    ))
163                    .route(RouteFn::new(
164                        http::Method::POST,
165                        login_path,
166                        crate::auth::login_post,
167                    ))
168                    .route(RouteFn::new(
169                        http::Method::POST,
170                        logout_path,
171                        crate::auth::logout_post,
172                    ));
173            }
174        }
175        if !search_handlers.is_empty() {
176            builder = builder.app_context(SearchRegistry(search_handlers));
177        }
178        // The mount prefix travels with the Router so generic handlers derive
179        // resource URLs from the declaration instead of sniffing the request
180        // path (item 6 / B4).
181        builder = builder.app_context(PanelPrefix(prefix.clone()));
182        if !nav_items.is_empty() {
183            builder = builder.app_context(nav_items);
184        }
185        if let Some(assets) = assets {
186            builder = builder.assets(assets);
187        }
188        if let Some(shell_assets) = shell_assets {
189            builder = builder.app_context(shell_assets);
190        }
191        if let Some(brand) = brand {
192            builder = builder.app_context(brand);
193        }
194        if let Some(enabled) = dark_mode {
195            builder = builder.app_context(DarkMode(enabled));
196        }
197        // Where uploaded bytes go: installed once, found by the form
198        // handlers and the multipart parser through the app context.
199        if let Some(uploads) = uploads {
200            builder = builder.app_context(uploads);
201        }
202        for (path, dir) in served_dirs {
203            // Files the panel serves share its origin, so each directory route
204            // is wrapped in the hardening layer that makes them inert
205            // The same path scopes the layer to that route only.
206            builder = builder
207                .layer(headers::ServedFileHeaders::new(&path))
208                .serve_dir(route_path(&path), dir);
209        }
210        for page in pages {
211            builder = builder.page(page);
212        }
213        for route in routes {
214            builder = builder.route(route);
215        }
216        // The panel root has no home page of its own; until custom pages exist,
217        // the prefix serves a redirect to the first resource's
218        // list so the mount point is never a dead URL.
219        if let Some(target) = root_target {
220            builder = builder
221                .app_context(RootRedirect(target))
222                .route(RouteFn::new(
223                    http::Method::GET,
224                    route_path(&prefix),
225                    panel_root_redirect,
226                ));
227        }
228        #[cfg(feature = "auth")]
229        {
230            builder = builder.app_context(auth);
231            if let Some(hint) = login_hint {
232                builder = builder.app_context(LoginHint(hint));
233            }
234        }
235        // The runtime layer registers last, outside every other pathless
236        // layer: a page re-run is a marked POST the layer rewrites into a
237        // GET for the page's own URL, and the layers it wraps must receive
238        // the rewritten GET rather than the discarded POST.
239        Ok(builder.runtime().build())
240    }
241}

249/// The panel root: a temporary redirect to the first declared resource's
250/// list, so the mount point is never a dead URL (custom pages remain future
251/// work; see `docs/guide/src/panel-and-routing.md`). Filament registers its
252/// home page here.
253pub(crate) fn panel_root_redirect(cx: &Cx, _body: Body) -> RouteFuture<'_> {
254    Box::pin(async move {
255        // Defense in depth: every panel handler re-checks the
256        // resolved user, so a missing or mis-mounted gate cannot leak the
257        // first resource's slug via the redirect target.
258        enforce_auth(cx)?;
259        let RootRedirect(target) = app_context::<RootRedirect>(cx);
260        Err(redirect(target.clone()).into())
261    })
262}

264/// Whether a path is a route pattern ending in a catch-all, which is the only
265/// shape [`DirectoryRoute`](topcoat::router::DirectoryRoute) accepts.
266///
267/// Checked where the path is declared rather than where it is used: upstream
268/// `serve_dir` panics on anything else, and `Panel::build` reports instead of
269/// panicking — but the path comes from the app, and it would panic
270/// first in [`route_path`] (which refuses to spell a route it cannot parse) and
271/// then inside `DirectoryRoute::new` (which needs the catch-all last). Asking
272/// both conditions here turns a typo into a build error instead of a panic
273/// during the build.
274pub(super) fn is_directory_pattern(path: &str) -> bool {
275    Path::from_str(path)
276        .ok()
277        .and_then(|parsed| parsed.segments().next_back())
278        .is_some_and(|segment| segment.as_catch_all().is_some())
279}

281/// Validate one path segment a panel derives routes from: a
282/// `Resource::slug()` override, or a segment of the panel prefix.
283///
284/// Both reach a route path and, through the panel, a response body. A hostile
285/// value — quote, backslash, CR/LF, `..`, slash, URL punctuation, a route
286/// pattern character — must fail at registration rather than at request time,
287/// so this is the export filename sanitizer's rule tightened to what a URL
288/// segment can be: the export drops the offending characters because it must
289/// still produce a download, while a route has no meaningful fallback.
290///
291/// The route pattern characters that a literal segment cannot carry (`{`, `}`,
292/// `(`, `)`) are rejected rather than escaped: `Path::from_str` treats
293/// `{`/`(` as the start of a parameter or group segment, so a balanced pair
294/// silently becomes a pattern and an unbalanced one panics [`route_path`]. `*`
295/// stays accepted — it is a literal in a static segment — and the catch-all
296/// spelling `{*name}` needs the `{` this rule already refuses.
297pub(super) fn validate_route_segment(kind: &str, segment: &str) -> Result<(), String> {
298    if segment.is_empty() {
299        return Err(format!("{kind}: path segment must not be empty"));
300    }
301    if segment == "." || segment == ".." {
302        return Err(format!(
303            "{kind} '{segment}': a path segment may not be '.' or '..'"
304        ));
305    }
306    if let Some(bad) = segment.chars().find(|c| {
307        c.is_control()
308            || c.is_whitespace()
309            || matches!(
310                c,
311                '"' | '\\' | '/' | '?' | '#' | '%' | '&' | '=' | '{' | '}' | '(' | ')'
312            )
313    }) {
314        return Err(format!(
315            "{kind} '{segment}': a path segment may not contain {bad:?} (quotes, backslashes, control characters, whitespace, URL punctuation and the route pattern characters '{{', '}}', '(' and ')' are rejected)"
316        ));
317    }
318    Ok(())
319}

321/// A resource's build-time declaration check: monomorphized once per
322/// declared resource by [`Panel::resource`], run by [`Panel::build`] with the
323/// app's values and no request.
324pub(super) type ResourceCheck = fn(&Cx) -> Result<(), String>;

326/// What a declared resource must be able to promise before the panel serves it.
327///
328/// The trait defaults every method but `table`, so a resource that overrides
329/// nothing else compiles and only fails when a user reaches the page that needs
330/// the missing piece. The essentials that are *declarations* — a tenant
331/// predicate for a gated resource and a page size the list can serve — are
332/// checked here, at build, and reported with the
333/// resource's type name, together with the agreement between the resource's
334/// `Form` and its `form()` schema ([`check_form_declaration`]).
335/// Runtime essentials (the record fns) keep their loud failure.
336///
337/// A declaration that panics is a boot failure too: `Resource::table` and
338/// `Resource::form` run code that panics on a mis-declaration, and this check's
339/// contract is a registration error the caller can log or exit on. The whole
340/// body is caught, because `R::Model::schema()` and the policy predicates are
341/// part of the same declaration, and the panic's own message is carried into
342/// the error. `AssertUnwindSafe` is sound because nothing observes the captured
343/// state after an unwind: `cx` is the build-time `validation_cx`, and the panic
344/// fails the whole `build`.
345pub(super) fn check_resource<R: Resource>(cx: &Cx) -> Result<(), String> {
346    caught::<R>(|| check_resource_inner::<R>(cx))
347}
348
349/// The message out of a caught panic payload.
350///
351/// The declaration panics this catches are `assert!`/`panic!("…")` with a
352/// formatted string, so `&str` and `String` cover every one of them; anything
353/// else is reported by shape rather than silently dropped.
354fn panic_message(payload: &(dyn std::any::Any + Send)) -> String {
355    if let Some(message) = payload.downcast_ref::<&str>() {
356        (*message).to_string()
357    } else if let Some(message) = payload.downcast_ref::<String>() {
358        message.clone()
359    } else {
360        "a non-string panic payload".to_string()
361    }
362}

364/// The body of [`check_resource`], unwound through `catch_unwind` so a
365/// mis-declared resource is a registration error rather than a boot panic.
366fn check_resource_inner<R: Resource>(cx: &Cx) -> Result<(), String> {
367    // A gated resource that supplies no tenant predicate is misdeclared, and
368    // the declaration is checkable without a request:
369    // `R::tenant_scope` is pure, and the default derivation answers by the
370    // model's *shape* — a `tenant_id` UUID field, found by name and type — not
371    // by the tenant value, so the nil UUID is enough to ask whether a predicate
372    // exists at all. Refusing here is what `build`'s contract promises a
373    // declaration error gets; the request-time error in `apply_tenant_scope`
374    // stays as the backstop for a resource whose predicate is only `None` for
375    // some tenants, and for app code that calls `scoped_query` outside a panel.
376    //
377    // Checked before the page essentials below because the gate and the scope
378    // govern every handler this resource registers, not just the list and
379    // create pages those checks are about.
380    if R::requires_tenant() && R::tenant_scope(uuid::Uuid::nil()).is_none() {
381        return Err(format!(
382            "resource `{}` requires a tenant, but the framework cannot scope it: `{}` declares no \
383             `tenant_id` UUID column to derive the filter from, and the resource does not override \
384             `tenant_scope` — declare the column, override `tenant_scope`, or drop \
385             `requires_tenant` and scope in `query` (GH #231)",
386            std::any::type_name::<R>(),
387            std::any::type_name::<R::Model>(),
388        ));
389    }
390    // The table carries its key and columns by construction; only the page
391    // size can still misdeclare, so the build refuses that here rather than
392    // at request time.
393    if let Some(missing) = R::table(cx).missing_essentials() {
394        return Err(format!(
395            "resource `{}` cannot serve its list: {missing}",
396            std::any::type_name::<R>()
397        ));
398    }
399    check_form_declaration::<R>(cx)
400}

402/// A resource with a record form declares its schema and runs
403/// [`check_form_inner`]. A resource whose form serves no pages
404/// ([`RecordForm::HAS_FORM`] false) declares no schema, and a policy that
405/// allows create or an edit link would lead nowhere.
406fn check_form_declaration<R: Resource>(cx: &Cx) -> Result<(), String> {
407    let resource = std::any::type_name::<R>();
408    let form = std::any::type_name::<R::Form>();
409    if <R::Form as RecordForm>::HAS_FORM {
410        // A form with fields and the empty `form()` default is a missing
411        // override; the key check below would only name the first field.
412        if R::form(cx).is_empty() && !<R::Form as RecordForm>::fields(cx).is_empty() {
413            return Err(format!(
414                "resource `{resource}` names record form `{form}` but does not override `form()`, \
415                 whose default declares no controls"
416            ));
417        }
418        return check_form_inner::<R>(cx);
419    }
420    if !R::form(cx).is_empty() {
421        return Err(format!(
422            "resource `{resource}` declares a form schema but its `Form`, `{form}`, serves no \
423             form — name the record form in `type Form`"
424        ));
425    }
426    let declared = match (R::can_create(cx), R::editable()) {
427        (true, true) => "create and edit",
428        (true, false) => "create",
429        (false, true) => "edit",
430        (false, false) => return Ok(()),
431    };
432    Err(format!(
433        "resource `{resource}` allows {declared} but has no form — name its record form in `type \
434         Form` and declare `form()`"
435    ))
436}

438fn check_form_inner<R: Resource>(cx: &Cx) -> Result<(), String> {
439    let form = R::form(cx);
440    let resource = std::any::type_name::<R>();
441    let fields = <R::Form as RecordForm>::fields(cx);
442    let controls = form.controls();
443    // Key agreement, reported control-first: a control no field binds is the
444    // direction that drops what the user typed.
445    for control in &controls {
446        let claims = fields
447            .iter()
448            .filter(|field| field.keys.contains(&control.name))
449            .count();
450        if claims == 0 {
451            return Err(format!(
452                "resource `{resource}` renders form control `{}` but no field of its record form \
453                 binds it, so what the user types there is never written",
454                control.name
455            ));
456        }
457        if claims > 1 {
458            return Err(format!(
459                "resource `{resource}` binds form control `{}` from more than one record-form field",
460                control.name
461            ));
462        }
463    }
464    for field in &fields {
465        for key in &field.keys {
466            if !controls.iter().any(|control| &control.name == key) {
467                return Err(format!(
468                    "resource `{resource}`'s record form field `{}` binds key `{key}`, but the form \
469                     declares no control for it",
470                    field.name
471                ));
472            }
473        }
474        // Blank agreement: an empty submission must resolve wherever the
475        // schema lets one through.
476        if field.answers_blank {
477            continue;
478        }
479        if let Some(control) = controls.iter().find(|control| {
480            field.keys.contains(&control.name)
481                && (!control.required || control.skipped_by.is_some())
482        }) {
483            let place = match control.skipped_by {
484                Some(SkippedBy::Repeater) => "sits inside a `Repeater`, so it may be posted empty",
485                Some(SkippedBy::VariantGroup) => {
486                    "sits inside a variant group, which a submission naming another variant \
487                     leaves unchecked"
488                }
489                None => "is optional",
490            };
491            return Err(format!(
492                "resource `{resource}`'s form control `{}` {place}, but record form field `{}` has \
493                 no blank answer — declare `#[record_form(blank = ..)]`, make the field an \
494                 `Option`, or make the control required",
495                control.name, field.name
496            ));
497        }
498    }
499    // Tenant ownership: the framework stamps a gated resource's tenant column
500    // on create; a form that claimed it would let the client choose.
501    if R::requires_tenant()
502        && let Some(column) = crate::tenancy::tenant_field_name::<R::Model>()
503        && let Some(field) = fields.iter().find(|field| field.keys.contains(&column))
504    {
505        return Err(format!(
506            "resource `{resource}` requires a tenant, but record form field `{}` claims its tenant \
507             column `{column}` — the framework stamps it on create; drop it from the form",
508            field.name
509        ));
510    }
511    // Column coverage: a create that leaves a non-nullable column unset fails
512    // at the driver on every submit, with no field to point the user at.
513    if R::can_create(cx) {
514        check_create_columns::<R>(&fields)?;
515    }
516    // `.unique()` is a promise the panel makes and the database has to keep
517    // (item 3): the marker turns the app-side pre-check on, so a field
518    // whose column carries no unique index makes the panel enforce a rule
519    // nothing else does — a duplicate the check lets through, or a rule the
520    // database never asked for. The declaration checks are the only place both
521    // halves are reachable without a request, so the pair is refused here
522    // rather than discovered by a user. It is checked whatever the policies say:
523    // a `unique()` marker is wrong on a form the panel would not even serve.
524    // `lens_field_unique` recognizes composite indexes too, which is what makes
525    // `#[unique(tenant_id, email)]` — the tenant-scoped arrangement the panel
526    // documents — pass.
527    let model = R::Model::schema();
528    let root = model.as_root_unwrap();
529    for (name, input) in form.text_inputs() {
530        if !input.is_unique() {
531            continue;
532        }
533        // A bound lens always resolves, so a name with no field at all is a
534        // mis-declared schema — but it is not worth a second error string: it
535        // fails the same way, one message below.
536        let backed = root
537            .fields
538            .iter()
539            .filter(|field| field.name.app_unwrap() == name)
540            .any(|field| crate::schema::lens_field_unique(field, root));
541        if !backed {
542            return Err(format!(
543                "resource `{}` marks form field `{name}` unique, but `{}::{name}` carries no unique index — add `#[unique]` (or `#[unique(..)]`) to the column or drop `.unique()`, which would otherwise check a rule the database does not enforce",
544                std::any::type_name::<R>(),
545                std::any::type_name::<R::Model>()
546            ));
547        }
548    }
549    Ok(())
550}

552/// Every non-nullable column a create must set is set by something: the
553/// record form, toasty (`#[auto]`, `#[default(..)]`), the tenant stamp, or the
554/// resource's own `CREATE_COLUMNS`.
555fn check_create_columns<R: Resource>(
556    fields: &[crate::form::FormField<<R::Form as RecordForm>::Field>],
557) -> Result<(), String> {
558    let resource = std::any::type_name::<R>();
559    let prefilled = crate::form::prefilled_fields::<R::Model>();
560    let tenant = R::requires_tenant()
561        .then(crate::tenancy::tenant_field_name::<R::Model>)
562        .flatten();
563    let model = R::Model::schema();
564    let root = model.as_root_unwrap();
565    for name in R::CREATE_COLUMNS {
566        if !root
567            .fields
568            .iter()
569            .any(|field| field.name.app.as_deref() == Some(*name))
570        {
571            return Err(format!(
572                "resource `{resource}` names `{name}` in `CREATE_COLUMNS`, but `{}` has no such \
573                 field",
574                std::any::type_name::<R::Model>()
575            ));
576        }
577    }
578    for (index, field) in root.fields.iter().enumerate() {
579        let Some(name) = field.name.app.as_deref() else {
580            continue;
581        };
582        let filled = field.nullable()
583            || field.is_relation()
584            || prefilled.get(index).copied().unwrap_or(false)
585            || tenant.as_deref() == Some(name)
586            || fields.iter().any(|claim| claim.name == name)
587            || R::CREATE_COLUMNS.contains(&name);
588        if !filled {
589            return Err(format!(

599/// Run a declaration check, turning a panic in the app's declarations into a
600/// registration error naming the resource.
601fn caught<R: Resource>(check: impl FnOnce() -> Result<(), String>) -> Result<(), String> {
602    match std::panic::catch_unwind(std::panic::AssertUnwindSafe(check)) {
603        Ok(result) => result,
604        Err(payload) => Err(format!(
605            "resource `{}` panicked while declaring itself: {}",
606            std::any::type_name::<R>(),
607            panic_message(payload.as_ref())
608        )),
609    }
610}
