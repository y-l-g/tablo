26/// Retry link for a failed streamed table load.
27///
28/// A malformed `?after=`/`?before=` cursor, or a cursor the query's ordering
29/// refuses, is the failure itself: retrying the identical URL loops forever, so drop
30/// pagination from the link and keep the rest of the state
31/// (search/sort/filters/grouping). Every other failure keeps pagination too
32/// so a transient blip retries the same evidence.
33pub(crate) fn retry_url_for_error(
34    state: &TableState,
35    error: &topcoat::Error,
36    path: &str,
37) -> String {
38    if crate::error::TabloError::is_cursor(error) {
39        state.without_cursor(path)
40    } else {
41        state.list_url(path)
42    }
43}

45/// The action chrome a resource's declarations imply: the one derivation
46/// [`wire_table_actions`] reads to decide which affordances the table it serves
47/// carries. Each column follows what also governs its route — the delete
48/// handlers check `can_delete_any`, only a record form registers the edit
49/// route, and only a `view` schema serves the detail page — and is answered
50/// once per request, so the streamed skeleton and the table agree.
51pub(crate) fn declared_chrome<R: Resource>(cx: &Cx) -> TableChrome {
52    TableChrome {
53        delete: R::can_delete_any(cx),
54        edit: <R::Form as RecordForm>::HAS_FORM,
55        view: R::viewed(cx),
56        actions: true,
57    }
58}

60/// Delete/bulk/edit action wiring shared by the streamed list and the
61/// live-search shard: the delete form posts to `{list url}/{id}/delete` and the
62/// bulk bar to `{list url}/bulk-delete`, both derived from the panel
63/// declaration (not the request path) so the URLs are right wherever the table
64/// renders.
65///
66/// Each affordance renders only where [`declared_chrome`] allows it, so no
67/// control ships whose route cannot answer it. The per-*record* gate rides the
68/// same call: the
69/// table's row policy pairs each action with exactly what its route checks —
70/// `can_view` for View, `can_view` + `can_update` for Edit, `can_view` +
71/// `can_delete` for Delete, and `can_view` + `can_run` for each custom action;
72/// the bulk checkbox renders when bulk delete or any bulk custom action allows
73/// the record. A row the predicate refuses
74/// renders no link and no checkbox, while the handler keeps its
75/// all-or-nothing check for a hand-crafted POST.
76///
77/// `live` selects the shard variant: the swapped region is everything except the
78/// toolbar the page owns eagerly (the live host owns those slots, so a swap must
79/// never nest invocations or duplicate inputs), hence the shard forces
80/// `.hide_search().hide_filter_bar()` while the streamed page keeps the declared
81/// table as-is. The filter bar joins the search toolbar there: a control rebuilt
82/// by its own rerun loses focus.
83pub(crate) fn wire_table_actions<R: Resource>(cx: &Cx, live: bool) -> Table<R::Model> {
84    wire_table::<R>(cx, live, declared_chrome::<R>(cx))
85}

87/// [`wire_table_actions`] with the affordances `chrome` names, a subset of
88/// [`declared_chrome`]: a read-only relation keeps only the View link.
89pub(crate) fn wire_table<R: Resource>(cx: &Cx, live: bool, chrome: TableChrome) -> Table<R::Model> {
90    let mut table = R::table(cx);
91    if live {
92        // The page draws the card around the hoisted bars and this output.
93        table = table.hide_search().hide_filter_bar().unframed();
94    }
95    // `Cx` is Arc-backed and `Clone`, so the projection owns one: the policy
96    // outlives the request borrow without copying request state.
97    let policy_cx = cx.clone();
98    table = table.row_actions(move |record| {
99        // Read once: every route pairs its own predicate with `can_view`, so a
100        // record that cannot be viewed allows no action.
101        let view = R::can_view(&policy_cx, record);
102        RowActions {
103            view,
104            edit: view && R::can_update(&policy_cx, record),
105            delete: view && R::can_delete(&policy_cx, record),
106        }
107    });
108    if chrome.delete {
109        table = table
110            .with_delete(list_url(cx, &R::slug()))
111            .with_bulk_delete(true);
112    }
113    if chrome.edit {
114        table = table.with_edit(list_url(cx, &R::slug()));
115    }
116    if chrome.view {
117        table = table.with_view(list_url(cx, &R::slug()));
118    }
119    if chrome.actions {
120        table = wire_custom_actions::<R>(cx, table);
121    }
122    table
123}

125/// Attach `R`'s custom actions to `table`, each gated per record by exactly
126/// what its route checks: [`Resource::can_view`] and the action's
127/// `can_run`.
128fn wire_custom_actions<R: Resource>(cx: &Cx, table: Table<R::Model>) -> Table<R::Model> {
129    let actions = R::actions();
130    if actions.entries().is_empty() {
131        return table;
132    }
133    let wired = actions
134        .entries()
135        .iter()
136        .map(|action| {
137            let can_run = action.can_run;
138            let policy_cx = cx.clone();
139            TableAction {
140                name: action.name,
141                label: (action.label)(),
142                row: action.row,
143                bulk: action.bulk,
144                allowed: std::sync::Arc::new(move |record: &R::Model| {
145                    R::can_view(&policy_cx, record) && can_run(&policy_cx, record)
146                }),
147            }
148        })
149        .collect();
150    table.with_custom_actions(list_url(cx, &R::slug()), wired)
151}

153/// Branded in-region table failure shared by the streamed list and the
154/// live-search shard: the trace line, the cursor-aware
155/// retry link ([`retry_url_for_error`]), and the `ErrorState` render are one
156/// copy so the three load sites cannot drift.
157///
158/// On a live table (`signals`) the retry stays in place instead of
159/// navigating, and re-runs the request that failed with the query it failed
160/// with: the click increments a retry token the shard reads, so the
161/// rerun does not depend on the query signals changing — a write of an
162/// unchanged value re-runs nothing. A cursor failure resets only the cursor,
163/// the same reset its `href` spells out; every other failure keeps the whole
164/// query. `href` stays as the no-JS fallback, so it retries the URL as it
165/// stands.
166pub(crate) fn table_error_view<'a, R: Resource>(
167    cx: &'a Cx,
168    state: &TableState,
169    error: &topcoat::Error,
170    path: &str,
171    signals: Option<&TableSignals>,
172) -> BoxView<'a> {
173    tracing::error!(resource = R::slug(), error = %error, "table load failed");
174    let retry_url = retry_url_for_error(state, error, path);
175    let action: BoxView<'a> = match signals {
176        Some(signals) => {
177            // Retry re-runs the shard. The token is the rerun's only
178            // cause: the shard reads it (declaring the dependency below), and
179            // every click increments it, so the write always changes even when
180            // the query signals already hold the values that failed.
181            // Keyed by the list, like the table's own signals.
182            let attempt = topcoat::runtime::signal(&cx.keyed(path), || 0u64);
183            let cursor_error = crate::error::TabloError::is_cursor(error);
184            let attrs = if cursor_error {
185                // The retry URL drops the cursor: write its query, so the rerun
186                // loads the first page of the same result set.
187                let query = signals.query.clone();
188                let next = crate::resource::query_of(&retry_url).to_string();
189                attributes! {
190                    cx =>
191                    href=(retry_url)
192                    data-retry-attempt=(attempt.get())
193                    @click=$(|e: Event| {
194                        e.prevent_default();
195                        query.set(next.clone());
196                        attempt.increment();
197                    })
198                }
199            } else {
200                attributes! {
201                    cx =>
202                    href=(retry_url)
203                    data-retry-attempt=(attempt.get())
204                    @click=$(|e: Event| {
205                        e.prevent_default();
206                        attempt.increment();
207                    })
208                }
209            };
210            view! { cx => <a (attrs)>"Retry"</a> }.boxed()
211        }
212        None => view! { cx => <a href=(retry_url)>"Retry"</a> }.boxed(),
213    };
214    view! {
215        cx =>
216        tablo_ui::error_state(
217            title: format!("Couldn't load {}", R::navigation_label()),
218            detail: "Something went wrong while loading the records.",
219            action: Some(action.into()),
220            attrs: attributes! { role="alert" }
221        )
222    }
223    .boxed()
224}

226/// The list page header: the resource's title and the Create entry point
227/// (Filament's List page `CreateAction`), a real link so no-JS keeps
228/// working, gated on `can_create`. The POST handler enforces the same policy.
229///
230/// A resource with no record form ([`RecordForm::HAS_FORM`]) has no create
231/// route, so its header links to none even when a request-scoped `can_create`
232/// allows it and the build-time check could not see it.
233fn list_header<'a, R: Resource>(cx: &'a Cx, title: &str, list_path: &str) -> BoxView<'a> {
234    let title = title.to_string();
235    let create_url = (<R::Form as RecordForm>::HAS_FORM && R::can_create(cx))
236        .then(|| create_page_url(list_path));
237    let create_label = format!("Create {}", R::label());
238    view! {
239        cx =>
240        tablo_ui::page_header(
241            tablo_ui::page_title((title))
242            if let Some(url) = create_url {
243                tablo_ui::page_actions(
244                    <a
245                        (crate::resource::runtime_link(cx, &url))
246                        class=(tablo_ui::button_variants(
247                            tablo_ui::ButtonVariant::Primary,
248                            tablo_ui::ButtonSize::Md,
249                        ))
250                    >
251                        icon(data: tablo_ui::icons::PLUS)
252                        (create_label)
253                    </a>
254                )
255            }
256        )
257    }
258    .boxed()
259}

261/// The list page every declared [`Resource`] gets at `{prefix}/{slug}`.
262///
263/// One generic handler drives all resources: resolve the [`TableState`] from
264/// the URL, scope through the tenant-scoped query,
265/// apply the table's search/sort/pagination declarations, render through
266/// `Resource::table`. The page title is the resource's navigation label.
267///
268/// The page streams: shell and header go out with the first content, while the
269/// table body (toolbar/filter/bulk/pager included) loads inside a `suspense`
270/// region that swaps in the skeleton → table without any client-side fetching
271/// (GH #98: the skeleton is thead + placeholders only, so chrome pops in with
272/// the swap by design).
273pub(crate) fn resource_list<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
274    Box::pin(HoistView::new(ThenView::new(async move {
275        gate::<R>(cx)?;
276        if !R::can_view_any(cx) {
277            return Err(forbidden().into());
278        }
279        // Ensure the CSRF cookie before streaming starts: streamed
280        // children can only read it via current_token.
281        crate::csrf::ensure_token(cx);
282        let state = TableState::from_cx(cx);
283        let table = wire_table_actions::<R>(cx, false);
284        let title = R::navigation_label();
285        let list_path = list_url(cx, &R::slug());
286        if table.is_live_search() {
287            return Ok(resource_list_live::<R>(cx, table, state, title, list_path));
288        }
289
290        // The skeleton table (`Table::render_skeleton`) streams
291        // while the rows load below.
292        // The load catches its own errors: post-stream
293        // the status line is fixed, so a failed load must render the branded
294        // ErrorState inside the region instead of truncating the body.
295        // Pre-stream failures (e.g. the skeleton itself) still propagate and
296        // map onto the response status. (For children that partially stream
297        // before failing, topcoat's `error_boundary` is the replace-in-place
298        // seam.)
299        //
300        // One normalization per request: the skeleton, the load,
301        // the render and the retry link all read the state this page parsed,
302        // so it normalizes here and every seam below takes the proof (GH
303        // #153: the retry link must not echo an unknown `?group_by=`).
304        let state = table.normalize_state(&state);
305        let skeleton = table.render_skeleton(cx, &state).await?;
306        let header = list_header::<R>(cx, &title, &list_path);
307        let lazy_rows = ThenView::new(async move {
308            let rendered = async {
309                let page = load_table_page::<R>(cx, &table, &state).await?;
310                table.render_with_state(cx, page, &state, &list_path).await
311            };
312            match rendered.await {
313                Ok(view) => Ok(view),
314                Err(error) => Ok(table_error_view::<R>(cx, &state, &error, &list_path, None)),
315            }
316        });
317
318        Ok(view! {
319            cx =>
320            tablo_ui::page(
321                (header)
322                tablo_ui::page_content(
323                    suspense(fallback: skeleton, (lazy_rows.boxed()))
324                )
325            )
326        }
327        .boxed())
328    })))
329}

331/// Live list page for `Table::live_search` tables: the page owns the
332/// interaction signals (`query`, the list's URL query, and `bulk`, the
333/// selection) and renders the search and filter bars eagerly above the
334/// streamed region while the `table_search` shard invocation fills the table
335/// below — one table per response, so rows can never duplicate. Search, sort,
336/// filters, and pagination write the query, so they re-render only the
337/// invocation output, morphing in place with focus and scroll surviving.
338pub(crate) fn resource_list_live<R: Resource>(
339    cx: &Cx,
340    table: Table<R::Model>,
341    state: TableState,
342    title: String,
343    list_path: String,
344) -> BoxView<'_> {
345    Box::pin(HoistView::new(ThenView::new(async move {
346        // Seeded with the request's query as written: the shard parses and
347        // normalizes it as this page did.
348        //
349        // Keyed by the list: runtime navigation carries every signal the next
350        // page shares with this one, and one call site would otherwise give
351        // every resource's list the same ids — one list's search would filter
352        // the next.
353        let signals = TableState::signals_for(&cx.keyed(list_path.as_str()), &request_query(cx));
354        // One normalization per request: the toolbar, the hoisted
355        // filter bar, the skeleton, the dialog and the retry link all read
356        // the state this page parsed, so it normalizes here and every seam
357        // below takes the proof (: no unknown `?group_by=` in a link).
358        let state = table.normalize_state(&state);
359        let host = if table.search_enabled() {
360            Some(
361                table
362                    .render_live_search_bar(cx, &state, &list_path, &signals)
363                    .await?,
364            )
365        } else {
366            None
367        };
368        // The filter bar is hoisted next to the search host: a
369        // `<select>` change re-renders the table, and a control inside the
370        // swapped region would lose focus and collapse its popup mid-change.
371        let filter_bar = if table.filter_bar_enabled() {
372            Some(
373                table
374                    .render_live_filter_bar(cx, &state, &list_path, &signals)
375                    .await?,
376            )
377        } else {
378            None
379        };
380        // The shard's table renders neither bar (`wire_table_actions(cx,
381        // true)`), so neither does the placeholder it replaces.
382        // Nor does it draw a card: the page's card holds the bars and the
383        // table together.
384        let table = table.hide_search().hide_filter_bar().unframed();
385        let skeleton = table.render_skeleton(cx, &state).await?;
386        // The delete confirmation dialog is not part of the swapped table
387        // region: a keystroke starts a new result set and must never carry
388        // (or re-open) a dialog, so the live page renders it eagerly once.
389        let delete_dialog = table.render_delete_dialog(cx, &state).await?;
390        let header = list_header::<R>(cx, &title, &list_path);
391        let lazy_rows = ThenView::new(async move {
392            // The retry link inside the table writes the same signals the
393            // toolbar does, so a bad cursor recovers in place.
394            let retry_signals = signals.clone();
395            let rendered = table.render_live_invocation(cx, &list_path, signals).await;
396            match rendered {
397                Ok(view) => Ok(view),
398                Err(error) => Ok(table_error_view::<R>(
399                    cx,
400                    &state,
401                    &error,
402                    &list_path,
403                    Some(&retry_signals),
404                )),
405            }
406        });
407
408        Ok(view! {
409            cx =>
410            tablo_ui::page(
411                (header)
412                tablo_ui::page_content(
413                    // The page draws the table's card (`TABLE_CARD_CLASS`) so the
414                    // hoisted bars share it with the table they drive.
415                    <div class=(TABLE_CARD_CLASS)>
416                        if let Some(host) = host {
417                            (host)
418                        }
419                        if let Some(bar) = filter_bar {
420                            (bar)
421                        }
422                        suspense(fallback: skeleton, (lazy_rows.boxed()))
423                    </div>
424                    if let Some(dialog) = delete_dialog {
425                        (dialog)
426                    }
427                )
428            )
429        })
430    })))
431}

433/// Load `R`'s list page for `state` from the tenant-scoped
434/// [`scoped_query`](crate::resource::scoped_query) — the data-loading half of
435/// [`resource_list`] and the live shard, kept separate so the page shell can
436/// stream before it. [`TablePage::load`] applies the table's declaration and
437/// the relations its columns include.
438pub(crate) async fn load_table_page<R: Resource>(
439    cx: &Cx,
440    table: &Table<R::Model>,
441    state: &TableState,
442) -> Result<TablePage<R::Model>> {
443    TablePage::load(cx, table, crate::resource::scoped_query::<R>(cx)?, state).await
444}
445
446/// [`load_table_page`] over the rows `scope` also admits: a relation table's
447/// rows that belong to its owner.
448pub(crate) async fn load_scoped_page<R: Resource>(
449    cx: &Cx,
450    table: &Table<R::Model>,
451    state: &TableState,
452    scope: toasty::stmt::Expr<bool>,
453) -> Result<TablePage<R::Model>> {
454    let query = crate::resource::scoped_query::<R>(cx)?.filter(scope);
455    TablePage::load(cx, table, query, state).await
456}
