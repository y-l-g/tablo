25/// Retry link for a failed streamed table load.
26///
27/// A malformed `?after=`/`?before=` cursor, a conflicting `after` + `before`
28/// pair, or a cursor the query's ordering refuses is the
29/// failure itself: retrying the identical URL loops forever, so drop
30/// pagination from the link and keep the rest of the state
31/// (search/sort/filters/grouping). Every other failure keeps pagination too
32/// so a transient blip retries the same evidence.
33pub(crate) fn retry_url_for_error(
34    state: &TableState,
35    error: &topcoat::Error,
36    path: &str,
37) -> String {
38    if crate::cursor::is_cursor_error(error) {
39        state.without_cursor(path)
40    } else {
41        state.list_url(path)
42    }
43}

45/// The action chrome a resource's declarations imply: the one derivation
46/// [`wire_table_actions`] reads to decide which affordances the table it serves
47/// carries. Each column follows what also governs its route — the delete
48/// handlers check `can_delete_any`, only a record form registers the edit
49/// route, and only a `view` schema serves the detail page — and is answered
50/// once per request, so the streamed skeleton and the table agree.
51pub(crate) fn declared_chrome<R: Resource>(cx: &Cx) -> TableChrome {
52    TableChrome {
53        delete: R::can_delete_any(cx),
54        edit: <R::Form as RecordForm>::HAS_FORM,
55        view: R::viewed(cx),
56    }
57}

59/// Delete/bulk/edit action wiring shared by the streamed list and the
60/// live-search shard: the delete form posts to `{list url}/{id}/delete` and the
61/// bulk bar to `{list url}/bulk-delete`, both derived from the panel
62/// declaration (not the request path) so the URLs are right wherever the table
63/// renders.
64///
65/// Each affordance renders only where [`declared_chrome`] allows it, so no
66/// control ships whose route cannot answer it. The per-*record* gate rides the
67/// same call: the
68/// table's row policy pairs each action with exactly what its route checks —
69/// `can_view` for View, `can_view` + `can_update` for Edit, `can_view` +
70/// `can_delete` for Delete and the bulk checkbox. A row the predicate refuses
71/// renders no link and no checkbox, while the handler keeps its
72/// all-or-nothing check for a hand-crafted POST.
73///
74/// `live` selects the shard variant: the swapped region is everything except the
75/// toolbar the page owns eagerly (the live host owns those slots, so a swap must
76/// never nest invocations or duplicate inputs), hence the shard forces
77/// `.search(false).filter_bar(false)` while the streamed page keeps the declared
78/// table as-is. The filter bar joins the search toolbar there: a control rebuilt
79/// by its own rerun loses focus.
80pub(crate) fn wire_table_actions<R: Resource>(cx: &Cx, live: bool) -> Table<R::Model> {
81    let mut table = R::table(cx);
82    if live {
83        table = table.search(false).filter_bar(false);
84    }
85    // `Cx` is Arc-backed and `Clone`, so the projection owns one: the policy
86    // outlives the request borrow without copying request state.
87    let policy_cx = cx.clone();
88    table = table.row_actions(move |record| {
89        // Read once: every route pairs its own predicate with `can_view`, so a
90        // record that cannot be viewed allows no action.
91        let view = R::can_view(&policy_cx, record);
92        RowActions {
93            view,
94            edit: view && R::can_update(&policy_cx, record),
95            delete: view && R::can_delete(&policy_cx, record),
96        }
97    });
98    let chrome = declared_chrome::<R>(cx);
99    if chrome.delete {
100        table = table
101            .with_delete(list_url(cx, &R::slug()))
102            .with_bulk_delete(true);
103    }
104    if chrome.edit {
105        table = table.with_edit(list_url(cx, &R::slug()));
106    }
107    if chrome.view {
108        table = table.with_view(list_url(cx, &R::slug()));
109    }
110    table
111}

113/// Branded in-region table failure shared by the streamed list and the
114/// live-search shard: the trace line, the cursor-aware
115/// retry link ([`retry_url_for_error`]), and the `ErrorState` render are one
116/// copy so the three load sites cannot drift.
117///
118/// On a live table (`signals`) the retry stays in place instead of
119/// navigating, and re-runs the request that failed with the query it failed
120/// with: the click increments a retry token the shard reads, so the
121/// rerun does not depend on the query signals changing — a write of an
122/// unchanged value re-runs nothing. A cursor failure resets only the cursor,
123/// the same reset its `href` spells out; every other failure keeps the whole
124/// query. `href` stays as the no-JS fallback, so it retries the URL as it
125/// stands.
126pub(crate) fn table_error_view<'a, R: Resource>(
127    cx: &'a Cx,
128    state: &TableState,
129    error: &topcoat::Error,
130    path: &str,
131    signals: Option<&TableSignals>,
132) -> BoxView<'a> {
133    tracing::error!(resource = R::slug(), error = %error, "table load failed");
134    let retry_url = retry_url_for_error(state, error, path);
135    let action: BoxView<'a> = match signals {
136        Some(signals) => {
137            // Retry re-runs the shard. The token is the rerun's only
138            // cause: the shard reads it (declaring the dependency below), and
139            // every click increments it, so the write always changes even when
140            // the query signals already hold the values that failed.
141            // Keyed by the list, like the table's own signals.
142            let attempt = topcoat::runtime::signal(&cx.keyed(path), || 0u64);
143            let cursor = signals.cursor.clone();
144            let none = crate::resource::cursor_none();
145            let cursor_error = crate::cursor::is_cursor_error(error);
146            let attrs = if cursor_error {
147                attributes! {
148                    cx =>
149                    href=(retry_url)
150                    data-retry-attempt=(attempt.get())
151                    @click=$(|e: Event| {
152                        e.prevent_default();
153                        cursor.set(none.clone());
154                        attempt.increment();
155                    })
156                }
157            } else {
158                attributes! {
159                    cx =>
160                    href=(retry_url)
161                    data-retry-attempt=(attempt.get())
162                    @click=$(|e: Event| {
163                        e.prevent_default();
164                        attempt.increment();
165                    })
166                }
167            };
168            view! { cx => <a (attrs)>"Retry"</a> }.boxed()
169        }
170        None => view! { cx => <a href=(retry_url)>"Retry"</a> }.boxed(),
171    };
172    view! {
173        cx =>
174        tablo_ui::error_state(
175            title: format!("Couldn't load {}", R::navigation_label()),
176            detail: "Something went wrong while loading the records.",
177            action: Some(action.into()),
178            attrs: attributes! { role="alert" }
179        )
180    }
181    .boxed()
182}

184/// The list page header: the resource's title and the Create entry point
185/// (Filament's List page `CreateAction`), a real link so no-JS keeps
186/// working, gated on `can_create`. The POST handler enforces the same policy.
187///
188/// A resource with no record form ([`RecordForm::HAS_FORM`]) has no create
189/// route, so its header links to none even when a request-scoped `can_create`
190/// allows it and the build-time check could not see it.
191fn list_header<'a, R: Resource>(cx: &'a Cx, title: &str, list_path: &str) -> BoxView<'a> {
192    let title = title.to_string();
193    let create_url = (<R::Form as RecordForm>::HAS_FORM && R::can_create(cx))
194        .then(|| create_page_url(list_path));
195    let create_label = format!("Create {}", R::navigation_label());
196    view! {
197        cx =>
198        tablo_ui::page_header(
199            <div class="flex items-center justify-between gap-4">
200                tablo_ui::page_title((title))
201                if let Some(url) = create_url {
202                    <a
203                        (crate::resource::runtime_link(cx, &url))
204                        class=(tablo_ui::button_variants(
205                            tablo_ui::ButtonVariant::Primary,
206                            tablo_ui::ButtonSize::Md,
207                        ))
208                    >
209                        (create_label)
210                    </a>
211                }
212            </div>
213        )
214    }
215    .boxed()
216}

218/// The list page every declared [`Resource`] gets at `{prefix}/{slug}`.
219///
220/// One generic handler drives all resources: resolve the [`TableState`] from
221/// the URL, scope through the tenant-scoped query,
222/// apply the table's search/sort/pagination declarations, render through
223/// `Resource::table`. The page title is the resource's navigation label.
224///
225/// The page streams: shell and header go out with the first content, while the
226/// table body (toolbar/filter/bulk/pager included) loads inside a `suspense`
227/// region that swaps in the skeleton → table without any client-side fetching
228/// (GH #98: the skeleton is thead + placeholders only, so chrome pops in with
229/// the swap by design).
230pub(crate) fn resource_list<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
231    Box::pin(HoistView::new(ThenView::new(async move {
232        gate::<R>(cx)?;
233        if !R::can_view_any(cx) {
234            return Err(forbidden().into());
235        }
236        // Ensure the CSRF cookie before streaming starts: streamed
237        // children can only read it via current_token.
238        crate::csrf::ensure_token(cx);
239        let state = TableState::from_cx(cx);
240        let table = wire_table_actions::<R>(cx, false);
241        let title = R::navigation_label();
242        let list_path = list_url(cx, &R::slug());
243        if table.is_live_search() {
244            return Ok(resource_list_live::<R>(cx, table, state, title, list_path));
245        }
246
247        // The skeleton table (`Table::render_skeleton_normalized`) streams
248        // while the rows load below.
249        // The load catches its own errors: post-stream
250        // the status line is fixed, so a failed load must render the branded
251        // ErrorState inside the region instead of truncating the body.
252        // Pre-stream failures (e.g. the skeleton itself) still propagate and
253        // map onto the response status. (For children that partially stream
254        // before failing, topcoat's `error_boundary` is the replace-in-place
255        // seam.)
256        //
257        // One normalization per request: the skeleton, the load,
258        // the render and the retry link all read the state this page parsed,
259        // so it normalizes here and every seam below takes the proof (GH
260        // #153: the retry link must not echo an unknown `?group_by=`).
261        let state = table.normalize_state(&state);
262        let skeleton = table.render_skeleton_normalized(cx, &state).await?;
263        let header = list_header::<R>(cx, &title, &list_path);
264        let lazy_rows = ThenView::new(async move {
265            let rendered = async {
266                let page = load_table_page::<R>(cx, &table, &state).await?;
267                table.render_normalized(cx, page, &state, &list_path).await
268            };
269            match rendered.await {
270                Ok(view) => Ok(view),
271                Err(error) => Ok(table_error_view::<R>(cx, &state, &error, &list_path, None)),
272            }
273        });
274
275        Ok(view! {
276            cx =>
277            tablo_ui::page(
278                (header)
279                tablo_ui::page_content(
280                    <div class="flex flex-col gap-4">
281                        suspense(fallback: skeleton, (lazy_rows.boxed()))
282                    </div>
283                )
284            )
285        }
286        .boxed())
287    })))
288}

290/// Live list page for `Table::live_search` tables: the
291/// page owns the interaction signals (`q`, `filters`, `sort`, `dir`,
292/// `cursor`, `group_by`, `bulk`) and renders the search toolbar eagerly above
293/// the streamed region while the `table_search` shard invocation fills the
294/// table below — one table per response, so rows can never duplicate. Every
295/// interaction writes a signal, so search, sort, filters, and pagination
296/// re-render only the invocation output, morphing in place with focus and
297/// scroll surviving.
298pub(crate) fn resource_list_live<R: Resource>(
299    cx: &Cx,
300    table: Table<R::Model>,
301    state: TableState,
302    title: String,
303    list_path: String,
304) -> BoxView<'_> {
305    Box::pin(HoistView::new(ThenView::new(async move {
306        // One state→signal conversion, seeded from the state the
307        // page parsed — before normalizing, so an unknown `?group_by=` seeds
308        // the signal as written and is dropped on the way back in.
309        //
310        // Keyed by the list: runtime navigation carries every signal the next
311        // page shares with this one, and one call site would otherwise give
312        // every resource's list the same ids — one list's search would filter
313        // the next.
314        let signals = state.to_signals(&cx.keyed(list_path.as_str()));
315        // One normalization per request: the toolbar, the hoisted
316        // filter bar, the skeleton, the dialog and the retry link all read
317        // the state this page parsed, so it normalizes here and every seam
318        // below takes the proof (: no unknown `?group_by=` in a link).
319        let state = table.normalize_state(&state);
320        let host = if table.search_enabled() {
321            Some(
322                table
323                    .render_live_search_bar_normalized(cx, &state, &list_path, &signals)
324                    .await?,
325            )
326        } else {
327            None
328        };
329        // The filter bar is hoisted next to the search host: a
330        // `<select>` change re-renders the table, and a control inside the
331        // swapped region would lose focus and collapse its popup mid-change.
332        let filter_bar = if table.filter_bar_enabled() {
333            Some(
334                table
335                    .render_live_filter_bar_normalized(cx, &state, &list_path, &signals)
336                    .await?,
337            )
338        } else {
339            None
340        };
341        let skeleton = table.render_skeleton_normalized(cx, &state).await?;
342        // The delete confirmation dialog is not part of the swapped table
343        // region: a keystroke starts a new result set and must never carry
344        // (or re-open) a dialog, so the live page renders it eagerly once.
345        let delete_dialog = table.render_delete_dialog_normalized(cx, &state).await?;
346        let header = list_header::<R>(cx, &title, &list_path);
347        let lazy_rows = ThenView::new(async move {
348            // The retry link inside the table writes the same signals the
349            // toolbar does, so a bad cursor recovers in place.
350            let retry_signals = signals.clone();
351            let rendered = table.render_live_invocation(cx, &list_path, signals).await;
352            match rendered {
353                Ok(view) => Ok(view),
354                Err(error) => Ok(table_error_view::<R>(
355                    cx,
356                    &state,
357                    &error,
358                    &list_path,
359                    Some(&retry_signals),
360                )),
361            }
362        });
363
364        Ok(view! {
365            cx =>
366            tablo_ui::page(
367                (header)
368                tablo_ui::page_content(
369                    <div class="flex flex-col gap-4">
370                        if let Some(host) = host {
371                            (host)
372                        }
373                        if let Some(bar) = filter_bar {
374                            (bar)
375                        }
376                        suspense(fallback: skeleton, (lazy_rows.boxed()))
377                        if let Some(dialog) = delete_dialog {
378                            (dialog)
379                        }
380                    </div>
381                )
382            )
383        })
384    })))
385}

387/// Resolve the declared table (search / filters / sort / pagination) against
388/// the tenant-scoped [`scoped_query_with`](crate::resource::scoped_query_with)
389/// — the data-loading half of
390/// [`resource_list`], kept separate so the page shell can stream before it.
391///
392/// The load asks for the includes the table's columns declared
393/// ([`Table::include_needs`]), so a resource that narrows its loaders
394/// loads exactly what the rendered cells read; the cursor-existence probes ask
395/// for none, because they only test whether a row exists. A resource that
396/// overrides nothing keeps its full
397/// [`query`](crate::resource::Resource::query) at both, the same safe default
398/// the export takes.
399///
400/// Resource lists must declare a page size: without
401/// [`Table::paginate`] the load would be an unbounded `exec`, so the missing
402/// declaration fails loudly here instead of silently loading the whole table.
403/// Page-owned tables (GH #154 §2) load through [`Table::load`] directly and
404/// keep the unbounded branch for previews.
405pub(crate) async fn load_table_page<R: Resource>(
406    cx: &Cx,
407    table: &Table<R::Model>,
408    state: &TableState,
409) -> Result<TablePage<R::Model>> {
410    if table.page_size().is_none() {
411        return Err(std::io::Error::other(
412            "resource list requires Table::paginate(..) — unbounded tables are previews only (GH #172)",
413        )
414        .into());
415    }
416    table
417        .load_with_probe(
418            cx,
419            crate::resource::scoped_query_with::<R>(cx, &table.include_needs())?,
420            crate::resource::scoped_query_with::<R>(cx, &crate::resource::IncludeNeeds::default())?,
421            state,
422        )
423        .await
424}
