22/// Returns the retry link for a failed table load, dropping pagination for cursor failures and
23/// keeping it otherwise.
24pub(crate) fn retry_url_for_error(
25    state: &TableState,
26    error: &topcoat::Error,
27    path: &str,
28) -> String {
29    if crate::error::TabloError::is_cursor(error) {
30        state.without_cursor(path)
31    } else {
32        state.list_url(path)
33    }
34}

36/// Derives the action chrome a resource's declarations imply.
37pub(crate) fn declared_chrome<R: Resource>(cx: &Cx) -> TableChrome {
38    TableChrome {
39        delete: can::<R>(cx, Ability::DeleteAny),
40        edit: <R::Form as RecordForm>::HAS_FORM,
41        view: declared::<R>(cx).viewed(),
42        actions: true,
43    }
44}

46/// Wires `R`'s delete, bulk, edit, view, and custom actions onto the table; `live` selects the
47/// shard variant.
48pub(crate) fn wire_table_actions<R: Resource>(cx: &Cx, live: bool) -> Table<R::Model> {
49    wire_table::<R>(cx, live, declared_chrome::<R>(cx))
50}

52/// Wires `chrome`'s affordances onto `R`'s table.
53pub(crate) fn wire_table<R: Resource>(cx: &Cx, live: bool, chrome: TableChrome) -> Table<R::Model> {
54    let mut table = declared::<R>(cx).table.clone();
55    if live {
56        table = table.hide_search().hide_filter_bar().unframed();
57    }
58    let policy_cx = cx.clone();
59    table = table.row_actions(move |record| {
60        // Every route pairs its predicate with `View`.
61        let view = can::<R>(&policy_cx, Ability::View(record));
62        RowActions {
63            view,
64            edit: view && can::<R>(&policy_cx, Ability::Update(record)),
65            delete: view && can::<R>(&policy_cx, Ability::Delete(record)),
66        }
67    });
68    if chrome.delete {
69        table = table
70            .with_delete(list_url(cx, &R::slug()))
71            .with_bulk_delete(true);
72    }
73    if chrome.edit {
74        table = table.with_edit(list_url(cx, &R::slug()));
75    }
76    if chrome.view {
77        table = table.with_view(list_url(cx, &R::slug()));
78    }
79    if chrome.actions {
80        table = wire_custom_actions::<R>(cx, table);
81    }
82    table
83}

85/// Attaches `R`'s custom actions to `table`, each gated per record by `View` and the action's
86/// `can_run`.
87fn wire_custom_actions<R: Resource>(cx: &Cx, table: Table<R::Model>) -> Table<R::Model> {
88    let actions = R::actions();
89    if actions.entries().is_empty() {
90        return table;
91    }
92    let wired = actions
93        .entries()
94        .iter()
95        .map(|action| {
96            let can_run = action.can_run;
97            let policy_cx = cx.clone();
98            TableAction {
99                name: action.name,
100                label: (action.label)(),
101                row: action.row,
102                bulk: action.bulk,
103                allowed: std::sync::Arc::new(move |record: &R::Model| {
104                    can::<R>(&policy_cx, Ability::View(record)) && can_run(&policy_cx, record)
105                }),
106            }
107        })
108        .collect();
109    table.with_custom_actions(list_url(cx, &R::slug()), wired)
110}

112/// Renders the branded in-region table failure with a cursor-aware retry link.
113pub(crate) fn table_error_view<'a, R: Resource>(
114    cx: &'a Cx,
115    state: &TableState,
116    error: &topcoat::Error,
117    path: &str,
118    signals: Option<&TableSignals>,
119) -> BoxView<'a> {
120    tracing::error!(resource = R::slug(), error = %error, "table load failed");
121    let retry_url = retry_url_for_error(state, error, path);
122    let action: BoxView<'a> = match signals {
123        Some(signals) => {
124            // Keyed by the list, like the table's own signals.
125            let attempt = topcoat::runtime::signal(&cx.keyed(path), || 0u64);
126            let cursor_error = crate::error::TabloError::is_cursor(error);
127            let attrs = if cursor_error {
128                let query = signals.query.clone();
129                let next = crate::resource::query_of(&retry_url).to_string();
130                attributes! {
131                    cx =>
132                    href=(retry_url)
133                    data-retry-attempt=(attempt.get())
134                    @click=$(|e: Event| {
135                        e.prevent_default();
136                        query.set(next.clone());
137                        attempt.increment();
138                    })
139                }
140            } else {
141                attributes! {
142                    cx =>
143                    href=(retry_url)
144                    data-retry-attempt=(attempt.get())
145                    @click=$(|e: Event| {
146                        e.prevent_default();
147                        attempt.increment();
148                    })
149                }
150            };
151            view! { cx => <a (attrs)>"Retry"</a> }.boxed()
152        }
153        None => view! { cx => <a href=(retry_url)>"Retry"</a> }.boxed(),
154    };
155    view! {
156        cx =>
157        tablo_ui::error_state(
158            title: format!("Couldn't load {}", R::navigation_label()),
159            detail: "Something went wrong while loading the records.",
160            action: Some(action.into()),
161            attrs: attributes! { role="alert" }
162        )
163    }
164    .boxed()
165}

167/// Renders the list page header with the resource's title and its Create link.
168fn list_header<'a, R: Resource>(cx: &'a Cx, title: &str, list_path: &str) -> BoxView<'a> {
169    let title = title.to_string();
170    let create_url = (<R::Form as RecordForm>::HAS_FORM && can::<R>(cx, Ability::Create))
171        .then(|| create_page_url(list_path));
172    let create_label = format!("Create {}", R::label());
173    view! {
174        cx =>
175        tablo_ui::page_header(
176            tablo_ui::page_title((title))
177            if let Some(url) = create_url {
178                tablo_ui::page_actions(
179                    <a
180                        (crate::resource::runtime_link(cx, &url))
181                        class=(tablo_ui::button_variants(
182                            tablo_ui::ButtonVariant::Primary,
183                            tablo_ui::ButtonSize::Md,
184                        ))
185                    >
186                        icon(data: tablo_ui::icons::PLUS)
187                        (create_label)
188                    </a>
189                )
190            }
191        )
192    }
193    .boxed()
194}

196/// Serves the list page every declared [`Resource`] gets at `{prefix}/{slug}`.
197pub(crate) fn resource_list<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
198    Box::pin(HoistView::new(ThenView::new(async move {
199        gate::<R>(cx)?;
200        if !can::<R>(cx, Ability::ViewAny) {
201            return Err(forbidden().into());
202        }
203        crate::csrf::ensure_token(cx);
204        let state = TableState::from_cx(cx);
205        let table = wire_table_actions::<R>(cx, false);
206        let title = R::navigation_label();
207        let list_path = list_url(cx, &R::slug());
208        if table.is_live_search() {
209            return Ok(resource_list_live::<R>(cx, table, state, title, list_path));
210        }
211
212        // Normalizes once per request (GH #153).
213        let state = table.normalize_state(&state);
214        let skeleton = table.render_skeleton(cx, &state).await?;
215        let header = list_header::<R>(cx, &title, &list_path);
216        let lazy_rows = ThenView::new(async move {
217            let rendered = async {
218                let page = load_table_page::<R>(cx, &table, &state).await?;
219                table.render_with_state(cx, page, &state, &list_path).await
220            };
221            match rendered.await {
222                Ok(view) => Ok(view),
223                Err(error) => Ok(table_error_view::<R>(cx, &state, &error, &list_path, None)),
224            }
225        });
226
227        Ok(view! {
228            cx =>
229            tablo_ui::page(
230                (header)
231                tablo_ui::page_content(
232                    suspense(fallback: skeleton, (lazy_rows.boxed()))
233                )
234            )
235        }
236        .boxed())
237    })))
238}

240/// Serves the live list page for `Table::live_search` tables.
241pub(crate) fn resource_list_live<R: Resource>(
242    cx: &Cx,
243    table: Table<R::Model>,
244    state: TableState,
245    title: String,
246    list_path: String,
247) -> BoxView<'_> {
248    Box::pin(HoistView::new(ThenView::new(async move {
249        // Keyed by the list so one list's search never filters the next.
250        let signals = TableState::signals_for(&cx.keyed(list_path.as_str()), &request_query(cx));
251        // Normalizes once per request.
252        let state = table.normalize_state(&state);
253        let host = if table.search_enabled() {
254            Some(
255                table
256                    .render_live_search_bar(cx, &state, &list_path, &signals)
257                    .await?,
258            )
259        } else {
260            None
261        };
262        let filter_bar = if table.filter_bar_enabled() {
263            Some(
264                table
265                    .render_live_filter_bar(cx, &state, &list_path, &signals)
266                    .await?,
267            )
268        } else {
269            None
270        };
271        let table = table.hide_search().hide_filter_bar().unframed();
272        let skeleton = table.render_skeleton(cx, &state).await?;
273        let delete_dialog = table.render_delete_dialog(cx, &state).await?;
274        let header = list_header::<R>(cx, &title, &list_path);
275        let lazy_rows = ThenView::new(async move {
276            let retry_signals = signals.clone();
277            let rendered = table.render_live_invocation(cx, &list_path, signals).await;
278            match rendered {
279                Ok(view) => Ok(view),
280                Err(error) => Ok(table_error_view::<R>(
281                    cx,
282                    &state,
283                    &error,
284                    &list_path,
285                    Some(&retry_signals),
286                )),
287            }
288        });
289
290        Ok(view! {
291            cx =>
292            tablo_ui::page(
293                (header)
294                tablo_ui::page_content(
295                    <div class=(TABLE_CARD_CLASS)>
296                        if let Some(host) = host {
297                            (host)
298                        }
299                        if let Some(bar) = filter_bar {
300                            (bar)
301                        }
302                        suspense(fallback: skeleton, (lazy_rows.boxed()))
303                    </div>
304                    if let Some(dialog) = delete_dialog {
305                        (dialog)
306                    }
307                )
308            )
309        })
310    })))
311}

313/// Loads `R`'s list page for `state` from the tenant-scoped query.
314pub(crate) async fn load_table_page<R: Resource>(
315    cx: &Cx,
316    table: &Table<R::Model>,
317    state: &TableState,
318) -> Result<TablePage<R::Model>> {
319    TablePage::load(cx, table, crate::resource::scoped_query::<R>(cx)?, state).await
320}

322/// Loads [`load_table_page`] over the rows `scope` admits.
323pub(crate) async fn load_scoped_page<R: Resource>(
324    cx: &Cx,
325    table: &Table<R::Model>,
326    state: &TableState,
327    scope: toasty::stmt::Expr<bool>,
328) -> Result<TablePage<R::Model>> {
329    let query = crate::resource::scoped_query::<R>(cx)?.filter(scope);
330    TablePage::load(cx, table, query, state).await
331}
