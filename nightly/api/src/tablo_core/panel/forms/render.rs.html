22/// What a form page shows around its form: the create page and the edit page
23/// differ only here.
24pub(super) struct FormChrome<'a> {
25    title: String,
26    submit_label: &'static str,
27    /// The record's public page, linked from the header.
28    public_url: Option<String>,
29    /// The record's relation tables, below the form.
30    relations: Vec<BoxView<'a>>,
31}
32
33impl<'a> FormChrome<'a> {
34    /// The create page: no record yet, so no public link and no relations.
35    pub(super) fn create<R: Resource>() -> Self {
36        Self {
37            title: format!("Create {}", R::label()),
38            submit_label: "Create",
39            public_url: None,
40            relations: Vec::new(),
41        }
42    }
43
44    /// The edit page of `record`.
45    pub(super) fn edit<R: Resource>(cx: &'a Cx, record: &R::Model) -> Self {
46        Self {
47            title: format!("Edit {}", R::label()),
48            submit_label: "Save",
49            public_url: R::public_url(cx, record),
50            relations: render_relations::<R>(cx, record, false),
51        }
52    }
53}

55/// Shared create/edit page shell (GH #73 multipart enctype, CSRF hidden
56/// input, inline error slot), with the page's [`FormChrome`] around it.
57///
58/// `carried` names the upload fields whose value is an uploader's answer rather
59/// than the record's: the shell renders each one's path as a hidden
60/// `keep_<field>` control, so the submit a corrected form makes can keep a file
61/// the browser's empty file input cannot resend.
62pub(super) async fn render_form_page<'a, R: Resource>(
63    cx: &'a Cx,
64    chrome: FormChrome<'a>,
65    values: &HashMap<String, String>,
66    errors: &FieldErrors,
67    carried: &HashSet<String>,
68) -> Result<BoxView<'a>> {
69    let FormChrome {
70        title,
71        submit_label,
72        public_url,
73        relations,
74    } = chrome;
75    let declared = declared::<R>(cx);
76    let schema = &declared.form;
77    let form_html = schema
78        .render(cx, crate::schema::Source::form(values, errors))
79        .await?;
80    // The form posts where it was served, keeping a validated `?return=` so
81    // the write lands where the page was opened from, and Cancel goes there.
82    let path = topcoat::router::request::uri(cx).path();
83    let return_to = return_target(cx);
84    let action = match &return_to {
85        Some(target) => crate::resource::with_return(path, target),
86        None => path.to_string(),
87    };
88    let cancel = return_to.unwrap_or_else(|| list_url(cx, &R::slug()));
89    // Browsers only send `<input type="file">` content as multipart.
90    let enctype: Option<String> = schema
91        .fields()
92        .any(|field| field.is_file())
93        .then(|| "multipart/form-data".to_string());
94    let csrf = crate::csrf::current_token(cx);
95    // The candidate paths, one hidden control each: the framework re-verifies
96    // them against the installed store before it uses one.
97    let mut carried_fields: Vec<BoxView<'a>> = Vec::new();
98    let mut carried_names: Vec<&String> = carried.iter().collect();
99    carried_names.sort();
100    for name in carried_names {
101        let Some(path) = values.get(name) else {
102            continue;
103        };
104        let control = format!("keep_{name}");
105        let path = path.clone();
106        carried_fields
107            .push(view! { cx => <input type="hidden" name=(control) value=(path)> }.boxed());
108    }
109    Ok(view! {
110        cx =>
111        tablo_ui::page(
112            tablo_ui::page_header(
113                tablo_ui::page_title((title.clone()))
114                if let Some(public) = public_url {
115                    tablo_ui::page_actions(
116                        <a
117                            href=(public)
118                            class=(tablo_ui::button_variants(
119                                tablo_ui::ButtonVariant::Outline,
120                                tablo_ui::ButtonSize::Md,
121                            ))
122                        >
123                            icon(data: tablo_ui::icons::EXTERNAL_LINK)
124                            "View public post"
125                        </a>
126                    )
127                }
128            )
129            tablo_ui::page_content(
130                <form
131                    method="post"
132                    action=(action)
133                    enctype=(enctype)
134                    class="flex flex-col gap-4"
135                >
136                    (crate::csrf::field(cx, &csrf))
137                    for carried in carried_fields {
138                        (carried)
139                    }
140                    (form_html)
141                    <div class="flex items-center gap-2">
142                        tablo_ui::button(
143                            variant: tablo_ui::ButtonVariant::Primary,
144                            attrs: attributes! { type="submit" },
145                            (submit_label)
146                        )
147                        <a
148                            (crate::resource::runtime_link(cx, &cancel))
149                            class=(tablo_ui::button_variants(
150                                tablo_ui::ButtonVariant::Outline,
151                                tablo_ui::ButtonSize::Md,
152                            ))
153                        >
154                            "Cancel"
155                        </a>
156                    </div>
157                </form>
158                for relation in relations {
159                    (relation)
160                }
161            )
162        )
163    }
164    .boxed())
165}

167/// Create page GET.
168///
169/// A query parameter that names a relationship control seeds it
170/// (`?post_id=…`): a relation's create link opens the child's form with the
171/// owner already chosen. It is a prefill only — the POST parses and checks
172/// what is submitted, like any other value the user could have typed — and
173/// only a relationship is seeded, so a link cannot prefill free text or a
174/// stored file into the form an admin submits.
175pub(crate) fn resource_create<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
176    Box::pin(HoistView::new(ThenView::new(async move {
177        gate::<R>(cx)?;
178        if !R::can_create(cx) {
179            return Err(forbidden().into());
180        }
181        crate::csrf::ensure_token(cx);
182        let html = render_form_page::<R>(
183            cx,
184            FormChrome::create::<R>(),
185            &seeded_values::<R>(cx),
186            &FieldErrors::new(),
187            &HashSet::new(),
188        )
189        .await?;
190        Ok(html)
191    })))
192}

194/// The create form's initial values: the request's query parameters that
195/// name a relationship control of `R`'s form, first occurrence wins.
196fn seeded_values<R: Resource>(cx: &Cx) -> HashMap<String, String> {
197    let declared = declared::<R>(cx);
198    let schema = &declared.form;
199    let seedable: Vec<&str> = schema
200        .fields()
201        .filter(|field| {
202            field
203                .as_choice()
204                .is_some_and(|choice| choice.is_relationship())
205        })
206        .map(|field| field.name())
207        .collect();
208    let query = topcoat::router::request::uri(cx)
209        .query()
210        .unwrap_or_default();
211    let mut values = HashMap::new();
212    for (name, value) in form_urlencoded::parse(query.as_bytes()) {
213        if seedable.contains(&name.as_ref()) {
214            values
215                .entry(name.into_owned())
216                .or_insert_with(|| value.into_owned());
217        }
218    }
219    values
220}
