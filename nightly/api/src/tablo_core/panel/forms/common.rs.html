18/// A decoded form body: the text values plus any file parts.
19pub(crate) struct FormParts {
20    pub(crate) values: HashMap<String, String>,
21    /// File parts by field name, staged for the installed
22    /// [`Uploader`](crate::Uploader) — empty when none is installed, because
23    /// then the bytes could only be dropped and today's drain-and-discard
24    /// is what keeps a large upload off the heap.
25    pub(crate) files: HashMap<String, crate::upload::StagedUpload>,
26    /// Field names that arrived as a multipart part carrying a `filename`
27    /// (chosen or empty). Only these may set a file field's value: a
28    /// text part or a url-encoded pair under the same name is client-typed, not
29    /// an upload.
30    pub(crate) file_part_names: HashSet<String>,
31}
32
33/// Max form/multipart body accepted: 10 MiB. It bounds the whole
34/// multipart stream, file bytes included — whether they are discarded or
35/// buffered for an installed [`Uploader`](crate::Uploader).
36pub(crate) const MAX_FORM_BYTES: usize = 10 * 1024 * 1024;

38/// Reject POST keys no declared Schema input owns (GH #89 mass-assignment
39/// allow-list). `csrf_token` is a handler key, not a field, so it is filtered
40/// before the check, as are `clear_<field>` flags for declared file
41/// fields (explicit-clear convention — `truthy`); absent keys are fine (an
42/// edit completes them from the stored record), unknown keys are a 400 — accepting
43/// `role`/`tenant_id` smuggling would let a generic record fn iterating
44/// `values` promote them to client-controlled writes.
45pub(super) fn reject_unknown_form_keys(
46    schema: &crate::schema::Schema,
47    values: &HashMap<String, String>,
48) -> Result<(), topcoat::Error> {
49    // One transport-key vocabulary: the same `strip_transport_keys`
50    // the record fns benefit from defines which keys the framework owns, so
51    // the allow-list and the strip cannot drift apart.
52    let mut filtered = values.clone();
53    strip_transport_keys(schema, &mut filtered);
54    let unknown = schema.unknown_keys(&filtered);
55    if unknown.is_empty() {
56        Ok(())
57    } else {
58        Err(topcoat::router::error::bad_request(format!(
59            "unknown field(s): {}",
60            unknown.join(", ")
61        ))
62        .into())
63    }
64}

66/// The one boolean-vocabulary check for framework form flags:
67/// `confirm=1|true`, `clear_<field>=1|true`. One vocabulary, not a per-handler
68/// set.
69pub(crate) fn truthy(v: &str) -> bool {
70    v == "1" || v == "true"
71}
72
73/// Strip framework transport keys from the submitted values before any
74/// record fn sees them: `csrf_token`, the `clear_<field>` flags and
75/// the `keep_<field>` candidates a re-rendered form carries are
76/// handler keys, not writable fields — a generic `Resource` impl iterating
77/// `values` (the exact threat model in the `unknown_keys` docs) must not
78/// receive them as writes. The framework strips once here, not per-app
79/// convention.
80pub(super) fn strip_transport_keys(
81    schema: &crate::schema::Schema,
82    values: &mut HashMap<String, String>,
83) {
84    let declared: HashSet<&str> = schema.fields().map(crate::schema::Field::name).collect();
85    // A schema field literally named `csrf_token` (or `clear_<upload>`) is a
86    // misconfiguration that would silently swallow its own value here — the
87    // declared-name check keeps such a field's value flowing (the collision
88    // is a build-time bug, not a transport key).
89    values.retain(|k, _| {
90        if k == crate::csrf::FIELD_NAME {
91            return declared.contains(k.as_str());
92        }
93        let field = k.strip_prefix("clear_").or_else(|| k.strip_prefix("keep_"));
94        match field {
95            Some(field) if schema.fields().any(|f| f.is_file() && f.name() == field) => {
96                declared.contains(k.as_str())
97            }
98            _ => true,
99        }
100    });
101}

103/// Drop any value a declared file field received from something other than a
104/// file part. The field's value is the uploader's answer, the stored
105/// value (edit backfill), or empty (clear) — never text the client typed, which
106/// would reach the record and render as the file's link.
107pub(super) fn drop_client_typed_uploads(
108    schema: &crate::schema::Schema,
109    file_part_names: &HashSet<String>,
110    values: &mut HashMap<String, String>,
111) {
112    for field in schema.fields().filter(|field| field.is_file()) {
113        if !file_part_names.contains(field.name()) {
114            values.remove(field.name());
115        }
116    }
117}

119/// Re-use the upload a re-rendered form carried.
120///
121/// A re-rendered form posts each carried upload's path back under
122/// `keep_<field>`, because the browser's file input is empty on the next
123/// attempt. The candidate is used only when the installed uploader still holds
124/// the path ([`crate::upload::holds`]): a client-typed value is never stored,
125/// which is the GH #277 rule the carry must not re-open. Without an installed
126/// uploader nothing can vouch for a path, so nothing is restored.
127///
128/// A field that carried a file of its own in this submission, or one the user
129/// cleared, keeps its own answer. Returns the field names whose value is an
130/// upload, so the caller can carry them through another re-render.
131pub(super) async fn restore_pending_uploads(
132    cx: &Cx,
133    schema: &crate::schema::Schema,
134    values: &mut HashMap<String, String>,
135) -> HashSet<String> {
136    let mut restored = HashSet::new();
137    for field in schema.fields().filter(|field| field.is_file()) {
138        let name = field.name();
139        let empty = values
140            .get(name)
141            .map(|value| value.trim().is_empty())
142            .unwrap_or(true);
143        let cleared = values
144            .get(&format!("clear_{name}"))
145            .is_some_and(|value| truthy(value));
146        if !empty || cleared {
147            continue;
148        }
149        let Some(candidate) = values.get(&format!("keep_{name}")) else {
150            continue;
151        };
152        let candidate = candidate.trim().to_string();
153        if candidate.is_empty() || !crate::upload::holds(cx, &candidate).await {
154            continue;
155        }
156        values.insert(name.to_string(), candidate);
157        restored.insert(name.to_string());
158    }
159    restored
160}

162/// Shared create/edit POST error tail: re-render the form with inline
163/// errors. Takes the open framework transaction by value and drops it before
164/// rendering — the re-rendered form reloads relationship options on
165/// its own handle, which would block on the pool while the tx holds it —
166/// so the drop is enforced here rather than trusted at each call site.
167//
168// The public link rides through to the re-rendered form for the same reason
169// as above.
170#[allow(clippy::too_many_arguments)]
171pub(super) async fn rerender_invalid_form<'a, R: Resource>(
172    cx: &'a Cx,
173    tx: toasty::Transaction<'_>,
174    title: String,
175    submit_label: &'static str,
176    values: &HashMap<String, String>,
177    errors: &HashMap<String, Vec<String>>,
178    carried: &HashSet<String>,
179    public_url: Option<String>,
180) -> Result<BoxView<'a>> {
181    drop(tx);
182    render_form_page::<R>(cx, title, submit_label, values, errors, carried, public_url).await
183}

185/// Edit page GET — hydrates the form from the record the tenant-scoped
186/// load returned.
187pub(crate) fn resource_edit<R: Resource>(cx: &Cx, _body: Body) -> BoxView<'_> {
188    Box::pin(HoistView::new(ThenView::new(async move {
189        gate::<R>(cx)?;
190        let mut db = db(cx);
191        let record = load_viewable::<R>(cx, &mut db).await?;
192        if !R::can_update(cx, &record) {
193            return Err(forbidden().into());
194        }
195        crate::csrf::ensure_token(cx);
196        let values = <R::Form as RecordForm>::hydrate(cx, &record);
197        let public = R::public_url(cx, &record);
198        let html = render_form_page::<R>(
199            cx,
200            format!("Edit {}", R::label()),
201            "Save",
202            &values,
203            &HashMap::new(),
204            &HashSet::new(),
205            public,
206        )
207        .await?;
208        Ok(html)
209    })))
210}
