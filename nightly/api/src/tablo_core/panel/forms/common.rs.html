29/// A decoded form body: the text values plus any file parts.
30pub(crate) struct FormParts {
31    pub(crate) values: HashMap<String, String>,
32    /// File parts by field name, staged for the installed
33    /// [`Uploader`](crate::Uploader) — empty when none is installed, because
34    /// then the bytes could only be dropped and today's drain-and-discard
35    /// is what keeps a large upload off the heap.
36    pub(crate) files: HashMap<String, crate::upload::StagedUpload>,
37    /// Field names that arrived as a multipart part carrying a `filename`
38    /// (chosen or empty). Only these may set a `FileUpload` value: a
39    /// text part or a url-encoded pair under the same name is client-typed, not
40    /// an upload.
41    pub(crate) file_part_names: HashSet<String>,
42}
43
44/// Max form/multipart body accepted: 10 MiB. It bounds the whole
45/// multipart stream, file bytes included — whether they are discarded or
46/// buffered for an installed [`Uploader`](crate::Uploader).
47pub(crate) const MAX_FORM_BYTES: usize = 10 * 1024 * 1024;
48
49/// Reject POST keys no declared Schema input owns (GH #89 mass-assignment
50/// allow-list). `csrf_token` is a handler key, not a field, so it is filtered
51/// before the check, as are `clear_<field>` flags for declared `FileUpload`
52/// fields (explicit-clear convention — `truthy`); absent keys are fine (an
53/// edit completes them from the stored record), unknown keys are a 400 — accepting
54/// `role`/`tenant_id` smuggling would let a generic record fn iterating
55/// `values` promote them to client-controlled writes.
56pub(super) fn reject_unknown_form_keys(
57    schema: &crate::schema::Schema,
58    values: &HashMap<String, String>,
59) -> Result<(), topcoat::Error> {
60    // One transport-key vocabulary: the same `strip_transport_keys`
61    // the record fns benefit from defines which keys the framework owns, so
62    // the allow-list and the strip cannot drift apart.
63    let mut filtered = values.clone();
64    strip_transport_keys(schema, &mut filtered);
65    let unknown = schema.unknown_keys(&filtered);
66    if unknown.is_empty() {
67        Ok(())
68    } else {
69        Err(topcoat::router::error::bad_request(format!(
70            "unknown field(s): {}",
71            unknown.join(", ")
72        ))
73        .into())
74    }
75}

77/// The one boolean-vocabulary check for framework form flags:
78/// `confirm=1|true`, `clear_<field>=1|true`. One vocabulary, not a per-handler
79/// set.
80pub(crate) fn truthy(v: &str) -> bool {
81    v == "1" || v == "true"
82}
83
84/// Strip framework transport keys from the submitted values before any
85/// record fn sees them: `csrf_token`, the `clear_<field>` flags and
86/// the `keep_<field>` candidates a re-rendered form carries are
87/// handler keys, not writable fields — a generic `Resource` impl iterating
88/// `values` (the exact threat model in the `unknown_keys` docs) must not
89/// receive them as writes. The framework strips once here, not per-app
90/// convention.
91pub(super) fn strip_transport_keys(
92    schema: &crate::schema::Schema,
93    values: &mut HashMap<String, String>,
94) {
95    let declared: std::collections::HashSet<String> = schema.field_names().into_iter().collect();
96    // A schema field literally named `csrf_token` (or `clear_<upload>`) is a
97    // misconfiguration that would silently swallow its own value here — the
98    // declared-name check keeps such a field's value flowing (the collision
99    // is a build-time bug, not a transport key).
100    values.retain(|k, _| {
101        if k == crate::csrf::FIELD_NAME {
102            return declared.contains(k.as_str());
103        }
104        let field = k.strip_prefix("clear_").or_else(|| k.strip_prefix("keep_"));
105        match field {
106            Some(field) if schema.file_uploads().contains_key(field) => {
107                declared.contains(k.as_str())
108            }
109            _ => true,
110        }
111    });
112}

114/// Drop any value a declared `FileUpload` received from something other than a
115/// file part. The field's value is the uploader's answer, the stored
116/// value (edit backfill), or empty (clear) — never text the client typed, which
117/// would reach the record and render as the file's link.
118pub(super) fn drop_client_typed_uploads(
119    schema: &crate::schema::Schema,
120    file_part_names: &HashSet<String>,
121    values: &mut HashMap<String, String>,
122) {
123    for name in schema.file_uploads().keys() {
124        if !file_part_names.contains(name) {
125            values.remove(name);
126        }
127    }
128}

130/// Re-use the upload a re-rendered form carried.
131///
132/// A re-rendered form posts each carried upload's path back under
133/// `keep_<field>`, because the browser's file input is empty on the next
134/// attempt. The candidate is used only when the installed uploader still holds
135/// the path ([`crate::upload::holds`]): a client-typed value is never stored,
136/// which is the GH #277 rule the carry must not re-open. Without an installed
137/// uploader nothing can vouch for a path, so nothing is restored.
138///
139/// A field that carried a file of its own in this submission, or one the user
140/// cleared, keeps its own answer. Returns the field names whose value is an
141/// upload, so the caller can carry them through another re-render.
142pub(super) async fn restore_pending_uploads(
143    cx: &Cx,
144    schema: &crate::schema::Schema,
145    values: &mut HashMap<String, String>,
146) -> HashSet<String> {
147    let mut restored = HashSet::new();
148    for name in schema.file_uploads().keys() {
149        let empty = values
150            .get(name)
151            .map(|value| value.trim().is_empty())
152            .unwrap_or(true);
153        let cleared = values
154            .get(&format!("clear_{name}"))
155            .is_some_and(|value| truthy(value));
156        if !empty || cleared {
157            continue;
158        }
159        let Some(candidate) = values.get(&format!("keep_{name}")) else {
160            continue;
161        };
162        let candidate = candidate.trim().to_string();
163        if candidate.is_empty() || !crate::upload::holds(cx, &candidate).await {
164            continue;
165        }
166        values.insert(name.clone(), candidate);
167        restored.insert(name.clone());
168    }
169    restored
170}

172/// Shared create/edit POST error tail: re-render the form with inline
173/// errors. Takes the open framework transaction by value and drops it before
174/// rendering — the re-rendered form reloads relationship options on
175/// its own handle, which would block on the pool while the tx holds it —
176/// so the drop is enforced here rather than trusted at each call site.
177//
178// The public link rides through to the re-rendered form for the same reason
179// as above.
180#[allow(clippy::too_many_arguments)]
181pub(super) async fn rerender_invalid_form<'a, R: FormResource>(
182    cx: &'a Cx,
183    tx: toasty::Transaction<'_>,
184    title: String,
185    submit_label: &'static str,
186    values: &HashMap<String, String>,
187    errors: &HashMap<String, Vec<String>>,
188    carried: &HashSet<String>,
189    public_url: Option<String>,
190) -> Result<BoxView<'a>> {
191    drop(tx);
192    render_form_page::<R>(cx, title, submit_label, values, errors, carried, public_url).await
193}

195/// Shared create/edit POST success tail: Post/Redirect/Get with a
196/// flash notification. The browser follows with a GET, and the flash cookie
197/// rides the error response (Topcoat flushes `Set-Cookie` on `Err` too,
198/// topcoat#408). Redirect target and notification are the caller's only
199/// deltas, so the redirect behavior cannot drift between create and edit.
200pub(super) fn redirect_after_write<R: Resource>(cx: &Cx, note: &'static str) -> topcoat::Error {
201    set_notification(cx, Notification::success(note));
202    see_other(list_url(cx, &R::slug())).into()
203}

205/// Edit page GET — hydrates the form from the record the tenant-scoped
206/// load returned.
207pub(crate) fn resource_edit<R: FormResource>(cx: &Cx, _body: Body) -> BoxView<'_> {
208    Box::pin(HoistView::new(ThenView::new(async move {
209        gate::<R>(cx)?;
210        let mut db = db(cx);
211        let record = load_viewable_narrowed::<R>(cx, &mut db).await?;
212        if !R::can_update(cx, &record) {
213            return Err(forbidden().into());
214        }
215        crate::csrf::ensure_token(cx);
216        let values = <R::Form as RecordForm>::hydrate(cx, &record);
217        let public = R::public_url(cx, &record);
218        let html = render_form_page::<R>(
219            cx,
220            format!("Edit {}", R::navigation_label()),
221            "Save",
222            &values,
223            &HashMap::new(),
224            &HashSet::new(),
225            public,
226        )
227        .await?;
228        Ok(html)
229    })))
230}

232mod tests {

235    /// One boolean vocabulary for framework form flags: `1` and
236    /// `true` are truthy everywhere (`confirm`, `clear_<field>`); `yes` was a
237    /// delete-only extra and is gone.
238    #[test]
239    fn truthy_accepts_one_vocabulary() {
240        assert!(truthy("1") && truthy("true"));
241        assert!(!truthy("yes") && !truthy("") && !truthy("on") && !truthy("TRUE"));
242    }
243
244    #[test]
245    fn reject_unknown_form_keys_allows_declared_plus_csrf() {
246        use crate::schema::{Schema, TextInput};
247
248        #[derive(Debug, toasty::Model)]
249        struct Member {
250            #[key]
251            #[auto]
252            id: uuid::Uuid,
253            name: String,
254        }
255        let schema = Schema::new(TextInput::r#for(Member::fields().name()));
256
257        // Declared keys + csrf_token pass.
258        let values = HashMap::from([
259            ("name".to_string(), "Ada".to_string()),
260            (
261                crate::csrf::FIELD_NAME.to_string(),
262                "some-token".to_string(),
263            ),
264        ]);
265        assert!(reject_unknown_form_keys(&schema, &values).is_ok());
266
267        // Absent keys are fine: an edit completes them from the stored record.
268        let values = HashMap::from([(
269            crate::csrf::FIELD_NAME.to_string(),
270            "some-token".to_string(),
271        )]);
272        assert!(reject_unknown_form_keys(&schema, &values).is_ok());
273
274        // role/tenant_id smuggling is a 400.
275        let values = HashMap::from([
276            ("name".to_string(), "Ada".to_string()),
277            ("role".to_string(), "admin".to_string()),
278            ("tenant_id".to_string(), "victim".to_string()),
279        ]);
280        assert!(reject_unknown_form_keys(&schema, &values).is_err());
281    }
