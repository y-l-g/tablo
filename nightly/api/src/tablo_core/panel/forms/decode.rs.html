17/// Helper: parse form bodies into a [`FormParts`] — `application/x-www-form-urlencoded`
18/// buffered, plus `multipart/form-data` streamed when a `FileUpload` is present.
19/// urlencoded decoding is delegated to `form_urlencoded` (already in the tree
20/// via topcoat): it splits pairs, decodes `+` as space, assembles multi-byte
21/// UTF-8 from `%XX` sequences (`%C3%A9` → `é`, not `Ã©`), and keeps encoded
22/// separators (`%26` → `&`) intact. Invalid UTF-8 degrades per-value (lossy)
23/// instead of discarding the whole form.
24///
25/// Multipart (file) parts stream through Topcoat's multer-based extractor.
26/// With no installed uploader the bytes are drained in chunks and
27/// discarded while the sanitized filename becomes the `String` value; with one
28/// installed they are buffered up to the same body cap and handed to it.
29/// Text parts store their content, and unknown content types fall
30/// back to urlencoded.
31pub(crate) async fn parse_form_body(cx: &Cx, body: Body) -> Result<FormParts, topcoat::Error> {
32    let content_type =
33        topcoat::context::try_request_context::<http::request::Parts>(cx).and_then(|parts| {
34            parts
35                .headers
36                .get(http::header::CONTENT_TYPE)
37                .and_then(|v| v.to_str().ok().map(|s| s.to_string()))
38        });
39    if content_type
40        .as_deref()
41        .is_some_and(is_multipart_content_type)
42    {
43        // Stage bytes only when something will consume them: the
44        // parser is the one place that knows whether an uploader exists.
45        return parse_multipart_values(cx, body, crate::upload::installed(cx)).await;
46    }
47    let bytes = Bytes::from_request(cx, body).await.map_err(|error| {
48        // A body over the route's limit is the extractor's 413, not a malformed
49        // form; any other read failure is the 400. The multipart half
50        // propagates the same over-limit error untouched.
51        if error.is::<topcoat::router::error::ContentTooLargeError>() {
52            error
53        } else {
54            topcoat::router::error::bad_request("cannot read form body").into()
55        }
56    })?;
57    Ok(FormParts {
58        values: form_values_from_request_parts(content_type.as_deref(), bytes.as_ref())?,
59        files: HashMap::new(),
60        file_part_names: HashSet::new(),
61    })
62}

64/// Whether a content type is `multipart/form-data` (parameters ignored).
65fn is_multipart_content_type(ct: &str) -> bool {
66    ct.split(';')
67        .next()
68        .is_some_and(|mime| mime.trim().eq_ignore_ascii_case("multipart/form-data"))
69}

71/// Streamed multipart half of [`parse_form_body`].
72///
73/// Fields stream one at a time with constant memory: text fields buffer
74/// (bounded by the request body limit), file fields drain-and-discard while
75/// only the sanitized filename is kept — or, when `capture` is set because an
76/// uploader is installed, buffer up to the same cap so it can store them.
77/// Duplicate part names are last-wins; nameless parts are skipped. A
78/// missing boundary is a 400, an over-limit body a 413 — both classified by the
79/// extractor, never silent fallbacks. Every byte the stream carries is also
80/// counted against [`MAX_FORM_BYTES`]: file reads and skipped parts
81/// go through the counter chunk by chunk, text fields join it after their
82/// (extractor-bounded) read, so a large upload cannot be read chunk-by-chunk
83/// holding the handler even if the extractor's limit stops wrapping the stream.
84async fn parse_multipart_values(
85    cx: &Cx,
86    body: Body,
87    capture: bool,
88) -> Result<FormParts, topcoat::Error> {
89    use topcoat::router::{content::multipart::Multipart, request::FromRequest};
90
91    let mut out = FormParts {
92        values: HashMap::new(),
93        files: HashMap::new(),
94        file_part_names: HashSet::new(),
95    };
96    let mut bytes_seen = 0usize;
97    let mut multipart = Multipart::from_request(cx, body).await?;
98    while let Some(mut field) = multipart.next_field().await? {
99        let Some(name) = field.name().map(str::to_string) else {
100            // Nameless parts carry bytes too: drain them through the counter
101            // so the accounting covers the whole request stream.
102            read_bounded(&mut field, &mut bytes_seen, None).await?;
103            continue;
104        };
105        if name.is_empty() {
106            read_bounded(&mut field, &mut bytes_seen, None).await?;
107            continue;
108        }
109        // RFC 6266: `filename*=` (decoded) takes precedence over `filename=`.
110        // Multer surfaces the plain `filename=` first, so the raw header is
111        // read for `filename*=` before falling back.
112        let filename =
113            filename_star_from_headers(&field).or_else(|| field.file_name().map(str::to_string));
114        match filename {
115            Some(f) if !f.is_empty() => {
116                let sanitized = sanitize_filename(&f);
117                // Duplicate part names are last-write-wins, bytes included: a
118                // later part replaces whatever an earlier one under the same
119                // name staged, so a name that sanitizes to empty cannot leave
120                // the earlier part's bytes behind.
121                out.files.remove(&name);
122                // Bytes are staged only for a name the framework would persist
123                // (a rejected name sanitizes to empty) and only when
124                // an uploader is installed to store them. Otherwise
125                // drain to advance the stream.
126                if capture && !sanitized.is_empty() {
127                    let mut bytes = Vec::new();
128                    read_bounded(&mut field, &mut bytes_seen, Some(&mut bytes)).await?;
129                    out.files.insert(
130                        name.clone(),
131                        crate::upload::StagedUpload {
132                            filename: sanitized.clone(),
133                            bytes,
134                        },
135                    );
136                } else {
137                    read_bounded(&mut field, &mut bytes_seen, None).await?;
138                }
139                // A chosen file is the one thing that may set a `FileUpload`
140                // value.
141                out.file_part_names.insert(name.clone());
142                out.values.insert(name, sanitized);
143            }
144            Some(_) => {
145                // Empty filename (no file chosen) → empty value so `required`
146                // validation fires instead of treating it as missing. It is
147                // still a file part: the browser submits every file
148                // input, and "keep" on edit must not read as a forged text
149                // value. It chose no file, so it discards bytes an earlier part
150                // staged under the same name.
151                read_bounded(&mut field, &mut bytes_seen, None).await?;
152                out.files.remove(&name);
153                out.file_part_names.insert(name.clone());
154                out.values.insert(name, String::new());
155            }
156            None => {
157                // Text fields buffer (extractor-bounded); the read joins the
158                // same counter so the backstop sees the per-request total.
159                let text = field.text().await?;
160                count_form_bytes(&mut bytes_seen, text.len())?;
161                // A later text part under a name an earlier file part used
162                // takes the name out of the file-part set, so the drop removes
163                // the client-typed value instead of storing it.
164                out.file_part_names.remove(&name);
165                out.files.remove(&name);
166                out.values.insert(name, text);
167            }
168        }
169    }
170    Ok(out)
171}

173/// Read one multipart field chunk-by-chunk, accounting every byte against
174/// [`MAX_FORM_BYTES`]: enforcement normally happens in the extractor
175/// (`BodyLimit` wraps the multipart stream), but the reader owns its own
176/// counter so an over-cap upload 413s here too instead of holding the handler.
177///
178/// `keep` decides the destination, not the accounting: `None` drains and
179/// discards (the default path, constant memory), `Some(sink)` buffers for an
180/// installed [`Uploader`](crate::Uploader). One loop, so the two
181/// paths cannot disagree about the cap — and the buffered bytes are bounded by
182/// that same cap, so installing an uploader trades the discard for at most one
183/// body's worth of heap rather than widening the contract.
184async fn read_bounded(
185    field: &mut topcoat::router::content::multipart::Field<'_>,
186    bytes_seen: &mut usize,
187    keep: Option<&mut Vec<u8>>,
188) -> Result<(), topcoat::Error> {
189    let mut keep = keep;
190    while let Some(chunk) = field.chunk().await? {
191        count_form_bytes(bytes_seen, chunk.len())?;
192        if let Some(sink) = keep.as_deref_mut() {
193            sink.extend_from_slice(&chunk);
194        }
195    }
196    Ok(())
197}

199/// Account one drained chunk against the form-body cap. Extracted
200/// so the 413 mapping is testable at the boundary without building a
201/// multipart body — through the router the extractor's own limit classifies
202/// the same body first, so the counter only answers when that limit stops
203/// applying (defense-in-depth alongside `BodyLimit`, never a competing cap).
204fn count_form_bytes(bytes_seen: &mut usize, chunk_len: usize) -> Result<(), topcoat::Error> {
205    // checked_add: the accumulator cannot overflow a usize at real chunk
206    // sizes, but wrapping would silently disable the cap in release builds.
207    let Some(total) = bytes_seen.checked_add(chunk_len) else {
208        return Err(topcoat::router::error::content_too_large().into());
209    };
210    *bytes_seen = total;
211    if *bytes_seen > MAX_FORM_BYTES {
212        return Err(topcoat::router::error::content_too_large().into());
213    }
214    Ok(())
215}

217/// RFC 5987 `filename*=` from a field's raw `Content-Disposition` header.
218fn filename_star_from_headers(
219    field: &topcoat::router::content::multipart::Field<'_>,
220) -> Option<String> {
221    let raw = field
222        .headers()
223        .get(http::header::CONTENT_DISPOSITION)?
224        .to_str()
225        .ok()?;
226    raw.split(';').find_map(|seg| {
227        let seg = seg.trim();
228        seg.get(..10)
229            .filter(|h| h.eq_ignore_ascii_case("filename*="))
230            .and_then(|_| decode_rfc5987(seg[10..].trim()))
231    })
232}

234/// Pure urlencoded half of [`parse_form_body`] — testable without
235/// a request. Rejects bodies over `MAX_FORM_BYTES` with 413. Multipart
236/// never reaches here: it streams via [`parse_multipart_values`], where a
237/// missing boundary is a 400 and an over-limit body a 413 (both classified
238/// by the extractor).
239///
240/// The length check is a deliberate second layer: through the router
241/// `Bytes::from_request` buffers via `to_bytes(body, body_limit(cx))`, which
242/// enforces the `BodyLimit::max(MAX_FORM_BYTES)` layer
243/// [`Panel::build`](crate::panel::Panel::build) installs, so this branch is
244/// unreachable there. It is the urlencoded symmetric backstop to the multipart
245/// [`count_form_bytes`] counter, and the only pin of the 10 MiB
246/// urlencoded contract at unit level — a bare `CxTestBuilder` carries no
247/// `BodyLimitKind`, so `body_limit(cx)` falls back to Topcoat's 2 MiB default
248/// and cannot pin this cap. Do not collapse into [`form_values_from_bytes`]
249/// without restoring both properties.
250fn form_values_from_request_parts(
251    content_type: Option<&str>,
252    bytes: &[u8],
253) -> Result<HashMap<String, String>, topcoat::Error> {
254    if bytes.len() > MAX_FORM_BYTES {
255        return Err(topcoat::router::error::content_too_large().into());
256    }
257    debug_assert!(
258        content_type.is_none_or(|ct| !is_multipart_content_type(ct)),
259        "multipart must stream via parse_multipart_values, not buffer here (GH #90)"
260    );
261    Ok(form_values_from_bytes(bytes))
262}

264/// Sanitize a client-supplied filename to a basename.
265///
266/// Strips directory components (`../../etc/passwd` → `passwd`,
267/// `/abs/path` → `path`, `C:\fakepath\x` → `x`), trims whitespace, drops
268/// control chars, and caps length at 255 bytes. Empty stays empty so
269/// `required` validation fires.
270///
271/// Names that could never be a safely persisted file are rejected to empty
272/// (before persistence lands): `.` and `..`, and Windows reserved
273/// device names (`con`, `nul`, `com1` — also with an extension, and
274/// case-insensitive). v1 stores only the basename `String` and never touches
275/// the filesystem, so today this is latent; the required validation then
276/// surfaces the empty value as an inline form error (on edit, the
277/// untouched-file backfill preserves the stored value instead — an
278/// explicitly rejected name falls back to "keep").
279fn sanitize_filename(raw: &str) -> String {
280    let base = raw.rsplit(['/', '\\']).next().unwrap_or(raw).trim();
281    let clean: String = base.chars().filter(|c| !c.is_control()).collect();
282    let trimmed = clean.trim();
283    if trimmed.is_empty() {
284        return String::new();
285    }
286    if trimmed == "." || trimmed == ".." || is_windows_reserved_name(trimmed) {
287        return String::new();
288    }
289    // Cap at 255 bytes (common filename limit), preserving the tail. The cut
290    // point is walked forward to a char boundary: slicing a multibyte char
291    // would panic (a >255-byte non-ASCII filename is attacker-controlled).
292    if trimmed.len() > 255 {
293        let mut start = trimmed.len() - 255;
294        while !trimmed.is_char_boundary(start) {
295            start += 1;
296        }
297        trimmed[start..].to_string()
298    } else {
299        trimmed.to_string()
300    }
301}

303/// Windows reserved device names: the stem before the first dot is
304/// reserved case-insensitively — `con`, `nul`, `aux`, `prn`, `com1`–`com9`,
305/// `lpt1`–`lpt9` — so `con.txt` cannot become a persisted basename either.
306fn is_windows_reserved_name(name: &str) -> bool {
307    let stem = match name.split_once('.') {
308        Some((stem, _)) => stem,
309        None => name,
310    };
311    let stem = stem.to_ascii_uppercase();
312    if matches!(stem.as_str(), "CON" | "PRN" | "AUX" | "NUL") {
313        return true;
314    }
315    let Some(n) = stem
316        .strip_prefix("COM")
317        .or_else(|| stem.strip_prefix("LPT"))
318    else {
319        return false;
320    };
321    n.parse::<u8>().is_ok_and(|n| (1..=9).contains(&n))
322}

324/// Decode an RFC 5987/6266 `filename*=UTF-8''...` value.
325///
326/// Three rules fail the whole value to `None`, so the caller falls back to
327/// `filename=` instead of storing a mangled name: the charset is not UTF-8, a
328/// `%` does not start a `pct-encoded` triplet (two hex digits), or the decoded
329/// bytes are not valid UTF-8.
330fn decode_rfc5987(value: &str) -> Option<String> {
331    let (charset, rest) = value.split_once('\'')?;
332    let (_lang, encoded) = rest.split_once('\'')?;
333    if !charset.eq_ignore_ascii_case("utf-8") {
334        return None;
335    }
336    if !is_pct_encoded(encoded) {
337        return None;
338    }
339    percent_decode_str(encoded)
340        .decode_utf8()
341        .ok()
342        .map(|decoded| decoded.into_owned())
343}

345/// Whether every `%` in `value` starts an RFC 5987 `pct-encoded` triplet (`%`
346/// followed by two hex digits).
347fn is_pct_encoded(value: &str) -> bool {
348    let bytes = value.as_bytes();
349    let mut i = 0;
350    while i < bytes.len() {
351        if bytes[i] == b'%' {
352            if !bytes
353                .get(i + 1..i + 3)
354                .is_some_and(|hex| hex.iter().all(u8::is_ascii_hexdigit))
355            {
356                return false;
357            }
358            i += 3;
359        } else {
360            i += 1;
361        }
362    }
363    true
364}

560    async fn multipart_stages_file_bytes_only_when_capturing() {
561        let boundary = "----CaptureBoundary";
562        let body = format!(
563            "--{b}\r\nContent-Disposition: form-data; name=\"title\"\r\n\r\nHello\r\n\
564             --{b}\r\nContent-Disposition: form-data; name=\"image_path\"; filename=\"photo.jpg\"\r\nContent-Type: image/jpeg\r\n\r\nBINARYBYTES\r\n\
565             --{b}--\r\n",
566            b = boundary
567        );
568
569        // Capturing: the part's bytes and sanitized name reach the uploader's
570        // staging area, and the value keeps the basename until an uploader
571        // replaces it (the handler's step).
572        let captured = multipart_parts(&multipart_type(boundary), body.clone().into_bytes(), true)
573            .await
574            .unwrap();
575        let staged = captured.files.get("image_path").expect("staged file part");
576        assert_eq!(staged.filename, "photo.jpg");
577        assert_eq!(staged.bytes, b"BINARYBYTES");
578        assert_eq!(
579            captured.values.get("image_path").map(String::as_str),
580            Some("photo.jpg")
581        );
582        assert!(
583            !captured.files.contains_key("title"),
584            "a text part is not a file part, got {:?}",
585            captured.files.keys()
586        );
587
588        // Not capturing: same values, no bytes held.
589        let drained = multipart_parts(&multipart_type(boundary), body.into_bytes(), false)
590            .await
591            .unwrap();
592        assert!(drained.files.is_empty(), "no uploader, no buffering");
593        assert_eq!(
594            drained.values.get("image_path").map(String::as_str),
595            Some("photo.jpg")
596        );
597    }

613        // Traversal filename lands sanitized.

623            .await
624            .unwrap();

626            got.get("image_path").map(String::as_str),
627            Some("€photo.jpg"),
628            "filename*=UTF-8 must decode and win, got {got:?}"
629        );

631        let body = "--B\r\nContent-Disposition: form-data; name=\"image_path\"; filename=\"plain.jpg\"; filename*=latin-1''%E9.jpg\r\nContent-Type: image/jpeg\r\n\r\nBYTES\r\n--B--\r\n";
632        let got = multipart_values(&multipart_type("B"), body.as_bytes().to_vec())

675    #[test]
676    fn filenames_sanitize_to_basename_and_dispatch_guards_size() {
677        assert_eq!(sanitize_filename("upload.jpg"), "upload.jpg");
678        assert_eq!(sanitize_filename("../../../etc/cron.d/x"), "x");
679        assert_eq!(sanitize_filename("/abs/path"), "path");
680        assert_eq!(sanitize_filename("C:\\fakepath\\x"), "x");
681        assert_eq!(sanitize_filename(""), "");
682        // Names that could never be a safe persisted file are rejected to
683        // empty: dot/dot-dot, and Windows reserved device names —
684        // case-insensitively and with an extension too.
685        assert_eq!(sanitize_filename("."), "");
686        assert_eq!(sanitize_filename(".."), "");
687        assert_eq!(sanitize_filename("../.."), "");
688        assert_eq!(sanitize_filename("..."), "...");
689        assert_eq!(sanitize_filename("con"), "");
690        assert_eq!(sanitize_filename("NUL"), "");
691        assert_eq!(sanitize_filename("Com1.txt"), "");
692        assert_eq!(sanitize_filename("lpt9"), "");
693        assert_eq!(sanitize_filename("console.txt"), "console.txt");
694        assert_eq!(sanitize_filename("companion"), "companion");
695        assert_eq!(sanitize_filename("...."), "....");
696        // Cap keeps the tail without splitting a multibyte char: a naive
697        // `[len - 255..]` slice panics here (the cut lands inside `é`).
698        let multibyte = format!("{}{}", "é".repeat(200), "a".repeat(200));
699        let capped = sanitize_filename(&multibyte);
700        assert!(
701            capped.len() <= 255,
702            "cap must bound bytes, got {}",
703            capped.len()
704        );
705        assert!(
706            capped.ends_with('a'),
707            "tail must be preserved, got {capped:?}"
708        );
709        // Over-cap body is rejected before buffering into maps.
710        let big = vec![b'a'; MAX_FORM_BYTES + 1];
711        assert!(
712            form_values_from_request_parts(Some("application/x-www-form-urlencoded"), &big)
713                .is_err()
714        );
715        // Normal urlencoded still parses.
716        let ok =
717            form_values_from_request_parts(Some("application/x-www-form-urlencoded"), b"name=Ada")
718                .unwrap();
719        assert_eq!(ok.get("name").map(String::as_str), Some("Ada"));
720    }
