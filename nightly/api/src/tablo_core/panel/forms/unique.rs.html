9/// App-side uniqueness check over the form's `unique()`-marked text inputs.
10///
11/// Generic over every marked field. Queries through the tenant-scoped query and
12/// returns `field_name → ["<Label> has already been taken"]` per duplicated
13/// value. `current` holds the record's own hydrated values on edit: a field
14/// whose submitted value normalises to the same stored value belongs to this
15/// record and is skipped, so a typed field's re-spelled equivalent is not a
16/// duplicate.
17///
18/// Empty submits are never probed: a `unique()` field is required (see
19/// [`crate::schema::TextInput::unique`]), so `validate` has already answered
20/// `"<Label> is required"` and this check has nothing left to say.
21///
22/// The probe binds the leaf's own type: a typed field parses the submission and
23/// compares the parsed value, so a value unique as text but not as its declared
24/// type is still refused.
25///
26/// Known limits (upstream gap #117): races with concurrent inserts, and a
27/// `unique()` field whose index carries components outside the tenant-scoped
28/// query's scope is not checked exactly — a composite index such as
29/// `#[unique(tenant_id, email)]` on a tenant-scoped resource is. `unique`
30/// exists on `TextInput` only.
31pub(super) async fn check_unique<R: Resource>(
32    cx: &Cx,
33    schema: &crate::schema::Schema,
34    values: &HashMap<String, String>,
35    current: &HashMap<String, String>,
36    ex: &mut dyn toasty::Executor,
37) -> Result<HashMap<String, Vec<String>>, topcoat::Error> {
38    let mut errors: HashMap<String, Vec<String>> = HashMap::new();
39    // Groups the submission leaves out are not checked:
40    // `validate` treats an all-empty repeater group and a hidden variant group
41    // as untouched through the same classification, so a stored value must not
42    // flag a group the user never saw.
43    let skip = schema.absent_fields(values);
44    for (name, input) in schema.text_inputs() {
45        if !input.is_unique() || skip.contains(&name) {
46            continue;
47        }
48        let Some(submitted) = values.get(&name).map(|s| s.trim().to_string()) else {
49            continue;
50        };
51        // Empty values are never probed: a `unique` field is
52        // required, so validation has already refused this submit — and `""` is
53        // still a value the framework stores (never NULL), so a probe
54        // would only rediscover the constraint the form just enforced.
55        if submitted.is_empty() {
56            continue;
57        }
58        // Unchanged on edit → this record's own value, not a duplicate. Both
59        // sides normalise through the leaf's own rule: a typed
60        // field's re-spelled equivalent — `01` for `1`, an upper-case UUID for
61        // its lower-case form — is the same value, so the probe is skipped. A
62        // text comparison would call it changed, probe this record's own row
63        // and refuse the save.
64        let unchanged = current.get(&name).is_some_and(|kept| {
65            matches!(
66                (input.normalize(kept), input.normalize(&submitted)),
67                (Ok(kept), Ok(submitted)) if kept == submitted
68            )
69        });
70        if unchanged {
71            continue;
72        }
73        // The leaf's own binding: a typed field parses the
74        // submission first, so the probe compares the value the record will
75        // store rather than its spelling. A typed submission that does not
76        // parse has no value to compare — validation refused it first.
77        let Some(filter) = input.eq_filter::<R::Model>(&submitted) else {
78            continue;
79        };
80        // Inside the handler's tx: the check observes the same
81        // snapshot as the write that follows. A failing probe fails the
82        // submit — swallowing it would write past a check that
83        // never ran. The probe runs through the tenant-scoped query and reads
84        // only the record's own columns, so it passes an empty include set.
85        let rows =
86            crate::resource::scoped_query_with::<R>(cx, &crate::resource::IncludeNeeds::default())?
87                .filter(filter)
88                .limit(1)
89                .exec(&mut *ex)
90                .await
91                .map_err(crate::db::unavailable)?;
92        if !rows.is_empty() {
93            errors.insert(
94                name,
95                vec![format!("{} has already been taken", input.label_str())],
96            );
97        }
98    }
99    Ok(errors)
100}

462    /// the app-side unique probe binds the leaf's declared type. The
463    /// stored token's canonical spelling is lower case, so an upper-case
464    /// submission is a different string and the same `Uuid`: a text comparison
465    /// finds no duplicate — and on this non-text column it cannot run at all —
466    /// while the typed comparison refuses the submit.
467    #[tokio::test]
468    async fn a_typed_unique_field_probes_the_declared_type() {
469        const TOKEN: &str = "0f8fad5b-d9cb-469f-a165-70867728950e";
470
471        let db = Db::builder()
472            .models(toasty::models!(Tagged))
473            .connect("sqlite::memory:")
474            .await
475            .unwrap();
476        db.push_schema().await.unwrap();
477        let mut db_q = db.clone();
478        toasty::create!(Tagged {
479            name: "one".to_string(),
480            token: uuid::Uuid::parse_str(TOKEN).unwrap(),
481        })
482        .exec(&mut db_q)
483        .await
484        .unwrap();
485        let router = form_panel_for::<TaggedResource>(db.clone())
486            .build()
487            .expect("panel builds");
488
489        let csrf = uuid::Uuid::new_v4().to_string();
490        let request = |body: String| {
491            http::Request::builder()
492                .method(http::Method::POST)
493                .uri("/admin/tagged/create")
494                .header(
495                    http::header::CONTENT_TYPE,
496                    "application/x-www-form-urlencoded",
497                )
498                .header(
499                    http::header::COOKIE,
500                    format!("{}={csrf}", crate::csrf::COOKIE_NAME),
501                )
502                .body(Body::from(body))
503                .unwrap()
504        };
505
506        // The upper-case spelling is not the stored one, so a text probe sees
507        // no duplicate; the typed probe sees the same `Uuid`.
508        let response = router
509            .handle(request(format!(
510                "name=two&token={}&csrf_token={csrf}",
511                TOKEN.to_uppercase()
512            )))
513            .await;
514        assert_eq!(
515            response.status(),
516            200,
517            "the duplicate must re-render, not create"
518        );
519        let html = response_html(response).await;
520        assert!(
521            html.contains("Token has already been taken"),
522            "the typed probe must see the duplicate, got {html}"
523        );
524        let mut db_q = db.clone();
525        assert_eq!(
526            Tagged::all().exec(&mut db_q).await.unwrap().len(),
527            1,
528            "a refused create writes nothing"
529        );
530
531        // The other direction: a genuinely different token still creates.
532        let response = router
533            .handle(request(format!(
534                "name=two&token=3f8fad5b-d9cb-469f-a165-70867728950e&csrf_token={csrf}"
535            )))
536            .await;
537        assert_eq!(
538            response.status(),
539            303,
540            "a distinct value must create, not flag a duplicate"
541        );
542        let mut db_q = db.clone();
543        assert_eq!(
544            Tagged::all().exec(&mut db_q).await.unwrap().len(),
545            2,
546            "the accepted create writes its row"
547        );
548    }

550    /// the edit exclusion normalises both sides through the leaf's own
551    /// rule, so a re-spelled equivalent of the record's own value is that value
552    /// and the save succeeds; another record's value still refuses.
553    #[tokio::test]
554    async fn a_typed_unique_field_skips_the_records_own_value_on_edit() {
555        const MINE: &str = "0f8fad5b-d9cb-469f-a165-70867728950e";
556        const THEIRS: &str = "3f8fad5b-d9cb-469f-a165-70867728950e";
557
558        let db = Db::builder()
559            .models(toasty::models!(Tagged))
560            .connect("sqlite::memory:")
561            .await
562            .unwrap();
563        db.push_schema().await.unwrap();
564        let mut db_q = db.clone();
565        let mine = toasty::create!(Tagged {
566            name: "mine".to_string(),
567            token: uuid::Uuid::parse_str(MINE).unwrap(),
568        })
569        .exec(&mut db_q)
570        .await
571        .unwrap();
572        toasty::create!(Tagged {
573            name: "theirs".to_string(),
574            token: uuid::Uuid::parse_str(THEIRS).unwrap(),
575        })
576        .exec(&mut db_q)
577        .await
578        .unwrap();
579        let router = form_panel_for::<TaggedResource>(db.clone())
580            .build()
581            .expect("panel builds");
582
583        let csrf = uuid::Uuid::new_v4().to_string();
584        let url = format!("/admin/tagged/{}/edit", mine.id);
585        let edit = |token: &str| {
586            http::Request::builder()
587                .method(http::Method::POST)
588                .uri(&url)
589                .header(
590                    http::header::CONTENT_TYPE,
591                    "application/x-www-form-urlencoded",
592                )
593                .header(
594                    http::header::COOKIE,
595                    format!("{}={csrf}", crate::csrf::COOKIE_NAME),
596                )
597                .body(Body::from(format!(
598                    "name=mine&token={token}&csrf_token={csrf}"
599                )))
600                .unwrap()
601        };
602
603        // The record's own token, re-spelled: the same value, so the save
604        // succeeds instead of probing this record's own row.
605        let response = router.handle(edit(&MINE.to_uppercase())).await;
606        assert!(
607            response.status().is_redirection(),
608            "re-spelling the record's own value must save, got {} {}",
609            response.status(),
610            response_html(response).await
611        );
612        let mut db_q = db.clone();
613        let saved = Tagged::filter(Tagged::fields().id().eq(mine.id))
614            .first()
615            .exec(&mut db_q)
616            .await
617            .unwrap()
618            .expect("the edited record");
619        assert_eq!(
620            saved.token,
621            uuid::Uuid::parse_str(MINE).unwrap(),
622            "the re-spelled value is stored canonically"
623        );
624
625        // Another record holds the submitted token: refused, nothing written.
626        let response = router.handle(edit(THEIRS)).await;
627        assert_eq!(
628            response.status(),
629            200,
630            "another record's value must refuse the edit"
631        );
632        let html = response_html(response).await;
633        assert!(
634            html.contains("Token has already been taken"),
635            "the typed probe must see the other record, got {html}"
636        );
637        let mut db_q = db.clone();
638        let unchanged = Tagged::filter(Tagged::fields().id().eq(mine.id))
639            .first()
640            .exec(&mut db_q)
641            .await
642            .unwrap()
643            .expect("the refused record");
644        assert_eq!(
645            unchanged.token,
646            uuid::Uuid::parse_str(MINE).unwrap(),
647            "a refused edit writes nothing"
648        );
649    }
