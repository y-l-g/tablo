1//! App-side uniqueness probe over `unique()`-marked text fields.
2
3use std::collections::HashMap;
4
5use topcoat::{Result, context::Cx};
6
7use crate::{form::FieldErrors, resource::Resource};
8
9/// Refuses duplicated `unique()` values with `<Label> has already been taken`, skipping empty
10/// submits and the record's own value (#117).
11pub(super) async fn check_unique<R: Resource>(
12    cx: &Cx,
13    schema: &crate::schema::Schema,
14    values: &HashMap<String, String>,
15    current: &HashMap<String, String>,
16    ex: &mut dyn toasty::Executor,
17) -> Result<FieldErrors, topcoat::Error> {
18    let mut errors = FieldErrors::new();
19    // Skips groups the submission leaves out.
20    let skip = schema.absent_fields(values);
21    for field in schema.fields() {
22        let name = field.name();
23        if !field.is_unique() || skip.contains(name) {
24            continue;
25        }
26        let Some(submitted) = values.get(name).map(|s| s.trim().to_string()) else {
27            continue;
28        };
29        // Never probes empty values; validation already refused them.
30        if submitted.is_empty() {
31            continue;
32        }
33        // Skips values normalising to the record's own stored value.
34        let unchanged = current.get(name).is_some_and(|kept| {
35            matches!(
36                (field.normalize(kept), field.normalize(&submitted)),
37                (Ok(kept), Ok(submitted)) if kept == submitted
38            )
39        });
40        if unchanged {
41            continue;
42        }
43        // Probes the parsed value for typed fields.
44        let Some(filter) = field.eq_filter(&submitted) else {
45            continue;
46        };
47        // Probes inside the write transaction through the tenant-scoped query.
48        let rows = crate::resource::scoped_query::<R>(cx)?
49            .filter(filter)
50            .limit(1)
51            .exec(&mut *ex)
52            .await
53            .map_err(crate::error::unavailable)?;
54        if !rows.is_empty() {
55            errors.add(
56                name,
57                format!("{} has already been taken", field.label_str()),
58            );
59        }
60    }
61    Ok(errors)
62}
