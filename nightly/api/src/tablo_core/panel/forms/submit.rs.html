34/// Failure-toast wording for the create/update handlers: one place,
35/// so the two paths cannot drift.
36const WRITE_CREATE: &str = "create the record";
37const WRITE_UPDATE: &str = "save the changes";

39/// The staged submission both write handlers carry into their transaction: the
40/// declared schema, the completed values, the validation errors so far, the
41/// upload paths a re-render keeps, and the keys the submission named.
42struct Submission {
43    schema: Arc<Schema>,
44    values: HashMap<String, String>,
45    errors: FieldErrors,
46    carried: HashSet<String>,
47    named: HashSet<String>,
48}

50/// Stage a create/edit submission: reject undeclared keys, take file values
51/// only from file parts, store the uploads outside the transaction, restore the
52/// paths a re-rendered form carried, strip the transport keys, record the keys
53/// the submission names, complete the rest from `advisory`, and validate —
54/// required and unique-free checks first, then the async relationship
55/// existence check.
56///
57/// `advisory` is the edit path's pre-transaction snapshot. A create passes
58/// `None`, so nothing is completed and an absent key validates as `""`.
59async fn prepare_submission<R: Resource>(
60    cx: &Cx,
61    parts: FormParts,
62    advisory: Option<&R::Model>,
63) -> Result<Submission, topcoat::Error> {
64    let schema = Arc::clone(&declared::<R>(cx).form);
65    reject_unknown_form_keys(&schema, &parts.values)?;
66    let FormParts {
67        mut values,
68        files,
69        file_part_names,
70    } = parts;
71    let stored = advisory
72        .map(|advisory| <R::Form as RecordForm>::hydrate(cx, advisory))
73        .unwrap_or_default();
74    // A declared file field takes its value only from a file part:
75    // a text part or a url-encoded pair under the same name is client-typed,
76    // not an upload, and would otherwise reach the record and render as the
77    // file's link.
78    drop_client_typed_uploads(&schema, &file_part_names, &mut values);
79    // Uploaded bytes become stored paths before validation, and outside the
80    // transaction: an upload is a side effect in another system, so a
81    // rolled-back transaction must not have to undo it, and a store that
82    // rejects the file must be able to answer inline.
83    let (upload_errors, mut carried) =
84        crate::upload::store_uploads(cx, &schema, &files, &mut values).await;
85    // A form re-rendered after a failed submit carries the path its store just
86    // answered; the uploader must still hold it. A restored field is
87    // non-empty, so the untouched check below leaves it alone.
88    carried.extend(restore_pending_uploads(cx, &schema, &mut values).await);
89    // An untouched file input is not named: the edit form renders an empty file
90    // input (browsers never pre-fill it), so an empty part means "keep", not
91    // "clear". An explicit `clear_<field>=1` names it empty; a chosen file
92    // still wins over the clear, because a replacement is not a removal.
93    for field in schema.fields().filter(|field| field.is_file()) {
94        let name = field.name();
95        let cleared = values
96            .get(&format!("clear_{name}"))
97            .is_some_and(|v| truthy(v));
98        let empty = values
99            .get(name)
100            .map(|v| v.trim().is_empty())
101            .unwrap_or(true);
102        if !cleared && empty && stored.get(name).is_some_and(|v| !v.trim().is_empty()) {
103            values.remove(name);
104        }
105    }
106    // Transport keys never reach the record fn; see `strip_transport_keys`.
107    strip_transport_keys(&schema, &mut values);
108    let named: HashSet<String> = values.keys().cloned().collect();
109    complete(&schema, &mut values, &named, &stored);
110    let mut errors = schema.validate_async(cx, &values).await;
111    // A rejected upload owns its field's error slot: "required" would restate
112    // the symptom (nothing was stored) and hide the reason.
113    errors.replace(upload_errors);
114    Ok(Submission {
115        schema,
116        values,
117        errors,
118        carried,
119        named,
120    })
121}

123/// Fill every declared key the submission did not name from `stored`, the
124/// record's form projection, and drop one `stored` does not hold. On create
125/// `stored` is empty and an unnamed key stays absent.
126fn complete(
127    schema: &Schema,
128    values: &mut HashMap<String, String>,
129    named: &HashSet<String>,
130    stored: &HashMap<String, String>,
131) {
132    for field in schema.fields() {
133        let name = field.name();
134        if named.contains(name) {
135            continue;
136        }
137        match stored.get(name) {
138            Some(value) => values.insert(name.to_string(), value.clone()),
139            None => values.remove(name),
140        };
141    }
142}

144/// Parse the completed values into the resource's form and run its
145/// `validate_record`, adding each failure to `errors`.
146///
147/// A parse failure is added only to a key with no error yet, so a blank
148/// required control shows the schema's message once. `validate_record` needs a
149/// whole form, so it runs only when every field parsed.
150///
151/// # Errors
152///
153/// An error keyed to something this submission renders nowhere — a control the
154/// schema does not declare, a repeater group's label the schema does not carry,
155/// or a field of a variant group the submission names hides. No slot owns the
156/// message, and the write must not proceed past it.
157fn parse_form<R: Resource>(
158    cx: &Cx,
159    schema: &Schema,
160    values: &HashMap<String, String>,
161    errors: &mut FieldErrors,
162) -> Result<Option<R::Form>, topcoat::Error> {
163    // Typed fields parse their own spelling, not the browser's.
164    let mut normalized = values.clone();
165    schema.normalize_values(&mut normalized);
166    match <R::Form as RecordForm>::parse(cx, &normalized) {
167        Ok(form) => {
168            for error in R::validate_record(cx, &form).iter() {
169                if !schema.renders_error_key(values, &error.key) {
170                    return Err(unrenderable_error::<R>(
171                        "validate_record",
172                        &error.key,
173                        &error.message,
174                    ));
175                }
176                errors.push(error.clone());
177            }
178            Ok(Some(form))
179        }
180        Err(failures) => {
181            let controls = schema.controls();
182            for mut failure in failures {
183                if !schema.renders_error_key(values, &failure.key) {
184                    return Err(unrenderable_error::<R>(
185                        "the parse",
186                        &failure.key,
187                        &failure.message,
188                    ));
189                }
190                if errors.contains_key(&failure.key) {
191                    continue;
192                }
193                // A blank required control shows the wording the schema
194                // declares for it, when it declares one.
195                if failure.kind == FieldErrorKind::Required
196                    && let Some(wording) = controls
197                        .iter()
198                        .find(|control| control.name == failure.key)
199                        .and_then(|control| control.required_error.clone())
200                {
201                    failure.message = wording;
202                }
203                errors.push(failure);
204            }
205            Ok(None)
206        }
207    }
208}

210/// Re-check the submission's relationship keys inside the write's
211/// transaction, once nothing else refused it: the related records must still
212/// be in the tenant-scoped queries and viewable as the write sees them. A
213/// refusal re-renders the form with the field's error, like the pre-write
214/// check, and nothing is written.
215async fn recheck_relationships(
216    cx: &Cx,
217    schema: &Schema,
218    values: &HashMap<String, String>,
219    errors: &mut FieldErrors,
220    ex: &mut dyn toasty::Executor,
221) {
222    if errors.is_empty() {
223        errors.extend(schema.recheck_relationships(cx, values, ex).await);
224    }
225}

227/// Refuse an error whose key this submission renders nowhere: no slot would
228/// carry the message, and writing anyway would drop it.
229fn unrenderable_error<R: Resource>(source: &str, key: &str, message: &str) -> topcoat::Error {
230    TabloError::Declaration(format!(
231        "{source} refused {key:?}, which `{}` renders nowhere for this submission: {message}",
232        std::any::type_name::<R::Form>()
233    ))
234    .into()
235}

237/// The form fields with at least one key the submission named.
238fn named_fields<F: RecordForm>(cx: &Cx, named: &HashSet<String>) -> Vec<F::Field> {
239    F::fields(&DeclCx::from_cx(cx))
240        .into_iter()
241        .filter(|field| field.keys.iter().any(|key| named.contains(key)))
242        .map(|field| field.field)
243        .collect()
244}

246pub(crate) fn resource_create_post<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
247    Box::pin(HoistView::new(ThenView::new(async move {
248        gate::<R>(cx)?;
249        if !can::<R>(cx, Ability::Create) {
250            return Err(forbidden().into());
251        }
252        let parts = parse_form_body(cx, body).await?;
253        crate::csrf::verify(cx, &parts.values)?;
254        // A create has no stored value to keep, so it stages no advisory
255        // snapshot: a rejected file leaves its field empty beside the reason.
256        let Submission {
257            schema,
258            values,
259            mut errors,
260            carried,
261            ..
262        } = prepare_submission::<R>(cx, parts, None).await?;
263        // Framework-owned transaction, opened only after validation so that
264        // `validate_async` loaders still run before it opens (see `crate::db`
265        // pool discipline). The unique check and the write observe one snapshot
266        // and commit atomically; dropping `tx` without commit rolls back.
267        let mut db = db(cx);
268        let mut tx = db.transaction().await.map_err(crate::error::unavailable)?;
269        // App-side unique check over every `unique()`-marked input — the only
270        // error layer until toasty exposes a unique-violation predicate
271        // (upstream gap #117; never string-match driver error messages).
272        errors.extend(check_unique::<R>(cx, &schema, &values, &HashMap::new(), &mut tx).await?);
273        let form = parse_form::<R>(cx, &schema, &values, &mut errors)?;
274        recheck_relationships(cx, &schema, &values, &mut errors, &mut tx).await;
275        let Some(form) = form.filter(|_| errors.is_empty()) else {
276            return rerender_invalid_form::<R>(
277                cx,
278                tx,
279                FormChrome::create::<R>(),
280                &values,
281                &errors,
282                &carried,
283            )
284            .await;
285        };
286        // The row the record fn returns is what `after_commit` names for this
287        // write — the key is the database's to generate, so the row is the
288        // only place the framework can learn it.
289        let written = R::create_record(cx, form, &mut tx).await;
290        commit_write::<R, _>(cx, tx, written, Committed::created, "Created", WRITE_CREATE).await
291    })))
292}

294/// Edit page POST — validates, checks `View` + `Update`, and writes
295/// the fields the submission named.
296///
297/// Requires both `View` and `Update` (matching GET, deny-by-default):
298/// a view-denied but writable record must not be mutable by direct POST.
299pub(crate) fn resource_edit_post<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
300    Box::pin(HoistView::new(ThenView::new(async move {
301        gate::<R>(cx)?;
302        let parts = parse_form_body(cx, body).await?;
303        crate::csrf::verify(cx, &parts.values)?;
304        let id = topcoat::router::path_param_segment(cx, "id").to_string();
305        // Advisory load on a pooled handle: feeds the completion validation
306        // reads. The body is already parsed and CSRF-verified, so the load
307        // never runs for a forged POST. The authoritative load + policy check
308        // happens inside the framework transaction; validation's
309        // `validate_async` loaders run before it opens (see `crate::db` pool
310        // discipline).
311        let mut db0 = db(cx);
312        let advisory = find_by_key::<R>(cx, &id, &mut db0).await?;
313        if !can::<R>(cx, Ability::View(&advisory)) {
314            return Err(forbidden().into());
315        }
316        if !can::<R>(cx, Ability::Update(&advisory)) {
317            return Err(forbidden().into());
318        }
319        let Submission {
320            schema,
321            mut values,
322            mut errors,
323            carried,
324            named,
325        } = prepare_submission::<R>(cx, parts, Some(&advisory)).await?;
326        // Authoritative load inside the framework transaction (#86):
327        // policy is checked on this snapshot and the same record flows into
328        // the write — never a silent re-load outside the checked snapshot.
329        let mut db = db(cx);
330        let mut tx = db.transaction().await.map_err(crate::error::unavailable)?;
331        let record = find_by_key::<R>(cx, &id, &mut tx).await?;
332        if !can::<R>(cx, Ability::View(&record)) {
333            return Err(forbidden().into());
334        }
335        if !can::<R>(cx, Ability::Update(&record)) {
336            return Err(forbidden().into());
337        }
338        // The unnamed keys come from this snapshot, not the advisory one: the
339        // parsed form reads what the write sees, and no unnamed key is ever
340        // written back.
341        let stored = <R::Form as RecordForm>::hydrate(cx, &record);
342        complete(&schema, &mut values, &named, &stored);
343        errors.extend(check_unique::<R>(cx, &schema, &values, &stored, &mut tx).await?);
344        let form = parse_form::<R>(cx, &schema, &values, &mut errors)?;
345        recheck_relationships(cx, &schema, &values, &mut errors, &mut tx).await;
346        let Some(form) = form.filter(|_| errors.is_empty()) else {
347            return rerender_invalid_form::<R>(
348                cx,
349                tx,
350                FormChrome::edit::<R>(cx, &record),
351                &values,
352                &errors,
353                &carried,
354            )
355            .await;
356        };
357        let posted = Posted::new(form, named_fields::<R::Form>(cx, &named));
358        let written = R::update_record(cx, record, posted, &mut tx).await;
359        commit_write::<R, _>(cx, tx, written, Committed::updated, "Updated", WRITE_UPDATE).await
360    })))
361}
