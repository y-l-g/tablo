28/// Failure-toast wording for the create/update handlers: one place,
29/// so the two paths cannot drift.
30const WRITE_CREATE: &str = "create the record";
31const WRITE_UPDATE: &str = "save the changes";
32
33/// The staged submission both write handlers carry into their transaction: the
34/// declared schema, the completed values, the validation errors so far, the
35/// upload paths a re-render keeps, and the keys the submission named.
36struct Submission {
37    schema: Schema,
38    values: HashMap<String, String>,
39    errors: HashMap<String, Vec<String>>,
40    carried: HashSet<String>,
41    named: HashSet<String>,
42}
43
44/// Stage a create/edit submission: reject undeclared keys, take file values
45/// only from file parts, store the uploads outside the transaction, restore the
46/// paths a re-rendered form carried, strip the transport keys, record the keys
47/// the submission names, complete the rest from `advisory`, and validate —
48/// required and unique-free checks first, then the async relationship
49/// existence check.
50///
51/// `advisory` is the edit path's pre-transaction snapshot. A create passes
52/// `None`, so nothing is completed and an absent key validates as `""`.
53async fn prepare_submission<R: Resource>(
54    cx: &Cx,
55    parts: FormParts,
56    advisory: Option<&R::Model>,
57) -> Result<Submission, topcoat::Error> {
58    let schema = R::form(cx);
59    reject_unknown_form_keys(&schema, &parts.values)?;
60    let FormParts {
61        mut values,
62        files,
63        file_part_names,
64    } = parts;
65    let stored = advisory
66        .map(|advisory| <R::Form as RecordForm>::hydrate(cx, advisory))
67        .unwrap_or_default();
68    // A declared `FileUpload` takes its value only from a file part:
69    // a text part or a url-encoded pair under the same name is client-typed,
70    // not an upload, and would otherwise reach the record and render as the
71    // file's link.
72    drop_client_typed_uploads(&schema, &file_part_names, &mut values);
73    // Uploaded bytes become stored paths before validation, and outside the
74    // transaction: an upload is a side effect in another system, so a
75    // rolled-back transaction must not have to undo it, and a store that
76    // rejects the file must be able to answer inline.
77    let (upload_errors, mut carried) =
78        crate::upload::store_uploads(cx, &schema, &files, &mut values).await;
79    // A form re-rendered after a failed submit carries the path its store just
80    // answered; the uploader must still hold it. A restored field is
81    // non-empty, so the untouched check below leaves it alone.
82    carried.extend(restore_pending_uploads(cx, &schema, &mut values).await);
83    // An untouched file input is not named: the edit form renders an empty file
84    // input (browsers never pre-fill it), so an empty part means "keep", not
85    // "clear". An explicit `clear_<field>=1` names it empty; a chosen file
86    // still wins over the clear, because a replacement is not a removal.
87    for name in schema.file_uploads().keys() {
88        let cleared = values
89            .get(&format!("clear_{name}"))
90            .is_some_and(|v| truthy(v));
91        let empty = values
92            .get(name)
93            .map(|v| v.trim().is_empty())
94            .unwrap_or(true);
95        if !cleared && empty && stored.get(name).is_some_and(|v| !v.trim().is_empty()) {
96            values.remove(name);
97        }
98    }
99    // Transport keys never reach the record fn; see `strip_transport_keys`.
100    strip_transport_keys(&schema, &mut values);
101    let named: HashSet<String> = values.keys().cloned().collect();
102    complete(&schema, &mut values, &named, &stored);
103    let mut errors = schema.validate_async(cx, &values).await;
104    // A rejected upload owns its field's error slot: "required" would restate
105    // the symptom (nothing was stored) and hide the reason.
106    errors.extend(upload_errors);
107    Ok(Submission {
108        schema,
109        values,
110        errors,
111        carried,
112        named,
113    })
114}

116/// Fill every declared key the submission did not name from `stored`, the
117/// record's form projection, and drop one `stored` does not hold. On create
118/// `stored` is empty and an unnamed key stays absent.
119fn complete(
120    schema: &Schema,
121    values: &mut HashMap<String, String>,
122    named: &HashSet<String>,
123    stored: &HashMap<String, String>,
124) {
125    for name in schema.field_names() {
126        if named.contains(&name) {
127            continue;
128        }
129        match stored.get(&name) {
130            Some(value) => values.insert(name, value.clone()),
131            None => values.remove(&name),
132        };
133    }
134}

136/// Parse the completed values into the resource's form and run its
137/// `validate_record`, adding each failure to `errors`.
138///
139/// A parse failure is added only to a key with no error yet, so a blank
140/// required control shows the schema's message once. `validate_record` needs a
141/// whole form, so it runs only when every field parsed.
142///
143/// # Errors
144///
145/// A `validate_record` error on a field `RecordForm::fields` binds to no key:
146/// it cannot render, and the write must not proceed past it.
147fn parse_form<R: Resource>(
148    cx: &Cx,
149    schema: &Schema,
150    values: &HashMap<String, String>,
151    errors: &mut HashMap<String, Vec<String>>,
152) -> Result<Option<R::Form>, topcoat::Error> {
153    // Typed fields parse their own spelling, not the browser's.
154    let mut normalized = values.clone();
155    schema.normalize_values(&mut normalized);
156    match <R::Form as RecordForm>::parse(cx, &normalized) {
157        Ok(form) => {
158            let fields = <R::Form as RecordForm>::fields(cx);
159            for (field, message) in R::validate_record(cx, &form).iter() {
160                // A field's errors render under its first key: a scalar's own,
161                // an embedded enum's discriminant.
162                let Some(key) = fields
163                    .iter()
164                    .find(|claim| claim.field == *field)
165                    .and_then(|claim| claim.keys.first())
166                else {
167                    // A rule refused a field the form lists no key for: there is
168                    // nowhere to render it, and writing anyway would drop it.
169                    return Err(std::io::Error::other(format!(
170                        "validate_record refused {field:?}, which `{}::fields` binds to no key: \
171                         {message}",
172                        std::any::type_name::<R::Form>()
173                    ))
174                    .into());
175                };
176                errors.entry(key.clone()).or_default().push(message.clone());
177            }
178            Ok(Some(form))
179        }
180        Err(failures) => {
181            let controls = schema.controls();
182            for failure in failures {
183                if errors.contains_key(&failure.key) {
184                    continue;
185                }
186                let message = match failure.kind {
187                    FieldErrorKind::Required => controls
188                        .iter()
189                        .find(|control| control.name == failure.key)
190                        .and_then(|control| control.required_error.clone())
191                        .unwrap_or(failure.message),
192                    FieldErrorKind::Invalid => failure.message,
193                };
194                errors.insert(failure.key, vec![message]);
195            }
196            Ok(None)
197        }
198    }
199}

201/// The form fields with at least one key the submission named.
202fn named_fields<F: RecordForm>(cx: &Cx, named: &HashSet<String>) -> Vec<F::Field> {
203    F::fields(cx)
204        .into_iter()
205        .filter(|field| field.keys.iter().any(|key| named.contains(key)))
206        .map(|field| field.field)
207        .collect()
208}

210pub(crate) fn resource_create_post<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
211    Box::pin(HoistView::new(ThenView::new(async move {
212        gate::<R>(cx)?;
213        if !R::can_create(cx) {
214            return Err(forbidden().into());
215        }
216        let parts = parse_form_body(cx, body).await?;
217        crate::csrf::verify(cx, &parts.values)?;
218        // A create has no stored value to keep, so it stages no advisory
219        // snapshot: a rejected file leaves its field empty beside the reason.
220        let Submission {
221            schema,
222            values,
223            mut errors,
224            carried,
225            ..
226        } = prepare_submission::<R>(cx, parts, None).await?;
227        // Framework-owned transaction, opened only after validation so that
228        // `validate_async` loaders still run before it opens (see `crate::db`
229        // pool discipline). The unique check and the write observe one snapshot
230        // and commit atomically; dropping `tx` without commit rolls back.
231        let mut db = db(cx);
232        let mut tx = db.transaction().await.map_err(crate::db::unavailable)?;
233        // App-side unique check over every `unique()`-marked input — the only
234        // error layer until toasty exposes a unique-violation predicate
235        // (upstream gap #117; never string-match driver error messages).
236        for (name, errs) in
237            check_unique::<R>(cx, &schema, &values, &HashMap::new(), &mut tx).await?
238        {
239            errors.entry(name).or_default().extend(errs);
240        }
241        let form = parse_form::<R>(cx, &schema, &values, &mut errors)?;
242        let Some(form) = form.filter(|_| errors.is_empty()) else {
243            return rerender_invalid_form::<R>(
244                cx,
245                tx,
246                format!("Create {}", R::label()),
247                "Create",
248                &values,
249                &errors,
250                &carried,
251                None,
252            )
253            .await;
254        };
255        // The row the record fn returns is what `after_commit` names for this
256        // write — the key is the database's to generate, so the row is the
257        // only place the framework can learn it.
258        let written = R::create_record(cx, form, &mut tx).await;
259        commit_write::<R, _>(cx, tx, written, Committed::created, "Created", WRITE_CREATE).await
260    })))
261}

263/// Edit page POST — validates, checks `can_view` + `can_update`, and writes
264/// the fields the submission named.
265///
266/// Requires both `can_view` and `can_update` (matching GET, deny-by-default):
267/// a view-denied but writable record must not be mutable by direct POST.
268pub(crate) fn resource_edit_post<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
269    Box::pin(HoistView::new(ThenView::new(async move {
270        gate::<R>(cx)?;
271        let parts = parse_form_body(cx, body).await?;
272        crate::csrf::verify(cx, &parts.values)?;
273        let id = topcoat::router::path_param_segment(cx, "id").to_string();
274        // Advisory load on a pooled handle: feeds the completion validation
275        // reads. The body is already parsed and CSRF-verified, so the load
276        // never runs for a forged POST. The authoritative load + policy check
277        // happens inside the framework transaction; validation's
278        // `validate_async` loaders run before it opens (see `crate::db` pool
279        // discipline).
280        let mut db0 = db(cx);
281        let advisory = find_by_key::<R>(cx, &id, &mut db0).await?;
282        if !R::can_view(cx, &advisory) {
283            return Err(forbidden().into());
284        }
285        if !R::can_update(cx, &advisory) {
286            return Err(forbidden().into());
287        }
288        let Submission {
289            schema,
290            mut values,
291            mut errors,
292            carried,
293            named,
294        } = prepare_submission::<R>(cx, parts, Some(&advisory)).await?;
295        // Authoritative load inside the framework transaction (#86):
296        // policy is checked on this snapshot and the same record flows into
297        // the write — never a silent re-load outside the checked snapshot.
298        let mut db = db(cx);
299        let mut tx = db.transaction().await.map_err(crate::db::unavailable)?;
300        let record = find_by_key::<R>(cx, &id, &mut tx).await?;
301        if !R::can_view(cx, &record) {
302            return Err(forbidden().into());
303        }
304        if !R::can_update(cx, &record) {
305            return Err(forbidden().into());
306        }
307        // The unnamed keys come from this snapshot, not the advisory one: the
308        // parsed form reads what the write sees, and no unnamed key is ever
309        // written back.
310        let stored = <R::Form as RecordForm>::hydrate(cx, &record);
311        complete(&schema, &mut values, &named, &stored);
312        for (name, errs) in check_unique::<R>(cx, &schema, &values, &stored, &mut tx).await? {
313            errors.entry(name).or_default().extend(errs);
314        }
315        let form = parse_form::<R>(cx, &schema, &values, &mut errors)?;
316        let Some(form) = form.filter(|_| errors.is_empty()) else {
317            let public = R::public_url(cx, &record);
318            return rerender_invalid_form::<R>(
319                cx,
320                tx,
321                format!("Edit {}", R::label()),
322                "Save",
323                &values,
324                &errors,
325                &carried,
326                public,
327            )
328            .await;
329        };
330        let posted = Posted::new(form, named_fields::<R::Form>(cx, &named));
331        let written = R::update_record(cx, record, posted, &mut tx).await;
332        commit_write::<R, _>(cx, tx, written, Committed::updated, "Updated", WRITE_UPDATE).await
333    })))
334}
