1//! URLs into the request's panel, so app code never spells a prefix.
2//!
3//! Each helper answers for the panel serving the request: the one whose
4//! prefix the request is under, or the only panel when the router mounts one.
5//! A resource or page that panel does not register has no URL there, and the
6//! helper returns `None`.
7//!
8//! ```ignore
9//! let posts = tablo::url::resource::<PostResource>(cx);  // Some("/admin/posts")
10//! let media = tablo::url::page::<MediaLibraryPage>(cx);  // Some("/admin/media")
11//! let home = tablo::url::panel(cx);                      // Some("/admin")
12//! ```
13
14use std::any::TypeId;
15
16use topcoat::context::Cx;
17
18use super::state::current;
19use crate::{Page, resource::Resource};
20
21/// The request's panel prefix, e.g. `/admin`.
22pub fn panel(cx: &Cx) -> Option<String> {
23    current(cx).map(|panel| panel.prefix.clone())
24}
25
26/// `R`'s list URL in the request's panel, e.g. `/admin/posts`.
27pub fn resource<R: Resource>(cx: &Cx) -> Option<String> {
28    registered::<R>(cx)
29}
30
31/// `P`'s URL in the request's panel: `{prefix}/{slug}`, or the prefix itself
32/// for the panel's [`home`](super::Panel::home) page.
33pub fn page<P: Page>(cx: &Cx) -> Option<String> {
34    registered::<P>(cx)
35}
36
37fn registered<T: 'static>(cx: &Cx) -> Option<String> {
38    current(cx).and_then(|panel| panel.urls.get(&TypeId::of::<T>()).cloned())
39}
