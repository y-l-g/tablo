27/// Failure-toast wording for the delete handlers.
28const WRITE_BULK_DELETE: &str = "delete the selected rows";

30/// Bulk delete POST — ids via `ids` form field (comma-separated).
31///
32/// Identity is the typed PK fetch alone: the display key
33/// is never re-matched, so non-canonical keys (uppercase UUID,
34/// email key) cannot 404 a batch whose rows exist. Bounded by
35/// `MAX_BULK_IDS` so the `IN` list cannot be amplified into a DoS.
36/// Fetch, policy checks, and deletes share one framework transaction
37/// a mid-loop failure deletes zero rows.
38pub(crate) fn resource_bulk_delete<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
39    Box::pin(HoistView::new(ThenView::<_, BoxView<'_>>::new(
40        async move {
41            gate::<R>(cx)?;
42            // The whole-resource half of the policy, before the body is read:
43            // a resource that allows no delete renders no delete chrome.
44            if !R::can_delete_any(cx) {
45                return Err(forbidden().into());
46            }
47            // Delete/bulk-delete carry no file parts: only the values half is read.
48            let values = parse_form_body(cx, body).await?.values;
49            crate::csrf::verify(cx, &values)?;
50            // Confirmation marker, mirroring the row delete: the bulk
51            // bar's dialog carries `confirm=1`, so a POST without it did not
52            // come from the confirming control. Checked after CSRF verification
53            // and before any DB work — a forged POST answers 400
54            // without touching a connection.
55            if !values.get("confirm").is_some_and(|v| truthy(v)) {
56                return Err(topcoat::router::error::bad_request(
57                    "bulk delete requires confirmation",
58                )
59                .into());
60            }
61            let ids_raw = values.get("ids").cloned().unwrap_or_default();
62            let ids = parse_bulk_ids(&ids_raw, MAX_BULK_IDS);
63            if ids.is_empty() {
64                // No ids is a validation miss, not a raw 400 page:
65                // the bulk bar cannot submit without a selection, so only a
66                // crafted (or stale) POST gets here — answer like any other
67                // mutation, with the list and the reason.
68                set_notification(cx, Notification::error("Select at least one row to delete"));
69                return Err(see_other(list_url(cx, &R::slug())).into());
70            }
71            if ids.len() > MAX_BULK_IDS {
72                return Err(topcoat::router::error::bad_request(format!(
73                    "too many ids (max {MAX_BULK_IDS})"
74                ))
75                .into());
76            }
77            // Fetch only the requested rows through the tenancy-scoped seam:
78            // one `pk IN (…)` query replaces the #75 item-1
79            // fetch-everything-then-match loop. A malformed id cannot exist and
80            // maps to 404; a missing/wrong-tenant id makes the fetch come back
81            // short and 404s as well.
82            let keys: Vec<&str> = ids.iter().map(String::as_str).collect();
83            let Some(pk_filter) = crate::schema::pk_in_expr::<R::Model>(&keys) else {
84                if let Some(error) = composite_pk_error::<R>() {
85                    return Err(error);
86                }
87                return Err(topcoat::router::error::not_found().into());
88            };
89            let mut db = db(cx);
90            let mut tx = db.transaction().await.map_err(crate::db::unavailable)?;
91            // The batch fetch reads only the records' own columns:
92            // the policy predicates and the write below never touch a relation.
93            let rows = crate::resource::scoped_query::<R>(cx)?
94                .filter(pk_filter)
95                .exec(&mut tx)
96                .await
97                .map_err(crate::db::unavailable)?;
98            if rows.len() != ids.len() {
99                return Err(topcoat::router::error::not_found().into());
100            }
101            for rec in &rows {
102                // Edit contract on every row: viewing precedes
103                // deleting, same as the edit GET/POST pair.
104                if !R::can_view(cx, rec) {
105                    return Err(forbidden().into());
106                }
107                if !R::can_delete(cx, rec) {
108                    return Err(forbidden().into());
109                }
110            }
111            // All checks passed — perform bulk delete inside the tx, then
112            // commit once. Any error drops `tx` uncommitted: zero rows
113            // deleted, never half-applied. The hook names the whole batch: a
114            // bulk delete is one write, so it is one `after_commit` call.
115            let written = R::bulk_delete_records(cx, &rows, &mut tx)
116                .await
117                .map(|()| rows);
118            commit_write::<R, _>(
119                cx,
120                tx,
121                written,
122                Committed::deleted,
123                "Bulk deleted",
124                WRITE_BULK_DELETE,
125            )
126            .await
127        },
128    )))
129}

131/// Max ids accepted by bulk delete: bounds the `IN` list.
132const MAX_BULK_IDS: usize = 400;
133
134/// Parse + dedupe bulk `ids` while preserving order, so a repeated id can't
135/// make the fetched-rows count check misfire.
136///
137/// `max` bounds the parse itself, not just the final list: a 10 MiB
138/// body of distinct ids stops at `max + 1` entries (which the handler then
139/// rejects with 400) instead of allocating millions of strings while the
140/// `MAX_BULK_IDS` check waits for the parse to finish. Deduping uses a set, so
141/// the scan stays linear in the number of ids.
142///
143/// Known limit: the split happens after url-decoding, so a
144/// `String`-PK id containing a literal comma (`%2C`) splits into phantom
145/// ids and the batch 404s. Comma-bearing string PKs need a different
146/// transport (future work); all other PK types are comma-free.
147fn parse_bulk_ids(raw: &str, max: usize) -> Vec<String> {
148    let mut ids: Vec<String> = Vec::new();
149    let mut seen = std::collections::HashSet::new();
150    for s in raw.split(',').map(str::trim).filter(|s| !s.is_empty()) {
151        if seen.insert(s) {
152            ids.push(s.to_string());
153            if ids.len() > max {
154                break;
155            }
156        }
157    }
158    ids
159}
