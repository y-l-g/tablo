28/// Failure-toast wording for the delete handlers.
29const WRITE_BULK_DELETE: &str = "delete the selected rows";
30
31/// Serves the bulk delete POST over the `ids` form field.
32pub(crate) fn resource_bulk_delete<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
33    Box::pin(HoistView::new(ThenView::<_, BoxView<'_>>::new(
34        async move {
35            gate::<R>(cx)?;
36            if !can::<R>(cx, Ability::DeleteAny) {
37                return Err(forbidden().into());
38            }
39            let values = parse_form_body(cx, body).await?.values;
40            crate::csrf::verify(cx, &values)?;
41            // Requires the `confirm=1` marker.
42            if !values.get("confirm").is_some_and(|v| truthy(v)) {
43                return Err(topcoat::router::error::bad_request(
44                    "bulk delete requires confirmation",
45                )
46                .into());
47            }
48            let ids_raw = values.get("ids").cloned().unwrap_or_default();
49            let ids = parse_bulk_ids(&ids_raw, MAX_BULK_IDS);
50            if ids.is_empty() {
51                set_notification(cx, Notification::error("Select at least one row to delete"));
52                return Err(see_other(landing_url(cx, &R::slug())).into());
53            }
54            if ids.len() > MAX_BULK_IDS {
55                return Err(topcoat::router::error::bad_request(format!(
56                    "too many ids (max {MAX_BULK_IDS})"
57                ))
58                .into());
59            }
60            // Fetches only the requested rows through the tenancy-scoped query.
61            let keys: Vec<&str> = ids.iter().map(String::as_str).collect();
62            let Some(pk_filter) = crate::schema::pk_in_expr::<R::Model>(&keys) else {
63                if let Some(error) = composite_pk_error::<R>() {
64                    return Err(error);
65                }
66                return Err(topcoat::router::error::not_found().into());
67            };
68            let mut db = db(cx);
69            let mut tx = db.transaction().await.map_err(crate::error::unavailable)?;
70            let rows = crate::resource::scoped_query::<R>(cx)?
71                .filter(pk_filter)
72                .exec(&mut tx)
73                .await
74                .map_err(crate::error::unavailable)?;
75            if rows.len() != ids.len() {
76                return Err(topcoat::router::error::not_found().into());
77            }
78            // Viewing precedes deleting on every row.
79            if rows.iter().any(|rec| !can::<R>(cx, Ability::View(rec))) {
80                return Err(forbidden().into());
81            }
82            let refused = rows
83                .iter()
84                .filter(|rec| !can::<R>(cx, Ability::Delete(rec)))
85                .count();
86            if refused > 0 {
87                let noun = if refused == 1 { "record" } else { "records" };
88                set_notification(
89                    cx,
90                    Notification::error(format!(
91                        "{refused} selected {noun} cannot be deleted; nothing was deleted"
92                    )),
93                );
94                return Err(see_other(landing_url(cx, &R::slug())).into());
95            }
96            let written = R::bulk_delete_records(cx, &rows, &mut tx)
97                .await
98                .map(|()| rows);
99            commit_write::<R, _>(
100                cx,
101                tx,
102                written,
103                Committed::deleted,
104                "Bulk deleted",
105                WRITE_BULK_DELETE,
106            )
107            .await
108        },
109    )))
110}

112/// Bounds the bulk delete `IN` list.
113pub(super) const MAX_BULK_IDS: usize = 400;
114
115/// Parses and dedupes bulk `ids` while preserving order.
116pub(super) fn parse_bulk_ids(raw: &str, max: usize) -> Vec<String> {
117    let mut ids: Vec<String> = Vec::new();
118    let mut seen = std::collections::HashSet::new();
119    for s in raw.split(',').map(str::trim).filter(|s| !s.is_empty()) {
120        if seen.insert(s) {
121            ids.push(s.to_string());
122            if ids.len() > max {
123                break;
124            }
125        }
126    }
127    ids
128}
