17/// Relationship option search endpoint (D2/D5).
18///
19/// `GET {parent_list_url}/options?field=&q=` — server-side narrowing for
20/// tables above the option cap. `field` allow-lists to a declared searchable
21/// relationship choice in `Resource::form` (400 otherwise); a non-searchable
22/// choice keeps the cap error and never calls here. `q` is trimmed and
23/// clamped to the shared query bound; empty `q` returns the bounded head.
24///
25/// Gates: `auth::guard` + `enforce_tenant::<R>` (parent), then the related
26/// gates inside the search (`ViewAny` + tenant + `View` filtering
27/// before labels). Parent form policy (`Create` / `View`+`Update`)
28/// stays on the form pages themselves: requiring parent `ViewAny` here
29/// would lock create-only users out of a form they may use, and adds no
30/// visibility the related list does not already expose. `Denied` → 403, driver failure → 500,
31/// filtered overflow → 200 with a "keep typing" hint option (client keeps its hint element).
32/// Success → 200 `text/html` with `<option>` markup, bounded to
33/// `MAX_RELATIONSHIP_OPTIONS`, values are typed PK strings, labels escaped.
34pub(crate) fn resource_options<R: Resource>(cx: &Cx, _body: Body) -> RouteFuture<'_> {
35    Box::pin(async move {
36        gate::<R>(cx)?;
37        let (field, q) = options_query(cx);
38        let field = field.trim();
39        if field.is_empty() {
40            return Err(topcoat::router::error::bad_request("missing field").into());
41        }
42        let q = clamp_query_term(&q);
43        let declared = declared::<R>(cx);
44        let form = &declared.form;
45        let Some(select) = form
46            .fields()
47            .find(|declared| declared.name() == field)
48            .and_then(|declared| declared.as_choice())
49        else {
50            return Err(topcoat::router::error::bad_request("unknown field").into());
51        };
52        if !select.is_relationship() {
53            return Err(topcoat::router::error::bad_request("not a relationship select").into());
54        }
55        if !select.is_searchable() {
56            return Err(topcoat::router::error::bad_request("not searchable").into());
57        }
58        match select.search_options(cx, &q).await {
59            Ok(opts) => {
60                let options: Vec<_> = opts
61                    .into_iter()
62                    .map(|(value, label)| option_view(cx, value, label, false))
63                    .collect();
64                let html = view! {
65                    cx =>
66                    for option in options {
67                        (option)
68                    }
69                }
70                .single()
71                .await?
72                .render(cx);
73                let res = http::Response::builder()
74                    .status(200)
75                    .header(http::header::CONTENT_TYPE, "text/html; charset=utf-8")
76                    .header("x-content-type-options", "nosniff")
77                    .body(Body::from(html))?;
78                Ok(res)
79            }
80            Err(OptionLoadError::Denied) => Err(forbidden().into()),
81            Err(OptionLoadError::LoadFailed) => {
82                Err(TabloError::Infrastructure("option search failed").into())
83            }
84            // Permanent: the related resource cannot be scoped at
85            // all, so the search cannot succeed until the declaration is
86            // fixed — a 500 that says so, not a retry.
87            Err(OptionLoadError::Misdeclared) => Err(TabloError::Declaration(
88                "option search unavailable: the related resource requires a tenant the framework \
89                 cannot scope"
90                    .to_string(),
91            )
92            .into()),
93            Err(OptionLoadError::Overflow) => {
94                let html = "<option value=\"\" disabled>Too many results — keep typing</option>"
95                    .to_string();
96                let res = http::Response::builder()
97                    .status(200)
98                    .header(http::header::CONTENT_TYPE, "text/html; charset=utf-8")
99                    .header("x-content-type-options", "nosniff")
100                    .body(Body::from(html))?;
101                Ok(res)
102            }
103        }
104    })
105}

107/// Parse `?field=` + `?q=` for the options endpoint (first-wins, like the
108/// table state parser).
109fn options_query(cx: &Cx) -> (String, String) {
110    let Some(parts) = topcoat::context::try_request_context::<http::request::Parts>(cx) else {
111        return (String::new(), String::new());
112    };
113    let query = parts.uri.query().unwrap_or("");
114    let mut field = String::new();
115    let mut q = String::new();
116    let mut seen_field = false;
117    let mut seen_q = false;
118    for (k, v) in form_urlencoded::parse(query.as_bytes()) {
119        if k == "field" && !seen_field {
120            field = v.into_owned();
121            seen_field = true;
122        } else if k == "q" && !seen_q {
123            q = v.into_owned();
124            seen_q = true;
125        }
126    }
127    (field, q)
128}
