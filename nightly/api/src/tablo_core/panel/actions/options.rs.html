15/// Relationship option search endpoint (D2/D5).
16///
17/// `GET {parent_list_url}/options?field=&q=` — server-side narrowing for
18/// tables above the option cap. `field` allow-lists to a declared searchable
19/// relationship `Select` in `Resource::form` (400 otherwise); non-searchable
20/// selects keep today's cap error and never call here. `q` is trimmed and
21/// clamped to the shared query bound; empty `q` returns the bounded head.
22///
23/// Gates: `enforce_auth` + `enforce_tenant::<R>` (parent), then the related
24/// gates inside the search (`can_view_any` + tenant + `can_view` filtering
25/// before labels). Parent form policy (`can_create` / `can_view`+`can_update`)
26/// stays on the form pages themselves: requiring parent `can_view_any` here
27/// would lock create-only users out of a form they may use, and adds no
28/// visibility the related list does not already expose. `Denied` → 403, driver failure → 500,
29/// filtered overflow → 200 with a "keep typing" hint option (client keeps its hint element).
30/// Success → 200 `text/html` with `<option>` markup, bounded to
31/// `MAX_RELATIONSHIP_OPTIONS`, values are typed PK strings, labels escaped.
32pub(crate) fn resource_options<R: Resource>(cx: &Cx, _body: Body) -> RouteFuture<'_> {
33    Box::pin(async move {
34        gate::<R>(cx)?;
35        let (field, q) = options_query(cx);
36        let field = field.trim();
37        if field.is_empty() {
38            return Err(topcoat::router::error::bad_request("missing field").into());
39        }
40        let q = clamp_query_term(&q);
41        let form = R::form(cx);
42        let selects = form.select_inputs();
43        let Some(select) = selects.get(field) else {
44            return Err(topcoat::router::error::bad_request("unknown field").into());
45        };
46        if !select.is_relationship() {
47            return Err(topcoat::router::error::bad_request("not a relationship select").into());
48        }
49        if !select.is_searchable() {
50            return Err(topcoat::router::error::bad_request("not searchable").into());
51        }
52        match select.search_options(cx, &q).await {
53            Ok(opts) => {
54                let mut html = String::with_capacity(opts.len() * 32);
55                for (v, lab) in opts {
56                    html.push_str(&format!(
57                        "<option value=\"{}\">{}</option>",
58                        escape_option(&v),
59                        escape_option(&lab)
60                    ));
61                }
62                let res = http::Response::builder()
63                    .status(200)
64                    .header(http::header::CONTENT_TYPE, "text/html; charset=utf-8")
65                    .header("x-content-type-options", "nosniff")
66                    .body(Body::from(html))
67                    .map_err(|e| std::io::Error::other(e.to_string()))?;
68                Ok(res)
69            }
70            Err(OptionLoadError::Denied) => Err(forbidden().into()),
71            Err(OptionLoadError::LoadFailed) => Err(topcoat::Error::from(std::io::Error::other(
72                "option search failed",
73            ))),
74            // Permanent: the related resource cannot be scoped at
75            // all, so the search cannot succeed until the declaration is
76            // fixed — a 500 that says so, not a retry.
77            Err(OptionLoadError::Misdeclared) => Err(topcoat::Error::from(std::io::Error::other(
78                "option search unavailable: the related resource requires a tenant the                      framework cannot scope (GH #223)",
79            ))),
80            Err(OptionLoadError::Overflow) => {
81                let html = "<option value=\"\" disabled>Too many results — keep typing</option>"
82                    .to_string();
83                let res = http::Response::builder()
84                    .status(200)
85                    .header(http::header::CONTENT_TYPE, "text/html; charset=utf-8")
86                    .header("x-content-type-options", "nosniff")
87                    .body(Body::from(html))
88                    .map_err(|e| std::io::Error::other(e.to_string()))?;
89                Ok(res)
90            }
91        }
92    })
93}

95/// Parse `?field=` + `?q=` for the options endpoint (first-wins, like the
96/// table state parser).
97fn options_query(cx: &Cx) -> (String, String) {
98    let Some(parts) = topcoat::context::try_request_context::<http::request::Parts>(cx) else {
99        return (String::new(), String::new());
100    };
101    let query = parts.uri.query().unwrap_or("");
102    let mut field = String::new();
103    let mut q = String::new();
104    let mut seen_field = false;
105    let mut seen_q = false;
106    for (k, v) in form_urlencoded::parse(query.as_bytes()) {
107        if k == "field" && !seen_field {
108            field = v.into_owned();
109            seen_field = true;
110        } else if k == "q" && !seen_q {
111            q = v.into_owned();
112            seen_q = true;
113        }
114    }
115    (field, q)
116}

118/// Escape a value/label for `<option>` markup.
119fn escape_option(s: &str) -> String {
120    let mut out = String::with_capacity(s.len());
121    for c in s.chars() {
122        match c {
123            '&' => out.push_str("&amp;"),
124            '<' => out.push_str("&lt;"),
125            '>' => out.push_str("&gt;"),
126            '"' => out.push_str("&quot;"),
127            '\'' => out.push_str("&#39;"),
128            _ => out.push(c),
129        }
130    }
131    out
132}
