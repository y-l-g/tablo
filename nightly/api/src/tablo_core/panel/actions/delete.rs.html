24/// Failure-toast wording for the delete handlers.
25const WRITE_DELETE: &str = "delete the record";
26
27/// Delete action POST — confirmation-marked, policy-checked, and run in the
28/// framework transaction: the checked record flows into the write.
29///
30/// The confirmation is the row's alert dialog on the list page: the
31/// Delete link opens `?delete=<key>` and the dialog's form POSTs here with
32/// `confirm=1`. Authentication comes before any DB work: the CSRF
33/// check and the confirmation marker run first, so a forged POST answers 403
34/// without opening a transaction, holding a pooled connection across the body
35/// read, or probing record existence (create/bulk-delete ordering).
36/// The dialog itself is deliberately fetch-free and policy-blind: it carries
37/// no record data and embeds only the caller's own CSRF token, and the
38/// policy/tenancy checks run against the loaded record here.
39pub(crate) fn resource_delete<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
40    Box::pin(HoistView::new(ThenView::<_, BoxView<'_>>::new(
41        async move {
42            gate::<R>(cx)?;
43            // The whole-resource half of the policy, before the body is read:
44            // a resource that allows no delete renders no delete chrome.
45            if !can::<R>(cx, Ability::DeleteAny) {
46                return Err(forbidden().into());
47            }
48            // Delete/bulk-delete carry no file parts: only the values half is read.
49            let values = parse_form_body(cx, body).await?.values;
50            crate::csrf::verify(cx, &values)?;
51            let confirmed = values.get("confirm").is_some_and(|v| truthy(v));
52            if !confirmed {
53                // The confirmation UI is the list-page alert dialog:
54                // the row link opens `?delete=<key>` and the dialog's form carries
55                // `confirm=1`. This route only accepts that confirmed POST, so a
56                // missing marker is a malformed client, not a user path.
57                return Err(
58                    topcoat::router::error::bad_request("delete requires confirmation").into(),
59                );
60            }
61            // Confirmed and authenticated: open the transaction only now (GH
62            // #144), fetch through the tenant-scoped query, check Policy against the
63            // loaded record, and delete inside the tx — commit makes the checked
64            // delete durable, any error rolls it back. Delete takes
65            // the edit contract: `View` plus
66            // `Delete` — a record that cannot be viewed cannot be deleted
67            // by UUID-guessing the route.
68            let mut db = db(cx);
69            let mut tx = db.transaction().await.map_err(crate::error::unavailable)?;
70            let id = topcoat::router::path_param_segment(cx, "id").to_string();
71            // The delete path reads only the record's own columns:
72            // `View`/`Delete` are Rust predicates over those, and
73            // `delete_record` reads the same snapshot.
74            let record = find_by_key::<R>(cx, &id, &mut tx).await?;
75            if !can::<R>(cx, Ability::View(&record)) {
76                return Err(forbidden().into());
77            }
78            if !can::<R>(cx, Ability::Delete(&record)) {
79                return Err(forbidden().into());
80            }
81            // The hook names what was removed: the pre-delete snapshot, since
82            // the row is gone by the time it runs.
83            let written = R::delete_record(cx, &record, &mut tx)
84                .await
85                .map(|()| vec![record]);
86            commit_write::<R, _>(cx, tx, written, Committed::deleted, "Deleted", WRITE_DELETE).await
87        },
88    )))
89}
