22/// Failure-toast wording for the delete handlers.
23const WRITE_DELETE: &str = "delete the record";
24
25/// Delete action POST — confirmation-marked, policy-checked, and run in the
26/// framework transaction: the checked record flows into the write.
27///
28/// The confirmation is the row's alert dialog on the list page: the
29/// Delete link opens `?delete=<key>` and the dialog's form POSTs here with
30/// `confirm=1`. Authentication comes before any DB work: the CSRF
31/// check and the confirmation marker run first, so a forged POST answers 403
32/// without opening a transaction, holding a pooled connection across the body
33/// read, or probing record existence (create/bulk-delete ordering).
34/// The dialog itself is deliberately fetch-free and policy-blind: it carries
35/// no record data and embeds only the caller's own CSRF token, and the
36/// policy/tenancy checks run against the loaded record here.
37pub(crate) fn resource_delete<R: Resource>(cx: &Cx, body: Body) -> BoxView<'_> {
38    Box::pin(HoistView::new(ThenView::<_, BoxView<'_>>::new(
39        async move {
40            gate::<R>(cx)?;
41            // The whole-resource half of the policy, before the body is read:
42            // a resource that allows no delete renders no delete chrome.
43            if !R::can_delete_any(cx) {
44                return Err(forbidden().into());
45            }
46            // Delete/bulk-delete carry no file parts: only the values half is read.
47            let values = parse_form_body(cx, body).await?.values;
48            crate::csrf::verify(cx, &values)?;
49            let confirmed = values.get("confirm").is_some_and(|v| truthy(v));
50            if !confirmed {
51                // The confirmation UI is the list-page alert dialog:
52                // the row link opens `?delete=<key>` and the dialog's form carries
53                // `confirm=1`. This route only accepts that confirmed POST, so a
54                // missing marker is a malformed client, not a user path.
55                return Err(
56                    topcoat::router::error::bad_request("delete requires confirmation").into(),
57                );
58            }
59            // Confirmed and authenticated: open the transaction only now (GH
60            // #144), fetch through the tenant-scoped query, check Policy against the
61            // loaded record, and delete inside the tx — commit makes the checked
62            // delete durable, any error rolls it back. Delete takes
63            // the edit contract: `can_view` plus
64            // `can_delete` — a record that cannot be viewed cannot be deleted
65            // by UUID-guessing the route.
66            let mut db = db(cx);
67            let mut tx = db.transaction().await.map_err(crate::db::unavailable)?;
68            let id = topcoat::router::path_param_segment(cx, "id").to_string();
69            // The delete path reads only the record's own columns:
70            // `can_view`/`can_delete` are Rust predicates over those, and
71            // `delete_record` reads the same snapshot.
72            let record = find_by_key_narrowed::<R>(cx, &id, &mut tx).await?;
73            if !R::can_view(cx, &record) {
74                return Err(forbidden().into());
75            }
76            if !R::can_delete(cx, &record) {
77                return Err(forbidden().into());
78            }
79            // The hook names what was removed: the pre-delete snapshot, since
80            // the row is gone by the time it runs.
81            let written = R::delete_record(cx, &record, &mut tx)
82                .await
83                .map(|()| vec![record]);
84            commit_write::<R, _>(cx, tx, written, Committed::deleted, "Deleted", WRITE_DELETE).await
85        },
86    )))
87}
