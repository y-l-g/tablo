1//! Record fetchers through the tenancy-scoped query seam.
2//!
3//! One row by URL key — parsed against the model's primary-key type, so a
4//! malformed or unknown id is a 404 — plus the policy-checked loader the
5//! record pages share.
6
7use topcoat::{Result, context::Cx};
8
9use crate::resource::Resource;
10
11/// Fetch one record by its URL `id` through the tenancy-scoped query seam.
12///
13/// The string id is parsed against the model's primary-key type and the PK
14/// filter is ANDed onto the tenant-scoped
15/// [`scoped_query`](crate::resource::scoped_query) (ADR-0002), so
16/// tenancy and soft-delete scoping both hold. Fetches the one row by key
17/// instead of loading every row and matching keys in memory — O(N) rows per
18/// edit/delete, leaking the whole table before the policy check.
19///
20/// A malformed or unknown id maps to 404, not a query error.
21///
22/// Runs on the caller's executor: mutation handlers pass the open framework
23/// transaction so the fetched snapshot is the checked snapshot.
24pub(crate) async fn find_by_key<R: Resource>(
25    cx: &Cx,
26    id: &str,
27    ex: &mut dyn toasty::Executor,
28) -> Result<R::Model> {
29    find_by_key_in::<R>(id, ex, || crate::resource::scoped_query::<R>(cx)).await
30}
31
32/// [`find_by_key`] for a loader that reads only the record's own columns: the
33/// same PK filter over
34/// [`scoped_query_with`](crate::resource::scoped_query_with) with an empty
35/// [`IncludeNeeds`](crate::resource::IncludeNeeds), so the edit handler and
36/// delete do not load the relations the record's list or detail page reads. A
37/// resource that overrides nothing keeps its full
38/// [`query`](crate::resource::Resource::query).
39pub(crate) async fn find_by_key_narrowed<R: Resource>(
40    cx: &Cx,
41    id: &str,
42    ex: &mut dyn toasty::Executor,
43) -> Result<R::Model> {
44    find_by_key_in::<R>(id, ex, || {
45        crate::resource::scoped_query_with::<R>(cx, &crate::resource::IncludeNeeds::default())
46    })
47    .await
48}

50/// The error a resource with a composite primary key reports when the URL or
51/// batch carries no single-key representation. `None` means the model
52/// has a single-column key.
53pub(super) fn composite_pk_error<R: Resource>() -> Option<topcoat::Error> {
54    if !crate::schema::pk_is_composite::<R::Model>() {
55        return None;
56    }
57    tracing::error!(
58        resource = R::slug(),
59        "composite primary key has no URL representation"
60    );
61    Some(topcoat::Error::from(std::io::Error::other(format!(
62        "resource '{}' has a composite primary key, which has no URL representation (GH #95)",
63        R::slug()
64    ))))
65}

67/// The shared body of [`find_by_key`] and [`find_by_key_narrowed`]: parse the
68/// URL id against the model's primary key, then fetch the one row through
69/// `seed`.
70///
71/// `seed` is a closure so the composite-PK misdeclaration is reported before
72/// the scoped query is built.
73async fn find_by_key_in<R: Resource>(
74    id: &str,
75    ex: &mut dyn toasty::Executor,
76    seed: impl FnOnce() -> Result<toasty::stmt::Query<toasty::stmt::List<R::Model>>>,
77) -> Result<R::Model> {
78    let Some(expr) = crate::schema::pk_eq_expr::<R::Model>(id) else {
79        // Composite PKs have no URL representation: fail loudly so
80        // the misconfiguration surfaces instead of 404ing every id.
81        if let Some(error) = composite_pk_error::<R>() {
82            return Err(error);
83        }
84        return Err(topcoat::router::error::not_found().into());
85    };
86    seed()?
87        .filter(expr)
88        .first()
89        .exec(&mut *ex)
90        .await
91        .map_err(crate::db::unavailable)?
92        .ok_or_else(topcoat::router::error::not_found)
93        .map_err(Into::into)
94}

96/// Load the record the request names, scoped and policy-checked.
97///
98/// Reads the `{id}` path param, loads through the tenant-scoped query (which
99/// turns an unknown *or* out-of-scope id into one 404), and returns 403 unless
100/// `can_view` accepts the loaded snapshot.
101///
102/// Callers run [`gate`](super::super::gate) first, add their own policy on top
103/// (`can_update` for the edit page), and 404 a page the resource does not
104/// declare (`R::viewed` for the view page).
105pub(crate) async fn load_viewable<R: Resource>(
106    cx: &Cx,
107    ex: &mut dyn toasty::Executor,
108) -> Result<R::Model> {
109    load_viewable_in::<R>(cx, ex, false).await
110}
111
112/// [`load_viewable`] for a loader that reads only the record's own columns:
113/// the edit page hydrates its fields from the record, so it does
114/// not load the relations the record's list or detail page reads.
115pub(crate) async fn load_viewable_narrowed<R: Resource>(
116    cx: &Cx,
117    ex: &mut dyn toasty::Executor,
118) -> Result<R::Model> {
119    load_viewable_in::<R>(cx, ex, true).await
120}
121
122/// The shared body of [`load_viewable`] and [`load_viewable_narrowed`]: read
123/// the `{id}` path param, load through the tenant-scoped query, then 403
124/// unless `can_view` accepts the snapshot.
125///
126/// `narrowed` selects the loader that reads only the record's own columns.
127async fn load_viewable_in<R: Resource>(
128    cx: &Cx,
129    ex: &mut dyn toasty::Executor,
130    narrowed: bool,
131) -> Result<R::Model> {
132    let id = topcoat::router::path_param_segment(cx, "id").to_string();
133    let record = if narrowed {
134        find_by_key_narrowed::<R>(cx, &id, ex).await?
135    } else {
136        find_by_key::<R>(cx, &id, ex).await?
137    };
138    if !R::can_view(cx, &record) {
139        return Err(topcoat::router::error::forbidden().into());
140    }
141    Ok(record)
142}
