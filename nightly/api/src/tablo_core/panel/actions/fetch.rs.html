1//! Record fetchers through the tenancy-scoped query seam.
2//!
3//! One row by URL key — parsed against the model's primary-key type, so a
4//! malformed or unknown id is a 404 — plus the policy-checked loader the
5//! record pages share.
6
7use topcoat::{Result, context::Cx};
8
9use crate::{
10    error::TabloError,
11    policy::{Ability, can},
12    resource::Resource,
13};

15/// Fetch one record by its URL `id` through the tenancy-scoped query seam.
16///
17/// The string id is parsed against the model's primary-key type and the PK
18/// filter is ANDed onto the tenant-scoped
19/// [`scoped_query`](crate::resource::scoped_query) (ADR-0002), so
20/// tenancy and soft-delete scoping both hold. Fetches the one row by key
21/// instead of loading every row and matching keys in memory — O(N) rows per
22/// edit/delete, leaking the whole table before the policy check.
23///
24/// A malformed or unknown id maps to 404, not a query error.
25///
26/// Runs on the caller's executor: mutation handlers pass the open framework
27/// transaction so the fetched snapshot is the checked snapshot.
28pub(crate) async fn find_by_key<R: Resource>(
29    cx: &Cx,
30    id: &str,
31    ex: &mut dyn toasty::Executor,
32) -> Result<R::Model> {
33    find_by_key_in::<R>(id, ex, || crate::resource::scoped_query::<R>(cx)).await
34}

36/// The error a resource with a composite primary key reports when the URL or
37/// batch carries no single-key representation. `None` means the model
38/// has a single-column key.
39pub(super) fn composite_pk_error<R: Resource>() -> Option<topcoat::Error> {
40    if !crate::schema::pk_is_composite::<R::Model>() {
41        return None;
42    }
43    tracing::error!(
44        resource = R::slug(),
45        "composite primary key has no URL representation"
46    );
47    Some(
48        TabloError::Declaration(format!(
49            "resource '{}' has a composite primary key, which has no URL representation",
50            R::slug()
51        ))
52        .into(),
53    )
54}

56/// The shared body of [`find_by_key`] and [`load_detail`]: parse the URL id
57/// against the model's primary key, then fetch the one row through `seed`.
58///
59/// `seed` is a closure so the composite-PK misdeclaration is reported before
60/// the scoped query is built.
61async fn find_by_key_in<R: Resource>(
62    id: &str,
63    ex: &mut dyn toasty::Executor,
64    seed: impl FnOnce() -> Result<toasty::stmt::Query<toasty::stmt::List<R::Model>>>,
65) -> Result<R::Model> {
66    let Some(expr) = crate::schema::pk_eq_expr::<R::Model>(id) else {
67        // Composite PKs have no URL representation: fail loudly so
68        // the misconfiguration surfaces instead of 404ing every id.
69        if let Some(error) = composite_pk_error::<R>() {
70            return Err(error);
71        }
72        return Err(topcoat::router::error::not_found().into());
73    };
74    seed()?
75        .filter(expr)
76        .first()
77        .exec(&mut *ex)
78        .await
79        .map_err(crate::error::unavailable)?
80        .ok_or_else(topcoat::router::error::not_found)
81        .map_err(Into::into)
82}

84/// Load the record the request names, scoped and policy-checked.
85///
86/// Reads the `{id}` path param, loads through the tenant-scoped query (which
87/// turns an unknown *or* out-of-scope id into one 404), and returns 403 unless
88/// `View` accepts the loaded snapshot.
89///
90/// Callers run [`gate`](super::super::gate) first and add their own policy on
91/// top (`Update` for the edit page).
92pub(crate) async fn load_viewable<R: Resource>(
93    cx: &Cx,
94    ex: &mut dyn toasty::Executor,
95) -> Result<R::Model> {
96    let id = topcoat::router::path_param_segment(cx, "id").to_string();
97    check_viewable::<R>(cx, find_by_key::<R>(cx, &id, ex).await?)
98}

100/// [`load_viewable`] for the detail page: the same key, scope and policy, over
101/// [`Resource::view_query`] so the relations the page reads off the record
102/// arrive loaded. The caller
103/// 404s a resource that declares no detail page (`R::viewed`).
104pub(crate) async fn load_detail<R: Resource>(
105    cx: &Cx,
106    ex: &mut dyn toasty::Executor,
107) -> Result<R::Model> {
108    let id = topcoat::router::path_param_segment(cx, "id").to_string();
109    let record =
110        find_by_key_in::<R>(&id, ex, || crate::resource::scoped_view_query::<R>(cx)).await?;
111    check_viewable::<R>(cx, record)
112}
113
114/// 403 unless `View` accepts the loaded snapshot.
115fn check_viewable<R: Resource>(cx: &Cx, record: R::Model) -> Result<R::Model> {
116    if !can::<R>(cx, Ability::View(&record)) {
117        return Err(topcoat::router::error::forbidden().into());
118    }
119    Ok(record)
120}
