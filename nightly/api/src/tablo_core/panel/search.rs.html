34/// A monomorphized live-search table loader, one per declared resource.
35///
36/// `#[shard]` inventory only discovers concrete fns, so the single
37/// concrete [`table_search`] shard dispatches through this registry instead
38/// of going generic. Built by [`Panel::resource`], keyed by list path.
39pub(crate) type SearchFn = Arc<
40    dyn for<'a> Fn(
41            &'a Cx,
42            String,
43            TableSignals,
44        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
45        + Send
46        + Sync,
47>;

49/// A relation table's live-search request: the owner's key in form spelling
50/// (`seed`), the record page the table renders on (`page`, the links' target
51/// and the writes' return), and whether that page shows rows without write
52/// actions (`read_only`, the detail page).
53pub(crate) struct RelationRequest {
54    pub(crate) seed: String,
55    pub(crate) page: String,
56    pub(crate) read_only: bool,
57}

59/// A monomorphized live-search relation loader, one per declared relation.
60///
61/// Built by [`Relation::has_many`](crate::resource::Relation::has_many), which
62/// is the one site that names both the child resource and the typed foreign
63/// key, and keyed by `(parent slug, child slug)`. The seed travels as a string
64/// because shard arguments cross the browser; the handler parses it back
65/// through the key's [`FormScalar`] spelling, so a tampered seed that does not
66/// parse is refused rather than loaded.
67pub(crate) type RelationSearchFn = Arc<
68    dyn for<'a> Fn(
69            &'a Cx,
70            RelationRequest,
71            TableSignals,
72        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
73        + Send
74        + Sync,
75>;

77/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
78/// then the same load + render the streamed list uses.
79///
80/// The table catches its own load errors: a tampered cursor in the query
81/// fails to decode inside the shard invocation, and the invocation must render
82/// the branded in-region `ErrorState` + retry link (via
83/// `super::list::table_error_view`, same as the streamed list) instead of
84/// erroring the shard. Auth/tenancy/policy failures still propagate — they
85/// are not table evidence.
86pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
87    Arc::new(
88        |cx: &Cx,
89         path: String,
90         signals: TableSignals|
91         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
92            Box::pin(async move {
93                gate::<R>(cx)?;
94                if !can::<R>(cx, Ability::ViewAny) {
95                    return Err(forbidden().into());
96                }
97                let table = wire_table_actions::<R>(cx, true);
98                // The GET path's parser over the client-owned query, then one
99                // normalization (GH #148): an unknown `group_by` must not echo
100                // through the retry link. The live page renders the delete
101                // dialog outside the swapped region, so the shard drops it.
102                let mut state = TableState::from_query(&signals.query.get());
103                state.delete = None;
104                state.open = None;
105                let state = table.normalize_state(&state);
106                // The retry link inside a failed table writes the same query
107                // signal the controls do, so keep a handle for it.
108                let retry_signals = signals.clone();
109                let rendered = async {
110                    let page = load_table_page::<R>(cx, &table, &state).await?;
111                    table.render_live(cx, page, &state, &path, signals).await
112                };
113                match rendered.await {
114                    Ok(view) => Ok(view),
115                    Err(error) => Ok(table_error_view::<R>(
116                        cx,
117                        &state,
118                        &error,
119                        &path,
120                        Some(&retry_signals),
121                    )),
122                }
123            })
124        },
125    )
126}

128/// Monomorphize `C`'s relation loader over the typed foreign key into a
129/// [`RelationSearchFn`]: tenancy + policy gate, then the same scoped load +
130/// live render the streamed relation uses.
131///
132/// `read_only` selects the row chrome, exactly as the page does: a tampered
133/// flag only changes which links render, never what the routes answer — every
134/// write route re-checks its own policy. `page` must sit under the panel
135/// prefix, like every page the panel serves; anything else is refused rather
136/// than reflected into the table's links. The scope narrows the child rows to
137/// the owner, so the output is a subset of what `C`'s list serves the same
138/// caller — a tampered seed discloses no row the list hides.
139pub(crate) fn relation_search_handler_for<C: Resource, T>(
140    foreign_key: FieldLens<C::Model, T>,
141) -> RelationSearchFn
142where
143    T: IntoExpr<T> + FormScalar + Send + Sync + 'static,
144{
145    Arc::new(
146        move |cx: &Cx,
147              ctx: RelationRequest,
148              signals: TableSignals|
149              -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
150            let foreign_key = foreign_key.clone();
151            Box::pin(async move {
152                gate::<C>(cx)?;
153                if !can::<C>(cx, Ability::ViewAny) {
154                    return Err(forbidden().into());
155                }
156                let prefix = panel_prefix(cx);
157                if ctx
158                    .page
159                    .strip_prefix(prefix.as_str())
160                    .is_none_or(|rest| !rest.is_empty() && !rest.starts_with(['/', '?']))
161                {
162                    return Err(topcoat::router::error::bad_request("unknown relation page").into());
163                }
164                let owner = T::parse_form(ctx.seed.trim())
165                    .map_err(|_| topcoat::router::error::bad_request("unknown relation owner"))?;
166                let scope = foreign_key.eq(owner);
167                let chrome = super::relations::relation_chrome::<C>(cx, ctx.read_only);
168                let table = super::list::wire_table::<C>(cx, true, chrome);
169                // The GET path's prefixed parser over the client-owned query,
170                // then one normalization: an unknown `group_by` must not echo
171                // through the retry link. The page renders the delete dialog
172                // outside the swapped region, so the shard drops it.
173                let mut state = TableState::from_query_prefixed(&signals.query.get(), &C::slug());
174                state.delete = None;
175                state.open = None;
176                let state = table.normalize_state(&state);
177                let table = table.returning_to(state.list_url(&ctx.page));
178                // The retry link inside a failed table writes the same query
179                // signal the controls do, so keep a handle for it.
180                let retry_signals = signals.clone();
181                let rendered = async {
182                    let rows =
183                        super::list::load_scoped_page::<C>(cx, &table, &state, scope).await?;
184                    table
185                        .render_live(cx, rows, &state, &ctx.page, signals)
186                        .await
187                };
188                match rendered.await {
189                    Ok(view) => Ok(view),
190                    Err(error) => Ok(table_error_view::<C>(
191                        cx,
192                        &state,
193                        &error,
194                        &ctx.page,
195                        Some(&retry_signals),
196                    )),
197                }
198            })
199        },
200    )
201}

203/// Resolve the registered live-search handler for `path`, answering the gate
204/// first (defense in depth): the registry lookup runs only for an
205/// authenticated request, so an unknown `path` cannot be distinguished from a
206/// registered one by an unauthenticated probe (404-vs-401 oracle).
207fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
208    crate::auth::guard(cx)?;
209    current(cx)
210        .and_then(|panel| panel.search.get(path).cloned())
211        .ok_or_else(|| topcoat::router::error::not_found().into())
212}

214/// Live table interactions: re-renders one resource's table as its signals
215/// change, morphing in place per Topcoat #392 (focus, scroll and typing
216/// survive; rows carry stable `id`s).
217///
218/// The shard owns no state: the page creates the signals ([`TableSignals`]),
219/// renders its controls against them, and passes their handles here. Search,
220/// sort, filters and pagination all write the `query` signal, so one
221/// dependency re-renders the table without a navigation or a scroll jump.
222///
223/// Every arg is untrusted shard input: `path` must name a registered list, and
224/// the query is parsed by [`TableState::from_query`] with the GET path's
225/// bounds. Authorization mirrors the list page (the resource's tenancy +
226/// `ViewAny`, row scoping via the tenant-scoped query); shard POSTs carry
227/// no CSRF token, and none is needed for this read-only rerun.
228#[shard("/_topcoat/runtime/shards/tablo-table-search")]
229pub(crate) async fn table_search(
230    cx: &Cx,
231    path: String,
232    query: topcoat::runtime::Signal<String>,
233    bulk: topcoat::runtime::Signal<String>,
234) -> Result<impl View> {
235    let entry = search_entry(cx, &path)?;
236    entry(cx, path, TableSignals { query, bulk }).await
237}

239/// The endpoint [`table_search`] is served at.
240///
241/// A shard that declares no path is served at a build-random one (topcoat#441);
242/// naming it keeps the endpoint stable across builds and legible in logs and
243/// tests. The `/tablo-` prefix separates it from a generated path, and the
244/// whole path stays under `/_topcoat/runtime`, so the panel's auth gate covers
245/// the shard and an unauthenticated rerun answers 401 rather than redirecting to
246/// the login page.

249/// `table_search_endpoint_is_the_named_path` pins the two together.
250pub(crate) const TABLE_SEARCH_PATH: &str = "/_topcoat/runtime/shards/tablo-table-search";

252/// Resolve the registered live-search relation handler for the
253/// (`parent`, `child`) pair, answering the gate first (defense in depth):
254/// the registry lookup runs only for an authenticated request, so an unknown
255/// pair cannot be distinguished from a registered one by an unauthenticated
256/// probe (404-vs-401 oracle).
257fn relation_entry(cx: &Cx, parent: &str, child: &str) -> Result<RelationSearchFn> {
258    crate::auth::guard(cx)?;
259    current(cx)
260        .and_then(|panel| {
261            panel
262                .relations
263                .get(&(parent.to_string(), child.to_string()))
264                .cloned()
265        })
266        .ok_or_else(|| topcoat::router::error::not_found().into())
267}

269/// Live relation-table interactions: re-renders one record page's relation
270/// table as its signals change, morphing in place per Topcoat #392 (focus,
271/// scroll and typing survive; rows carry stable `id`s).
272///
273/// The shard owns no state: the page creates the signals ([`TableSignals`])
274/// keyed by page and relation key, renders the search and filter bars against
275/// them above the swapped region, and passes their handles here. Search, sort,
276/// filters and pagination all write the `query` signal, so one dependency
277/// re-renders the table without a navigation or a scroll jump.
278///
279/// Every arg is untrusted shard input: `scope` must name a registered
280/// (`parent`, `child`) pair and the owner's seed as `{parent}/{child}/{seed}`,
281/// `page` must sit under the panel prefix, and the query is parsed by
282/// [`TableState::from_query_prefixed`] with the GET path's bounds.
283/// Authorization mirrors the relation table (the related resource's tenancy +
284/// `ViewAny`, row scoping via the tenant-scoped query plus the owner's
285/// scope); shard POSTs carry no CSRF token, and none is needed for this
286/// read-only rerun.
287#[shard("/_topcoat/runtime/shards/tablo-table-relation-search")]
288pub(crate) async fn table_relation_search(
289    cx: &Cx,
290    scope: String,
291    page: String,
292    read_only: bool,
293    query: topcoat::runtime::Signal<String>,
294    bulk: topcoat::runtime::Signal<String>,
295) -> Result<impl View> {
296    // Slugs never carry `/` (the panel refuses them at registration), so the
297    // pair splits off the front and the seed keeps the rest, slashes
298    // included; anything else misses the registry as 404.
299    let mut parts = scope.splitn(3, '/');
300    let (parent, child, seed) = (
301        parts.next().unwrap_or_default(),
302        parts.next().unwrap_or_default(),
303        parts.next().unwrap_or_default(),
304    );
305    let entry = relation_entry(cx, parent, child)?;
306    entry(
307        cx,
308        RelationRequest {
309            seed: seed.to_string(),
310            page,
311            read_only,
312        },
313        TableSignals { query, bulk },
314    )
315    .await
316}

327/// The largest shard request [`ShardPanel`] reads: arguments and signal
328/// values, a list path and a query string each, far below it.
329const MAX_SHARD_BYTES: usize = 64 * 1024;
330
331/// Puts the panel a live table's shard re-renders for on the request.
332///
333/// A shard is served at one runtime path for every panel, so no panel's
334/// prefix layer runs for its re-render. The page names the panel in the
335/// shard's arguments — the list path [`table_search`] takes, the record page
336/// [`table_relation_search`] takes — and this layer reads that argument from
337/// the body Topcoat's runtime posts (`{"args": [..], ..}`), finds the panel
338/// serving it, and hands the request on with that panel and the same bytes.
339/// An argument no panel serves leaves the request as it came, and the shard's
340/// registry lookup answers 404.
341pub(crate) struct ShardPanel {
342    path: PathBuf,
343    /// The index of the argument holding a path under the panel's prefix.
344    arg: usize,
345}
346
347impl ShardPanel {
348    pub(crate) fn new(path: &str, arg: usize) -> Self {
349        Self {
350            path: route_path(path),
351            arg,
352        }
353    }
354}
355
356/// The part of a shard request [`ShardPanel`] reads.
357#[derive(serde::Deserialize)]
358struct ShardArgs {
359    args: Vec<serde_json::Value>,
360}
361
362impl Layer for ShardPanel {
363    fn path(&self) -> Option<&Path> {
364        Some(&self.path)
365    }
366
367    fn handle<'a>(&'a self, cx: &'a Cx, body: Body, next: Next<'a>) -> LayerFuture<'a> {
368        Box::pin(async move {
369            let bytes = http_body_util::Limited::new(body, MAX_SHARD_BYTES)
370                .collect()
371                .await
372                .map_err(|_| content_too_large())?
373                .to_bytes();
374            let panel = serde_json::from_slice::<ShardArgs>(&bytes)
375                .ok()
376                .and_then(|request| request.args.get(self.arg)?.as_str().map(str::to_owned))
377                .and_then(|path: String| panels(cx)?.by_path(&path).cloned());
378            let body = Body::from(bytes);
379            match panel {
380                Some(panel) => next.run(&cx.with(CurrentPanel(panel)), body).await,
381                None => next.run(cx, body).await,
382            }
383        })
384    }
385}
