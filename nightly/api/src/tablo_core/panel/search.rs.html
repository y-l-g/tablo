22/// One live-table shard invocation: the list path the page asks for
23/// and the interaction signals it owns.
24///
25/// The one argument the shard's *handler* takes. The `#[shard]` entry packs
26/// its wire parameters into this, and every seam below — the registry lookup,
27/// the state rebuild, the load, the render, and the retry link — reads the
28/// same value, so a new interaction dimension never changes a signature
29/// here.
30pub(crate) struct TableSearchArgs {
31    /// The list path to rerun: resolved through the registry (an allow-list,
32    /// never a raw route).
33    pub(crate) path: String,
34    /// The page's signals, untrusted by the time the shard reads them back.
35    pub(crate) signals: TableSignals,
36}

38/// A monomorphized live-search table loader, one per declared resource.
39///
40/// `#[shard]` inventory only discovers concrete fns, so the single
41/// concrete [`table_search`] shard dispatches through this registry instead
42/// of going generic. Built by [`Panel::resource`], keyed by list path.
43pub(crate) type SearchFn = Arc<
44    dyn for<'a> Fn(
45            &'a Cx,
46            TableSearchArgs,
47        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
48        + Send
49        + Sync,
50>;
51
52/// Live-search handlers installed on the app context by [`Panel::build`].
53#[derive(Clone, Default)]
54pub(crate) struct SearchRegistry(pub(crate) HashMap<String, SearchFn>);

56/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
57/// then the same load + render the streamed list uses.
58///
59/// The table catches its own load errors: a tampered `after=` /
60/// `before=` signal fails to decode inside the shard invocation, and the
61/// invocation must render the branded in-region `ErrorState` + retry link
62/// (via `super::list::table_error_view`, same as the streamed list) instead of
63/// erroring the shard. Auth/tenancy/policy failures still propagate — they
64/// are not table evidence.
65pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
66    Arc::new(
67        |cx: &Cx,
68         args: TableSearchArgs|
69         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
70            Box::pin(async move {
71                gate::<R>(cx)?;
72                if !R::can_view_any(cx) {
73                    return Err(forbidden().into());
74                }
75                let table = wire_table_actions::<R>(cx, true);
76                let TableSearchArgs { path, signals } = args;
77                // One shared bound and one normalization per request (GH #148):
78                // `TableSignals::to_state` applies the same `q` clamp and
79                // `filters` bound the GET path applies. The shard `group_by` arg
80                // is client input — an
81                // unknown value must not echo through the retry link.
82                // The render below takes the proof and does not
83                // normalize again.
84                let state = table.normalize_state(&signals.to_state());
85                // The retry link inside a failed table writes the same signals
86                // the toolbar does, so keep a handle for it.
87                let retry_signals = signals.clone();
88                let rendered = async {
89                    let page = load_table_page::<R>(cx, &table, &state).await?;
90                    table
91                        .render_live_normalized(cx, page, &state, &path, signals)
92                        .await
93                };
94                match rendered.await {
95                    Ok(view) => Ok(view),
96                    Err(error) => Ok(table_error_view::<R>(
97                        cx,
98                        &state,
99                        &error,
100                        &path,
101                        Some(&retry_signals),
102                    )),
103                }
104            })
105        },
106    )
107}

109/// Resolve the registered live-search handler for `path`, answering the gate
110/// first (defense in depth): the registry lookup runs only for an
111/// authenticated request, so an unknown `path` cannot be distinguished from a
112/// registered one by an unauthenticated probe (404-vs-401 oracle).
113fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
114    enforce_auth(cx)?;
115    topcoat::context::try_app_context::<SearchRegistry>(cx)
116        .and_then(|reg| reg.0.get(path).cloned())
117        .ok_or_else(|| topcoat::router::error::not_found().into())
118}

120/// Live table interactions: re-renders one resource's table as its signals
121/// change, morphing in place per Topcoat #392 (focus, scroll and typing
122/// survive; rows carry stable `id`s).
123///
124/// The shard owns no state: the page creates the signals ([`TableSignals`]),
125/// renders the toolbar against them, and passes their handles here. Search,
126/// sort, filters and pagination all write those signals, so one dependency
127/// graph re-renders the table without a navigation or a scroll jump.
128///
129/// Every arg is untrusted shard input: `path` must name a registered list, and
130/// every signal value is clamped or re-parsed through
131/// [`TableSignals::to_state`] like the GET path. Authorization mirrors the list
132/// page (`requires_tenant` + `can_view_any`, row scoping via the tenant-scoped
133/// query); shard POSTs carry no CSRF token, and none is needed for this
134/// read-only rerun.
135///
136/// The module exists only to carry `allow(too_many_arguments)`: the shard's
137/// arity *is* the interaction list. The wire stays scalar by choice — a struct
138/// cannot travel as a shard argument (topcoat requires the `expr!` vocabulary,
139/// and a struct has no `Surrogated` surrogate the browser's `cx.hydrate` can
140/// rebuild), and packing the dimensions into a list would couple the browser to
141/// this server's ordering.
142#[allow(clippy::too_many_arguments)]
143mod shard_body {
144    use super::*;
145
146    #[shard("/_topcoat/runtime/shards/tablo-table-search")]
147    pub(crate) async fn table_search(
148        cx: &Cx,
149        path: String,
150        q: topcoat::runtime::Signal<String>,
151        filters: topcoat::runtime::Signal<String>,
152        sort: topcoat::runtime::Signal<String>,
153        dir: topcoat::runtime::Signal<String>,
154        cursor: topcoat::runtime::Signal<String>,
155        group_by: topcoat::runtime::Signal<String>,
156        bulk: topcoat::runtime::Signal<String>,
157    ) -> Result<impl View> {
158        let entry = search_entry(cx, &path)?;
159        // One argument struct from here down: the handler owns the
160        // wire arity, nothing below it does.
161        entry(
162            cx,
163            TableSearchArgs {
164                path,
165                signals: TableSignals {
166                    q,
167                    filters,
168                    sort,
169                    dir,
170                    cursor,
171                    group_by,
172                    bulk,
173                },
174            },
175        )
176        .await
177    }
178}
179pub(crate) use shard_body::table_search;

181/// The endpoint [`table_search`] is served at.
182///
183/// A shard that declares no path is served at a build-random one (topcoat#441);
184/// naming it keeps the endpoint stable across builds and legible in logs and
185/// tests. The `/tablo-` prefix separates it from a generated path, and the
186/// whole path stays under `/_topcoat/runtime`, so the panel's auth gate covers
187/// the shard and an unauthenticated rerun answers 401 rather than redirecting to
188/// the login page.
189///
190/// The literal in [`table_search`]'s attribute is the same path;
191/// `table_search_endpoint_is_the_named_path` pins the two together.
192#[cfg(test)]
