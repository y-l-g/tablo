22/// A monomorphized live-search table loader, one per declared resource.
23///
24/// `#[shard]` inventory only discovers concrete fns, so the single
25/// concrete [`table_search`] shard dispatches through this registry instead
26/// of going generic. Built by [`Panel::resource`], keyed by list path.
27pub(crate) type SearchFn = Arc<
28    dyn for<'a> Fn(
29            &'a Cx,
30            String,
31            TableSignals,
32        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
33        + Send
34        + Sync,
35>;
36
37/// Live-search handlers installed on the app context by [`Panel::build`].
38#[derive(Clone, Default)]
39pub(crate) struct SearchRegistry(pub(crate) HashMap<String, SearchFn>);
40
41/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
42/// then the same load + render the streamed list uses.
43///
44/// The table catches its own load errors: a tampered cursor in the query
45/// fails to decode inside the shard invocation, and the invocation must render
46/// the branded in-region `ErrorState` + retry link (via
47/// `super::list::table_error_view`, same as the streamed list) instead of
48/// erroring the shard. Auth/tenancy/policy failures still propagate — they
49/// are not table evidence.
50pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
51    Arc::new(
52        |cx: &Cx,
53         path: String,
54         signals: TableSignals|
55         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
56            Box::pin(async move {
57                gate::<R>(cx)?;
58                if !R::can_view_any(cx) {
59                    return Err(forbidden().into());
60                }
61                let table = wire_table_actions::<R>(cx, true);
62                // The GET path's parser over the client-owned query, then one
63                // normalization (GH #148): an unknown `group_by` must not echo
64                // through the retry link. The live page renders the delete
65                // dialog outside the swapped region, so the shard drops it.
66                let mut state = TableState::from_query(&signals.query.get());
67                state.delete = None;
68                state.open = None;
69                let state = table.normalize_state(&state);
70                // The retry link inside a failed table writes the same query
71                // signal the controls do, so keep a handle for it.
72                let retry_signals = signals.clone();
73                let rendered = async {
74                    let page = load_table_page::<R>(cx, &table, &state).await?;
75                    table.render_live(cx, page, &state, &path, signals).await
76                };
77                match rendered.await {
78                    Ok(view) => Ok(view),
79                    Err(error) => Ok(table_error_view::<R>(
80                        cx,
81                        &state,
82                        &error,
83                        &path,
84                        Some(&retry_signals),
85                    )),
86                }
87            })
88        },
89    )
90}

92/// Resolve the registered live-search handler for `path`, answering the gate
93/// first (defense in depth): the registry lookup runs only for an
94/// authenticated request, so an unknown `path` cannot be distinguished from a
95/// registered one by an unauthenticated probe (404-vs-401 oracle).
96fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
97    enforce_auth(cx)?;
98    topcoat::context::try_app_context::<SearchRegistry>(cx)
99        .and_then(|reg| reg.0.get(path).cloned())
100        .ok_or_else(|| topcoat::router::error::not_found().into())
101}

103/// Live table interactions: re-renders one resource's table as its signals
104/// change, morphing in place per Topcoat #392 (focus, scroll and typing
105/// survive; rows carry stable `id`s).
106///
107/// The shard owns no state: the page creates the signals ([`TableSignals`]),
108/// renders its controls against them, and passes their handles here. Search,
109/// sort, filters and pagination all write the `query` signal, so one
110/// dependency re-renders the table without a navigation or a scroll jump.
111///
112/// Every arg is untrusted shard input: `path` must name a registered list, and
113/// the query is parsed by [`TableState::from_query`] with the GET path's
114/// bounds. Authorization mirrors the list page (`requires_tenant` +
115/// `can_view_any`, row scoping via the tenant-scoped query); shard POSTs carry
116/// no CSRF token, and none is needed for this read-only rerun.
117#[shard("/_topcoat/runtime/shards/tablo-table-search")]
118pub(crate) async fn table_search(
119    cx: &Cx,
120    path: String,
121    query: topcoat::runtime::Signal<String>,
122    bulk: topcoat::runtime::Signal<String>,
123) -> Result<impl View> {
124    let entry = search_entry(cx, &path)?;
125    entry(cx, path, TableSignals { query, bulk }).await
126}

128/// The endpoint [`table_search`] is served at.
129///
130/// A shard that declares no path is served at a build-random one (topcoat#441);
131/// naming it keeps the endpoint stable across builds and legible in logs and
132/// tests. The `/tablo-` prefix separates it from a generated path, and the
133/// whole path stays under `/_topcoat/runtime`, so the panel's auth gate covers
134/// the shard and an unauthenticated rerun answers 401 rather than redirecting to
135/// the login page.
136///
137/// The literal in [`table_search`]'s attribute is the same path;
138/// `table_search_endpoint_is_the_named_path` pins the two together.
139#[cfg(test)]
140pub(crate) const TABLE_SEARCH_PATH: &str = "/_topcoat/runtime/shards/tablo-table-search";
