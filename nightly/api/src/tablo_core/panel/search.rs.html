27/// A monomorphized live-search table loader, one per declared resource.
28///
29/// `#[shard]` inventory only discovers concrete fns, so the single
30/// concrete [`table_search`] shard dispatches through this registry instead
31/// of going generic. Built by [`Panel::resource`], keyed by list path.
32pub(crate) type SearchFn = Arc<
33    dyn for<'a> Fn(
34            &'a Cx,
35            String,
36            TableSignals,
37        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
38        + Send
39        + Sync,
40>;

42/// Live-search handlers installed on the app context by [`Panel::build`].
43#[derive(Clone, Default)]
44pub(crate) struct SearchRegistry(pub(crate) HashMap<String, SearchFn>);

46/// A relation table's live-search request: the owner's key in form spelling
47/// (`seed`), the record page the table renders on (`page`, the links' target
48/// and the writes' return), and whether that page shows rows without write
49/// actions (`read_only`, the detail page).
50pub(crate) struct RelationRequest {
51    pub(crate) seed: String,
52    pub(crate) page: String,
53    pub(crate) read_only: bool,
54}

56/// A monomorphized live-search relation loader, one per declared relation.
57///
58/// Built by [`Relation::has_many`](crate::resource::Relation::has_many), which
59/// is the one site that names both the child resource and the typed foreign
60/// key, and keyed by `(parent slug, child slug)`. The seed travels as a string
61/// because shard arguments cross the browser; the handler parses it back
62/// through the key's [`FormScalar`] spelling, so a tampered seed that does not
63/// parse is refused rather than loaded.

65    dyn for<'a> Fn(
66            &'a Cx,
67            RelationRequest,
68            TableSignals,
69        ) -> Pin<Box<dyn Future<Output = Result<BoxView<'a>>> + Send + 'a>>
70        + Send
71        + Sync,
72>;

79/// Monomorphize `R`'s table loader into a [`SearchFn`]: tenancy + policy gate,
80/// then the same load + render the streamed list uses.
81///
82/// The table catches its own load errors: a tampered cursor in the query
83/// fails to decode inside the shard invocation, and the invocation must render
84/// the branded in-region `ErrorState` + retry link (via
85/// `super::list::table_error_view`, same as the streamed list) instead of
86/// erroring the shard. Auth/tenancy/policy failures still propagate — they
87/// are not table evidence.
88pub(crate) fn search_handler_for<R: Resource>() -> SearchFn {
89    Arc::new(
90        |cx: &Cx,
91         path: String,
92         signals: TableSignals|
93         -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
94            Box::pin(async move {
95                gate::<R>(cx)?;
96                if !R::can_view_any(cx) {
97                    return Err(forbidden().into());
98                }
99                let table = wire_table_actions::<R>(cx, true);
100                // The GET path's parser over the client-owned query, then one
101                // normalization (GH #148): an unknown `group_by` must not echo
102                // through the retry link. The live page renders the delete
103                // dialog outside the swapped region, so the shard drops it.
104                let mut state = TableState::from_query(&signals.query.get());
105                state.delete = None;
106                state.open = None;
107                let state = table.normalize_state(&state);
108                // The retry link inside a failed table writes the same query
109                // signal the controls do, so keep a handle for it.
110                let retry_signals = signals.clone();
111                let rendered = async {
112                    let page = load_table_page::<R>(cx, &table, &state).await?;
113                    table.render_live(cx, page, &state, &path, signals).await
114                };
115                match rendered.await {
116                    Ok(view) => Ok(view),
117                    Err(error) => Ok(table_error_view::<R>(
118                        cx,
119                        &state,
120                        &error,
121                        &path,
122                        Some(&retry_signals),
123                    )),
124                }
125            })
126        },
127    )
128}

130/// Monomorphize `C`'s relation loader over the typed foreign key into a
131/// [`RelationSearchFn`]: tenancy + policy gate, then the same scoped load +
132/// live render the streamed relation uses.
133///
134/// `read_only` selects the row chrome, exactly as the page does: a tampered
135/// flag only changes which links render, never what the routes answer — every
136/// write route re-checks its own policy. `page` must sit under the panel
137/// prefix, like every page the panel serves; anything else is refused rather
138/// than reflected into the table's links. The scope narrows the child rows to
139/// the owner, so the output is a subset of what `C`'s list serves the same
140/// caller — a tampered seed discloses no row the list hides.
141pub(crate) fn relation_search_handler_for<C: Resource, T>(
142    foreign_key: FieldLens<C::Model, T>,
143) -> RelationSearchFn
144where
145    T: IntoExpr<T> + FormScalar + Send + Sync + 'static,
146{
147    Arc::new(
148        move |cx: &Cx,
149              ctx: RelationRequest,
150              signals: TableSignals|
151              -> Pin<Box<dyn Future<Output = Result<BoxView<'_>>> + Send + '_>> {
152            let foreign_key = foreign_key.clone();
153            Box::pin(async move {
154                gate::<C>(cx)?;
155                if !C::can_view_any(cx) {
156                    return Err(forbidden().into());
157                }
158                let prefix = panel_prefix(cx);
159                if ctx
160                    .page
161                    .strip_prefix(prefix.as_str())
162                    .is_none_or(|rest| !rest.is_empty() && !rest.starts_with(['/', '?']))
163                {
164                    return Err(topcoat::router::error::bad_request("unknown relation page").into());
165                }
166                let owner = T::parse_form(ctx.seed.trim())
167                    .map_err(|_| topcoat::router::error::bad_request("unknown relation owner"))?;
168                let scope = foreign_key.eq(owner);
169                let chrome = super::relations::relation_chrome::<C>(cx, ctx.read_only);
170                let table = super::list::wire_table::<C>(cx, true, chrome);
171                // The GET path's prefixed parser over the client-owned query,
172                // then one normalization: an unknown `group_by` must not echo
173                // through the retry link. The page renders the delete dialog
174                // outside the swapped region, so the shard drops it.
175                let mut state = TableState::from_query_prefixed(&signals.query.get(), &C::slug());
176                state.delete = None;
177                state.open = None;
178                let state = table.normalize_state(&state);
179                let table = table.returning_to(state.list_url(&ctx.page));
180                // The retry link inside a failed table writes the same query
181                // signal the controls do, so keep a handle for it.
182                let retry_signals = signals.clone();
183                let rendered = async {
184                    let rows =
185                        super::list::load_scoped_page::<C>(cx, &table, &state, scope).await?;
186                    table
187                        .render_live(cx, rows, &state, &ctx.page, signals)
188                        .await
189                };
190                match rendered.await {
191                    Ok(view) => Ok(view),
192                    Err(error) => Ok(table_error_view::<C>(
193                        cx,
194                        &state,
195                        &error,
196                        &ctx.page,
197                        Some(&retry_signals),
198                    )),
199                }
200            })
201        },
202    )
203}

205/// Resolve the registered live-search handler for `path`, answering the gate
206/// first (defense in depth): the registry lookup runs only for an
207/// authenticated request, so an unknown `path` cannot be distinguished from a
208/// registered one by an unauthenticated probe (404-vs-401 oracle).
209fn search_entry(cx: &Cx, path: &str) -> Result<SearchFn> {
210    enforce_auth(cx)?;
211    topcoat::context::try_app_context::<SearchRegistry>(cx)
212        .and_then(|reg| reg.0.get(path).cloned())
213        .ok_or_else(|| topcoat::router::error::not_found().into())
214}

216/// Live table interactions: re-renders one resource's table as its signals
217/// change, morphing in place per Topcoat #392 (focus, scroll and typing
218/// survive; rows carry stable `id`s).
219///
220/// The shard owns no state: the page creates the signals ([`TableSignals`]),
221/// renders its controls against them, and passes their handles here. Search,
222/// sort, filters and pagination all write the `query` signal, so one
223/// dependency re-renders the table without a navigation or a scroll jump.
224///
225/// Every arg is untrusted shard input: `path` must name a registered list, and
226/// the query is parsed by [`TableState::from_query`] with the GET path's
227/// bounds. Authorization mirrors the list page (`requires_tenant` +
228/// `can_view_any`, row scoping via the tenant-scoped query); shard POSTs carry
229/// no CSRF token, and none is needed for this read-only rerun.
230#[shard("/_topcoat/runtime/shards/tablo-table-search")]
231pub(crate) async fn table_search(
232    cx: &Cx,
233    path: String,
234    query: topcoat::runtime::Signal<String>,
235    bulk: topcoat::runtime::Signal<String>,
236) -> Result<impl View> {
237    let entry = search_entry(cx, &path)?;
238    entry(cx, path, TableSignals { query, bulk }).await
239}

241/// The endpoint [`table_search`] is served at.
242///
243/// A shard that declares no path is served at a build-random one (topcoat#441);
244/// naming it keeps the endpoint stable across builds and legible in logs and
245/// tests. The `/tablo-` prefix separates it from a generated path, and the
246/// whole path stays under `/_topcoat/runtime`, so the panel's auth gate covers
247/// the shard and an unauthenticated rerun answers 401 rather than redirecting to
248/// the login page.
249///
250/// The literal in [`table_search`]'s attribute is the same path;
251/// `table_search_endpoint_is_the_named_path` pins the two together.
252#[cfg(test)]
253pub(crate) const TABLE_SEARCH_PATH: &str = "/_topcoat/runtime/shards/tablo-table-search";

255/// Resolve the registered live-search relation handler for the
256/// (`parent`, `child`) pair, answering the gate first (defense in depth):
257/// the registry lookup runs only for an authenticated request, so an unknown
258/// pair cannot be distinguished from a registered one by an unauthenticated
259/// probe (404-vs-401 oracle).
260fn relation_entry(cx: &Cx, parent: &str, child: &str) -> Result<RelationSearchFn> {
261    enforce_auth(cx)?;
262    topcoat::context::try_app_context::<RelationRegistry>(cx)
263        .and_then(|reg| reg.0.get(&(parent.to_string(), child.to_string())).cloned())
264        .ok_or_else(|| topcoat::router::error::not_found().into())
265}

267/// Live relation-table interactions: re-renders one record page's relation
268/// table as its signals change, morphing in place per Topcoat #392 (focus,
269/// scroll and typing survive; rows carry stable `id`s).
270///
271/// The shard owns no state: the page creates the signals ([`TableSignals`])
272/// keyed by page and relation key, renders the search and filter bars against
273/// them above the swapped region, and passes their handles here. Search, sort,
274/// filters and pagination all write the `query` signal, so one dependency
275/// re-renders the table without a navigation or a scroll jump.
276///
277/// Every arg is untrusted shard input: `scope` must name a registered
278/// (`parent`, `child`) pair and the owner's seed as `{parent}/{child}/{seed}`,
279/// `page` must sit under the panel prefix, and the query is parsed by
280/// [`TableState::from_query_prefixed`] with the GET path's bounds.
281/// Authorization mirrors the relation table (`requires_tenant` +
282/// `can_view_any`, row scoping via the tenant-scoped query plus the owner's
283/// scope); shard POSTs carry no CSRF token, and none is needed for this
284/// read-only rerun.
285#[shard("/_topcoat/runtime/shards/tablo-table-relation-search")]
286pub(crate) async fn table_relation_search(
287    cx: &Cx,
288    scope: String,
289    page: String,
290    read_only: bool,
291    query: topcoat::runtime::Signal<String>,
292    bulk: topcoat::runtime::Signal<String>,
293) -> Result<impl View> {
294    // Slugs never carry `/` (the panel refuses them at registration), so the
295    // pair splits off the front and the seed keeps the rest, slashes
296    // included; anything else misses the registry as 404.
297    let mut parts = scope.splitn(3, '/');
298    let (parent, child, seed) = (
299        parts.next().unwrap_or_default(),
300        parts.next().unwrap_or_default(),
301        parts.next().unwrap_or_default(),
302    );
303    let entry = relation_entry(cx, parent, child)?;
304    entry(
305        cx,
306        RelationRequest {
307            seed: seed.to_string(),
308            page,
309            read_only,
310        },
311        TableSignals { query, bulk },
312    )
313    .await
314}
