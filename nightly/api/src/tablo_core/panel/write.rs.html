1//! The write tail every mutation shares: create, update, delete, and bulk
2//! delete end in [`commit_write`].
3
4use topcoat::{context::Cx, router::error::see_other, view::BoxView};

8    notification::{Notification, notify_write_failure, set_notification},
9    resource::{Committed, Resource},
10};

12/// Commit the transaction, run the after-commit hook on what the record fn
13/// wrote, and redirect to the list with the success flash. A failed write or
14/// commit maps to the caller's failure toast and an opaque error.
15///
16/// `written` is the record fn's result, carrying what the hook receives;
17/// `committed` names the mutation, `note` the success flash, and `failure`
18/// the toast.
19pub(crate) async fn commit_write<'a, R: Resource, T>(
20    cx: &'a Cx,
21    tx: toasty::Transaction<'_>,
22    written: Result<T, topcoat::Error>,
23    committed: impl FnOnce(T) -> Committed<R::Model>,
24    note: &'static str,
25    failure: &'static str,
26) -> Result<BoxView<'a>, topcoat::Error> {
27    match written {
28        Ok(value) => match tx.commit().await {
29            Ok(()) => {
30                // Post-commit, so the effect cannot survive a rollback, and
31                // the tx is gone, so the hook may open its own handle.
32                crate::resource::run_after_commit::<R>(cx, committed(value)).await;
33                Err(redirect_after_write::<R>(cx, note))
34            }
35            Err(error) => {
36                notify_write_failure(cx, failure);
37                Err(crate::error::unavailable(error))
38            }
39        },
40        // A record fn's error is not echoed raw: the driver's text goes to the
41        // log through the opaque mapping, and an app-authored error keeps its
42        // own. On a create or update this includes a unique violation that
43        // slipped past the app-side check (a concurrent write): Toasty exposes
44        // no unique-violation predicate (upstream gap #117), so it cannot be
45        // classified as an inline field error here.
46        Err(error) => {
47            notify_write_failure(cx, failure);
48            Err(crate::error::driver_failure(
49                error,
50                crate::error::DATABASE_UNAVAILABLE,
51            ))
52        }
53    }
54}

56/// Post/Redirect/Get with a flash notification. The browser follows with a
57/// GET, and the flash cookie rides the error response (Topcoat flushes
58/// `Set-Cookie` on `Err` too, topcoat#408), so every mutation redirects the
59/// same way.
60fn redirect_after_write<R: Resource>(cx: &Cx, note: &'static str) -> topcoat::Error {
61    set_notification(cx, Notification::success(note));
62    see_other(list_url(cx, &R::slug())).into()
63}
