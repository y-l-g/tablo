27/// `extra` plus the attribute that sends a sidebar link through runtime
28/// navigation. The menu button writes the `href` itself, so this carries none.
29fn sidebar_link(cx: &Cx, mut extra: Attributes) -> Attributes {
30    let mut attrs = crate::resource::runtime_link(cx, "");
31    attrs.remove("href");
32    extra.extend(attrs);
33    extra
34}

36/// The value of the request's `name` cookie.
37///
38/// Parsed from the raw `Cookie` header on purpose: `topcoat::cookie::cookies`
39/// panics when the cookie router layer is absent (tests, minimal routers), and
40/// the shell must render everywhere.
41fn request_cookie(cx: &Cx, name: &str) -> Option<String> {
42    let header = try_request_context::<http::request::Parts>(cx)?
43        .headers
44        .get(COOKIE)?
45        .to_str()
46        .ok()?;
47    header.split(';').find_map(|part| {
48        let (key, value) = part.trim().split_once('=')?;
49        (key == name).then(|| value.to_string())
50    })
51}

53/// Branding for the admin shell (panel header + sidebar header).
54#[derive(Debug, Clone)]
55pub struct Brand {
56    /// Display name (e.g. `"Acme"`).
57    pub name: String,
58    /// Optional logo URL (e.g. `"/logo.svg"`). Rendered as an `<img>` when present.
59    pub logo: Option<String>,
60}

62impl Brand {
63    /// Create a brand with the given name (GH #102: surrounding whitespace is
64    /// trimmed so `" Acme "` cannot break the `flex h-16` header).
65    pub fn new(name: impl Into<String>) -> Self {
66        Self {
67            name: name.into().trim().to_string(),
68            logo: None,
69        }
70    }
71
72    /// Attach a logo URL (GH #102: blank values are ignored so an empty
73    /// `logo("")` falls back to the name-only render instead of a
74    /// broken-image icon).
75    pub fn logo(mut self, logo: impl Into<String>) -> Self {
76        let logo = logo.into().trim().to_string();
77        if !logo.is_empty() {
78            self.logo = Some(logo);
79        }
80        self
81    }

84/// Whether the shell starts in dark mode for a visitor with no stored choice.
85/// Persisted via `theme.js` (`localStorage` + `theme` cookie).
86///
87/// Precedence: the stored preference wins in **both** directions over this
88/// build-time default. The server renders `<html class>` from the `theme`
89/// cookie when it carries `dark` or `light`, and from this default otherwise;
90/// the blocking `theme_init_script` then applies `localStorage`, then the
91/// cookie, with this default as its fallback. Rendering the cookie matters for
92/// runtime navigation, which copies `<html>`'s attributes from the next page
93/// and runs no script (GH #184, GH #395).
94#[derive(Debug, Clone, Copy)]
95pub struct DarkMode(pub bool);

146            }

149            Ok(view! {
150                cx =>
151                <div class="flex items-center gap-2 font-semibold text-foreground">
152                    (name)
153                </div>
154            }
155            .boxed())
156        }

159    async fn sidebar_navigation<'a>(
160        cx: &'a Cx,
161        nav_items: &[NavigationItem],
162        current_path: &str,
163        mobile_open: Signal<bool>,
164    ) -> Result<BoxView<'a>> {
165        use tablo_ui::{
166            sidebar_group, sidebar_group_content, sidebar_group_label, sidebar_menu,
167            sidebar_menu_button, sidebar_menu_item,
168        };
169        let mut nav_items = nav_items.to_vec();
170        // Stable order: explicit `order` first, declaration order
171        // breaking ties — a resource's `navigation()` override interleaves by
172        // setting it.
173        nav_items.sort_by_key(|item| item.order);
174        // A Panel resolves every item it owns (`Panel::resource`, `page`,
175        // `home`); one that
176        // reaches the sidebar unresolved has no URL to render, which is a
177        // framework bug rather than user error.
178        debug_assert!(
179            nav_items.iter().all(|item| item.url().is_some()),
180            "navigation items are resolved by the Panel that owns them"
181        );
182        // One active entry: a home entry at the bare prefix matches every path
183        // under it, so the longest matching URL wins, and the first such entry
184        // among any that share it.
185        let active = nav_items
186            .iter()
187            .enumerate()
188            .filter(|(_, item)| item.is_current_path(current_path))
189            .min_by_key(|(_, item)| std::cmp::Reverse(item.url().map_or(0, str::len)))
190            .map(|(index, _)| index);
191
192        Ok(view! {
193            cx =>
194            sidebar_group(
195                sidebar_group_label("Navigation")
196                sidebar_group_content(
197                    sidebar_menu(
198                        for (index, item) in nav_items.iter().enumerate() {
199                            let is_active = active == Some(index);
200                            let attrs = sidebar_link(
201                                cx,
202                                attributes! {
203                                    // Tapping a link in the mobile sheet closes
204                                    // it; on desktop the navigation is the effect.
205                                    @click=$(|_e: Event| mobile_open.set(false))
206                                },
207                            );
208                            sidebar_menu_item(
209                                sidebar_menu_button(
210                                    active: is_active,
211                                    href: item.url(),
212                                    tooltip: Some(item.label.as_str()),
213                                    attrs: attrs,
214                                    <span>(item.label.clone())</span>
215                                )
216                            )
217                        }
218                    )
219                )
220            )
221        }
222        .boxed())
223    }

225    /// Whether the persisted `sidebar_state` cookie asks for an expanded
226    /// desktop sidebar (default: expanded).
227    ///
228    /// The value only seeds the runtime
229    /// signal's initial `data-state`; after hydration the browser owns the
230    /// state, and `assets/sidebar.js` mirrors changes back to the cookie.
231    fn sidebar_starts_open(cx: &Cx) -> bool {
232        request_cookie(cx, "sidebar_state").as_deref() != Some("collapsed")
233    }

235    /// Render the Filament-grade Shell that frames every admin page.
236    ///
237    /// Composes Topcoat's upstream `sidebar` primitives (ADR-0007): the
238    /// desktop panel and the mobile sheet share one navigation rendering, and
239    /// the open state lives in runtime signals — `open` seeds from the
240    /// `sidebar_state` cookie for the first paint, the triggers carry
241    /// `@click` handlers, and `assets/sidebar.js` persists changes back to
242    /// the cookie. Includes dark-mode toggle (Ghost button, persisted by
243    /// `theme.js` to `localStorage` + the `theme` cookie) and the toast stack
244    /// (shadcn/Sonner surface, fixed bottom-right). Additive `class` is
245    /// allowed on the outer container only (narrow seam).
246    ///
247    /// Asset note: desktop persistence needs `assets/sidebar.js`
248    /// (`tablo_ui::SIDEBAR_JS`), which [`Self::layout_shell`]'s document —
249    /// not this function — emits (scripts are owned by the document). The
250    /// mobile sheet (`#mobile-sidebar-sheet`) instead dismisses through its
251    /// own runtime `@keydown`/`@click` handlers, so it needs no asset; the
252    /// vendored `sheet`/`sidebar` primitives carry no note themselves
253    /// (ADR-0007 sync guard). See ADR-0014.
254    pub async fn render_shell<'a>(
255        cx: &'a Cx,
256        nav_items: &[NavigationItem],
257        current_path: &str,
258        slot: Child<'a>,
259        extra_class: Option<String>,
260    ) -> Result<BoxView<'a>> {
261        // A hoisting body: the sidebar signals are declared while this view
262        // resolves, and a page re-run resumes them from the client. The owned
263        // copies pin the caller's navigation to the lazy body's lifetime.
264        let nav_items = nav_items.to_vec();
265        let current_path = current_path.to_string();
266        Ok(Box::pin(HoistView::new(ThenView::new(async move {
267            Self::render_shell_body(cx, &nav_items, &current_path, slot, extra_class).await
268        }))))
269    }

271    async fn render_shell_body<'a>(
272        cx: &'a Cx,
273        nav_items: &[NavigationItem],
274        current_path: &str,
275        slot: Child<'a>,
276        extra_class: Option<String>,
277    ) -> Result<BoxView<'a>> {
278        use tablo_ui::{
279            SeparatorOrientation, SidebarCollapsible, separator, sidebar, sidebar_content,
280            sidebar_footer, sidebar_header, sidebar_inset, sidebar_provider, sidebar_trigger,
281        };
282
283        let sidebar_open = signal(cx, || Self::sidebar_starts_open(cx));
284        let mobile_open = signal(cx, || false);
285        let outer_class = extra_class.clone().unwrap_or_default();
286        let header_title = topcoat::context::try_app_context::<Brand>(cx)
287            .map(|b| b.name.clone())
288            .unwrap_or_else(|| "Tablo".to_string());
289        // One navigation tree: the upstream sidebar renders its children once
290        // and shares them between the desktop panel and the mobile sheet.
291        let navigation =
292            Self::sidebar_navigation(cx, nav_items, current_path, mobile_open.clone()).await?;
293        let sidebar_brand = Self::render_brand(cx).await?;
294        let sidebar_theme_toggle = Self::theme_toggle(cx).await?;
295        let header_theme_toggle = Self::theme_toggle(cx).await?;
296        // Signed-in identity + logout control, present only with a session
297        // (ADR-0013). `ensure_token` runs before any streaming starts so the
298        // logout form always carries a matching CSRF pair.
299        let account_view: BoxView<'_> = match crate::auth::current_user(cx) {
300            Some(user) => {
301                let csrf = crate::csrf::ensure_token(cx);
302                let logout = crate::auth::logout_url(cx);
303                view! {
304                    cx =>
305                    <div class="flex items-center gap-2">
306                        <span class="text-sm text-muted-foreground">
307                            (user.display_name)
308                        </span>
309                        <form method="post" action=(logout)>
310                            (crate::csrf::field(cx, &csrf))
311                            <button
312                                type="submit"
313                                class="text-sm text-muted-foreground underline"
314                            >
315                                "Sign out"
316                            </button>
317                        </form>
318                    </div>
319                }
320                .boxed()
321            }
322            None => view! { cx => <span></span> }.boxed(),
323        };
324        let notification_view: BoxView<'_> = match take_notification(cx) {
325            Some(notification) => {
326                crate::notification::render_notification(cx, notification, Default::default())
327                    .await?
328            }
329            // No `<span>` placeholder: the toaster renders an `<ol>`,
330            // which permits only `li`/`script`/`template` children — the empty
331            // view renders nothing.
332            None => ().boxed(),
333        };
334        // The page owns the live-toast signals; resolve the same handles here
335        // (same helper, same request identity) and hand them to the shard
336        // (GH #154 §3).
337        let LiveToast {
338            status: toast_status,
339            title: toast_title,
340            description: toast_description,
341            serial: toast_serial,
342        } = live_toast(cx);
343
344        Ok(view! {
345            cx =>
346            sidebar_provider(
347                attrs: attributes! { class=(outer_class) },
348                // `sidebar_rail` is intentionally not rendered: the header
349                // `sidebar_trigger` is the explicit toggle, and the rail's
350                // edge hit-area reads as stray chrome as a primary toggle.
351                // Keep the component available (`tablo_ui::sidebar_rail`)
352                // for opt-in `variant=inset`/`floating` layouts.
353                sidebar(
354                    open: $(sidebar_open.get()),
355                    mobile_open: $(mobile_open.get()),
356                    collapsible: SidebarCollapsible::Offcanvas,
357                    sheet_attrs: attributes! {
358                        id="mobile-sidebar-sheet"
359                        aria-label="Navigation"
360                        @keydown=$(|e: Event| {
361                            if e.key == "Escape" {
362                                mobile_open.set(false);
363                            }
364                        })
365                        @click=$(|e: Event| {
366                            if e.target.id == "mobile-sidebar-sheet" {
367                                mobile_open.set(false);
368                            }
369                        })
370                    },
371                    sidebar_header(
372                        <div class="flex items-center gap-2">
373                            (sidebar_brand)
374                            // Below `md` the sheet covers the inset header,
375                            // so the sheet's own header carries the close
376                            // control (upstream `examples/ui`); the mobile
377                            // trigger in the inset header is behind the veil.
378                            tablo_ui::button(
379                                variant: tablo_ui::ButtonVariant::Ghost,
380                                size: tablo_ui::ButtonSize::Icon,
381                                attrs: attributes! {
382                                    type="button"
383                                    class="md:hidden ml-auto"
384                                    aria-label="Close sidebar"
385                                    @click=$(|_e: Event| mobile_open.set(false))
386                                },
387                                icon(data: tablo_ui::icons::X)
388                            )
389                        </div>
390                    )
391                    sidebar_content((navigation))
392                    sidebar_footer((sidebar_theme_toggle))
393                )
394                sidebar_inset(
395                    sidebar_header(
396                        // The desktop trigger collapses the rail; below md the
397                        // mobile trigger opens the sheet instead (shadcn
398                        // SidebarTrigger pair, upstream `examples/ui`).
399                        sidebar_trigger(
400                            open: $(sidebar_open.get()),
401                            attrs: attributes! {
402                                class="max-md:hidden"
403                                aria-controls="mobile-sidebar-sheet"
404                                @click=$(|_e: Event| sidebar_open.toggle())
405                            }
406                        )
407                        sidebar_trigger(
408                            open: $(mobile_open.get()),
409                            attrs: attributes! {
410                                class="md:hidden"
411                                aria-controls="mobile-sidebar-sheet"
412                                @click=$(|_e: Event| mobile_open.toggle())
413                            }
414                        )
415                        separator(orientation: SeparatorOrientation::Vertical)
416                        <div class="font-semibold text-foreground">(header_title)</div>
417                        <div class="ml-auto flex items-center gap-2">
418                            (account_view)
419                            (header_theme_toggle)
420                        </div>
421                    )
422                    // `sidebar_inset` is the document's one `<main>`; a second
423                    // nested landmark is invalid and confuses landmark
424                    // navigation (upstream `examples/ui` uses a plain div).
425                    <div class="flex-1 mx-auto max-w-7xl w-full p-6">(slot)</div>
426                )
427                // Toast stack — the shadcn/Sonner surface, fixed bottom-right
428                // and a polite live region so streamed swaps are announced.
429                // `live_toaster` is the page-owned
430                // in-place transport (GH #154 §3); the flash cookie's toast
431                // rides beside it.
432                tablo_ui::toaster(
433                    (notification_view)
434                    live_toaster(
435                        status: $(toast_status),
436                        title: $(toast_title),
437                        description: $(toast_description),
438                        serial: $(toast_serial)
439                    )
440                )
441            ) // Scripts are owned by the document (layout_shell).
442        }
443        .boxed())
444    }

446    /// Convenience wrapper for `#[layout]` handlers: takes the layout's
447    /// `slot: Slot<'_>` and renders the complete HTML document around the
448    /// shell.
449    ///
450    /// The stylesheet and font are supplied to the Panel builder with
451    /// [`Self::shell_assets`]. A Panel without those values remains renderable
452    /// for tests and custom document owners, but does not pretend that a CSS
453    /// bundle exists. Errors from the page slot propagate unchanged when the
454    /// document view is resolved.
455    pub async fn layout_shell<'a>(cx: &'a Cx, slot: Slot<'a>) -> Result<impl View + 'a> {
456        use topcoat::{context::try_app_context, router::request::uri};
457        let current = uri(cx).path().to_string();
458        // Prefer declarative nav_items from Panel::resource, fallback to Home.
459        let nav_items = try_app_context::<Vec<NavigationItem>>(cx)
460            .cloned()
461            .unwrap_or_else(|| vec![NavigationItem::at("Home", super::gate::panel_prefix(cx))]);
462        let shell = Self::render_shell(cx, &nav_items, &current, slot, None).await?;
463        let brand_title = try_app_context::<Brand>(cx)
464            .map(|b| b.name.clone())
465            .unwrap_or_else(|| "Tablo".to_string());
466        Self::render_document(cx, brand_title, shell).await
467    }

469    /// The complete HTML document around a page the panel does not own.
470    ///
471    /// An app-level `#[layout]` outside the panel prefix takes the panel's
472    /// document from here, so a public page carries the same head as the admin
473    /// shell: the dev script, the theme init, and — where
474    /// [`Self::shell_assets`] registered them — the runtime script, the font,
475    /// the stylesheet, and the shell scripts. Without those values the head
476    /// degrades to the dev script and the theme init, which is what lets a
477    /// router built without `.assets(..)` render at all.
478    ///
479    /// `body` is the page's own content: the panel renders the `<body>` element,
480    /// its dark-mode `<html>` class, and the head; the page owns its chrome
481    /// inside. `title` is the page's own, where [`Self::layout_shell`] titles an
482    /// admin page with the panel's [`Brand`].
483    pub async fn document<'a>(
484        cx: &'a Cx,
485        title: impl Into<String>,
486        body: impl View + 'a,
487    ) -> Result<BoxView<'a>> {
488        Self::render_document(cx, title.into(), body.boxed()).await
489    }

491    /// The complete HTML document around a rendered body: assets, dark-mode
492    /// class, and title. [`Self::layout_shell`] frames the panel shell with
493    /// it; the standalone login page (ADR-0013) uses the same document so
494    /// brand and dark mode carry over.
495    ///
496    /// The eleven shell scripts ship `defer`red (deliberate all-load policy,
497    /// ADR-0014): parsing never waits for them, and every one is safe
498    /// deferred — document-level listeners install after parse, and the
499    /// `DOMContentLoaded` handlers still run, since deferred scripts execute
500    /// first. The blocking `theme_init_script` stays inline so the `dark`
501    /// class lands pre-paint.
502    pub(crate) async fn render_document<'a>(
503        cx: &'a Cx,
504        title: String,
505        body: BoxView<'a>,
506    ) -> Result<BoxView<'a>> {
507        use topcoat::context::try_app_context;
508        // The build-time default: the `<html class>` a first-time visitor gets,
509        // and the fallback the blocking script uses when nothing is stored.
510        let default_dark = try_app_context::<DarkMode>(cx).is_some_and(|dm| dm.0);
511        let head: BoxView<'_> = match try_app_context::<ShellAssets>(cx).copied() {
512            Some(ShellAssets { stylesheet, font }) => view! {
513                cx =>
514                topcoat::dev::script()
515                tablo_ui::theme_init_script(default_dark: default_dark)
516                topcoat::runtime::script()
517                topcoat::font::link(font: font)
518                <link rel="stylesheet" href=(stylesheet)>
519                <script src=(tablo_ui::SIDEBAR_JS) defer=""></script>
520                <script src=(tablo_ui::THEME_JS) defer=""></script>
521                <script src=(tablo_ui::DIALOG_JS) defer=""></script>
522                <script src=(tablo_ui::WIRE_JS) defer=""></script>
523                <script src=(tablo_ui::BULK_JS) defer=""></script>
524                <script src=(tablo_ui::FILTERS_JS) defer=""></script>
525                <script src=(tablo_ui::LIVE_SEARCH_JS) defer=""></script>
526                <script src=(tablo_ui::SELECTS_JS) defer=""></script>
527                <script src=(tablo_ui::VARIANT_JS) defer=""></script>
528                <script src=(tablo_ui::NOTIFICATION_JS) defer=""></script>
529                <script src=(tablo_ui::MUTATION_SUBMIT_JS) defer=""></script>
530            }
531            .boxed(),
532            None => view! {
533                cx =>
534                topcoat::dev::script()
535                tablo_ui::theme_init_script(default_dark: default_dark)
536            }
537            .boxed(),
538        };
539        let dark = match request_cookie(cx, "theme").as_deref() {
540            Some("dark") => true,
541            Some("light") => false,
542            _ => default_dark,
543        };
544        let html_class = dark.then_some("dark");
545        Ok(view! {
546            cx =>
547            <!DOCTYPE html>
548            <html class=(html_class)>
549                <head>
550                    <title>(title)</title>
551                    (head)
552                </head>
553                <body>(body)</body>
554            </html>
555        }
556        .boxed())
557    }
