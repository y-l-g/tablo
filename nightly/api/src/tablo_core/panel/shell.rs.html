27/// `extra` plus the attribute that sends a sidebar link through runtime
28/// navigation. The menu button writes the `href` itself, so this carries none.
29fn sidebar_link(cx: &Cx, mut extra: Attributes) -> Attributes {
30    let mut attrs = crate::resource::runtime_link(cx, "");
31    attrs.remove("href");
32    extra.extend(attrs);
33    extra
34}

36/// The value of the request's `name` cookie.
37///
38/// Parsed from the raw `Cookie` header on purpose: `topcoat::cookie::cookies`
39/// panics when the cookie router layer is absent (tests, minimal routers), and
40/// the shell must render everywhere.
41fn request_cookie(cx: &Cx, name: &str) -> Option<String> {
42    let header = try_request_context::<http::request::Parts>(cx)?
43        .headers
44        .get(COOKIE)?
45        .to_str()
46        .ok()?;
47    header.split(';').find_map(|part| {
48        let (key, value) = part.trim().split_once('=')?;
49        (key == name).then(|| value.to_string())
50    })
51}

53/// Branding for the admin shell: the sidebar header, the login card, and the
54/// topbar below `md`, where the sidebar folds into a sheet.
55#[derive(Debug, Clone)]
56pub struct Brand {
57    /// Display name (e.g. `"Acme"`).
58    pub name: String,
59    /// Optional logo URL (e.g. `"/logo.svg"`). Rendered as an `<img>` when present.
60    pub logo: Option<String>,
61}

63impl Brand {
64    /// Create a brand with the given name (GH #102: surrounding whitespace is
65    /// trimmed so `" Acme "` cannot break the `flex h-16` header).
66    pub fn new(name: impl Into<String>) -> Self {
67        Self {
68            name: name.into().trim().to_string(),
69            logo: None,
70        }
71    }
72
73    /// Attach a logo URL (GH #102: blank values are ignored so an empty
74    /// `logo("")` falls back to the name-only render instead of a
75    /// broken-image icon).
76    pub fn logo(mut self, logo: impl Into<String>) -> Self {
77        let logo = logo.into().trim().to_string();
78        if !logo.is_empty() {
79            self.logo = Some(logo);
80        }
81        self
82    }

98    /// The top bar's tenant switcher: a disclosure listing the user's
99    /// tenants, each a form that posts to the tenant-switch route. A user
100    /// with fewer than two tenants has nothing to switch, and gets nothing.
101    fn tenant_switcher<'a>(
102        cx: &'a Cx,
103        tenants: &[crate::tenancy::Membership],
104        csrf: &str,
105    ) -> BoxView<'a> {
106        if tenants.len() < 2 {
107            return ().boxed();
108        }
109        let current = crate::tenancy::membership(cx);
110        let label = current.map_or_else(|| "Select a tenant".to_string(), |m| m.name.clone());
111        let current = current.map(|membership| membership.tenant);
112        let action = crate::auth::tenant_url(cx);
113        let choices: Vec<BoxView<'a>> = tenants
114            .iter()
115            .map(|membership| {
116                let selected = current == Some(membership.tenant);
117                let value = membership.tenant.to_string();
118                let name = membership.name.clone();
119                let action = action.clone();
120                let csrf = crate::csrf::field(cx, csrf);
121                view! {
122                    cx =>
123                    <form method="post" action=(action)>
124                        (csrf)
125                        <input
126                            type="hidden"
127                            name=(crate::auth::TENANT_FIELD)
128                            value=(value)
129                        >
130                        <button
131                            type="submit"
132                            aria-current=(selected.then_some("true"))
133                            class="flex w-full items-center gap-2 rounded-sm px-2 py-1.5 text-left text-sm hover:bg-muted"
134                        >
135                            <span class="flex-1 truncate">(name)</span>
136                            if selected {
137                                icon(
138                                    data: tablo_ui::icons::CHECK,
139                                    attrs: attributes! { class="size-4" }
140                                )
141                            }
142                        </button>
143                    </form>
144                }
145                .boxed()
146            })
147            .collect();
148        view! {
149            cx =>
150            <details class="relative" data-tenant-switcher="">
151                <summary
152                    class="flex cursor-pointer list-none items-center gap-1 rounded-md px-2 py-1 text-sm font-medium text-foreground hover:bg-muted [&::-webkit-details-marker]:hidden"
153                >
154                    <span class="max-w-40 truncate">(label)</span>
155                    icon(
156                        data: tablo_ui::icons::CHEVRON_DOWN,
157                        attrs: attributes! { class="size-4 text-muted-foreground" }
158                    )
159                </summary>
160                <div
161                    class="absolute right-0 z-50 mt-1 min-w-48 rounded-md border border-border bg-popover p-1 text-popover-foreground shadow-md"
162                >
163                    for choice in choices {
164                        (choice)
165                    }
166                </div>
167            </details>
168        }
169        .boxed()
170    }

248    async fn sidebar_navigation<'a>(
249        cx: &'a Cx,
250        nav_items: &[NavigationItem],
251        current_path: &str,
252        mobile_open: Signal<bool>,
253    ) -> Result<BoxView<'a>> {
254        use tablo_ui::{
255            sidebar_group, sidebar_group_content, sidebar_group_label, sidebar_menu,
256            sidebar_menu_button, sidebar_menu_item,
257        };
258        let mut nav_items = nav_items.to_vec();
259        // Stable order: explicit `order` first, declaration order
260        // breaking ties — a resource's `navigation()` override interleaves by
261        // setting it.
262        nav_items.sort_by_key(|item| item.order);
263        // A Panel resolves every item it owns (`Panel::resource`, `page`,
264        // `home`); one that
265        // reaches the sidebar unresolved has no URL to render, which is a
266        // framework bug rather than user error.
267        debug_assert!(
268            nav_items.iter().all(|item| item.url().is_some()),
269            "navigation items are resolved by the Panel that owns them"
270        );
271        // One active entry: a home entry at the bare prefix matches every path
272        // under it, so the longest matching URL wins, and the first such entry
273        // among any that share it.
274        let active = nav_items
275            .iter()
276            .enumerate()
277            .filter(|(_, item)| item.is_current_path(current_path))
278            .min_by_key(|(_, item)| std::cmp::Reverse(item.url().map_or(0, str::len)))
279            .map(|(index, _)| index);
280
281        Ok(view! {
282            cx =>
283            sidebar_group(
284                sidebar_group_label("Navigation")
285                sidebar_group_content(
286                    sidebar_menu(
287                        for (index, item) in nav_items.iter().enumerate() {
288                            let is_active = active == Some(index);
289                            let attrs = sidebar_link(
290                                cx,
291                                attributes! {
292                                    // Tapping a link in the mobile sheet closes
293                                    // it; on desktop the navigation is the effect.
294                                    @click=$(|_e: Event| mobile_open.set(false))
295                                },
296                            );
297                            sidebar_menu_item(
298                                sidebar_menu_button(
299                                    active: is_active,
300                                    href: item.url(),
301                                    tooltip: Some(item.label.as_str()),
302                                    attrs: attrs,
303                                    if let Some(data) = item.icon.clone() {
304                                        icon(data: data)
305                                    }
306                                    <span>(item.label.clone())</span>
307                                )
308                            )
309                        }
310                    )
311                )
312            )
313        }
314        .boxed())
315    }

327    /// Render the Filament-grade Shell that frames every admin page.
328    ///
329    /// Composes Topcoat's upstream `sidebar` primitives (ADR-0007): the
330    /// desktop panel and the mobile sheet share one navigation rendering, and
331    /// the open state lives in runtime signals — `open` seeds from the
332    /// `sidebar_state` cookie for the first paint, the triggers carry
333    /// `@click` handlers, and `assets/sidebar.js` persists changes back to
334    /// the cookie. Includes dark-mode toggle (Ghost button, persisted by
335    /// `theme.js` to `localStorage` + the `theme` cookie) and the toast stack
336    /// (shadcn/Sonner surface, fixed bottom-right). Additive `class` is
337    /// allowed on the outer container only (narrow seam).
338    ///
339    /// Asset note: desktop persistence needs `assets/sidebar.js`
340    /// (`tablo_ui::SIDEBAR_JS`), which [`Self::layout_shell`]'s document —
341    /// not this function — emits (scripts are owned by the document). The
342    /// mobile sheet (`#mobile-sidebar-sheet`) instead dismisses through its
343    /// own runtime `@keydown`/`@click` handlers, so it needs no asset; the
344    /// vendored `sheet`/`sidebar` primitives carry no note themselves
345    /// (ADR-0007 sync guard). See ADR-0014.
346    pub async fn render_shell<'a>(
347        cx: &'a Cx,
348        nav_items: &[NavigationItem],
349        current_path: &str,
350        slot: Child<'a>,
351        extra_class: Option<String>,
352    ) -> Result<BoxView<'a>> {
353        // A hoisting body: the sidebar signals are declared while this view
354        // resolves, and a page re-run resumes them from the client. The owned
355        // copies pin the caller's navigation to the lazy body's lifetime.
356        let nav_items = nav_items.to_vec();
357        let current_path = current_path.to_string();
358        Ok(Box::pin(HoistView::new(ThenView::new(async move {
359            Self::render_shell_body(cx, &nav_items, &current_path, slot, extra_class).await
360        }))))
361    }

363    async fn render_shell_body<'a>(
364        cx: &'a Cx,
365        nav_items: &[NavigationItem],
366        current_path: &str,
367        slot: Child<'a>,
368        extra_class: Option<String>,
369    ) -> Result<BoxView<'a>> {
370        use tablo_ui::{
371            SeparatorOrientation, SidebarCollapsible, separator, sidebar, sidebar_content,
372            sidebar_header, sidebar_inset, sidebar_provider, sidebar_trigger,
373        };
374
375        let sidebar_open = signal(cx, || Self::sidebar_starts_open(cx));
376        let mobile_open = signal(cx, || false);
377        let outer_class = extra_class.clone().unwrap_or_default();
378        let header_title = brand_name(cx);
379        // One navigation tree: the upstream sidebar renders its children once
380        // and shares them between the desktop panel and the mobile sheet.
381        let navigation =
382            Self::sidebar_navigation(cx, nav_items, current_path, mobile_open.clone()).await?;
383        let sidebar_brand = Self::render_brand(cx).await?;
384        let theme_toggle = Self::theme_toggle(cx).await?;
385        // Signed-in identity + logout control, present only with a session
386        // (ADR-0013). `ensure_token` runs before any streaming starts so the
387        // logout form always carries a matching CSRF pair.
388        let account_view: BoxView<'_> = match crate::auth::signed(cx) {
389            Some(signed) => {
390                let csrf = crate::csrf::ensure_token(cx);
391                let logout = crate::auth::logout_url(cx);
392                let name = signed.user.display_name().to_string();
393                let initial = name.chars().next().map(String::from).unwrap_or_default();
394                let switcher = Self::tenant_switcher(cx, signed.user.tenants(), &csrf);
395                view! {
396                    cx =>
397                    <div class="flex items-center gap-2">
398                        (switcher)
399                        <span
400                            aria-hidden="true"
401                            class="flex size-7 shrink-0 items-center justify-center rounded-full bg-muted text-xs font-medium text-muted-foreground"
402                        >
403                            (initial)
404                        </span>
405                        <span class="max-sm:hidden text-sm font-medium text-foreground">
406                            (name)
407                        </span>
408                        <form method="post" action=(logout)>
409                            (crate::csrf::field(cx, &csrf))
410                            tablo_ui::button(
411                                variant: tablo_ui::ButtonVariant::Ghost,
412                                size: tablo_ui::ButtonSize::Icon,
413                                attrs: attributes! { type="submit" aria-label="Sign out" title="Sign out" },
414                                icon(data: tablo_ui::icons::LOG_OUT)
415                            )
416                        </form>
417                    </div>
418                }
419                .boxed()
420            }
421            None => view! { cx => <span></span> }.boxed(),
422        };
423        let notification_view: BoxView<'_> = match take_notification(cx) {
424            Some(notification) => {
425                crate::notification::render_notification(cx, notification, Default::default())
426                    .await?
427            }
428            // No `<span>` placeholder: the toaster renders an `<ol>`,
429            // which permits only `li`/`script`/`template` children — the empty
430            // view renders nothing.
431            None => ().boxed(),
432        };
433        // The page owns the live-toast signals; resolve the same handles here
434        // (same helper, same request identity) and hand them to the shard
435        // (GH #154 §3).
436        let LiveToast {
437            status: toast_status,
438            title: toast_title,
439            description: toast_description,
440            serial: toast_serial,
441        } = live_toast(cx);
442
443        Ok(view! {
444            cx =>
445            sidebar_provider(
446                attrs: attributes! { class=(outer_class) },
447                // `sidebar_rail` is intentionally not rendered: the header
448                // `sidebar_trigger` is the explicit toggle, and the rail's
449                // edge hit-area reads as stray chrome as a primary toggle.
450                // Keep the component available (`tablo_ui::sidebar_rail`)
451                // for opt-in `variant=inset`/`floating` layouts.
452                sidebar(
453                    open: $(sidebar_open.get()),
454                    mobile_open: $(mobile_open.get()),
455                    collapsible: SidebarCollapsible::Offcanvas,
456                    sheet_attrs: attributes! {
457                        id="mobile-sidebar-sheet"
458                        aria-label="Navigation"
459                        @keydown=$(|e: Event| {
460                            if e.key == "Escape" {
461                                mobile_open.set(false);
462                            }
463                        })
464                        @click=$(|e: Event| {
465                            if e.target.id == "mobile-sidebar-sheet" {
466                                mobile_open.set(false);
467                            }
468                        })
469                    },
470                    sidebar_header(
471                        <div class="flex items-center gap-2 px-2">
472                            (sidebar_brand)
473                            // Below `md` the sheet covers the inset header,
474                            // so the sheet's own header carries the close
475                            // control (upstream `examples/ui`); the mobile
476                            // trigger in the inset header is behind the veil.
477                            tablo_ui::button(
478                                variant: tablo_ui::ButtonVariant::Ghost,
479                                size: tablo_ui::ButtonSize::Icon,
480                                attrs: attributes! {
481                                    type="button"
482                                    class="md:hidden ml-auto"
483                                    aria-label="Close sidebar"
484                                    @click=$(|_e: Event| mobile_open.set(false))
485                                },
486                                icon(data: tablo_ui::icons::X)
487                            )
488                        </div>
489                    )
490                    sidebar_content((navigation))
491                )
492                sidebar_inset(
493                    sidebar_header(
494                        // The desktop trigger collapses the rail; below md the
495                        // mobile trigger opens the sheet instead (shadcn
496                        // SidebarTrigger pair, upstream `examples/ui`).
497                        sidebar_trigger(
498                            open: $(sidebar_open.get()),
499                            attrs: attributes! {
500                                class="max-md:hidden"
501                                aria-controls="mobile-sidebar-sheet"
502                                @click=$(|_e: Event| sidebar_open.toggle())
503                            }
504                        )
505                        sidebar_trigger(
506                            open: $(mobile_open.get()),
507                            attrs: attributes! {
508                                class="md:hidden"
509                                aria-controls="mobile-sidebar-sheet"
510                                @click=$(|_e: Event| mobile_open.toggle())
511                            }
512                        )
513                        // The sidebar carries the brand on desktop; below md
514                        // it hides in the sheet, so the topbar names the
515                        // panel there.
516                        <div class="md:hidden font-semibold text-foreground">
517                            (header_title)
518                        </div>
519                        // The separator stays a direct child of the topbar,
520                        // where `sidebar_inset` sizes a vertical rule.
521                        <div class="ml-auto">(theme_toggle)</div>
522                        separator(orientation: SeparatorOrientation::Vertical)
523                        (account_view)
524                    )
525                    // `sidebar_inset` is the document's one `<main>`; a second
526                    // nested landmark is invalid and confuses landmark
527                    // navigation (upstream `examples/ui` uses a plain div).
528                    // The page container (`tablo_ui::page`) owns the width
529                    // and the padding.
530                    <div class="flex flex-1 flex-col">(slot)</div>
531                )
532                // Toast stack — the shadcn/Sonner surface, fixed bottom-right
533                // and a polite live region so streamed swaps are announced.
534                // `live_toaster` is the page-owned
535                // in-place transport (GH #154 §3); the flash cookie's toast
536                // rides beside it.
537                tablo_ui::toaster(
538                    (notification_view)
539                    live_toaster(
540                        status: $(toast_status),
541                        title: $(toast_title),
542                        description: $(toast_description),
543                        serial: $(toast_serial)
544                    )
545                )
546            ) // Scripts are owned by the document (layout_shell).
547        }
548        .boxed())
549    }

551    /// The panel shell around a page, as a complete HTML document: the layout
552    /// every panel registers at its prefix unless [`Panel::layout`] replaces
553    /// it, and what a replacement calls to keep the shell around its own
554    /// markup.
555    ///
556    /// The stylesheet and font are supplied to the Panel builder with
557    /// [`Self::shell_assets`]. A Panel without those values remains renderable
558    /// for tests and custom document owners, but does not pretend that a CSS
559    /// bundle exists. Errors from the page slot propagate unchanged when the
560    /// document view is resolved.
561    pub fn layout_shell<'a>(cx: &'a Cx, slot: Slot<'a>) -> BoxView<'a> {
562        Box::pin(ThenView::new(Self::render_layout_shell(cx, slot)))
563    }

565    async fn render_layout_shell<'a>(cx: &'a Cx, slot: Slot<'a>) -> Result<BoxView<'a>> {
566        use topcoat::router::request::uri;
567        let path = uri(cx).path().to_string();
568        // The panel's sidebar, or a single Home entry outside any panel.
569        let nav_items = current(cx)
570            .map(|panel| panel.nav_items.clone())
571            .filter(|items| !items.is_empty())
572            .unwrap_or_else(|| vec![NavigationItem::at("Home", super::gate::panel_prefix(cx))]);
573        let shell = Self::render_shell(cx, &nav_items, &path, slot, None).await?;
574        Self::render_document(cx, brand_name(cx), shell).await
575    }

577    /// The complete HTML document around a page the panel does not own.
578    ///
579    /// An app-level `#[layout]` outside the panel prefix takes the panel's
580    /// document from here, so a public page carries the same head as the admin
581    /// shell: the dev script, the theme init, and — where
582    /// [`Self::shell_assets`] registered them — the runtime script, the font,
583    /// the stylesheet, and the shell scripts. Outside a prefix the panel is
584    /// the router's only one; a router that mounts several has no panel there,
585    /// and the head degrades to the dev script and the theme init, as it does
586    /// for a router built without `.assets(..)`.
587    ///
588    /// `body` is the page's own content: the panel renders the `<body>` element,
589    /// its dark-mode `<html>` class, and the head; the page owns its chrome
590    /// inside. `title` is the page's own, where [`Self::layout_shell`] titles an
591    /// admin page with the panel's [`Brand`].
592    pub async fn document<'a>(
593        cx: &'a Cx,
594        title: impl Into<String>,
595        body: impl View + 'a,
596    ) -> Result<BoxView<'a>> {
597        Self::render_document(cx, title.into(), body.boxed()).await
598    }

600    /// The complete HTML document around a rendered body: assets, dark-mode
601    /// class, and title. [`Self::layout_shell`] frames the panel shell with
602    /// it; the standalone login page (ADR-0013) uses the same document so
603    /// brand and dark mode carry over.
604    ///
605    /// The eleven shell scripts ship `defer`red (deliberate all-load policy,
606    /// ADR-0014): parsing never waits for them, and every one is safe
607    /// deferred — document-level listeners install after parse, and the
608    /// `DOMContentLoaded` handlers still run, since deferred scripts execute
609    /// first. The blocking `theme_init_script` stays inline so the `dark`
610    /// class lands pre-paint.
611    pub(crate) async fn render_document<'a>(
612        cx: &'a Cx,
613        title: String,
614        body: BoxView<'a>,
615    ) -> Result<BoxView<'a>> {
616        // The panel's default: the `<html class>` a first-time visitor gets,
617        // and the fallback the blocking script uses when nothing is stored.
618        let default_dark = current(cx).is_some_and(|panel| panel.dark_mode);
619        let head: BoxView<'_> = match current(cx).and_then(|panel| panel.shell_assets) {
620            Some(ShellAssets { stylesheet, font }) => view! {
621                cx =>
622                topcoat::dev::script()
623                tablo_ui::theme_init_script(default_dark: default_dark)
624                topcoat::runtime::script()
625                topcoat::font::link(font: font)
626                <link rel="stylesheet" href=(stylesheet)>
627                <script src=(tablo_ui::SIDEBAR_JS) defer=""></script>
628                <script src=(tablo_ui::THEME_JS) defer=""></script>
629                <script src=(tablo_ui::DIALOG_JS) defer=""></script>
630                <script src=(tablo_ui::WIRE_JS) defer=""></script>
631                <script src=(tablo_ui::BULK_JS) defer=""></script>
632                <script src=(tablo_ui::FILTERS_JS) defer=""></script>
633                <script src=(tablo_ui::LIVE_SEARCH_JS) defer=""></script>
634                <script src=(tablo_ui::SELECTS_JS) defer=""></script>
635                <script src=(tablo_ui::VARIANT_JS) defer=""></script>
636                <script src=(tablo_ui::NOTIFICATION_JS) defer=""></script>
637                <script src=(tablo_ui::MUTATION_SUBMIT_JS) defer=""></script>
638            }
639            .boxed(),
640            None => view! {
641                cx =>
642                topcoat::dev::script()
643                tablo_ui::theme_init_script(default_dark: default_dark)
644            }
645            .boxed(),
646        };
647        let dark = match request_cookie(cx, "theme").as_deref() {
648            Some("dark") => true,
649            Some("light") => false,
650            _ => default_dark,
651        };
652        let html_class = dark.then_some("dark");
653        Ok(view! {
654            cx =>
655            <!DOCTYPE html>
656            <html class=(html_class)>
657                <head>
658                    <title>(title)</title>
659                    (head)
660                </head>
661                <body>(body)</body>
662            </html>
663        }
664        .boxed())
665    }
666}

668/// The request's panel brand name, `Tablo` without one.
669fn brand_name(cx: &Cx) -> String {
670    current(cx)
671        .and_then(|panel| panel.brand.as_ref())
672        .map_or_else(|| "Tablo".to_string(), |brand| brand.name.clone())
673}
