1//! The `Db` glue: how Tablo code reaches the pooled Toasty database.
2//!
3//! `Db` is registered once on the app context (`app_context::<Db>`), and each
4//! request clones it — a cheap `Arc` bump — before running Toasty statements,
5//! which require `&mut Db`.
6
7use toasty::Db;
8use topcoat::context::{Cx, app_context};
9
10/// Returns the pooled [`Db`] registered on the app context.
11///
12/// Cloning the `Db` is cheap; Toasty statements need `&mut Db`, so callers do
13/// `let mut db = db(cx); ...exec(&mut db).await?`.
14///
15/// Pool discipline: while a framework transaction holds its
16/// connection, no other handle may run statements — with a single-connection
17/// pool (notably `sqlite::memory:`) a second handle blocks forever. Mutation
18/// handlers therefore keep the tx strictly around check + write + commit and
19/// `drop(tx)` before any re-render (form option loaders open their own
20/// handle).
21///
22/// List pages stream: the table query runs while the response body streams,
23/// so a live list body holds a pooled connection until drained or dropped.
24/// With a single-connection pool a second query waits while an earlier list
25/// body stays alive; drain or drop the body before the next query.
26#[inline]
27pub fn db(cx: &Cx) -> Db {
28    app_context::<Db>(cx).clone()
29}
30
31/// Map a database infrastructure failure (pool/tx open, probe/exec, commit)
32/// to an opaque 500: the driver/SQL text is logged for operators
33/// but never reaches the error page. The streamed list already holds this
34/// contract through its generic `ErrorState` (logged once at
35/// `table_error_view`); mutation/export paths must match it.
36///
37/// Only infra failures come here. App-hook errors (`create_record` et al.)
38/// and explicit guards (404s, 403s, config errors) keep their own mapping;
39/// [`hook_failure`] is the seam that separates the two when both arrive
40/// through the same record fn.
41pub(crate) fn unavailable(source: impl std::fmt::Display) -> topcoat::Error {
42    tracing::error!(error = %source, "database unavailable");
43    topcoat::Error::from(std::io::Error::other("database unavailable"))
44}
45
46/// Map a failed record hook to the error the response carries.
47///
48/// A hook (`create_record`, `update_record`, `delete_record`, …) is app code,
49/// so its error is one of two things: the app's own — a guard's 404, a config
50/// error — or the driver's. The app's own
51/// keeps the app's mapping (a guard must still answer 404, not a 500), while
52/// the driver's is an infra failure and takes [`unavailable`]: its text goes
53/// to the log, and the page never echoes it. The error's own type settles
54/// which one it is, never its message — Toasty owns [`toasty::Error`], so a
55/// downcast decides (the same seam the crate uses for
56/// `ContentTooLargeError` and `CursorDecodeError`).
57pub(crate) fn hook_failure(error: topcoat::Error) -> topcoat::Error {
58    if error.is::<toasty::Error>() {
59        unavailable(error)
60    } else {
61        error
62    }
63}
