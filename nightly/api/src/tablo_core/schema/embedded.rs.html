1//! Embedded **values**: a typed value and the flat form map, converted in one
2//! declared place, and the schema node that renders it.
3//!
4//! An embedded leaf binds one column: a lens through an embedded struct, an
5//! enum variant, or a `#[document]` resolves to its flattened storage column
6//! ([`ResolvedLens::new`](super::ResolvedLens::new)). A **value** binds as a
7//! whole. `#[derive(EmbeddedForm)]` builds its schema node — one resolved field
8//! per leaf, a nested node per `#[form(embed)]` value, and for an enum the
9//! variant control over the discriminant column plus one group per variant —
10//! and the codec reads and writes the flat map through that node's keys, so the
11//! app never spells a flattened name.
12//!
13//! A payload selects the variant only when the submission carries no
14//! discriminant at all (the create form has nothing to hydrate); a
15//! `#[shared(..)]` column belongs to several variants and never selects one. A
16//! named discriminant always wins, and an unknown one is refused.
17//!
18//! # What an app writes
19//!
20//! ```ignore
21//! #[derive(Clone, toasty::Embed, tablo_core::EmbeddedForm)]
22//! pub struct Seo { pub title: String, pub description: String }
23//!
24//! Section::new("SEO").schema(Seo::form(dx, Post::fields().seo()));
25//! record.seo.write_form(cx, Post::fields().seo(), &mut values);
26//! let seo = Seo::read_form(cx, Post::fields().seo(), &values)?;
27//! ```
28//!
29//! # What is not covered
30//!
31//! A `#[document]` inside an embedded value (its fields share one column, so no
32//! per-field binding; a `#[document]` leaf binds through `ResolvedLens::new`),
33//! a relation inside one, and an embedded enum nested inside an enum variant.
34//! Nesting inside *structs* works at any depth. With JavaScript off every
35//! variant group renders.
36//!
37//! Every leaf under an embedded step is **not required** by default: only the
38//! matching variant writes a variant payload column, so the resolver reports
39//! every embedded leaf nullable.
40

55/// An embedded value that can be read from, and written to, the flat form map.
56///
57/// Derive it with [`tablo_core::EmbeddedForm`](crate::EmbeddedForm), which also
58/// generates `form(dx, parent)`, the value's schema. The hidden methods are
59/// the derive's; the two provided ones are the codec.
60pub trait EmbeddedForm: Sized {
61    /// Write this value's leaves into `out`, under the columns the app schema
62    /// resolves for `parent`.
63    ///
64    /// For an enum, the **active variant's** leaves are written, plus its
65    /// discriminant: a value has one variant, and the form must say which.
66    fn write_form<M>(
67        &self,
68        cx: &Cx,
69        parent: impl Into<Path<M, Self>>,
70        out: &mut HashMap<String, String>,
71    ) where
72        M: toasty::schema::Model,
73    {
74        let schema = Self::build_schema(&DeclCx::from_cx(cx), parent.into());
75        self.write_node(schema.embedded_root(), out);
76    }
77
78    /// Read a value back from a submission.
79    ///
80    /// An embedded enum takes its variant from the discriminant key; a
81    /// submission naming one it does not declare is refused, and one that
82    /// carries no discriminant at all falls back to the first variant whose
83    /// own payload was submitted, else the first variant.
84    ///
85    /// # Errors
86    ///
87    /// Every leaf whose value its type refuses, and a discriminant that names
88    /// no variant.
89    fn read_form<M>(
90        cx: &Cx,
91        parent: impl Into<Path<M, Self>>,
92        values: &HashMap<String, String>,
93    ) -> std::result::Result<Self, Vec<FieldError>>
94    where
95        M: toasty::schema::Model,
96    {
97        let schema = Self::build_schema(&DeclCx::from_cx(cx), parent.into());
98        Self::read_node(schema.embedded_root(), values)
99    }
100
101    /// The value's schema: one node holding its resolved fields.
102    #[doc(hidden)]
103    fn build_schema<M>(dx: &DeclCx, parent: Path<M, Self>) -> Schema
104    where
105        M: toasty::schema::Model;
106
107    /// Whether every leaf answers a blank submission: a declared
108    /// `#[form(blank = ..)]`, the scalar's own blank answer, or — for a nested
109    /// value — every leaf of that value's.
110    ///
111    /// The panel's build check reads it: a control a submission can post empty
112    /// (an optional one, a variant group's payload, or a control inside a
113    /// `Repeater`) whose field answers none is a declaration the panel refuses
114    /// rather than a blank the parse would refuse at submit.
115    #[doc(hidden)]
116    fn answers_blank() -> bool;
117
118    /// Write through a built node.
119    #[doc(hidden)]
120    fn write_node(&self, node: &Embedded, out: &mut HashMap<String, String>);
121
122    /// Read through a built node.
123    #[doc(hidden)]
124    fn read_node(
125        node: &Embedded,
126        values: &HashMap<String, String>,
127    ) -> std::result::Result<Self, Vec<FieldError>>;
128}

130/// An embedded value's schema node: its resolved keys, the fields that render
131/// them, and for an enum its variant control and variant groups.
132#[doc(hidden)]
133#[derive(Debug)]
134pub struct Embedded {
135    shape: Shape,
136}
137
138#[derive(Debug)]
139enum Shape {
140    /// One member per struct field, in declaration order.
141    Struct(Vec<Member>),
142    Enum(EnumNode),
143}
144
145#[derive(Debug)]
146struct EnumNode {
147    /// The discriminant column.
148    key: String,
149    /// The variant control's field slot.
150    discriminant: usize,
151    /// The `#[shared(..)]` columns' field slots: each renders once, outside
152    /// every variant group, because the column belongs to several variants and
153    /// must stay editable whichever one is chosen.
154    shared: Vec<usize>,
155    variants: Vec<Variant>,
156}
157
158#[derive(Debug)]
159struct Variant {
160    /// The discriminant text this variant stores (`2`).
161    value: String,
162    /// One member per payload field, in declaration order.
163    members: Vec<Member>,
164}
165
166#[derive(Debug)]
167enum Member {
168    /// A leaf column: its key, and the field slot that renders it in place —
169    /// `None` for a shared column, which renders outside the variant groups.
170    Leaf {
171        key: String,
172        field: Option<usize>,
173    },
174    Nested(Embedded),
175}
176
177impl Embedded {
178    fn members(&self, variant: Option<usize>) -> &[Member] {
179        match (&self.shape, variant) {
180            (Shape::Struct(members), None) => members,
181            (Shape::Enum(e), Some(index)) => &e.variants[index].members,
182            _ => panic!("a struct member is addressed without a variant, an enum's with one"),
183        }
184    }
185
186    /// The key of leaf `index` (of `variant`, for an enum).
187    pub fn key(&self, variant: Option<usize>, index: usize) -> &str {
188        match &self.members(variant)[index] {
189            Member::Leaf { key, .. } => key,
190            Member::Nested(_) => panic!("member {index} is an embedded value, not a leaf"),
191        }
192    }
193
194    /// The nested value at member `index` (of `variant`, for an enum).
195    pub fn nested(&self, variant: Option<usize>, index: usize) -> &Embedded {
196        match &self.members(variant)[index] {
197            Member::Nested(nested) => nested,
198            Member::Leaf { .. } => panic!("member {index} is a leaf, not an embedded value"),
199        }
200    }
201
202    fn enum_node(&self) -> &EnumNode {
203        match &self.shape {
204            Shape::Enum(e) => e,
205            Shape::Struct(_) => panic!("a struct value has no variant"),
206        }
207    }
208
209    /// Write variant `index`'s discriminant into `out`.
210    pub fn write_variant(&self, index: usize, out: &mut HashMap<String, String>) {
211        let e = self.enum_node();
212        out.insert(e.key.clone(), e.variants[index].value.clone());
213    }
214
215    /// The variant a submission reads as.
216    ///
217    /// A discriminant the submission **names** always wins, and one it names
218    /// but the enum does not declare is refused rather than read as another
219    /// variant. A submission that names none falls back to the first variant,
220    /// in declaration order, with a payload of its own submitted — a shared
221    /// column belongs to several variants and never selects one — else the
222    /// first variant.
223    ///
224    /// # Errors
225    ///
226    /// A discriminant that names no variant, keyed by the discriminant column.
227    pub fn variant_index(
228        &self,
229        values: &HashMap<String, String>,
230    ) -> std::result::Result<usize, Vec<FieldError>> {
231        let e = self.enum_node();
232        let submitted = values.get(&e.key).map(|v| v.trim()).unwrap_or_default();
233        if submitted.is_empty() {
234            let own = |member: &Member| match member {
235                Member::Leaf { key, field } => field.is_some() && is_present(values, key),
236                Member::Nested(nested) => nested.any_present(values),
237            };
238            let inferred = e.variants.iter().position(|v| v.members.iter().any(own));
239            return Ok(inferred.unwrap_or(0));
240        }
241        e.variants
242            .iter()
243            .position(|v| v.value == submitted)
244            .ok_or_else(|| {
245                vec![FieldError::invalid(
246                    e.key.clone(),
247                    format!("`{submitted}` is not a valid variant"),
248                )]
249            })
250    }
251
252    /// Whether a submission mentions any key of this value: a leaf's non-empty
253    /// value, or an enum's discriminant.
254    fn any_present(&self, values: &HashMap<String, String>) -> bool {
255        let member = |member: &Member| match member {
256            Member::Leaf { key, .. } => is_present(values, key),
257            Member::Nested(nested) => nested.any_present(values),
258        };
259        match &self.shape {
260            Shape::Struct(members) => members.iter().any(member),
261            Shape::Enum(e) => {
262                is_present(values, &e.key)
263                    || e.variants.iter().any(|v| v.members.iter().any(member))
264            }
265        }
266    }

268    /// Every form key the value occupies, each once: an enum's discriminant
269    /// first, then the leaf columns in declaration order.
270    pub(crate) fn keys(&self) -> Vec<String> {
271        let mut out = Vec::new();
272        self.collect_keys(&mut out);
273        out
274    }
275
276    fn collect_keys(&self, out: &mut Vec<String>) {
277        let push = |members: &[Member], out: &mut Vec<String>| {
278            for member in members {
279                match member {
280                    Member::Leaf { key, .. } if !out.contains(key) => out.push(key.clone()),
281                    Member::Leaf { .. } => {}
282                    Member::Nested(nested) => nested.collect_keys(out),
283                }
284            }
285        };
286        match &self.shape {
287            Shape::Struct(members) => push(members, out),
288            Shape::Enum(e) => {
289                out.push(e.key.clone());
290                for variant in &e.variants {
291                    push(&variant.members, out);
292                }
293            }
294        }
295    }
296
297    /// Shift every field slot by `by` (see [`Node::offset`]).
298    pub(crate) fn offset(&mut self, by: usize) {
299        let members = |members: &mut Vec<Member>| {
300            for member in members {
301                match member {
302                    Member::Leaf { field, .. } => {
303                        if let Some(index) = field {
304                            *index += by;
305                        }
306                    }
307                    Member::Nested(nested) => nested.offset(by),
308                }
309            }
310        };
311        match &mut self.shape {
312            Shape::Struct(list) => members(list),
313            Shape::Enum(e) => {
314                e.discriminant += by;
315                for index in &mut e.shared {
316                    *index += by;
317                }
318                for variant in &mut e.variants {
319                    members(&mut variant.members);
320                }
321            }
322        }
323    }
324
325    /// Visit every field slot, with where the leaf sits in the form.
326    pub(crate) fn visit_fields(&self, place: LeafPlace, f: &mut impl FnMut(usize, LeafPlace)) {
327        fn members(members: &[Member], place: LeafPlace, f: &mut impl FnMut(usize, LeafPlace)) {
328            for member in members {
329                match member {
330                    Member::Leaf {
331                        field: Some(index), ..
332                    } => f(*index, place),
333                    Member::Leaf { field: None, .. } => {}
334                    Member::Nested(nested) => nested.visit_fields(place, f),
335                }
336            }
337        }
338        match &self.shape {
339            Shape::Struct(list) => members(list, place, f),
340            Shape::Enum(e) => {
341                f(e.discriminant, LeafPlace::Discriminant);
342                for index in &e.shared {
343                    f(*index, place);
344                }
345                for variant in &e.variants {
346                    members(&variant.members, LeafPlace::Payload, f);
347                }
348            }
349        }
350    }

352    /// The field slots of every variant group `values` hides: a group whose
353    /// variant is not the one the submission names. A submission naming no
354    /// variant hides nothing, because the payload fallback may read any group.
355    pub(crate) fn hidden_fields(&self, values: &HashMap<String, String>, out: &mut Vec<usize>) {
356        let nested = |members: &[Member], out: &mut Vec<usize>| {
357            for member in members {
358                if let Member::Nested(nested) = member {
359                    nested.hidden_fields(values, out);
360                }
361            }
362        };
363        match &self.shape {
364            Shape::Struct(members) => nested(members, out),
365            Shape::Enum(e) => {
366                let chosen = values.get(&e.key).map(|v| v.trim()).unwrap_or_default();
367                for variant in &e.variants {
368                    if !chosen.is_empty() && chosen != variant.value {
369                        for member in &variant.members {
370                            match member {
371                                Member::Leaf {
372                                    field: Some(index), ..
373                                } => out.push(*index),
374                                Member::Leaf { field: None, .. } => {}
375                                Member::Nested(nested) => nested
376                                    .visit_fields(LeafPlace::Payload, &mut |index, _| {
377                                        out.push(index)
378                                    }),
379                            }
380                        }
381                    } else {
382                        nested(&variant.members, out);
383                    }
384                }
385            }
386        }
387    }

389    /// Render the value: a struct's members in order; an enum's variant
390    /// control, its shared columns, then its variant groups.
391    ///
392    /// A form renders every variant group, each marked with `data-variant`
393    /// (the value it stores) and `data-variant-of` (the discriminant column),
394    /// which `variant.js` reads to keep only the chosen one visible. A view
395    /// renders only the stored variant's group and the shared columns it
396    /// declares: the rest hold no values.
397    pub(crate) async fn render<'a>(
398        &self,
399        cx: &'a Cx,
400        fields: &[Field],
401        source: &Source<'_>,
402    ) -> Result<BoxView<'a>> {
403        match &self.shape {
404            Shape::Struct(members) => render_members(cx, members, fields, source).await,
405            Shape::Enum(e) => {
406                let mut views = Vec::with_capacity(e.shared.len() + e.variants.len() + 1);
407                views.push(
408                    Node::Field(e.discriminant)
409                        .render(cx, fields, source)
410                        .await?,
411                );
412                let stored = source.value(&e.key).map(str::trim);
413                let stored_variant = e
414                    .variants
415                    .iter()
416                    .find(|variant| stored == Some(variant.value.as_str()));
417                for index in &e.shared {
418                    // A view shows a shared column only when the stored
419                    // variant declares it: another variant's column holds no
420                    // value on this record.
421                    let key = fields[*index].name();
422                    let declared = stored_variant.is_some_and(|variant| {
423                        variant
424                            .members
425                            .iter()
426                            .any(|member| matches!(member, Member::Leaf { key: k, .. } if k == key))
427                    });
428                    if source.mode() == Mode::View && !declared {
429                        continue;
430                    }
431                    views.push(Node::Field(*index).render(cx, fields, source).await?);
432                }
433                for variant in &e.variants {
434                    if source.mode() == Mode::View && stored != Some(variant.value.as_str()) {
435                        continue;
436                    }
437                    let members = render_members(cx, &variant.members, fields, source).await?;
438                    let value = variant.value.clone();
439                    let owner = e.key.clone();
440                    views.push(
441                        view! {
442                            cx =>
443                            ui_field_group(
444                                attrs: attributes! { data-variant=(value) data-variant-of=(owner) },
445                                (members)
446                            )
447                        }
448                        .boxed(),
449                    );
450                }
451                Ok(view! {
452                    cx =>
453                    for v in views {
454                        (v)
455                    }
456                }
457                .boxed())
458            }
459        }
460    }
461}

463/// The members of a struct or a variant group, in order; a shared leaf renders
464/// elsewhere.
465async fn render_members<'a>(
466    cx: &'a Cx,
467    members: &[Member],
468    fields: &[Field],
469    source: &Source<'_>,
470) -> Result<BoxView<'a>> {
471    let mut views = Vec::with_capacity(members.len());
472    for member in members {
473        match member {
474            Member::Leaf {
475                field: Some(index), ..
476            } => views.push(Node::Field(*index).render(cx, fields, source).await?),
477            Member::Leaf { field: None, .. } => {}
478            Member::Nested(nested) => {
479                views.push(Box::pin(nested.render(cx, fields, source)).await?)
480            }
481        }
482    }
483    Ok(view! {
484        cx =>
485        for v in views {
486            (v)
487        }
488    }
489    .boxed())
490}

499/// Builds an embedded value's schema node, one member at a time, in the order
500/// the derive declares them.
501#[doc(hidden)]
502pub struct EmbeddedBuilder {
503    fields: Vec<Field>,
504    shape: Shape,
505    /// The enum variant members are being added to.
506    variant: Option<usize>,
507}
508
509impl EmbeddedBuilder {
510    /// A struct value.
511    pub fn structure() -> Self {
512        Self {
513            fields: Vec::new(),
514            shape: Shape::Struct(Vec::new()),
515            variant: None,
516        }
517    }
518
519    /// An enum value at `parent`: its discriminant column and variants come
520    /// from the app schema `dx` carries, and its variant control is the first
521    /// field.
522    ///
523    /// Panics without an app schema, or when `parent` names no embedded
524    /// enum: every caller is a declaration, and a lens that resolves to nothing
525    /// is a wiring bug.
526    pub fn enumeration<M, T>(dx: &DeclCx, parent: Path<M, T>) -> Self
527    where
528        M: toasty::schema::Model,
529    {
530        let resolver = FieldResolver::new(dx);
531        assert!(
532            resolver.has_schema(),
533            "an embedded enum needs the app schema: build the `DeclCx` from a `Db`"
534        );
535        let shape = resolver.resolve_enum(parent).unwrap_or_else(|| {
536            panic!(
537                "{} is not an embedded enum in this app schema",
538                std::any::type_name::<T>()
539            )
540        });
541        let variants = shape
542            .variants
543            .iter()
544            .map(|(value, _)| Variant {
545                value: value.clone(),
546                members: Vec::new(),
547            })
548            .collect();
549        Self {
550            fields: vec![Field::discriminant(
551                shape.discriminant.clone(),
552                shape.variants,
553            )],
554            shape: Shape::Enum(EnumNode {
555                key: shape.discriminant,
556                discriminant: 0,
557                shared: Vec::new(),
558                variants,
559            }),
560            variant: None,
561        }
562    }
563
564    /// Start the next variant's members.
565    pub fn variant(&mut self) {
566        let next = self.variant.map_or(0, |index| index + 1);
567        let Shape::Enum(e) = &self.shape else {
568            panic!("a struct value has no variant");
569        };
570        assert!(
571            next < e.variants.len(),
572            "the type declares more variants than the app schema"
573        );
574        self.variant = Some(next);
575    }
576
577    fn members_mut(&mut self) -> &mut Vec<Member> {
578        match (&mut self.shape, self.variant) {
579            (Shape::Struct(members), _) => members,
580            (Shape::Enum(e), Some(index)) => &mut e.variants[index].members,
581            (Shape::Enum(_), None) => panic!("an enum member needs `variant()` first"),
582        }
583    }
584
585    /// A leaf rendered in place.
586    pub fn leaf(&mut self, field: impl Into<Field>) {
587        let field = field.into();
588        let key = field.name().to_string();
589        let index = self.fields.len();
590        self.fields.push(field);
591        self.members_mut().push(Member::Leaf {
592            key,
593            field: Some(index),
594        });
595    }
596
597    /// A `#[shared(..)]` leaf: the first variant declaring its column renders
598    /// it, once, outside the variant groups.
599    pub fn shared(&mut self, field: impl Into<Field>) {
600        let field = field.into();
601        let key = field.name().to_string();
602        let Shape::Enum(e) = &mut self.shape else {
603            panic!("a struct value has no shared column");
604        };
605        if !e
606            .shared
607            .iter()
608            .any(|index| self.fields[*index].name() == key)
609        {
610            e.shared.push(self.fields.len());
611            self.fields.push(field);
612        }
613        self.members_mut().push(Member::Leaf { key, field: None });
614    }
615
616    /// A nested value, from its own `build_schema`.
617    pub fn nested(&mut self, schema: Schema) {
618        let Schema { nodes, fields } = schema;
619        let Ok([Node::Embedded(mut nested)]) = <[Node; 1]>::try_from(nodes) else {
620            panic!("a nested value's schema is its one embedded node");
621        };
622        nested.offset(self.fields.len());
623        self.fields.extend(fields);
624        self.members_mut().push(Member::Nested(*nested));
625    }
626
627    /// The value's schema: one embedded node.
628    pub fn finish(self) -> Schema {
629        if let Shape::Enum(e) = &self.shape {
630            assert_eq!(
631                self.variant.map_or(0, |index| index + 1),
632                e.variants.len(),
633                "the type declares fewer variants than the app schema"
634            );
635        }
636        Schema {
637            nodes: vec![Node::Embedded(Box::new(Embedded { shape: self.shape }))],
638            fields: self.fields,
639        }
640    }
641}

654/// Read one leaf out of a submission, by its resolved key.
655///
656/// Trimmed; an absent or empty value is the member's declared blank answer,
657/// else the type's own, and a type with neither is refused inline — the rule
658/// ADR-0022 gives a record form's scalar.
659///
660/// # Errors
661///
662/// A blank the leaf has no answer for, and a value the type cannot parse,
663/// worded as the typed rule words it.
664#[doc(hidden)]
665pub fn parse_leaf<T>(
666    key: &str,
667    values: &HashMap<String, String>,
668    blank: Option<T>,
669) -> std::result::Result<T, FieldError>
670where
671    T: FormScalar,
672{
673    let trimmed = values.get(key).map(|raw| raw.trim()).unwrap_or("");
674    if trimmed.is_empty() {
675        return blank
676            .or_else(T::blank)
677            .ok_or_else(|| FieldError::required(key));
678    }
679    T::parse_form(trimmed).map_err(|message| FieldError::invalid(key, message))
680}
681
682/// Move `result`'s value out, or its errors into `errors`. Generated
683/// `read_node` bodies collect every leaf's error with it.
684#[doc(hidden)]
685pub fn take_leaf<T>(
686    result: std::result::Result<T, FieldError>,
687    errors: &mut Vec<FieldError>,
688) -> Option<T> {
689    result.map_err(|error| errors.push(error)).ok()
690}
691
692/// [`take_leaf`] for a nested value's read.
693#[doc(hidden)]
694pub fn take_value<T>(
695    result: std::result::Result<T, Vec<FieldError>>,
696    errors: &mut Vec<FieldError>,
697) -> Option<T> {
698    result.map_err(|nested| errors.extend(nested)).ok()
699}
