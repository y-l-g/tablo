1//! Unified Schema primitive — layout blocks that compose via `view!`.
2//!
3//! `Schema` is a container for `Section`, `Group`, `Grid`, `Tabs` and
4//! `Repeater` nodes. Each node renders through Topcoat's `view!` macro;
5//! `Schema::render` combines them. The API mirrors Filament's
6//! `Schema::new(( ... ))` tuple form via the `IntoSchema` trait.
7//!
8//! Bridge note: `lens_field` is the one walk reaching into `toasty_core`
9//! (upstream issue #114), alongside the `pk_*` bridge helpers and `cursor.rs`
10//! cursor values. It hands back the built `app::Field`, so field metadata no
11//! longer needs a helper per property; uniqueness comes from
12//! `lens_field_unique`, since Toasty keeps it on the model's index list rather
13//! than the field. Retire the walk when Toasty exposes it (upstream #183).

46/// One control as the record-form checks see it ([`Schema::controls`]).
47#[derive(Debug, Clone)]
48pub(crate) struct Control {
49    /// The key the control posts.
50    pub(crate) name: String,
51    /// Whether an empty submission fails the control's rules.
52    pub(crate) required: bool,
53    /// The message an empty submission produces, when it fails.
54    pub(crate) required_error: Option<String>,
55    /// The container that can skip the control's requiredness, if any: a
56    /// submission may then reach the parse with the control empty.
57    pub(crate) skipped_by: Option<SkippedBy>,
58}
59
60/// A container whose submission can leave its controls' requiredness
61/// unchecked.
62#[derive(Debug, Clone, Copy, PartialEq, Eq)]
63pub(crate) enum SkippedBy {
64    /// An all-empty `Repeater` group.
65    Repeater,
66    /// A variant group the submission's discriminant does not name.
67    VariantGroup,
68}

107    /// Render the schema read-only: the detail page's side of the
108    /// same declaration.
109    ///
110    /// Every field shows the record's stored value in place of its control, so
111    /// a detail page reuses the field types and layout blocks a form already
112    /// declares rather than a parallel infolist vocabulary. `values` is the
113    /// record hydrated exactly as the edit form hydrates it
114    /// ([`RecordForm::hydrate`](crate::RecordForm::hydrate) for a form
115    /// resource): what a user reads on the page is what the form would have
116    /// shown them.
117    pub async fn render_readonly<'a>(
118        &self,
119        cx: &'a Cx,
120        values: &HashMap<String, String>,
121    ) -> Result<BoxView<'a>> {
122        self.render_source(
123            cx,
124            &RenderSource {
125                values,
126                errors: &HashMap::new(),
127                mode: Mode::View,
128            },
129        )
130        .await
131    }

133    /// Rewrite submitted values into their fields' stored spelling.
134    ///
135    /// Runs after validation and before a record fn sees the map, so a typed
136    /// field's `Display` — not the browser's spelling — is what gets written.
137    /// That is what makes an untouched edit round-trip: the form hydrates a
138    /// stored value, the browser echoes it, and this puts back the same string
139    /// the record fn would have produced.
140    ///
141    /// A `Select` takes its trimmed submission: its presence rule and its
142    /// option-existence check both read `value.trim()`, so the trimmed value is
143    /// the one that passed, and storing the untrimmed spelling would store a
144    /// value no rule authorised.
145    ///
146    /// A parse failure here leaves the submission untouched and reports nothing:
147    /// it is unreachable from the handlers, and a silent rewrite would hide a
148    /// bypass rather than surface it. A field with no submission keeps its
149    /// absence, and an **empty** submission stays empty: a typed column has no
150    /// spelling for "no value", so empty is the presence rule's business —
151    /// `.required()` refuses it inline, and an optional typed field reaches the
152    /// record form's parse as `""`, which reads it as the field's blank answer.
153    pub fn normalize_values(&self, values: &mut HashMap<String, String>) {
154        for (name, input) in self.text_inputs() {
155            let Some(submitted) = values.get(&name) else {
156                continue;
157            };
158            if submitted.trim().is_empty() {
159                continue;
160            }
161            if let Ok(normalized) = input.normalize(submitted) {
162                values.insert(name, normalized);
163            }
164        }
165        for name in self.select_inputs().keys() {
166            let Some(submitted) = values.get(name).cloned() else {
167                continue;
168            };
169            let trimmed = submitted.trim();
170            if trimmed.len() != submitted.len() {
171                values.insert(name.clone(), trimmed.to_string());
172            }
173        }
174    }

176    /// Append another schema's nodes after this one's.
177    ///
178    /// [`Schema::new`] composes through `IntoSchema`, whose tuple form stops at
179    /// eight nodes; a derived embedded form has one control per leaf column and
180    /// composes nested values, so it builds its schema by appending instead. The
181    /// nodes keep their order, so a form reads in declaration order either way.
182    pub fn extend(mut self, other: Schema) -> Schema {
183        self.nodes.extend(other.nodes);
184        // The same guard `Schema::new` runs: a derived form is built by
185        // appending, so this is the only check for the shapes `new` cannot see.
186        self.assert_unique_field_names();
187        self
188    }

208    /// The one node walk: every node renders its static form.
209    pub(crate) async fn render_source<'a>(
210        &self,
211        cx: &'a Cx,
212        source: &RenderSource<'_>,
213    ) -> Result<BoxView<'a>> {
214        let mut views = Vec::with_capacity(self.nodes.len());
215        for node in &self.nodes {
216            views.push(Box::pin(node.render_source(cx, source)).await?.boxed());
217        }
218        Ok(view! {
219            cx =>
220            for v in views {
221                (v)
222            }
223        }
224        .boxed())
225    }

227    /// Every control, in declaration order, with what an empty submission
228    /// does to it: the schema's own required message when it refuses one, and
229    /// the container that can skip its requiredness — an all-empty `Repeater`
230    /// or a variant group the discriminant does not name
231    /// ([`walk_absent_groups`]).
232    pub(crate) fn controls(&self) -> Vec<Control> {
233        fn walk(nodes: &[Node], skipped_by: Option<SkippedBy>, out: &mut Vec<Control>) {
234            let empty = HashMap::new();
235            for node in nodes {
236                if let Some((name, errors)) = validate_leaf(node, &empty) {
237                    out.push(Control {
238                        name: name.to_string(),
239                        required: !errors.is_empty(),
240                        required_error: errors.into_iter().next(),
241                        skipped_by,
242                    });
243                }
244                if let Some(child) = node.children() {
245                    let inner = match node {
246                        Node::Repeater(_) => Some(SkippedBy::Repeater),
247                        Node::Group(group) if group.is_variant() => Some(SkippedBy::VariantGroup),
248                        _ => None,
249                    };
250                    walk(&child.nodes, skipped_by.or(inner), out);
251                }
252            }
253        }
254        let mut out = Vec::new();
255        walk(&self.nodes, None, &mut out);
256        out
257    }

259    /// Collect field names for validation (TextInput + Textarea + Select + FileUpload).
260    pub fn field_names(&self) -> Vec<String> {
261        let mut out = Vec::new();
262        for node in &self.nodes {
263            for_each_field(node, &mut |n| match n {
264                Node::TextInput(f) => out.push(f.field_name().to_string()),
265                Node::Textarea(f) => out.push(f.field_name().to_string()),
266                Node::Select(f) => out.push(f.field_name().to_string()),
267                Node::FileUpload(f) => out.push(f.field_name().to_string()),
268                _ => {}
269            });
270        }
271        out
272    }

274    /// Keys in `values` that no declared input owns, sorted.
275    ///
276    /// Framework-level allow-list seam, enforced by the create/edit POST
277    /// handlers (unknown keys → 400): record handlers already whitelist via
278    /// per-field `.get(..)`, but a generic impl iterating `values` would
279    /// silently promote `role`/`tenant_id`/handler keys (`csrf_token`,
280    /// `confirm`, `ids`) to client-controlled writes. The transport keys the
281    /// handlers own (`csrf_token`, `clear_<field>`, `keep_<field>`) are
282    /// stripped before the record fns run, so a generic impl cannot
283    /// promote those either; `confirm`/`ids` are only read, never written.
284    /// Callers should reject or ignore the rest (at least `debug_assert!` in
285    /// tests); handler keys must be filtered by the caller before calling this.
286    pub fn unknown_keys(&self, values: &HashMap<String, String>) -> Vec<String> {
287        use std::collections::HashSet;
288        let known: HashSet<String> = self.field_names().into_iter().collect();
289        let mut out: Vec<String> = values
290            .keys()
291            .filter(|k| !known.contains(k.as_str()))
292            .cloned()
293            .collect();
294        out.sort();
295        out
296    }

309    /// Every leaf `pick` selects, keyed by field name — the one walk behind the
310    /// per-kind accessors. `pick` answers a node's `(field name,
311    /// leaf)`, or `None` when the node is not that kind.
312    fn leaves<T>(&self, pick: impl Fn(&Node) -> Option<(&str, T)>) -> HashMap<String, T> {
313        let mut map = HashMap::new();
314        for node in &self.nodes {
315            for_each_field(node, &mut |n| {
316                if let Some((name, leaf)) = pick(n) {
317                    map.insert(name.to_string(), leaf);
318                }
319            });
320        }
321        map
322    }
323
324    /// Whether any leaf satisfies `pick` — the allocation-free counterpart of
325    /// [`Self::leaves`] for a yes/no question.
326    fn any_leaf(&self, pick: impl Fn(&Node) -> bool) -> bool {
327        let mut found = false;
328        for node in &self.nodes {
329            for_each_field(node, &mut |n| found |= pick(n));
330        }
331        found
332    }

365    /// Validate submitted values against declared inputs.
366    ///
367    /// Absent keys are treated as `""`. The edit handler completes every key
368    /// the submission did not post from the stored record before it validates,
369    /// so an omitted key validates as its stored value there
370    /// ([`Resource::update_record`](crate::Resource::update_record)). Use [`Self::unknown_keys`] to
371    /// allow-list POST keys.
372    ///
373    /// A field a submission hides is not validated: an all-empty
374    /// Repeater group is absent, and a variant group the submission's
375    /// discriminant does not name is not rendered by `variant.js`, so neither
376    /// can fail the submit for a value the user cannot see.
377    pub fn validate(&self, values: &HashMap<String, String>) -> HashMap<String, Vec<String>> {
378        // Classify the groups first: an all-empty group is
379        // "absent" — an untouched group submits empty strings (or omits the
380        // keys), both treated as absent — so its inner inputs must not fail
381        // the submit for any requiredness. A `required` repeater answers with
382        // its one label-keyed error instead, and required repeaters nested
383        // inside an absent group are suppressed with it. A partially filled
384        // group (any inner value non-empty) enforces inner `required` as
385        // usual. Whitespace-only values count as empty, matching the
386        // codebase-wide trim convention. A variant group the submission's
387        // discriminant does not name is hidden with its subtree, which is what
388        // makes validation agree with the render.
389        let mut errors: HashMap<String, Vec<String>> = HashMap::new();
390        let mut skip: HashSet<String> = HashSet::new();
391        walk_absent_groups(&self.nodes, values, &mut skip, &mut errors, false);
392        // One walk, one match per node (`validate_leaf`): the single place a
393        // field kind joins validation.
394        for node in &self.nodes {
395            for_each_field(node, &mut |n| {
396                let Some((name, errs)) = validate_leaf(n, values) else {
397                    return;
398                };
399                if !skip.contains(name) && !errs.is_empty() {
400                    errors.insert(name.to_string(), errs);
401                }
402            });
403        }
404        errors
405    }
406
407    /// Field names a submission leaves out of validation:
408    /// the same classification `validate` uses — an all-empty Repeater group
409    /// is absent, and a variant group the discriminant does not name is hidden
410    /// — minus the required-group errors, which validation already reported.
411    /// `check_unique` consults it so an untouched group is never unique-checked
412    /// while validation calls it clean, and `validate_async` so a hidden
413    /// group's select is not probed for existence.
414    pub(crate) fn absent_fields(&self, values: &HashMap<String, String>) -> HashSet<String> {
415        let mut skip = HashSet::new();
416        let mut discarded = HashMap::new();
417        walk_absent_groups(&self.nodes, values, &mut skip, &mut discarded, false);
418        skip
419    }

421    /// Async validation for Select relationship existence (tenancy-aware).
422    ///
423    /// A field `validate` skipped is skipped here too: an absent
424    /// repeater group or a hidden variant group holds no value the user can
425    /// see, so its select must not be probed for existence.
426    pub async fn validate_async(
427        &self,
428        cx: &Cx,
429        values: &HashMap<String, String>,
430    ) -> HashMap<String, Vec<String>> {
431        let mut errors = self.validate(values);
432        let absent = self.absent_fields(values);
433        for (name, sel) in self.select_inputs() {
434            if errors.contains_key(&name) || absent.contains(&name) {
435                continue;
436            }
437            if sel.relationship.is_some() || !sel.options_static.is_empty() {
438                let val = values.get(&name).map(|s| s.as_str()).unwrap_or("");
439                if !val.trim().is_empty() {
440                    // `validate` above already ran the required rule, so the
441                    // existence-only check is what is left to ask.
442                    let existence_errs = sel.validate_exists(cx, val).await;
443                    if !existence_errs.is_empty() {
444                        errors.insert(name, existence_errs);
445                    }
446                }
447            }
448        }
449        errors
450    }
