1//! Unified Schema primitive — fields and layout blocks that compose via `view!`.
2//!
3//! `Schema` holds a tree of `Section`, `Group`, `Grid`, and `Repeater`
4//! blocks, embedded values, and [`Field`] slots. Building it resolves every
5//! field once into one list, which rendering, validation, and the panel's
6//! checks read. The API mirrors Filament's `Schema::new(( ... ))` tuple form
7//! via the `IntoSchema` trait.
8//!
9//! Bridge note: `lens_field` is the one walk reaching into `toasty_core`
10//! (upstream issue #114), alongside the `pk_*` bridge helpers and `cursor.rs`
11//! cursor values. It hands back the built `app::Field`, so field metadata
12//! needs no helper per property; uniqueness comes from `lens_field_unique`,
13//! since Toasty keeps it on the model's index list rather than the field.
14//! Retire the walk when Toasty exposes it (upstream #183).

40/// One control as the record-form checks see it ([`Schema::controls`]).
41#[derive(Debug, Clone)]
42pub(crate) struct ControlCheck {
43    /// The key the control posts.
44    pub(crate) name: String,
45    /// Whether an empty submission fails the control's rules.
46    pub(crate) required: bool,
47    /// The message an empty submission produces, when it fails.
48    pub(crate) required_error: Option<String>,
49    /// Whether the control sits inside a `Repeater`, whose all-empty group
50    /// skips its requiredness: a submission may then reach the parse with the
51    /// control empty.
52    pub(crate) in_repeater: bool,
53}

55/// The container that composes fields and layout blocks.
56///
57/// The field list is resolved once, when the schema is built: every
58/// [`Field`] a block or an embedded value holds moves into it, and the tree
59/// holds slots into it.
60#[derive(Debug, Default)]
61pub struct Schema {
62    pub(crate) nodes: Vec<Node>,
63    pub(crate) fields: Vec<Field>,
64}
65
66impl Schema {
67    /// Whether this schema declares nothing to render.
68    ///
69    /// Public because [`Resource::view`](crate::resource::Resource::view) defaults to
70    /// this and [`Resource::viewed`](crate::resource::Resource::viewed) reads it
71    /// as "no detail page declared".
72    pub fn is_empty(&self) -> bool {
73        self.nodes.is_empty()
74    }

76    /// Build a `Schema` from any `IntoSchema` (a field, a block, a tuple, or a
77    /// `Schema`).
78    ///
79    /// Panics on duplicate field names: two inputs sharing one name
80    /// render two `<input name="x">`, POST one value to both, and collapse to
81    /// one validation rule.
82    pub fn new(children: impl IntoSchema) -> Self {
83        let schema = children.into_schema();
84        schema.assert_unique_field_names();
85        schema
86    }
87
88    /// An empty schema (no nodes).
89    pub fn empty() -> Self {
90        Self::default()
91    }

93    /// Every field, in declaration order.
94    pub fn fields(&self) -> impl Iterator<Item = &Field> {
95        self.fields.iter()
96    }

98    /// Render the schema from `source`: a form's controls
99    /// ([`Source::form`]) or a record's read-only values ([`Source::view`]).
100    pub async fn render<'a>(&self, cx: &'a Cx, source: Source<'_>) -> Result<BoxView<'a>> {
101        render_nodes(cx, &self.nodes, &self.fields, &source).await
102    }

104    /// Append `other`'s nodes and fields after this one's, re-slotting its
105    /// field slots past this schema's fields.
106    pub(crate) fn append(&mut self, other: Schema) {
107        let Schema { mut nodes, fields } = other;
108        let offset = self.fields.len();
109        for node in &mut nodes {
110            node.offset(offset);
111        }
112        self.fields.extend(fields);
113        self.nodes.extend(nodes);
114    }
115
116    /// The embedded node a derived value's schema holds.
117    pub(crate) fn embedded_root(&self) -> &embedded::Embedded {
118        match self.nodes.as_slice() {
119            [Node::Embedded(node)] => node,
120            _ => panic!("an embedded value's schema is its one embedded node"),
121        }
122    }

124    /// Rewrite submitted values into their fields' stored spelling.
125    ///
126    /// Runs after validation and before a record fn sees the map, so a typed
127    /// field's `Display` — not the browser's spelling — is what gets written.
128    /// That is what makes an untouched edit round-trip: the form hydrates a
129    /// stored value, the browser echoes it, and this puts back the same string
130    /// the record fn would have produced.
131    ///
132    /// A choice takes its trimmed submission: its presence rule and its
133    /// option-existence check both read `value.trim()`, so the trimmed value is
134    /// the one that passed, and storing the untrimmed spelling would store a
135    /// value no rule authorised.
136    ///
137    /// A parse failure here leaves the submission untouched and reports nothing:
138    /// it is unreachable from the handlers, and a silent rewrite would hide a
139    /// bypass rather than surface it. A field with no submission keeps its
140    /// absence, and an **empty** submission stays empty: a typed column has no
141    /// spelling for "no value", so empty is the presence rule's business —
142    /// `.required()` refuses it inline, and an optional typed field reaches the
143    /// record form's parse as `""`, which reads it as the field's blank answer.
144    pub fn normalize_values(&self, values: &mut HashMap<String, String>) {
145        for field in &self.fields {
146            let Some(submitted) = values.get_mut(field.name()) else {
147                continue;
148            };
149            if submitted.trim().is_empty() {
150                continue;
151            }
152            if let Ok(normalized) = field.normalize(submitted) {
153                *submitted = normalized;
154            }
155        }
156    }

158    /// Append another schema's nodes after this one's.
159    ///
160    /// [`Schema::new`] composes through `IntoSchema`, whose tuple form stops at
161    /// eight nodes; a longer form appends instead. The nodes keep their order,
162    /// so a form reads in declaration order either way.
163    pub fn extend(mut self, other: Schema) -> Schema {
164        self.append(other);
165        // The same guard `Schema::new` runs: this is the only check for the
166        // shapes `new` cannot see.
167        self.assert_unique_field_names();
168        self
169    }
170
171    /// Every field, in declaration order, with what an empty submission
172    /// does to it: the field's own required message when it refuses one, and
173    /// whether an all-empty `Repeater` can skip its requiredness
174    /// ([`walk_absent_groups`]).
175    ///
176    /// A variant group can skip requiredness too, but only an embedded value's
177    /// fields sit in one, and its record-form field reads an empty key as
178    /// `Default`, so the checks have nothing to ask of it.
179    pub(crate) fn controls(&self) -> Vec<ControlCheck> {
180        fn mark(nodes: &[Node], inside: bool, out: &mut [bool]) {
181            for node in nodes {
182                match node {
183                    Node::Repeater(r) => mark(&r.children.nodes, true, out),
184                    Node::Field(_) | Node::Embedded(_) => {
185                        node.visit_fields(&mut |index, _| out[index] = inside)
186                    }
187                    _ => mark(node.children().unwrap_or_default(), inside, out),
188                }
189            }
190        }
191        let mut in_repeater = vec![false; self.fields.len()];
192        mark(&self.nodes, false, &mut in_repeater);
193        self.fields
194            .iter()
195            .zip(in_repeater)
196            .map(|(field, in_repeater)| {
197                let errors = field.validate("");
198                ControlCheck {
199                    name: field.name().to_string(),
200                    required: !errors.is_empty(),
201                    required_error: errors.into_iter().next(),
202                    in_repeater,
203                }
204            })
205            .collect()
206    }

208    /// Keys in `values` that no declared input owns, sorted.
209    ///
210    /// Framework-level allow-list seam, enforced by the create/edit POST
211    /// handlers (unknown keys → 400): record handlers already whitelist via
212    /// per-field `.get(..)`, but a generic impl iterating `values` would
213    /// silently promote `role`/`tenant_id`/handler keys (`csrf_token`,
214    /// `confirm`, `ids`) to client-controlled writes. The transport keys the
215    /// handlers own (`csrf_token`, `clear_<field>`, `keep_<field>`) are
216    /// stripped before the record fns run, so a generic impl cannot
217    /// promote those either; `confirm`/`ids` are only read, never written.
218    /// Callers should reject or ignore the rest (at least `debug_assert!` in
219    /// tests); handler keys must be filtered by the caller before calling this.
220    pub fn unknown_keys(&self, values: &HashMap<String, String>) -> Vec<String> {
221        let known: HashSet<&str> = self.fields.iter().map(Field::name).collect();
222        let mut out: Vec<String> = values
223            .keys()
224            .filter(|k| !known.contains(k.as_str()))
225            .cloned()
226            .collect();
227        out.sort();
228        out
229    }

231    fn assert_unique_field_names(&self) {
232        let mut seen = HashSet::new();
233        for field in &self.fields {
234            assert!(
235                seen.insert(field.name()),
236                "duplicate field name '{}': each Schema input needs a distinct field (GH #100)",
237                field.name()
238            );
239        }
240    }

242    /// Validate submitted values against declared inputs.
243    ///
244    /// Absent keys are treated as `""`. The edit handler completes every key
245    /// the submission did not post from the stored record before it validates,
246    /// so an omitted key validates as its stored value there
247    /// ([`Resource::update_record`](crate::Resource::update_record)). Use [`Self::unknown_keys`] to
248    /// allow-list POST keys.
249    ///
250    /// A field a submission hides is not validated: an all-empty
251    /// Repeater group is absent, and a variant group the submission's
252    /// discriminant does not name is not rendered by `variant.js`, so neither
253    /// can fail the submit for a value the user cannot see.
254    pub fn validate(&self, values: &HashMap<String, String>) -> HashMap<String, Vec<String>> {
255        // Classify the groups first: an all-empty group is
256        // "absent" — an untouched group submits empty strings (or omits the
257        // keys), both treated as absent — so its inner inputs must not fail
258        // the submit for any requiredness. A `required` repeater answers with
259        // its one label-keyed error instead, and required repeaters nested
260        // inside an absent group are suppressed with it. A partially filled
261        // group (any inner value non-empty) enforces inner `required` as
262        // usual. Whitespace-only values count as empty, matching the
263        // codebase-wide trim convention. A variant group the submission's
264        // discriminant does not name is hidden with its subtree, which is what
265        // makes validation agree with the render.
266        let mut errors: HashMap<String, Vec<String>> = HashMap::new();
267        let mut skip: HashSet<String> = HashSet::new();
268        walk_absent_groups(
269            &self.nodes,
270            &self.fields,
271            values,
272            &mut skip,
273            &mut errors,
274            false,
275        );
276        for field in &self.fields {
277            if skip.contains(field.name()) {
278                continue;
279            }
280            let value = values.get(field.name()).map(String::as_str).unwrap_or("");
281            let errs = field.validate(value);
282            if !errs.is_empty() {
283                errors.insert(field.name().to_string(), errs);
284            }
285        }
286        errors
287    }
288
289    /// Field names a submission leaves out of validation:
290    /// the same classification `validate` uses — an all-empty Repeater group
291    /// is absent, and a variant group the discriminant does not name is hidden
292    /// — minus the required-group errors, which validation already reported.
293    /// `check_unique` consults it so an untouched group is never unique-checked
294    /// while validation calls it clean, and `validate_async` so a hidden
295    /// group's select is not probed for existence.
296    pub(crate) fn absent_fields(&self, values: &HashMap<String, String>) -> HashSet<String> {
297        let mut skip = HashSet::new();
298        let mut discarded = HashMap::new();
299        walk_absent_groups(
300            &self.nodes,
301            &self.fields,
302            values,
303            &mut skip,
304            &mut discarded,
305            false,
306        );
307        skip
308    }
309
310    /// [`Self::validate`], then each choice's option existence
311    /// (tenancy-aware for a relationship).
312    ///
313    /// A field `validate` skipped is skipped here too: an absent
314    /// repeater group or a hidden variant group holds no value the user can
315    /// see, so its choice must not be probed for existence.
316    pub async fn validate_async(
317        &self,
318        cx: &Cx,
319        values: &HashMap<String, String>,
320    ) -> HashMap<String, Vec<String>> {
321        let mut errors = self.validate(values);
322        let absent = self.absent_fields(values);
323        for field in &self.fields {
324            let name = field.name();
325            if errors.contains_key(name) || absent.contains(name) {
326                continue;
327            }
328            let Some(value) = values.get(name) else {
329                continue;
330            };
331            // `validate` above already ran the required rule, so the
332            // existence-only check is what is left to ask.
333            let existence = field.validate_exists(cx, value).await;
334            if !existence.is_empty() {
335                errors.insert(name.to_string(), existence);
336            }
337        }
338        errors
339    }
