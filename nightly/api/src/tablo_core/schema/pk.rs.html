3/// Parses a URL path segment into `M`'s primary-key value through `toasty_core` (upstream #114) and
4/// returns `None` when the id does not parse or the key is not a single primitive field.
5fn pk_field_value<M>(
6    id: &str,
7) -> Option<(toasty_core::schema::app::FieldId, toasty_core::stmt::Value)>
8where
9    M: toasty::schema::Model,
10{
11    let app_model = M::schema();
12    let root = app_model.as_root()?;
13    if root.primary_key.fields.len() != 1 {
14        return None;
15    }
16    let fid = root.primary_key.fields.first().copied()?;
17    let model_field = app_model.fields().get(fid.index)?;
18    let toasty_core::schema::app::FieldTy::Primitive(prim) = &model_field.ty else {
19        return None;
20    };
21    let value = match prim.ty {
22        toasty_core::stmt::Type::Uuid => toasty_core::stmt::Value::Uuid(id.parse().ok()?),
23        toasty_core::stmt::Type::String => toasty_core::stmt::Value::String(id.to_string()),
24        toasty_core::stmt::Type::Bool => toasty_core::stmt::Value::Bool(id.parse().ok()?),
25        toasty_core::stmt::Type::I8 => toasty_core::stmt::Value::I8(id.parse().ok()?),
26        toasty_core::stmt::Type::I16 => toasty_core::stmt::Value::I16(id.parse().ok()?),
27        toasty_core::stmt::Type::I32 => toasty_core::stmt::Value::I32(id.parse().ok()?),
28        toasty_core::stmt::Type::I64 => toasty_core::stmt::Value::I64(id.parse().ok()?),
29        toasty_core::stmt::Type::U8 => toasty_core::stmt::Value::U8(id.parse().ok()?),
30        toasty_core::stmt::Type::U16 => toasty_core::stmt::Value::U16(id.parse().ok()?),
31        toasty_core::stmt::Type::U32 => toasty_core::stmt::Value::U32(id.parse().ok()?),
32        toasty_core::stmt::Type::U64 => toasty_core::stmt::Value::U64(id.parse().ok()?),
33        toasty_core::stmt::Type::F32 => toasty_core::stmt::Value::F32(id.parse().ok()?),
34        toasty_core::stmt::Type::F64 => toasty_core::stmt::Value::F64(id.parse().ok()?),
35        toasty_core::stmt::Type::Bytes => toasty_core::stmt::Value::Bytes(id.as_bytes().to_vec()),
36        toasty_core::stmt::Type::Timestamp => toasty_core::stmt::Value::Timestamp(id.parse().ok()?),
37        toasty_core::stmt::Type::Date => toasty_core::stmt::Value::Date(id.parse().ok()?),
38        toasty_core::stmt::Type::Time => toasty_core::stmt::Value::Time(id.parse().ok()?),
39        toasty_core::stmt::Type::DateTime => toasty_core::stmt::Value::DateTime(id.parse().ok()?),
40        toasty_core::stmt::Type::Zoned => toasty_core::stmt::Value::Zoned(id.parse().ok()?),
41        _ => return None,
42    };
43    Some((fid, value))
44}

46/// Reports whether `M`'s primary key is composite, which has no URL representation.
47pub(crate) fn pk_is_composite<M>() -> bool
48where
49    M: toasty::schema::Model,
50{
51    M::schema()
52        .as_root()
53        .is_some_and(|root| root.primary_key.fields.len() > 1)
54}

56/// Builds the equality predicate on `M`'s primary key for a URL id and returns `None` when the id
57/// does not parse or the key is not a single primitive field.
58pub(crate) fn pk_eq_expr<M>(id: &str) -> Option<toasty::stmt::Expr<bool>>
59where
60    M: toasty::schema::Model,
61{
62    let (fid, value) = pk_field_value::<M>(id)?;
63    let cond = toasty_core::stmt::Expr::eq(
64        toasty_core::stmt::Expr::ref_self_field(fid),
65        toasty_core::stmt::Expr::from(value),
66    );
67    Some(toasty::stmt::Expr::from_untyped(cond))
68}

70/// Builds the `IN` predicate over primary keys for a bulk id list and returns `None` when any id
71/// fails to parse or the key is not a single primitive field.
72pub(crate) fn pk_in_expr<M>(ids: &[&str]) -> Option<toasty::stmt::Expr<bool>>
73where
74    M: toasty::schema::Model,
75{
76    let mut parsed = ids.iter().map(|id| pk_field_value::<M>(id));
77    let (fid, first) = parsed.next()??;
78    let mut values = vec![first];
79    for item in parsed {
80        let (f, v) = item?;
81        debug_assert_eq!(
82            f.index, fid.index,
83            "pk_in_expr: one model, one PK field — mixed fields are a bug"
84        );
85        values.push(v);
86    }
87    let cond = toasty_core::stmt::Expr::in_list(
88        toasty_core::stmt::Expr::ref_self_field(fid),
89        toasty_core::stmt::Expr::list(values),
90    );
91    Some(toasty::stmt::Expr::from_untyped(cond))
92}
