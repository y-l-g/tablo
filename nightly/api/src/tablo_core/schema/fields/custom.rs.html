1//! App-defined field controls: the [`Control`] trait, what it renders from
2//! ([`ControlInput`]), and the built-in [`Toggle`] written against it.
3
4use topcoat::{context::Cx, view::*};
5
6/// A form control a [`Field`](super::Field) renders through
7/// [`Field::custom`](super::Field::custom).
8///
9/// The field keeps what every field shares: its key, its label, the required
10/// rule, validation, the error slot, and the chrome around the control. The
11/// control renders the input itself, from a [`ControlInput`], and, on the
12/// detail page, the stored value.
13///
14/// The submission is read like any other field's: the form field named
15/// [`ControlInput::name`] is the value, last one wins. A control that submits
16/// nothing in some state (a checkbox left unchecked) renders a hidden field of
17/// the same name before it, carrying the value that state means, as
18/// [`Toggle`] does.
19///
20/// ```ignore
21/// struct Color;
22///
23/// impl Control for Color {
24///     fn render<'a>(&self, cx: &'a Cx, input: ControlInput) -> BoxView<'a> {
25///         let attrs = input.attributes(cx);
26///         view! { cx => <input type="color" (attrs)> }.boxed()
27///     }
28/// }
29///
30/// Field::custom(Theme::fields().accent(), Color)
31/// ```
32pub trait Control: Send + Sync {
33    /// The input, named and identified by [`ControlInput::name`].
34    fn render<'a>(&self, cx: &'a Cx, input: ControlInput) -> BoxView<'a>;
35
36    /// The stored value as the detail page shows it. Defaults to the value as
37    /// text.
38    fn display<'a>(&self, cx: &'a Cx, value: &str) -> BoxView<'a> {
39        let text = value.to_string();
40        view! {
41            cx =>
42            <div class="text-sm text-foreground wrap-anywhere whitespace-pre-wrap">
43                (text)
44            </div>
45        }
46        .boxed()
47    }
48}

50/// What a [`Control`] renders for one form: the field's key, its current
51/// value, and its validation state.
52#[derive(Debug, Clone)]
53pub struct ControlInput {
54    name: String,
55    value: Option<String>,
56    required: bool,
57    invalid: bool,
58    described_by: Option<String>,
59}
60
61impl ControlInput {
62    pub(crate) fn new(
63        name: &str,
64        value: Option<&str>,
65        required: bool,
66        invalid: bool,
67        described_by: Option<String>,
68    ) -> Self {
69        Self {
70            name: name.to_string(),
71            value: value.map(str::to_string),
72            required,
73            invalid,
74            described_by,
75        }
76    }
77
78    /// The field's key: the control's `name` and `id`, which the field's
79    /// label points at.
80    pub fn name(&self) -> &str {
81        &self.name
82    }
83
84    /// The value to show: the record's stored value on an edit, the
85    /// submission on a re-rendered form, `None` on an empty create.
86    pub fn value(&self) -> Option<&str> {
87        self.value.as_deref()
88    }
89
90    /// Whether an empty submission fails validation.
91    pub fn required(&self) -> bool {
92        self.required
93    }
94
95    /// Whether the field carries an error.
96    pub fn invalid(&self) -> bool {
97        self.invalid
98    }
99
100    /// The id of the error message while the field carries one: the
101    /// control's `aria-describedby`.
102    pub fn described_by(&self) -> Option<&str> {
103        self.described_by.as_deref()
104    }
105
106    /// The attributes a single input carries: `id`, `name`, `value`,
107    /// `required`, `aria-required`, `aria-invalid`, and `aria-describedby`
108    /// pointing at the error while there is one.
109    pub fn attributes(&self, cx: &Cx) -> Attributes {
110        let required = self.required;
111        attributes! {
112            cx =>
113            id=(self.name.clone())
114            name=(self.name.clone())
115            value=(self.value.clone())
116            required=(required)
117            aria-required=(required.then_some("true"))
118            aria-invalid=(if self.invalid { "true" } else { "false" })
119            aria-describedby=(self.described_by.clone())
120        }
121    }
122}

124/// A checkbox over a `bool` field, built on [`Control`] alone:
125/// [`Field::toggle`](super::Field::toggle) declares one.
126///
127/// A hidden `false` precedes the checkbox under the same name, so an
128/// unchecked box submits `false` rather than nothing; a checked one submits
129/// `true` after it, and the last value wins. The detail page reads
130/// `"Yes"`/`"No"`.
131#[derive(Debug, Clone, Copy, Default)]
132pub struct Toggle;
133
134impl Control for Toggle {
135    fn render<'a>(&self, cx: &'a Cx, input: ControlInput) -> BoxView<'a> {
136        let name = input.name().to_string();
137        let checked = input.value().is_some_and(|v| v.trim() == "true");
138        let invalid = if input.invalid() { "true" } else { "false" };
139        let described_by = input.described_by().map(str::to_string);
140        view! {
141            cx =>
142            <input type="hidden" name=(name.clone()) value="false">
143            <input
144                type="checkbox"
145                id=(name.clone())
146                name=(name)
147                value="true"
148                checked=(checked)
149                aria-invalid=(invalid)
150                aria-describedby=(described_by)
151                class="size-4 accent-primary"
152            >
153        }
154        .boxed()
155    }
156
157    fn display<'a>(&self, cx: &'a Cx, value: &str) -> BoxView<'a> {
158        let text = if value.trim() == "true" { "Yes" } else { "No" };
159        view! { cx => <div class="text-sm text-foreground">(text)</div> }.boxed()
160    }
161}
