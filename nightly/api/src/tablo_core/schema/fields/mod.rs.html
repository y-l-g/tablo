29/// One form field binding a lens to the control editing it, offering only that control's modifiers
30/// so a modifier on the wrong control does not compile.
31///
32/// ```ignore
33/// Field::text(User::fields().name()).placeholder("Ada Lovelace")   // TextField
34/// Field::text(User::fields().email()).email().unique()
35/// Field::text(User::fields().age())                  // typed: an `i64` column
36/// Field::text(Post::fields().body()).multiline(6)
37/// Field::choice(Post::fields().status()).options(Status::options()) // ChoiceField
38/// Field::choice(Post::fields().author_id()).relationship::<AuthorResource>(..)
39/// Field::file(Doc::fields().path())                                  // FileField
40/// Field::toggle(Post::fields().featured())                           // CustomField
41/// ```
42///
43/// ```compile_fail
44/// # #[derive(Debug, Clone, toasty::Model)]
45/// # struct User { #[key] #[auto] id: uuid::Uuid, name: String }
46/// # fn main() {
47/// tablo_core::Field::text(User::fields().name()).options(["admin", "member"]);
48/// # }
49/// ```
50pub struct Field {
51    name: String,
52    label: String,
53    required: bool,
54    /// Whether the column stores NULL for an empty submission, which a unique
55    /// index admits any number of times.
56    nullable: bool,
57    /// The email and scalar-parse rules, with their messages.
58    rules: Rules,
59    control: ControlKind,
60    /// The declaration error when the lens binds no column.
61    misdeclared: Option<String>,
62}
63
64/// The control a [`Field`] renders, with what only that control declares.
65pub(crate) enum ControlKind {
66    /// A one-line `<input>`, or a `<textarea>` when `rows` is set.
67    Text(TextControl),
68    /// A `<select>` over static options or a relationship's rows.
69    Choice(ChoiceControl),
70    /// A file input storing the uploaded file's path.
71    File,
72    /// An app's control, or the built-in [`Toggle`].
73    Custom(Arc<dyn Control>),
74}

76impl std::fmt::Debug for Field {
77    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
78        let control = match &self.control {
79            ControlKind::Text(_) => "text",
80            ControlKind::Choice(_) => "choice",
81            ControlKind::File => "file",
82            ControlKind::Custom(_) => "custom",
83        };
84        f.debug_struct("Field")
85            .field("name", &self.name)
86            .field("label", &self.label)
87            .field("required", &self.required)
88            .field("control", &control)
89            .finish()
90    }
91}

93impl Field {
94    fn bound<M, T>(lens: ResolvedLens<M, T>, rules: Rules, control: ControlKind) -> Self {
95        Self {
96            name: lens.name,
97            label: lens.label,
98            required: !lens.nullable,
99            nullable: lens.nullable,
100            rules,
101            control,
102            misdeclared: lens.misdeclared,
103        }
104    }

106    /// Binds any [`FormScalar`] column, storing the type's own spelling and probing uniqueness
107    /// through the lens.
108    pub fn text<M, T>(lens: impl Into<ResolvedLens<M, T>>) -> TextField
109    where
110        M: toasty::schema::Model,
111        T: FormScalar + toasty::stmt::IntoExpr<T> + 'static,
112    {
113        let lens = lens.into();
114        let control = TextControl::new::<M, T>(lens.path.clone(), lens.unique);
115        TextField(Self::bound(
116            lens,
117            Rules::new().scalar::<T>(),
118            ControlKind::Text(control),
119        ))
120    }

122    /// The text field `#[derive(EmbeddedForm)]` renders for a leaf.

126        M: toasty::schema::Model,

273    /// Renders the field's control or its stored value.
274    pub(crate) async fn render<'a>(
275        &self,
276        cx: &'a Cx,
277        value: Option<&str>,
278        error: Option<&str>,
279        mode: Mode,
280    ) -> Result<BoxView<'a>> {
281        if mode == Mode::View && value.is_none() {
282            debug_assert!(
283                false,
284                "view field `{}` has no value: neither `view_values` nor the record form's \
285                 `hydrate` supplies its key",
286                self.name
287            );
288            return render_value(cx, &self.label, Some("(missing)"), ValueKind::Prose);
289        }
290        match &self.control {
291            ControlKind::Text(text) => self.render_text(text, cx, value, error, mode),
292            ControlKind::Choice(choice) => {
293                Box::pin(self.render_choice(choice, cx, value, error, mode)).await
294            }
295            ControlKind::File => self.render_file(cx, value, error, mode),
296            ControlKind::Custom(control) => {
297                self.render_custom(control.as_ref(), cx, value, error, mode)
298            }
299        }
300    }
301
302    /// Renders an app [`Control`] in its shared chrome.
303    fn render_custom<'a>(
304        &self,
305        control: &dyn Control,
306        cx: &'a Cx,
307        value: Option<&str>,
308        error: Option<&str>,
309        mode: Mode,
310    ) -> Result<BoxView<'a>> {
311        if mode == Mode::View {
312            let shown = control.display(cx, value.unwrap_or_default());
313            return render_value_view(cx, &self.label, shown);
314        }
315        let required = self.is_required();
316        let chrome = FieldChrome::new(&self.name, error, None);
317        let input = ControlInput::new(
318            &self.name,
319            value,
320            required,
321            chrome.aria_invalid() == "true",
322            chrome.described_by(),
323        );
324        let rendered = control.render(cx, input);
325        render_field(cx, &chrome, &self.label, required, attributes! {}, rendered)
326    }

329/// How a read-only value is presented.
330#[derive(Debug, Clone, Copy, PartialEq, Eq)]
331pub(crate) enum ValueKind {
332    /// Wrapping text: a title, a body, an address.
333    Prose,
334    /// A stored path — no spaces to break at.
335    Machine,
336}
337
338/// The read-only half of a field: the label with the record's stored value under it.
339fn render_value<'a>(
340    cx: &'a Cx,
341    label: &str,
342    value: Option<&str>,
343    kind: ValueKind,
344) -> Result<BoxView<'a>> {
345    let text = value.unwrap_or_default().to_string();
346    let value_class = match kind {
347        ValueKind::Prose => "text-sm text-foreground wrap-anywhere whitespace-pre-wrap",
348        ValueKind::Machine => "text-sm text-foreground font-mono break-all whitespace-pre-wrap",
349    };
350    let value = view! { cx => <div class=(value_class)>(text)</div> }.boxed();
351    render_value_view(cx, label, value)
352}
353
354/// The field chrome around a read-only value that is not a plain string.
355fn render_value_view<'a>(cx: &'a Cx, label: &str, value: BoxView<'a>) -> Result<BoxView<'a>> {
356    let label = label.to_string();
357    Ok(view! {
358        cx =>
359        ui_field(
360            attrs: attributes! { class="ac-field" },
361            ui_field_content(
362                ui_field_title((label))
363                (value)
364            )
365        )
366    }
367    .boxed())
368}

370/// The validation state a form control renders.
371pub(crate) struct FieldChrome {
372    name: String,
373    error_id: String,
374    error_text: String,
375    has_error: bool,
376}
377
378impl FieldChrome {
379    pub(crate) fn new(name: &str, error: Option<&str>, fallback: Option<String>) -> Self {
380        let has_error = error.is_some() || fallback.is_some();
381        let error_text = match error {
382            Some(message) if !message.is_empty() => message.to_string(),
383            _ => fallback.unwrap_or_default(),
384        };
385        Self {
386            name: name.to_string(),
387            error_id: format!("{name}-error"),
388            error_text,
389            has_error,
390        }
391    }
392
393    pub(crate) fn aria_invalid(&self) -> &'static str {
394        if self.has_error { "true" } else { "false" }
395    }
396
397    pub(crate) fn described_by(&self) -> Option<String> {
398        self.has_error.then(|| self.error_id.clone())
399    }
400}
401
402/// The chrome every form control renders: wrapper, label, control, and error slot.
403pub(crate) fn render_field<'a>(
404    cx: &'a Cx,
405    chrome: &FieldChrome,
406    label: &str,
407    required: bool,
408    attributes: Attributes,
409    control: BoxView<'a>,
410) -> Result<BoxView<'a>> {
411    let name = chrome.name.clone();
412    let label_text = label.to_string();
413    let has_error = chrome.has_error;
414    let error_id = chrome.error_id.clone();
415    let error_text = chrome.error_text.clone();
416    let field_class = if has_error {
417        "ac-field ac-field--error"
418    } else {
419        "ac-field"
420    };
421    Ok(view! {
422        cx =>
423        ui_field(
424            attrs: attributes! {
425                class=(field_class)
426                data-invalid=(has_error.then_some("true"))
427                (attributes)
428            },
429            ui_field_label(
430                attrs: attributes! { for=(name) },
431                (label_text)
432                if required {
433                    <span class="text-destructive" aria-hidden="true">"*"</span>
434                }
435            )
436            (control)
437            if has_error {
438                ui_field_error(
439                    attrs: attributes! { id=(error_id) class="ac-error" aria-live="polite" },
440                    (error_text)
441                )
442            }
443        )
444    }
445    .boxed())
446}

461    /// The opening tag carrying `needle` (attributes render unordered, topcoat#122).
462    pub(super) fn tag_with<'h>(html: &'h str, needle: &str) -> &'h str {
463        let at = html
464            .find(needle)
465            .unwrap_or_else(|| panic!("no {needle} in {html}"));
466        opening_tag_at(html, html[..at].rfind('<').expect("its opening tag"))
467    }
468
469    /// Slices the opening tag at `start`, honouring quoting (topcoat#122).
470    pub(super) fn opening_tag_at(html: &str, start: usize) -> &str {
471        let mut quoted = false;
472        for (offset, byte) in html.as_bytes()[start..].iter().enumerate() {
473            match byte {
474                b'"' => quoted = !quoted,
475                b'>' if !quoted => return &html[start..start + offset],
476                _ => {}
477            }
478        }
479        panic!("unterminated tag at byte {start} in {html}");
480    }
481
482    /// The sorted attributes of the tag carrying `needle` (unordered, topcoat#122).
483    pub(super) fn attributes_of(html: &str, needle: &str) -> Vec<String> {
484        let mut quoted = false;
485        let mut attrs: Vec<String> = Vec::new();
486        let mut current = String::new();
487        for ch in tag_with(html, needle).chars() {
488            match ch {
489                '"' => {
490                    quoted = !quoted;
491                    current.push(ch);
492                }
493                ch if ch.is_whitespace() && !quoted => {
494                    if !current.is_empty() {
495                        attrs.push(std::mem::take(&mut current));
496                    }
497                }
498                ch => current.push(ch),
499            }
500        }
501        if !current.is_empty() {
502            attrs.push(current);
503        }
504        attrs.remove(0); // the tag name
505        attrs.sort();
506        attrs
507    }
