33/// One form field: a key, a label, the rules a submission meets, and the
34/// control that edits it.
35///
36/// Three constructors pick the control, and each binds any lens a
37/// [`ResolvedLens`] accepts — a column (`Post::fields().title()`) or an
38/// embedded leaf (`ResolvedLens::new(cx, Post::fields().seo().title())`):
39///
40/// ```ignore
41/// Field::text(User::fields().name()).placeholder("Ada Lovelace")
42/// Field::text(User::fields().email()).email().unique()
43/// Field::text(User::fields().age())                  // typed: an `i64` column
44/// Field::text(Post::fields().body()).multiline(6)
45/// Field::choice(Post::fields().status()).options(vec!["draft".into(), "published".into()])
46/// Field::choice(Post::fields().author_id()).relationship::<AuthorResource>(..)
47/// Field::file(Doc::fields().path())
48/// ```
49///
50/// `required` defaults from the column's nullability (GH #100): a
51/// non-nullable column is required, so an empty submit fails inline instead of
52/// at the driver. `.optional()` opts out and `.required()` opts back in.
53///
54/// A modifier belongs to one control: `email`, `unique`, `placeholder`, and
55/// `multiline` to text; `options`, `options_with_labels`, `relationship`, and
56/// `searchable` to choice. Calling one on another control is a declaration
57/// bug and panics, naming the field.
58///
59/// [`toggle`](Self::toggle) renders a checkbox, and [`custom`](Self::custom)
60/// renders any [`Control`] an app writes.
61pub struct Field {
62    name: String,
63    label: String,
64    required: bool,
65    /// Whether the column stores NULL for an empty submission, which a unique
66    /// index admits any number of times.
67    nullable: bool,
68    /// The email and scalar-parse rules, with their messages.
69    rules: Rules,
70    control: ControlKind,
71}
72
73/// The control a [`Field`] renders, with what only that control declares.
74///
75/// Text, choice, and file are the framework's own: the unique probe, the
76/// relationship option endpoint, and the multipart parser read them.
77/// Everything else is a [`Control`].
78pub(crate) enum ControlKind {
79    /// A one-line `<input>`, or a `<textarea>` when `rows` is set.
80    Text(TextControl),
81    /// A `<select>` over static options or a relationship's rows.
82    Choice(ChoiceControl),
83    /// A file input storing the uploaded file's path.
84    File,
85    /// An app's control, or the built-in [`Toggle`].
86    Custom(Arc<dyn Control>),
87}

89impl std::fmt::Debug for Field {
90    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
91        // The probe and the loaders are closures with no useful Debug;
92        // everything a reader needs is the field's identity and its control.
93        let control = match &self.control {
94            ControlKind::Text(_) => "text",
95            ControlKind::Choice(_) => "choice",
96            ControlKind::File => "file",
97            ControlKind::Custom(_) => "custom",
98        };
99        f.debug_struct("Field")
100            .field("name", &self.name)
101            .field("label", &self.label)
102            .field("required", &self.required)
103            .field("control", &control)
104            .finish()
105    }
106}

109    /// A text field over a column of any [`FormScalar`] type: `String`, a
110    /// [`TypedValue`](crate::schema::TypedValue) type, or an `Option` of one.
111    ///
112    /// The control renders the stored value, and a typed column adds its
113    /// spelling rule:
114    ///
115    /// - a submission the type refuses is an inline field error naming the input (`` `2024-13-01`
116    ///   is not a valid timestamp ``);
117    /// - what is stored is the type's own spelling of the parsed value, so a value re-submitted
118    ///   unchanged is written back in the shape it was read.
119    ///
120    /// The control's `type` is the type's `INPUT_TYPE`: `datetime-local` for a
121    /// `jiff::Timestamp`, whose stored instant renders in UTC and whose
122    /// submission is read back as UTC, and `text` otherwise (`email` overrides
123    /// it). `unique` defaults from the column's unique index (GH #183).
124    ///
125    /// `T` is [`IntoExpr`](toasty::stmt::IntoExpr) of itself so the unique
126    /// probe compares the parsed value through the lens rather than its text.
127    pub fn text<M, T>(lens: impl Into<ResolvedLens<M, T>>) -> Self
128    where
129        M: toasty::schema::Model,
130        T: FormScalar + toasty::stmt::IntoExpr<T> + 'static,
131    {
132        let lens = lens.into();
133        let control = TextControl::new::<M, T>(lens.path, lens.unique);
134        Self {
135            name: lens.name,
136            label: lens.label,
137            required: !lens.nullable,
138            nullable: lens.nullable,
139            rules: Rules::new().scalar::<T>(),
140            control: ControlKind::Text(control),
141        }
142    }

358    /// Load the options from a related source's tenant-scoped query (choice).
359    ///
360    /// `R` is any [`OptionSource`] — every `Resource` is one. The first
361    /// argument is the related resource's `query` fn (`AuthorResource::query`),
362    /// a type-inference witness only: the loader calls the source's scoped
363    /// query, so the tenant gate and the derived tenant filter apply. The
364    /// second projects a record to the model's **primary key**, whose
365    /// `Display` is the `<option value>`; the third maps it to its label. A
366    /// projection to anything but the key type fails to compile, and an edit
367    /// form hydrates the foreign key with the same string.
368    ///
369    /// Policy-checked: the related source must allow `can_view_any` and, when
370    /// it declares `requires_tenant`, have a resolved tenant; each loaded row
371    /// is filtered through `can_view`. A denial fails the load closed — no
372    /// options and not the stored value, `{label} is not available` on GET,
373    /// and a submit that carries a value fails with that message.
374    ///
375    /// Bounded and memoized: at most one row past `MAX_RELATIONSHIP_OPTIONS`
376    /// loads per `(request, tenant)`. Past the cap a searchable field degrades
377    /// to type-to-search with a targeted existence check, and a non-searchable
378    /// one reports `could not load options, retry`.
379    pub fn relationship<R>(
380        mut self,
381        query: fn(&Cx) -> toasty::stmt::Query<toasty::stmt::List<R::Model>>,
382        value: impl Fn(&R::Model) -> RelatedPrimaryKey<R> + Send + Sync + 'static,
383        label: impl Fn(&R::Model) -> String + Send + Sync + 'static,
384    ) -> Self
385    where
386        R: OptionSource + 'static,
387        RelatedPrimaryKey<R>: std::fmt::Display,
388    {
389        let _ = query;
390        self.choice_mut("relationship").relationship =
391            Some(choice::Relationship::new::<R>(value, label));
392        self
393    }

469    /// Render the field: its control in `Mode::Form`, its stored value in
470    /// `Mode::View`.
471    ///
472    /// A view field whose key the values do not carry renders `(missing)` and
473    /// fails a `debug_assert!`: neither the resource's `view_values` nor its
474    /// record form supplies the key, so the page would otherwise show a blank
475    /// that reads as an empty value. It is the contract the list columns keep
476    /// for an unloaded relation (ADR-0011).
477    pub(crate) async fn render<'a>(
478        &self,
479        cx: &'a Cx,
480        value: Option<&str>,
481        error: Option<&str>,
482        mode: Mode,
483    ) -> Result<BoxView<'a>> {
484        if mode == Mode::View && value.is_none() {
485            debug_assert!(
486                false,
487                "view field `{}` has no value: neither `view_values` nor the record form's \
488                 `hydrate` supplies its key",
489                self.name
490            );
491            return render_value(cx, &self.label, Some("(missing)"), ValueKind::Prose);
492        }
493        match &self.control {
494            ControlKind::Text(text) => self.render_text(text, cx, value, error, mode),
495            ControlKind::Choice(choice) => {
496                Box::pin(self.render_choice(choice, cx, value, error, mode)).await
497            }
498            ControlKind::File => self.render_file(cx, value, error, mode),
499            ControlKind::Custom(control) => {
500                self.render_custom(control.as_ref(), cx, value, error, mode)
501            }
502        }
503    }

505    /// Render an app [`Control`]: its `display` on the detail page, its
506    /// input inside the shared field chrome on a form.
507    fn render_custom<'a>(
508        &self,
509        control: &dyn Control,
510        cx: &'a Cx,
511        value: Option<&str>,
512        error: Option<&str>,
513        mode: Mode,
514    ) -> Result<BoxView<'a>> {
515        if mode == Mode::View {
516            let shown = control.display(cx, value.unwrap_or_default());
517            return render_value_view(cx, &self.label, shown);
518        }
519        let required = self.is_required();
520        let chrome = FieldChrome::new(&self.name, error, None);
521        let input = ControlInput::new(
522            &self.name,
523            value,
524            required,
525            chrome.aria_invalid() == "true",
526            chrome.described_by(),
527        );
528        let rendered = control.render(cx, input);
529        render_field(cx, &chrome, &self.label, required, attributes! {}, rendered)
530    }

555/// How a read-only value is presented.
556///
557/// Two shapes, because the difference is content, not styling: prose wraps at
558/// spaces (and breaks a token too long for its line), and a stored path has no
559/// spaces to break at, so it breaks anywhere and sets in mono.
560#[derive(Debug, Clone, Copy, PartialEq, Eq)]
561pub(crate) enum ValueKind {
562    /// Wrapping text: a title, a body, an address.
563    Prose,
564    /// A stored path — no spaces to break at.
565    Machine,
566}
567
568/// The read-only half of a field: the label with the record's stored
569/// value under it, no control and no validation slot.
570///
571/// Every control renders its view through this, so a detail page reads
572/// uniformly. The label is the same `field_label` the form uses, inside the
573/// same `field` family, so a field is recognisable across the two pages.
574///
575/// An empty value renders empty: a `String` column stores `""` and a NULL
576/// hydrates as `""`, so the page cannot tell "no value" from an empty one.
577fn render_value<'a>(
578    cx: &'a Cx,
579    label: &str,
580    value: Option<&str>,
581    kind: ValueKind,
582) -> Result<BoxView<'a>> {
583    let text = value.unwrap_or_default().to_string();
584    let value_class = match kind {
585        ValueKind::Prose => "text-sm text-foreground wrap-anywhere whitespace-pre-wrap",
586        ValueKind::Machine => "text-sm text-foreground font-mono break-all whitespace-pre-wrap",
587    };
588    let value = view! { cx => <div class=(value_class)>(text)</div> }.boxed();
589    render_value_view(cx, label, value)
590}
591
592/// The field chrome `render_value` puts around a rendered value, for
593/// a field whose read-only value is not a plain string.
594///
595/// A file field renders its stored path as a link and supplies that view
596/// here, so the label, the `field` family and the `ac-field` marker stay the
597/// ones every other read-only field renders through.
598fn render_value_view<'a>(cx: &'a Cx, label: &str, value: BoxView<'a>) -> Result<BoxView<'a>> {
599    let label = label.to_string();
600    Ok(view! {
601        cx =>
602        ui_field(
603            attrs: attributes! { class="ac-field" },
604            ui_field_content(
605                ui_field_title((label))
606                (value)
607            )
608        )
609    }
610    .boxed())
611}

627impl FieldChrome {
628    pub(crate) fn new(name: &str, error: Option<&str>, fallback: Option<String>) -> Self {
629        let has_error = error.is_some() || fallback.is_some();
630        // An empty message is no message: the field's own fallback wording
631        // renders instead, if it has one.
632        let error_text = match error {
633            Some(message) if !message.is_empty() => message.to_string(),
634            _ => fallback.unwrap_or_default(),
635        };
636        Self {
637            name: name.to_string(),
638            error_id: format!("{name}-error"),
639            error_text,
640            has_error,
641        }
642    }
643
644    /// The control's `aria-invalid`: `"true"` also colors the field's label.
645    pub(crate) fn aria_invalid(&self) -> &'static str {
646        if self.has_error { "true" } else { "false" }
647    }
648
649    /// The control's `aria-describedby`, pointing at the error slot while the
650    /// field is invalid. Owned because a rendered view outlives this value.
651    pub(crate) fn described_by(&self) -> Option<String> {
652        self.has_error.then(|| self.error_id.clone())
653    }
654}
655
656/// The chrome every form control renders: the `field` wrapper carrying
657/// `ac-field` / `ac-field--error`, the label with the required marker, the
658/// control, and the error slot.
659///
660/// `attributes` carries the extra wrapper attributes a control needs — the
661/// choice option and filter hooks.
662pub(crate) fn render_field<'a>(
663    cx: &'a Cx,
664    chrome: &FieldChrome,
665    label: &str,
666    required: bool,
667    attributes: Attributes,
668    control: BoxView<'a>,
669) -> Result<BoxView<'a>> {
670    let name = chrome.name.clone();
671    let label_text = label.to_string();
672    let has_error = chrome.has_error;
673    let error_id = chrome.error_id.clone();
674    let error_text = chrome.error_text.clone();
675    let field_class = if has_error {
676        "ac-field ac-field--error"
677    } else {
678        "ac-field"
679    };
680    Ok(view! {
681        cx =>
682        ui_field(
683            attrs: attributes! {
684                class=(field_class)
685                data-invalid=(has_error.then_some("true"))
686                (attributes)
687            },
688            ui_field_label(
689                attrs: attributes! { for=(name) },
690                (label_text)
691                if required {
692                    <span class="text-destructive" aria-hidden="true">"*"</span>
693                }
694            )
695            (control)
696            if has_error {
697                ui_field_error(
698                    attrs: attributes! { id=(error_id) class="ac-error" aria-live="polite" },
699                    (error_text)
700                )
701            }
702        )
703    }
704    .boxed())
705}

720    /// The whole opening tag carrying `needle` — how a test asserts on an
721    /// element whose attributes render in no guaranteed order (topcoat#122)
722    /// without depending on a marker attribute nothing consumes.
723    pub(super) fn tag_with<'h>(html: &'h str, needle: &str) -> &'h str {
724        let at = html
725            .find(needle)
726            .unwrap_or_else(|| panic!("no {needle} in {html}"));
727        opening_tag_at(html, html[..at].rfind('<').expect("its opening tag"))
728    }
729
730    /// The opening tag that starts at `start`, sliced up to the `>` closing it.
731    ///
732    /// `Attributes` renders in no guaranteed order (topcoat#122), so a test
733    /// locates a tag by whichever attribute it can and asserts on the whole
734    /// tag. Quoting is honoured, so a `>` inside an attribute value (Tailwind
735    /// selectors carry them) does not end the slice.
736    pub(super) fn opening_tag_at(html: &str, start: usize) -> &str {
737        let mut quoted = false;
738        for (offset, byte) in html.as_bytes()[start..].iter().enumerate() {
739            match byte {
740                b'"' => quoted = !quoted,
741                b'>' if !quoted => return &html[start..start + offset],
742                _ => {}
743            }
744        }
745        panic!("unterminated tag at byte {start} in {html}");
746    }
747
748    /// The attributes of the opening tag carrying `needle`, sorted — quoting is
749    /// honoured, so a Tailwind class value stays one token.
750    ///
751    /// `Attributes` renders in no guaranteed order (topcoat#122), so two
752    /// renders of the same markup compare as sets of `name="value"` tokens.
753    pub(super) fn attributes_of(html: &str, needle: &str) -> Vec<String> {
754        let mut quoted = false;
755        let mut attrs: Vec<String> = Vec::new();
756        let mut current = String::new();
757        for ch in tag_with(html, needle).chars() {
758            match ch {
759                '"' => {
760                    quoted = !quoted;
761                    current.push(ch);
762                }
763                ch if ch.is_whitespace() && !quoted => {
764                    if !current.is_empty() {
765                        attrs.push(std::mem::take(&mut current));
766                    }
767                }
768                ch => current.push(ch),
769            }
770        }
771        if !current.is_empty() {
772            attrs.push(current);
773        }
774        attrs.remove(0); // the tag name
775        attrs.sort();
776        attrs
777    }
