3use tablo_ui::{checkbox as ui_checkbox, input as ui_input, label as ui_label};
4use topcoat::{Result, context::Cx, view::*};

6use super::{
7    super::tree::Mode, Field, FieldChrome, ValueKind, render_field, render_value, render_value_view,
8};

10impl Field {
11    /// Render a file field: the stored path as a link in `Mode::View`, the
12    /// file input with the stored path and its clear toggle otherwise.
13    pub(super) fn render_file<'a>(
14        &self,
15        cx: &'a Cx,
16        value: Option<&str>,
17        errors: &[String],
18        mode: Mode,
19    ) -> Result<BoxView<'a>> {
20        // The detail page shows the stored path, never a file control: an
21        // empty file input on an edit is the panel's "keep the stored file"
22        // affordance, which is a statement about a form, not about a record.
23        if mode == Mode::View {
24            return stored_upload_value(cx, &self.label, value);
25        }
26        let name = self.name.clone();
27        // An edit hydrates the stored path; a create does not. The control
28        // is required only when nothing is stored, since a file input cannot
29        // be pre-filled.
30        let stored = stored_path(value);
31        let is_edit = stored.is_some();
32        let control_required = self.required && !is_edit;
33        let chrome = FieldChrome::new(&name, errors, None);
34        let hint_id = format!("{name}-hint");
35        // The clear flag is a framework transport key, not a field:
36        // it names the stored value's owner and is stripped before any record
37        // fn, so it can never be written as a field of its own.
38        let clear_name = format!("clear_{name}");
39        // The stored value as a link to the file it names. Nothing
40        // here guesses a URL convention — the app decides what it stores (the
41        // uploader's return value) — and a value that is not a rooted path or
42        // an `http(s)` URL renders as text rather than as a clickable scheme.
43        let stored_display: Option<BoxView<'a>> =
44            stored.map(|current| stored_upload_row(cx, current));
45        let aria_invalid = chrome.aria_invalid();
46        let described_by = chrome
47            .described_by()
48            .or_else(|| is_edit.then(|| hint_id.clone()));
49        let control = view! {
50            cx =>
51            if let Some(row) = stored_display {
52                // The stored path is visible, so "there is no file" is no
53                // longer ambiguous, and the empty control reads as "leave
54                // it alone" rather than "this field is broken".
55                (row)
56            }
57            // The `input` primitive styles `type="file"` through its
58            // `file:` classes and carries the `aria-invalid` error styling.
59            ui_input(
60                attrs: attributes! {
61                    id=(name.clone())
62                    type="file"
63                    name=(name.clone())
64                    required=(control_required)
65                    aria-required=(control_required.then_some("true"))
66                    aria-invalid=(aria_invalid)
67                    aria-describedby=(described_by)
68                }
69            )
70            if is_edit {
71                <div class="text-xs text-muted-foreground" id=(hint_id.clone())>
72                    "Leave empty to keep the current file."
73                </div>
74                // The one control that says "remove it" rather than "leave
75                // it alone". It carries `value="1"` so the
76                // framework's own `truthy` vocabulary reads it, and it is a
77                // declared transport key, so a generic record fn never sees
78                // it.
79                <div class="mt-2 flex items-center gap-2">
80                    ui_checkbox(
81                        attrs: attributes! {
82                            id=(clear_name.clone())
83                            name=(clear_name.clone())
84                            value="1"
85                        }
86                    )
87                    ui_label(
88                        attrs: attributes! {
89                            for=(clear_name.clone())
90                            class="text-xs text-muted-foreground"
91                        },
92                        "Remove the current file"
93                    )
94                </div>
95            }
96        }
97        .boxed();
98        render_field(
99            cx,
100            &chrome,
101            &self.label,
102            control_required,
103            attributes! {},
104            control,
105        )
106    }
107}

109/// The stored path a file field shows, if it has one.
110///
111/// A value that is absent or only whitespace is not a file: the form renders a
112/// create rather than an edit with an empty "Current:" line, and the read-only
113/// value renders without a link.
114fn stored_path(value: Option<&str>) -> Option<String> {
115    value
116        .map(|v| v.trim().to_string())
117        .filter(|v| !v.is_empty())
118}
119
120/// Whether a stored value may become an `href`: a rooted path
121/// (`/uploads/x.png`, not the scheme-relative `//host`) or an absolute
122/// `http(s)` URL. Anything else — a bare basename, `javascript:`, `data:` —
123/// renders as text: the framework stores what it is handed, so the render is
124/// where a scheme is refused.
125fn is_linkable(path: &str) -> bool {
126    let lower = path.trim_start().to_ascii_lowercase();
127    (lower.starts_with('/') && !lower.starts_with("//"))
128        || lower.starts_with("https://")
129        || lower.starts_with("http://")
130}
131
132/// The `Current: …` row a file field shows for a stored value.
133///
134/// Takes the path by value: the rendered view has to outlive the field's
135/// render, and a rendering coroutine may not hold a borrow of it.
136fn stored_upload_row<'a>(cx: &'a Cx, path: String) -> BoxView<'a> {
137    // The link is the only way to reach the file, and the path is what it
138    // says; each node owns its own copy of it.
139    let text = path.clone();
140    // A value that is not a safe URL renders as plain text: the
141    // wrapper and the label stay, only the anchor goes.
142    let inner: BoxView<'a> = if is_linkable(&path) {
143        let href = path.clone();
144        view! {
145            cx =>
146            <a class="font-medium text-foreground underline" href=(href)>(text)</a>
147        }
148        .boxed()
149    } else {
150        view! { cx => <span class="font-medium text-foreground">(text)</span> }.boxed()
151    };
152    view! {
153        cx =>
154        <div class="text-xs text-muted-foreground" data-file-current=(path)>
155            "Current: "
156            (inner)
157        </div>
158    }
159    .boxed()
160}

162/// A file field read rather than edited: the label over the stored
163/// path, as a link to the file when it is one.
164///
165/// The reader asks the same "is the stored value right?" question the editor
166/// asks, and following the link is how they answer it. `underline` is the
167/// affordance: the value sits in body colour inside the field chrome, so
168/// without it a stored path reads as text. The chrome is the one
169/// `render_value_view` gives every read-only field, so a detail page stays
170/// uniform.
171fn stored_upload_value<'a>(cx: &'a Cx, label: &str, value: Option<&str>) -> Result<BoxView<'a>> {
172    let Some(path) = stored_path(value) else {
173        return render_value(cx, label, value, ValueKind::Machine);
174    };
175    // A value that is not a safe URL is not a link: it renders
176    // through the same machine-value path an empty value takes, so the detail
177    // page shows the stored text without an `href` to click.
178    if !is_linkable(&path) {
179        return render_value(cx, label, Some(&path), ValueKind::Machine);
180    }
181    let href = path.clone();
182    let link = view! {
183        cx =>
184        <a
185            class="text-sm font-mono break-all whitespace-pre-wrap underline"
186            href=(href)
187        >
188            (path)
189        </a>
190    }
191    .boxed();
192    render_value_view(cx, label, link)
193}
