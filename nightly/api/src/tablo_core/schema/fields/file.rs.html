1//! The file control: a file input storing the uploaded file's path.

3use tablo_ui::{checkbox as ui_checkbox, input as ui_input, label as ui_label};
4use topcoat::{Result, context::Cx, view::*};
5
6use super::{
7    super::tree::Mode, Field, FieldChrome, ValueKind, render_field, render_value, render_value_view,
8};
9
10impl Field {
11    /// Renders a file field's input and its stored path.
12    pub(super) fn render_file<'a>(
13        &self,
14        cx: &'a Cx,
15        value: Option<&str>,
16        error: Option<&str>,
17        mode: Mode,
18    ) -> Result<BoxView<'a>> {
19        if mode == Mode::View {
20            return stored_upload_value(cx, &self.label, value);
21        }
22        let name = self.name.clone();
23        // Required only while nothing is stored.
24        let stored = stored_path(value);
25        let is_edit = stored.is_some();
26        let control_required = self.required && !is_edit;
27        let chrome = FieldChrome::new(&name, error, None);
28        let hint_id = format!("{name}-hint");
29        // The clear flag is a transport key stripped before any record fn.
30        let clear_name = format!("clear_{name}");
31        // A value that is not a rooted path or `http(s)` URL renders as text.
32        let stored_display: Option<BoxView<'a>> =
33            stored.map(|current| stored_upload_row(cx, current));
34        let aria_invalid = chrome.aria_invalid();
35        let described_by = chrome
36            .described_by()
37            .or_else(|| is_edit.then(|| hint_id.clone()));
38        let control = view! {
39            cx =>
40            if let Some(row) = stored_display {
41                (row)
42            }
43            ui_input(
44                attrs: attributes! {
45                    id=(name.clone())
46                    type="file"
47                    name=(name.clone())
48                    required=(control_required)
49                    aria-required=(control_required.then_some("true"))
50                    aria-invalid=(aria_invalid)
51                    aria-describedby=(described_by)
52                }
53            )
54            if is_edit {
55                <div class="text-xs text-muted-foreground" id=(hint_id.clone())>
56                    "Leave empty to keep the current file."
57                </div>
58                <div class="mt-2 flex items-center gap-2">
59                    ui_checkbox(
60                        attrs: attributes! {
61                            id=(clear_name.clone())
62                            name=(clear_name.clone())
63                            value="1"
64                        }
65                    )
66                    ui_label(
67                        attrs: attributes! {
68                            for=(clear_name.clone())
69                            class="text-xs text-muted-foreground"
70                        },
71                        "Remove the current file"
72                    )
73                </div>
74            }
75        }
76        .boxed();
77        render_field(
78            cx,
79            &chrome,
80            &self.label,
81            control_required,
82            attributes! {},
83            control,
84        )
85    }
86}

88/// The stored path a file field shows, if the value is present and not blank.
89fn stored_path(value: Option<&str>) -> Option<String> {
90    value
91        .map(|v| v.trim().to_string())
92        .filter(|v| !v.is_empty())
93}
94
95/// Whether a stored value may become an `href`; a rooted path or absolute `http(s)` URL may,
96/// anything else renders as text.
97fn is_linkable(path: &str) -> bool {
98    let lower = path.trim_start().to_ascii_lowercase();
99    (lower.starts_with('/') && !lower.starts_with("//"))
100        || lower.starts_with("https://")
101        || lower.starts_with("http://")
102}
103
104/// The `Current: …` row a file field shows for a stored value.
105fn stored_upload_row<'a>(cx: &'a Cx, path: String) -> BoxView<'a> {
106    let text = path.clone();
107    let inner: BoxView<'a> = if is_linkable(&path) {
108        let href = path.clone();
109        view! {
110            cx =>
111            <a class="font-medium text-foreground underline" href=(href)>(text)</a>
112        }
113        .boxed()
114    } else {
115        view! { cx => <span class="font-medium text-foreground">(text)</span> }.boxed()
116    };
117    view! {
118        cx =>
119        <div class="text-xs text-muted-foreground" data-file-current=(path)>
120            "Current: "
121            (inner)
122        </div>
123    }
124    .boxed()
125}

127/// A file field read rather than edited: the label over the stored path, as a link when it is one.
128fn stored_upload_value<'a>(cx: &'a Cx, label: &str, value: Option<&str>) -> Result<BoxView<'a>> {
129    let Some(path) = stored_path(value) else {
130        return render_value(cx, label, value, ValueKind::Machine);
131    };
132    // A value that is not a safe URL is not a link.
133    if !is_linkable(&path) {
134        return render_value(cx, label, Some(&path), ValueKind::Machine);
135    }
136    let href = path.clone();
137    let link = view! {
138        cx =>
139        <a
140            class="text-sm font-mono break-all whitespace-pre-wrap underline"
141            href=(href)
142        >
143            (path)
144        </a>
145    }
146    .boxed();
147    render_value_view(cx, label, link)
148}
