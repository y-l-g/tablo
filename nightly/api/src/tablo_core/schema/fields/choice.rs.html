20/// What a choice field declares beyond presence.
21#[derive(Clone, Default)]
22pub(crate) struct ChoiceControl {
23    pub(super) searchable: bool,
24    /// Whether this is an embedded enum's variant control: it renders
25    /// `data-variant-select`, which `variant.js` follows to show only the
26    /// chosen variant's group, and a read-only page shows the variant's name.
27    pub(super) discriminant: bool,
28    /// Static `(value, label)` options.
29    pub(super) options: Vec<(String, String)>,
30    pub(super) relationship: Option<Relationship>,
31}

33/// A relationship's three loaders: the bounded option load, the server-side
34/// search, and the targeted existence check past the cap.
35#[derive(Clone)]
36pub(crate) struct Relationship {
37    load: RelationshipLoader,
38    search: RelationshipSearchLoader,
39    check: RelationshipChecker,
40}
41
42/// One `<option>`: the markup a choice field, the filter bar and the
43/// relationship options endpoint share, so the value and the label are
44/// escaped by the view layer in one place.
45pub(crate) fn option_view<'a>(
46    cx: &'a Cx,
47    value: String,
48    label: String,
49    selected: bool,
50) -> BoxView<'a> {
51    view! { cx => <option value=(value) selected=(selected)>(label)</option> }.boxed()
52}
53
54impl Relationship {
55    /// The loaders for source `R`, projecting each row to its key and label.
56    pub(super) fn new<R>(
57        value: impl Fn(&R::Model) -> RelatedPrimaryKey<R> + Send + Sync + 'static,
58        label: impl Fn(&R::Model) -> String + Send + Sync + 'static,
59    ) -> Self
60    where
61        R: OptionSource + 'static,
62        RelatedPrimaryKey<R>: std::fmt::Display,
63    {
64        let project = std::sync::Arc::new(move |record: &R::Model| {
65            (value(record).to_string(), label(record))
66        });
67        let search_project = project.clone();
68        let load = std::sync::Arc::new(move |cx: &Cx| {
69            let project = project.clone();
70            let cx = cx.clone();
71            Box::pin(async move {
72                // `#[memoize(as_ref)]` hands back a borrow, so clone the
73                // (small) error back into this future's owned result.
74                let records = related_records::<R>(&cx, crate::tenancy::tenant_id(&cx))
75                    .await
76                    .map_err(|error| error.clone())?;
77                Ok(records.iter().map(|record| project(record)).collect())
78            }) as RelationshipLoadFuture
79        }) as RelationshipLoader;
80        let search = std::sync::Arc::new(move |cx: &Cx, q: String| {
81            let project = search_project.clone();
82            let cx = cx.clone();
83            Box::pin(async move {
84                let records = related_records_search::<R>(&cx, q)
85                    .await
86                    .map_err(|error| error.clone())?;
87                Ok(records.iter().map(|record| project(record)).collect())
88            }) as RelationshipLoadFuture
89        }) as RelationshipSearchLoader;
90        let check = std::sync::Arc::new(move |cx: &Cx, v: String| {
91            let cx = cx.clone();
92            Box::pin(async move {
93                related_record_check::<R>(&cx, v)
94                    .await
95                    .map_err(|error| error.clone())
96            }) as RelationshipCheckFuture
97        }) as RelationshipChecker;
98        Self {
99            load,
100            search,
101            check,
102        }
103    }
104}

106impl ChoiceControl {
107    pub(crate) fn is_searchable(&self) -> bool {
108        self.searchable
109    }
110
111    pub(crate) fn is_relationship(&self) -> bool {
112        self.relationship.is_some()
113    }
114
115    /// Server-side option search for the endpoint (D1/D5).
116    ///
117    /// Reuses the related table's searchable columns and bounds to
118    /// `MAX_RELATIONSHIP_OPTIONS`. Returns `Overflow` when the filtered set
119    /// still exceeds the cap (the caller renders "keep typing").
120    pub(crate) async fn search_options(
121        &self,
122        cx: &Cx,
123        q: &str,
124    ) -> Result<Vec<(String, String)>, OptionLoadError> {
125        match &self.relationship {
126            Some(relationship) => (relationship.search)(cx, q.to_string()).await,
127            None => Ok(self.options.clone()),
128        }
129    }
130
131    async fn load_options(&self, cx: &Cx) -> Result<Vec<(String, String)>, OptionLoadError> {
132        match &self.relationship {
133            Some(relationship) => (relationship.load)(cx).await,
134            None => Ok(self.options.clone()),
135        }
136    }
137
138    /// Existence-only check: whether a non-empty `value` matches an option.
139    ///
140    /// A loader failure is a field error rather than an empty-options
141    /// passthrough that would fail at FK write time. A policy denial reads
142    /// "not available" — retrying cannot fix a permission decision, and
143    /// "invalid" would misattribute it to the submitted value. An overflowed
144    /// load uses the targeted check for a searchable field (a legitimate key
145    /// beyond the cap validates) and keeps the retry error otherwise.
146    pub(super) async fn validate_exists(&self, cx: &Cx, label: &str, value: &str) -> Vec<String> {
147        let trimmed = value.trim();
148        if trimmed.is_empty() {
149            return Vec::new();
150        }
151        let Some(relationship) = &self.relationship else {
152            if !self.options.is_empty() && !self.options.iter().any(|(v, _)| v == trimmed) {
153                return vec![format!("{label} is invalid")];
154            }
155            return Vec::new();
156        };
157        let failure = match (relationship.load)(cx).await {
158            Ok(opts) if opts.iter().any(|(v, _)| v == trimmed) => return Vec::new(),
159            Ok(_) => "is invalid",
160            Err(OptionLoadError::Denied) => "is not available",
161            Err(OptionLoadError::Overflow) if self.searchable => {
162                match (relationship.check)(cx, trimmed.to_string()).await {
163                    Ok(RelatedCheck::FoundViewable) => return Vec::new(),
164                    Ok(RelatedCheck::FoundHidden | RelatedCheck::NotFound) => "is invalid",
165                    Err(error) => load_failure(&error),
166                }
167            }
168            Err(error) => load_failure(&error),
169        };
170        vec![format!("{label} {failure}")]
171    }
172}
173
174/// The wording of a failed option load.
175fn load_failure(error: &OptionLoadError) -> &'static str {
176    match error {
177        OptionLoadError::Denied => "is not available",
178        OptionLoadError::LoadFailed | OptionLoadError::Overflow => "could not load options, retry",
179        // A misdeclaration is permanent: retrying cannot fix it, so it is
180        // reported without the retry wording.
181        OptionLoadError::Misdeclared => "could not load options",
182    }
183}

185impl Field {
186    /// Render a choice field: a read-only option label in `Mode::View`, the
187    /// select (and its filter) otherwise.
188    pub(super) async fn render_choice<'a>(
189        &self,
190        choice: &ChoiceControl,
191        cx: &'a Cx,
192        value: Option<&str>,
193        error: Option<&str>,
194        mode: Mode,
195    ) -> Result<BoxView<'a>> {
196        if mode == Mode::View {
197            // View mode resolves a static option label and never loads
198            // options: a detail page renders one record, so a relationship's
199            // option load would be a query per page, and its scoped/denied
200            // paths police a *choice* the page is not offering. A relationship
201            // therefore shows its stored key, and the detail page's own
202            // includes are what make a related record readable.
203            let stored = value.unwrap_or("").trim();
204            let named = choice
205                .options
206                .iter()
207                .find(|(v, _)| v == stored)
208                .map(|(_, label)| label.clone());
209            // A variant control reads as the variant's **name** — `Published`,
210            // never the `3` the column holds (ADR-0016). A value the schema
211            // does not declare has no name to show and renders nothing.
212            if choice.discriminant {
213                return match named {
214                    Some(name) => render_value(cx, &self.label, Some(&name), ValueKind::Prose),
215                    None => Ok(().boxed()),
216                };
217            }
218            let shown = named.unwrap_or_else(|| stored.to_string());
219            return render_value(cx, &self.label, Some(&shown), ValueKind::Prose);
220        }
221        let name = self.name.clone();
222        let required = self.required;
223        let searchable = choice.searchable;
224        let current = value.unwrap_or("").trim().to_string();
225        let loaded = choice.load_options(cx).await;
226        // A policy denial does not re-render the stored value: the related
227        // rows are not viewable, so neither is their label, and the submit
228        // fails closed with "not available". The denial also surfaces on GET
229        // (when the caller carries no error yet): the select has no options to
230        // pick, so the empty control explains itself. A failed or overflowed
231        // load keeps the stored key selectable, so an edit does not blank the
232        // relation into a required error; a searchable field then degrades to
233        // type-to-search with a hint, a non-searchable one to the retry path.
234        let denied = matches!(&loaded, Err(OptionLoadError::Denied));
235        let overflowed = matches!(&loaded, Err(OptionLoadError::Overflow));
236        let overflow_searchable = overflowed && searchable && choice.is_relationship();
237        let keep_current_value = matches!(
238            &loaded,
239            Err(OptionLoadError::LoadFailed)
240                | Err(OptionLoadError::Overflow)
241                | Err(OptionLoadError::Misdeclared)
242        );
243        let mut options = loaded.unwrap_or_default();
244        if keep_current_value && !current.is_empty() && !options.iter().any(|(v, _)| v == &current)
245        {
246            options.push((current.clone(), current.clone()));
247        }
248        let chrome = FieldChrome::new(
249            &name,
250            error,
251            denied.then(|| format!("{} is not available", self.label)),
252        );
253        let mut option_views: Vec<BoxView<'a>> = vec![option_view(
254            cx,
255            String::new(),
256            "-- Select --".to_string(),
257            current.is_empty(),
258        )];
259        for (val, lab) in &options {
260            option_views.push(option_view(cx, val.clone(), lab.clone(), current == *val));
261        }
262        // The `select` primitive brings the same `aria-invalid` error styling
263        // and focus ring as the `input` primitive, plus the chevron and the
264        // customizable picker.
265        let list_id = format!("{name}-options-list");
266        let filter_label = format!("Filter {} options", self.label);
267        // Server fetch only past the cap: a bounded searchable set keeps the
268        // client-side label filter, so `data-options-server` follows the
269        // overflow state. `selects.js` branches on it.
270        let options_field = overflow_searchable.then(|| name.clone());
271        let options_server = overflow_searchable.then_some("true");
272        let variant_of = choice.discriminant.then(|| name.clone());
273        let overflow_hint = "Too many options — type to search".to_string();
274        let aria_invalid = chrome.aria_invalid();
275        let described_by = chrome.described_by();
276        let control = view! {
277            cx =>
278            if searchable {
279                // The filter input and its suggestion list are one combobox.
280                // The list is what makes the filter visible: the native
281                // `<select>` popup is browser chrome the script cannot narrow
282                // (the primitive opts into `appearance: base-select`, where
283                // `option[hidden]` has no effect), so `selects.js` renders its
284                // own filtered list here, keeps `aria-expanded` and
285                // `aria-activedescendant` in step, and writes the chosen value
286                // onto the select. Without the script the input is inert and
287                // the plain select keeps working.
288                <div class="relative" data-options-combobox="">
289                    ui_input(
290                        attrs: attributes! {
291                            type="search"
292                            role="combobox"
293                            aria-expanded="false"
294                            aria-controls=(list_id.clone())
295                            aria-autocomplete="list"
296                            aria-label=(filter_label.clone())
297                            placeholder="Filter…"
298                            data-options-filter=""
299                            class="h-9"
300                            autocomplete="off"
301                        }
302                    )
303                    <ul
304                        id=(list_id.clone())
305                        data-options-list=""
306                        role="listbox"
307                        aria-label=(filter_label.clone())
308                        hidden=""
309                        class="absolute z-20 mt-1 max-h-60 w-full overflow-y-auto rounded-lg border border-border bg-popover p-1 text-sm text-popover-foreground shadow-sm"
310                    ></ul>
311                </div>
312            }
313            if overflow_searchable {
314                <div class="text-xs text-muted-foreground">(overflow_hint)</div>
315            }
316            ui_select(
317                attrs: attributes! {
318                    id=(name.clone())
319                    name=(name.clone())
320                    required=(required)
321                    aria-required=(required.then_some("true"))
322                    aria-invalid=(aria_invalid)
323                    aria-describedby=(described_by)
324                    data-variant-select=(variant_of)
325                },
326                for opt in option_views {
327                    (opt)
328                }
329            )
330        }
331        .boxed();
332        render_field(
333            cx,
334            &chrome,
335            &self.label,
336            required,
337            attributes! {
338                cx =>
339                data-select-filterable=""
340                data-options-field=(options_field)
341                data-options-server=(options_server)
342            },
343            control,
344        )
345    }
