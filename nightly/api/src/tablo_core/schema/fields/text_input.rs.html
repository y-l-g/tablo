15/// The equality expression a typed leaf's unique probe binds.
16///
17/// Built where the declared type is known — the lens constructor — so a typed
18/// field compares as its declared type rather than as its text. `None` means
19/// the submitted value does not parse into that type: validation has already
20/// refused it, and the probe has nothing to compare.
21type EqProbe = std::sync::Arc<dyn Fn(&str) -> Option<toasty::stmt::Expr<bool>> + Send + Sync>;
22
23/// The equality probe a typed leaf binds: the submission is parsed into the
24/// declared type and compared through that type's own path, so a value that is
25/// unique as text but not as the type (or the reverse) is checked for what the
26/// record will store.
27///
28/// The index resolves inside the closure rather than here: the name is the
29/// leaf's **app field name**, and for a context-bound leaf that is its flattened
30/// storage column (`seo_title`), which the leaf's own lens root does not name —
31/// toasty matches app field names — so resolving it eagerly would panic for an
32/// embedded typed leaf. Only a `unique()` marker reaches the closure.
33///
34/// `IntoExpr` is what lets the comparison name the value: it is implemented for
35/// every scalar toasty stores, and the panel's typed constructors require it
36/// for the same reason they require `TypedValue` — a value that cannot become an
37/// expression cannot be compared against its column.
38fn eq_probe_typed<M, T>(name: &str) -> EqProbe
39where
40    M: toasty::schema::Model,
41    T: TypedValue + toasty::stmt::IntoExpr<T> + 'static,
42{
43    let name = name.to_string();
44    let timestamp = is_timestamp::<T>();
45    std::sync::Arc::new(move |value: &str| {
46        // A timestamp submission may be the control's zone-less shape;
47        // validation normalises it to storage spelling first.
48        if timestamp {
49            let storage = parse_timestamp_storage(value).ok()?;
50            let parsed = storage.parse::<T>().ok().filter(T::accepts)?;
51            let index = M::field_name_to_id(&name).index;
52            return Some(M::path_field::<T>(index).eq(parsed));
53        }
54        let parsed = value.parse::<T>().ok().filter(T::accepts)?;
55        let index = M::field_name_to_id(&name).index;
56        Some(M::path_field::<T>(index).eq(parsed))
57    })
58}

60/// Typed text field bound to a Toasty field lens. The lens is the single
61/// source of truth for the field name and type, so `TextInput::for(User::fields().name())`
62/// fails to compile if the column does not exist (ADR-0001).
63#[derive(Clone)]
64pub struct TextInput {
65    name: String,
66    label: String,
67    required: bool,
68    unique: bool,
69    placeholder: Option<String>,
70    /// The email and typed-parse rules, with their messages.
71    rules: Rules,
72    /// The typed leaf's unique probe, absent on a text leaf: a text
73    /// leaf's comparison is built from the model the handler queries.
74    typed_probe: Option<EqProbe>,
75    /// The control's `type`: `datetime-local` for a timestamp leaf, `text`
76    /// otherwise (`email` overrides at render). A timestamp submits a
77    /// zone-less `datetime-local` string the typed rule reads as UTC.
78    input_type: &'static str,
79}

81impl std::fmt::Debug for TextInput {
82    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
83        // The parser is a closure with no useful Debug; everything a reader
84        // needs is the field's identity and which rules it declares.
85        f.debug_struct("TextInput")
86            .field("name", &self.name)
87            .field("label", &self.label)
88            .field("required", &self.required)
89            .field("is_email", &self.rules.is_email())
90            .field("unique", &self.unique)
91            .field("placeholder", &self.placeholder)
92            .field("typed", &self.rules.is_typed())
93            .field("input_type", &self.input_type)
94            .finish()
95    }
96}

99    /// Create a `TextInput` bound to the given field lens.
100    ///
101    /// Only `String` lenses compile: binding a non-text field (a `Uuid` key,
102    /// a `bool`, …) fails at compile time, mirroring `TextColumn`.
103    ///
104    /// `required` and `unique` default from the field's metadata (GH #100,
105    /// GH #183): a non-nullable column is required, and a field backed by a
106    /// single-field unique index is unique — so neither has to be restated by
107    /// hand. Both stay overridable with `.optional()` / `.unique()`.
108    pub fn r#for<M>(path: toasty::stmt::Path<M, String>) -> Self
109    where
110        M: toasty::schema::Model,
111    {
112        let model = M::schema();
113        let field = lens_field(path, &model);
114        let label_str = lens_label(&field);
115        let unique = lens_field_unique(&field, model.as_root_unwrap());
116        let name = field.name.app_unwrap().to_string();
117        Self {
118            name,
119            label: label_str,
120            // Non-nullable columns are required by default: an
121            // empty submit would die at the driver instead of failing
122            // inline. Override with `.optional()` for nullable columns.
123            required: !field.nullable(),
124            unique,
125            placeholder: None,
126            rules: Rules::new(),
127            typed_probe: None,
128            input_type: "text",
129        }
130    }

132    /// Create a `TextInput` bound to a lens inside an embedded struct or a
133    /// `#[document]`.
134    ///
135    /// The plain `Self::r#for` resolves a lens against the model alone, which
136    /// is why it can only bind a top-level field: the owned `app::Model` cannot
137    /// see the embedded models, so a path like `Post::fields().seo().title()`
138    /// is rejected as a traversal lens. This resolves through the request's app
139    /// schema instead, so the leaf arrives as its **flattened storage column**
140    /// (`seo_title`) — the name the form posts and the record fn reads.
141    ///
142    /// An embedded leaf is never `required` by default: the resolver reports
143    /// `nullable=true` by binding policy, since only the matching enum variant
144    /// writes a variant payload column. That is the binding default, not a
145    /// storage fact — the flattened column of a required embedded struct is
146    /// `NOT NULL`. Opt in with [`.required()`](Self::required).
147    ///
148    /// Without a `Db` in context (a bare `CxTestBuilder`) this behaves exactly
149    /// like `Self::r#for` and rejects the traversal lens loudly, so a test
150    /// cannot silently bind the wrong column.
151    pub fn r#for_context<M>(cx: &Cx, path: toasty::stmt::Path<M, String>) -> Self
152    where
153        M: toasty::schema::Model,
154    {
155        let leaf = FieldResolver::from_cx(cx).resolve(path);
156        Self {
157            name: leaf.name,
158            label: leaf.label,
159            required: !leaf.nullable,
160            unique: false,
161            placeholder: None,
162            rules: Rules::new(),
163            typed_probe: None,
164            input_type: "text",
165        }
166    }

168    /// Create a `TextInput` bound to a lens whose leaf is **not** a `String`.
169    ///
170    /// `Self::r#for` takes a `Path<M, String>`, which makes a wrong lens a
171    /// compile error rather than a runtime mismatch (ADR-0001) — and also makes
172    /// a typed column unbindable. This constructor keeps that guarantee: the
173    /// lens must address one field of the model and `T` must be the leaf's
174    /// actual type, so `TextInput::typed::<User, Uuid>(User::fields().name())`
175    /// does not compile either. It adds the value's spelling rule:
176    ///
177    /// - the control renders the value's `Display`;
178    /// - a submission `T` cannot parse is an **inline field error** naming the offending input (``
179    ///   `2024-13-01` is not a valid timestamp ``);
180    /// - what is stored is `T`'s `Display` of the parsed value, so a value re-submitted unchanged
181    ///   is written back in the shape it was read.
182    ///
183    /// The record fn still receives `String`s: the panel's value map is
184    /// text-keyed, and a typed field is a *validated* string, not a second
185    /// channel.
186    ///
187    /// `TypedValue` is implemented for the types a panel binds — the integer
188    /// types, `bool`, `f32`, `f64`, `Uuid`, `jiff::Timestamp` — rather than as a
189    /// blanket over `FromStr`, because the error names what was expected; `T`
190    /// must be [`toasty::stmt::IntoExpr`] of itself so the unique probe compares
191    /// through the parsed value rather than its text.
192    ///
193    /// A `jiff::Timestamp` leaf renders `type="datetime-local"`: the control
194    /// carries no zone, so the stored instant renders in UTC and a submission
195    /// is read back as UTC.
196    pub fn typed<M, T>(path: toasty::stmt::Path<M, T>) -> Self
197    where
198        M: toasty::schema::Model,
199        T: TypedValue + toasty::stmt::IntoExpr<T> + 'static,
200    {
201        let model = M::schema();
202        let field = lens_field(path, &model);
203        let label_str = lens_label(&field);
204        // A typed field declares no index and is not marked unique by default;
205        // when the app marks it, the probe binds the declared type.
206        let name = field.name.app_unwrap().to_string();
207        let probe = eq_probe_typed::<M, T>(&name);
208        let input_type = if is_timestamp::<T>() {
209            "datetime-local"
210        } else {
211            "text"
212        };
213        Self {
214            name,
215            label: label_str,
216            required: !field.nullable(),
217            unique: false,
218            placeholder: None,
219            rules: Rules::new().typed::<T>(),
220            typed_probe: Some(probe),
221            input_type,
222        }
223    }

225    /// [`Self::typed`] for a lens inside an embedded struct or a `#[document]`,
226    /// resolving through the request's app schema exactly as
227    /// `Self::r#for_context` does.
228    ///
229    /// A leaf under an embedded step is never required by default: the resolver
230    /// reports `nullable=true` by binding policy, since only the matching enum
231    /// variant writes a variant payload column. That is the binding default,
232    /// not a storage fact — the flattened column of a required embedded struct
233    /// is `NOT NULL`. Opt in with [`.required()`](Self::required).
234    pub fn typed_context<M, T>(cx: &Cx, path: toasty::stmt::Path<M, T>) -> Self
235    where
236        M: toasty::schema::Model,
237        T: TypedValue + toasty::stmt::IntoExpr<T> + 'static,
238    {
239        let leaf = FieldResolver::from_cx(cx).resolve(path);
240        let probe = eq_probe_typed::<M, T>(&leaf.name);
241        let input_type = if is_timestamp::<T>() {
242            "datetime-local"
243        } else {
244            "text"
245        };
246        Self {
247            name: leaf.name,
248            label: leaf.label,
249            required: !leaf.nullable,
250            unique: false,
251            placeholder: None,
252            rules: Rules::new().typed::<T>(),
253            typed_probe: Some(probe),
254            input_type,
255        }
256    }

275    /// Mark the field as backed by a unique constraint, which the app-side
276    /// pre-check probes before the write.
277    ///
278    /// **Uniqueness implies presence**: the framework stores `""`,
279    /// never NULL, so an empty value is one the index admits only
280    /// once — an empty submit is refused inline as `"<Label> is required"`
281    /// instead of being written, and the probe never sees it. `.optional()`
282    /// does not lift that rule, whichever order the two are called in. The
283    /// reasoning (and the rejected alternative) is recorded in the ADR-0010
284    /// amendment of 2026-09-21.
285    ///
286    /// Non-`TextInput` fields declare no uniqueness (see `Textarea::r#for`),
287    /// so nothing else changes.
288    pub fn unique(mut self) -> Self {
289        self.unique = true;
290        // The marker carries presence itself, so `.optional().unique()` and
291        // `.unique().optional()` mean the same thing.
292        self.required = true;
293        self
294    }

328    /// The equality expression the app-side unique check probes with.
329    ///
330    /// A text leaf compares its submission's text through `M`'s own path — the
331    /// model the handler queries, which is also the model a context-bound
332    /// leaf's flattened column belongs to. A typed leaf instead
333    /// parses the submission into its declared type and compares that, so the
334    /// probe sees the value the record will store rather than its spelling —
335    /// `01` and `1` are one value to an integer column. `None` when a typed
336    /// submission does not parse: validation has already refused it, and there
337    /// is nothing left to compare.
338    pub(crate) fn eq_filter<M>(&self, value: &str) -> Option<toasty::stmt::Expr<bool>>
339    where
340        M: toasty::schema::Model,
341    {
342        let Some(probe) = &self.typed_probe else {
343            let index = M::field_name_to_id(&self.name).index;
344            return Some(M::path_field::<String>(index).eq(value.to_string()));
345        };
346        probe(value)
347    }

365    /// Static render: the create/edit path's control, with `value` rendered
366    /// into the `value` attribute and `errors` into the error slot.
367    pub(crate) async fn render_with<'a>(
368        &self,
369        cx: &'a Cx,
370        value: Option<&str>,
371        errors: &[String],
372        mode: Mode,
373    ) -> Result<BoxView<'a>> {
374        if mode == Mode::View {
375            return render_value(cx, &self.label, value, ValueKind::Machine);
376        }
377        let name = self.name.clone();
378        // The marker reads the same predicate validation uses, so a unique
379        // field is never refused for emptiness while rendering as optional.
380        // `self.required` alone would do exactly that.
381        let required = self.is_required();
382        let placeholder = self.placeholder.clone();
383        let input_type = if self.rules.is_email() {
384            "email"
385        } else {
386            self.input_type
387        };
388        // A timestamp renders its UTC `datetime-local` spelling, not the
389        // stored RFC 3339: the control carries no zone.
390        let value_owned = value.map(|s| {
391            if self.input_type == "datetime-local" {
392                format_timestamp_input(s)
393            } else {
394                s.to_string()
395            }
396        });
397        let chrome = FieldChrome::new(&self.name, errors, None);
398        let aria_invalid = chrome.aria_invalid();
399        let described_by = chrome.described_by();
400        // Beautiful rendering via the upstream `field` family (topcoat#420):
401        // label + control + reserved error slot, the label following the
402        // field's invalid state, and `aria-invalid` driving the control's
403        // error border/ring.
404        let control = view! {
405            cx =>
406            ui_input(
407                attrs: attributes! {
408                    id=(name.clone())
409                    type=(input_type)
410                    name=(name.clone())
411                    value=(value_owned.clone())
412                    placeholder=(placeholder.clone())
413                    required=(required)
414                    aria-required=(required.then_some("true"))
415                    aria-invalid=(aria_invalid)
416                    aria-describedby=(described_by)
417                }
418            )
419        }
420        .boxed();
421        render_field(cx, &chrome, &self.label, required, attributes! {}, control)
422    }
