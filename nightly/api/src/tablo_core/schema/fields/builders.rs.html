1//! The typed field builders the [`Field`] constructors return.
2//!
3//! Each builder wraps a `Field` whose control is its own kind and offers only
4//! that control's modifiers, so `.placeholder(..)` on a choice is a compile
5//! error rather than a declaration the panel refuses.

12/// `label`, `required` and `optional`, shared by every builder.
13macro_rules! common_modifiers {
14    ($builder:ident) => {
15        impl $builder {
16            /// Override the label.
17            pub fn label(mut self, label: impl Into<String>) -> Self {
18                self.0.label = label.into();
19                self
20            }
21
22            /// Refuse an empty submission.
23            pub fn required(mut self) -> Self {
24                self.0.required = true;
25                self
26            }
27
28            /// Accept an empty submission: for a nullable column, or for a
29            /// column a record fn fills when the form leaves it empty. The
30            /// browser-side `required` attribute drops too.
31            pub fn optional(mut self) -> Self {
32                self.0.required = false;
33                self
34            }
35
36            /// The key the control posts.
37            pub fn name(&self) -> &str {
38                &self.0.name
39            }
40        }
41
42        impl From<$builder> for Field {
43            fn from(builder: $builder) -> Field {
44                builder.0
45            }
46        }
47
48        impl IntoSchema for $builder {
49            fn into_schema(self) -> Schema {
50                self.0.into_schema()
51            }
52        }
53    };
54}

74impl TextField {
75    fn text(&mut self) -> &mut TextControl {
76        match &mut self.0.control {
77            ControlKind::Text(text) => text,
78            _ => unreachable!("a TextField holds a text control"),
79        }
80    }
81
82    /// Validate the value as an email address, and render `type="email"`.
83    pub fn email(mut self) -> Self {
84        self.0.rules.set_email();
85        self
86    }
87
88    /// Mark the field as backed by a unique index, which the app-side
89    /// pre-check probes before the write.
90    ///
91    /// **Uniqueness implies presence on a non-nullable column**: an empty
92    /// `String` stores `""`, which the index admits only once, so an empty
93    /// submit is refused inline as `"<Label> is required"` instead of being
94    /// written, and the probe never sees it. `.optional()` does not lift that
95    /// rule, whichever order the two are called in (ADR-0010). A nullable
96    /// column (`Option<T>`) stores NULL for an empty submit, which the index
97    /// admits any number of times, so it stays optional when declared so.
98    pub fn unique(mut self) -> Self {
99        self.text().unique = true;
100        self
101    }
102
103    /// The control's placeholder text.
104    pub fn placeholder(mut self, placeholder: impl Into<String>) -> Self {
105        self.text().placeholder = Some(placeholder.into());
106        self
107    }
108
109    /// Render a `<textarea>` of `rows` lines rather than a one-line input,
110    /// for a column holding prose.
111    pub fn multiline(mut self, rows: u32) -> Self {
112        self.text().rows = Some(rows);
113        self
114    }
115}

117impl ChoiceField {
118    fn choice(&mut self) -> &mut ChoiceControl {
119        match &mut self.0.control {
120            ControlKind::Choice(choice) => choice,
121            _ => unreachable!("a ChoiceField holds a choice control"),
122        }
123    }
124
125    /// Static options: values that are their own label
126    /// (`["draft", "published"]`), `(value, label)` pairs, or a
127    /// [`#[derive(Options)]`](crate::Options) enum's
128    /// [`options()`](crate::schema::Options::options).
129    pub fn options(mut self, options: impl IntoOptions) -> Self {
130        self.choice().options = options.into_options();
131        self
132    }
133
134    /// Filter the options as the user types.
135    ///
136    /// Renders a filter input and a suggestion listbox above the select.
137    /// Typing narrows the list by label substring for a bounded set, and a pick
138    /// writes the chosen option onto the select, which stays the form control.
139    /// Past the option cap, a relationship fetches
140    /// `GET {parent_list_url}/options?field=&q=` (debounced, in-flight
141    /// requests aborted, selection preserved) and re-renders the list from the
142    /// answer, searching the related table's `searchable()` columns; a
143    /// non-searchable relationship keeps the cap error.
144    ///
145    /// Needs `assets/selects.js` (`tablo_ui::SELECTS_JS`), emitted by
146    /// `Panel::render_document` on every document with shell assets
147    /// (ADR-0014); without it the input is inert and the plain select keeps
148    /// working.
149    pub fn searchable(mut self) -> Self {
150        self.choice().searchable = true;
151        self
152    }
153
154    /// Load the options from a related source's tenant-scoped query.
155    ///
156    /// `R` is any [`OptionSource`]; every `Resource` is one. The loader calls
157    /// the source's scoped query, so its tenant gate and filter apply. `value`
158    /// projects a record to the model's **primary key**, whose `Display` is the
159    /// `<option value>`, and `label` maps it to its label. A projection to
160    /// anything but the key type fails to compile, and an edit form hydrates the
161    /// foreign key with the same string.
162    ///
163    /// Policy-checked: the related source's policy must allow `ViewAny` and,
164    /// when the source is tenant-scoped, the request must have a tenant; each
165    /// loaded row is filtered through `View`. A denial fails the load closed —
166    /// no options and not the stored value, `{label} is not available` on GET,
167    /// and a submit that carries a value fails with that message.
168    ///
169    /// Re-checked inside the write: the create and edit handlers resolve the
170    /// submitted key again through the source's tenant-scoped query in the
171    /// write's transaction, so a record deleted, moved to another tenant or
172    /// hidden since the form validated refuses the write with the same field
173    /// error.
174    ///
175    /// Bounded and memoized: at most one row past `MAX_RELATIONSHIP_OPTIONS`
176    /// loads per `(request, tenant)`. Past the cap a searchable field degrades
177    /// to type-to-search with a targeted existence check, and a non-searchable
178    /// one reports `could not load options, retry`.
179    pub fn relationship<R>(
180        mut self,
181        value: impl Fn(&R::Model) -> RelatedPrimaryKey<R> + Send + Sync + 'static,
182        label: impl Fn(&R::Model) -> String + Send + Sync + 'static,
183    ) -> Self
184    where
185        R: OptionSource + 'static,
186        RelatedPrimaryKey<R>: std::fmt::Display,
187    {
188        self.choice().relationship = Some(choice::Relationship::new::<R>(value, label));
189        self
190    }
191}

193/// A choice's static options as `(value, label)` pairs, in display order.
194///
195/// Implemented for a list of values that are their own label, for a list of
196/// pairs, and for an array of string slices.
197pub trait IntoOptions {
198    /// The `(value, label)` pairs.
199    fn into_options(self) -> Vec<(String, String)>;
200}
201
202impl IntoOptions for Vec<String> {
203    fn into_options(self) -> Vec<(String, String)> {
204        self.into_iter()
205            .map(|value| (value.clone(), value))
206            .collect()
207    }
208}
209
210impl IntoOptions for Vec<(String, String)> {
211    fn into_options(self) -> Vec<(String, String)> {
212        self
213    }
214}
215
216impl<const N: usize> IntoOptions for [&str; N] {
217    fn into_options(self) -> Vec<(String, String)> {
218        self.into_iter()
219            .map(|value| (value.to_string(), value.to_string()))
220            .collect()
221    }
222}
