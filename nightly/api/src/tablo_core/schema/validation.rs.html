7/// Parses a typed column's text submission and names the type in the error it produces.
8pub trait TypedValue: std::fmt::Display + std::str::FromStr {
9    /// What this type is called in a validation error.
10    const NOUN: &'static str;
11
12    /// The `type` attribute of the text control that edits it.
13    const INPUT_TYPE: &'static str = "text";
14
15    /// Reads a trimmed, non-empty submission, or `None` when the type refuses it.
16    fn parse_input(value: &str) -> Option<Self> {
17        value.parse().ok()
18    }
19}

21/// Binds the integer types a typed field accepts.
22macro_rules! typed_whole_number {
23    ($($ty:ty),* $(,)?) => {
24        $(
25            impl TypedValue for $ty {
26                const NOUN: &'static str = "whole number";
27            }
28        )*
29    };
30}

67/// Binds a timestamp to a `datetime-local` control read back as UTC.
68impl TypedValue for jiff::Timestamp {
69    const NOUN: &'static str = "timestamp";
70    const INPUT_TYPE: &'static str = "datetime-local";
71
72    /// Parses RFC 3339 plus the `datetime-local` shapes a browser sends, read as UTC.
73    fn parse_input(value: &str) -> Option<Self> {
74        if let Ok(parsed) = value.parse::<jiff::Timestamp>() {
75            return Some(parsed);
76        }
77        normalize_datetime_local(value)?.parse().ok()
78    }
79}

81/// Converts a `datetime-local` value to the RFC 3339 string a timestamp parses, or `None` when the
82/// shape is not one the control sends.
83fn normalize_datetime_local(value: &str) -> Option<String> {
84    let t = value.find('T')?;
85    let after_t = &value[t + 1..];
86    if after_t.contains('+') || after_t.contains('-') {
87        return None;
88    }
89    if value.ends_with(['Z', 'z']) {
90        return None;
91    }
92    if value.len() == 16 {
93        Some(format!("{value}:00Z"))
94    } else {
95        Some(format!("{value}Z"))
96    }
97}

99/// Formats a stored timestamp as the `datetime-local` value its control renders, rendering anything
100/// else empty.
101pub(crate) fn format_timestamp_input(storage: &str) -> String {
102    let trimmed = storage.trim();
103    if trimmed.is_empty() {
104        return String::new();
105    }
106    match trimmed.parse::<jiff::Timestamp>() {
107        Ok(parsed) => parsed.strftime("%Y-%m-%dT%H:%M").to_string(),
108        Err(_) => String::new(),
109    }
110}

112/// Reads a submitted string back as the stored spelling or the error message.
113type ValueParser = fn(&str) -> Result<String, String>;
114
115/// Binds the parser for scalar type `T`, rejecting what `T` refuses and storing what `T`'s own form
116/// spelling produces.
117fn scalar_parser<T: FormScalar>(value: &str) -> Result<String, String> {
118    T::parse_form(value).map(|parsed| parsed.to_form())
119}

121/// Holds the rules a field declares on top of presence and the wording of every message they
122/// produce.
123#[derive(Clone, Copy, Default)]
124pub(crate) struct Rules {
125    email: bool,
126    parser: Option<ValueParser>,
127}
128
129impl Rules {
130    /// Holds a field with no declared rule: presence alone.
131    pub(crate) fn new() -> Self {
132        Self::default()
133    }
134
135    /// Adds the parse rule of the scalar type `T`.
136    pub(crate) fn scalar<T: FormScalar>(mut self) -> Self {
137        self.parser = Some(scalar_parser::<T>);
138        self
139    }
140
141    /// Turns on the email rule.
142    pub(crate) fn set_email(&mut self) {
143        self.email = true;
144    }
145
146    /// Reports whether the email rule is on.
147    pub(crate) fn is_email(&self) -> bool {
148        self.email
149    }
150
151    /// Validates `value` in rule order and skips the email and typed-parse rules on an empty
152    /// submit.
153    pub(crate) fn validate(
154        &self,
155        key: &str,
156        label: &str,
157        required: bool,
158        value: &str,
159    ) -> Vec<FieldError> {
160        let v = value.trim();
161        let mut errs = Vec::new();
162        if required && v.is_empty() {
163            errs.push(FieldError::unanswered(key, required_error(label)));
164        }
165        if self.email && !v.is_empty() && !is_email(v) {
166            errs.push(FieldError::invalid(
167                key,
168                format!("{label} must be a valid email"),
169            ));
170        }
171        if !v.is_empty()
172            && errs.is_empty()
173            && let Some(parser) = &self.parser
174            && let Err(message) = parser(v)
175        {
176            errs.push(FieldError::invalid(key, message));
177        }
178        errs
179    }
180
181    /// Returns the stored spelling of an already-validated submission and reports a failure rather
182    /// than guessing.
183    pub(crate) fn normalize(&self, value: &str) -> Result<String, String> {
184        let v = value.trim();
185        match &self.parser {
186            Some(parser) => parser(v),
187            None => Ok(v.to_string()),
188        }
189    }
190}
191
192/// Returns the message for an empty submit.
193pub(crate) fn required_error(label: &str) -> String {
194    format!("{label} is required")
195}

197/// Caps the longest address the rule accepts.
198const EMAIL_MAX_LENGTH: usize = 254;
199
200/// Reports whether `value` is an address the email rule accepts, requiring a TLD and refusing
201/// display text.
202fn is_email(value: &str) -> bool {
203    value.len() <= EMAIL_MAX_LENGTH
204        && EmailAddress::parse_with_options(
205            value,
206            Options::default()
207                .with_required_tld()
208                .without_display_text(),
209        )
210        .is_ok()
211}
