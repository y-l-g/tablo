1//! Tenancy via the `Cx` scoped value `Tenant(id)`.
2//!
3//! The framework owns the tenant *filter*, not only the tenant gate. For a
4//! resource whose `requires_tenant()` is `true`, every loader runs
5//! [`scoped_query`](crate::resource::scoped_query) — the resource's own
6//! [`query`](crate::resource::Resource::query) with `tenant_id = <tenant>`
7//! ANDed onto it — and the column is discovered from the model's schema here
8//! (`derived_tenant_filter`, the default body of `Resource::tenant_scope`). A
9//! gated resource therefore cannot serve unscoped rows by forgetting an
10//! override, and `Resource::query` stays the app's *non-tenant* scoping seam.
11//!
12//! Discovery is narrow and fails closed: the model must declare a field whose
13//! application name is `tenant_id` and whose type is a UUID. Anything else
14//! reads as "no tenant column", and a gated resource that hits that is refused
15//! when the panel is mounted. A predicate that is only `None` for some tenants
16//! still answers an error naming the resource rather than querying unscoped.
17//!
18//! The authenticated user's tenant is the production source: [`tenant_id`]
19//! answers the tenant the panel's signed-in user carries.
20//! A server-set `Tenant` request extension takes precedence, so app middleware
21//! and `Router::handle` tests can override it deliberately. No request header
22//! supplies a tenant: learning another tenant's UUID does not make anyone that
23//! tenant.
24
25use toasty::stmt::Expr;
26use topcoat::context::{Cx, try_request_context};
27
28/// Request-scoped tenant identifier.
29///
30/// App code may put it on the `Cx` with `cx.with(Tenant(id))`, and server code
31/// and tests may carry it as a request extension; either wins over the signed-in
32/// user's tenant.
33#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
34pub struct Tenant(pub uuid::Uuid);
35
36/// Returns the tenant id from `cx`, if present.
37///
38/// Checks a `Tenant` request extension first (a server-set override — the
39/// deliberate seam app middleware and `Router::handle` tests use), then a
40/// `Tenant` scoped value app code put on the `Cx`, then the tenant of the
41/// panel's signed-in [`current_user`](crate::auth::current_user). No request
42/// header is consulted.
43pub fn tenant_id(cx: &Cx) -> Option<uuid::Uuid> {
44    if let Some(parts) = try_request_context::<http::request::Parts>(cx)
45        && let Some(t) = parts.extensions.get::<Tenant>()
46    {
47        return Some(t.0);
48    }
49    if let Some(t) = try_request_context::<Tenant>(cx) {
50        return Some(t.0);
51    }
52    crate::auth::current_user(cx).and_then(|user| user.tenant_id)
53}
54
55/// Requires a tenant, returning an error if missing (for tenancy-gated resources).
56pub fn require_tenant(cx: &Cx) -> Result<uuid::Uuid, topcoat::Error> {
57    tenant_id(cx).ok_or_else(|| topcoat::router::error::forbidden().into())
58}

60/// The application name of the column the framework scopes a gated resource by.
61///
62/// The convention is public — it is the contract a model signs up to when its
63/// resource declares
64/// [`requires_tenant`](crate::resource::Resource::requires_tenant) — and this
65/// is the one place the discovery reads it.
66const TENANT_FIELD: &str = "tenant_id";
67
68/// Position of `M`'s tenant column in its own schema, or `None` when `M`
69/// declares none the framework can recognize.
70///
71/// Found by **name and type** over [`Model::schema`]'s field list: a primitive
72/// field whose application name is `tenant_id` and whose type is
73/// [`toasty::stmt::Type::Uuid`]. The index is taken from the field's own
74/// [`FieldId`](toasty::schema::app::FieldId) — the index
75/// [`Model::path_field`] addresses — rather than from the position in the
76/// vector, so the filter and the generated `tenant_id()` accessor cannot drift.
77///
78/// Name-based discovery is the fragile half, so a miss is `None` and every
79/// caller treats it as *fail closed*: nothing falls back to the unscoped query,
80/// and nothing guesses. A `tenant_id` declared as `String`, or a tenant column
81/// spelled any other way, is invisible here on purpose — comparing a UUID
82/// against it would be a driver-level type error at best and a cross-tenant
83/// match at worst.
84pub(crate) fn tenant_field_index<M: toasty::schema::Model>() -> Option<usize> {
85    M::schema()
86        .fields()
87        .iter()
88        .find(|field| {
89            field.name.app.as_deref() == Some(TENANT_FIELD)
90                && field
91                    .ty
92                    .as_primitive()
93                    .is_some_and(|primitive| primitive.ty.is_uuid())
94        })
95        .map(|field| field.id.index)
96}
97
98/// The form key of `M`'s tenant column, when [`tenant_field_index`] finds one.
99pub(crate) fn tenant_field_name<M: toasty::schema::Model>() -> Option<String> {
100    let index = tenant_field_index::<M>()?;
101    M::schema()
102        .fields()
103        .get(index)
104        .and_then(|field| field.name.app.clone())
105}

107/// The **derived** `tenant_id = tenant` over `M`, or `None` when
108/// [`tenant_field_index`] finds no tenant column.
109///
110/// This is the default body of
111/// [`Resource::tenant_scope`](crate::resource::Resource::tenant_scope) — the
112/// public hook a resource overrides when its rows inherit their tenant instead
113/// of carrying one — so the name says *derived*: it is the name-based
114/// convenience, not the only way to scope a gated resource.
115///
116/// Built generically on purpose: `Model::path_field` addresses the column by
117/// index and `Value::Uuid` types the comparison, so no generated accessor — and
118/// no per-resource copy of the filter — is needed.
119pub(crate) fn derived_tenant_filter<M: toasty::schema::Model>(
120    tenant: uuid::Uuid,
121) -> Option<Expr<bool>> {
122    let index = tenant_field_index::<M>()?;
123    Some(M::path_field::<uuid::Uuid>(index).eq(tenant))
124}
