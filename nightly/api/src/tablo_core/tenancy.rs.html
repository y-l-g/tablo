1//! Tenancy via the `Cx` scoped value `Tenant(id)`.
2//!
3//! The framework owns the tenant *filter*, not only the tenant gate. For a
4//! resource whose `requires_tenant()` is `true`, every loader runs
5//! [`scoped_query`](crate::resource::scoped_query) — the resource's own
6//! [`query`](crate::resource::Resource::query) with `tenant_id = <tenant>`
7//! ANDed onto it — and the column is discovered from the model's schema here
8//! (`derived_tenant_filter`, the default body of `Resource::tenant_scope`). A
9//! gated resource therefore cannot serve unscoped rows by forgetting an
10//! override, and `Resource::query` stays the app's *non-tenant* scoping seam.
11//!
12//! Discovery is narrow and fails closed: the model must declare a field whose
13//! application name is `tenant_id` and whose type is a UUID. Anything else
14//! reads as "no tenant column", and a gated resource that hits that is refused
15//! by `Panel::build` at boot. A predicate that is only `None` for some tenants
16//! still answers an error naming the resource rather than querying unscoped.
17//!
18//! The authenticated user's tenant is the production source: the auth layer
19//! injects `Tenant` into the request `Cx` when the logged-in user carries one.
20//! A server-set `Tenant` request extension takes precedence, so app middleware
21//! and `Router::handle` tests can override it deliberately. No request header
22//! supplies a tenant: learning another tenant's UUID does not make anyone that
23//! tenant.
24
25use toasty::stmt::Expr;
26use topcoat::context::{Cx, try_request_context};
27
28/// Request-scoped tenant identifier.
29///
30/// The auth layer sets it via `cx.with(Tenant(id))` from the logged-in user;
31/// server code and tests may also carry it as a request extension.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
33pub struct Tenant(pub uuid::Uuid);
34
35/// Returns the tenant id from `cx`, if present.
36///
37/// Checks a `Tenant` request extension first (a server-set override — the
38/// deliberate seam app middleware and `Router::handle` tests use), then the
39/// `Tenant` scoped value the auth layer injects from the authenticated user.
40/// No request header is consulted.
41pub fn tenant_id(cx: &Cx) -> Option<uuid::Uuid> {
42    if let Some(parts) = try_request_context::<http::request::Parts>(cx)
43        && let Some(t) = parts.extensions.get::<Tenant>()
44    {
45        return Some(t.0);
46    }
47    try_request_context::<Tenant>(cx).map(|t| t.0)
48}
49
50/// Requires a tenant, returning an error if missing (for tenancy-gated resources).
51pub fn require_tenant(cx: &Cx) -> Result<uuid::Uuid, topcoat::Error> {
52    tenant_id(cx).ok_or_else(|| topcoat::router::error::forbidden().into())
53}

55/// The application name of the column the framework scopes a gated resource by.
56///
57/// The convention is public — it is the contract a model signs up to when its
58/// resource declares
59/// [`requires_tenant`](crate::resource::Resource::requires_tenant) — and this
60/// is the one place the discovery reads it.
61const TENANT_FIELD: &str = "tenant_id";
62
63/// Position of `M`'s tenant column in its own schema, or `None` when `M`
64/// declares none the framework can recognize.
65///
66/// Found by **name and type** over [`Model::schema`]'s field list: a primitive
67/// field whose application name is `tenant_id` and whose type is
68/// [`toasty::stmt::Type::Uuid`]. The index is taken from the field's own
69/// [`FieldId`](toasty::schema::app::FieldId) — the index
70/// [`Model::path_field`] addresses — rather than from the position in the
71/// vector, so the filter and the generated `tenant_id()` accessor cannot drift.
72///
73/// Name-based discovery is the fragile half, so a miss is `None` and every
74/// caller treats it as *fail closed*: nothing falls back to the unscoped query,
75/// and nothing guesses. A `tenant_id` declared as `String`, or a tenant column
76/// spelled any other way, is invisible here on purpose — comparing a UUID
77/// against it would be a driver-level type error at best and a cross-tenant
78/// match at worst.
79pub(crate) fn tenant_field_index<M: toasty::schema::Model>() -> Option<usize> {
80    M::schema()
81        .fields()
82        .iter()
83        .find(|field| {
84            field.name.app.as_deref() == Some(TENANT_FIELD)
85                && field
86                    .ty
87                    .as_primitive()
88                    .is_some_and(|primitive| primitive.ty.is_uuid())
89        })
90        .map(|field| field.id.index)
91}
92
93/// The form key of `M`'s tenant column, when [`tenant_field_index`] finds one.
94pub(crate) fn tenant_field_name<M: toasty::schema::Model>() -> Option<String> {
95    let index = tenant_field_index::<M>()?;
96    M::schema()
97        .fields()
98        .get(index)
99        .and_then(|field| field.name.app.clone())
100}
101
102/// The **derived** `tenant_id = tenant` over `M`, or `None` when
103/// [`tenant_field_index`] finds no tenant column.
104///
105/// This is the default body of
106/// [`Resource::tenant_scope`](crate::resource::Resource::tenant_scope) — the
107/// public hook a resource overrides when its rows inherit their tenant instead
108/// of carrying one — so the name says *derived*: it is the name-based
109/// convenience, not the only way to scope a gated resource.
110///
111/// Built generically on purpose: `Model::path_field` addresses the column by
112/// index and `Value::Uuid` types the comparison, so no generated accessor — and
113/// no per-resource copy of the filter — is needed.
114pub(crate) fn derived_tenant_filter<M: toasty::schema::Model>(
115    tenant: uuid::Uuid,
116) -> Option<Expr<bool>> {
117    let index = tenant_field_index::<M>()?;
118    Some(M::path_field::<uuid::Uuid>(index).eq(tenant))
119}
