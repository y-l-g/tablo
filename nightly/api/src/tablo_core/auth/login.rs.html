18/// Max body the login route accepts: 64 KiB.
19///
20/// A credential form submits an email, a password, a `next` and a CSRF token,
21/// all short strings; the panel's 10 MiB form cap is for multipart uploads,
22/// which the login route never carries. 64 KiB leaves room for the fields plus
23/// percent-encoding growth while keeping the one unauthenticated POST route
24/// from buffering a megabyte-scale body.
25pub(crate) const MAX_LOGIN_BYTES: usize = 64 * 1024;

27/// Form field carrying the login identifier (the shipped default reads it as
28/// an email address).
29pub const LOGIN_FIELD: &str = "email";
30/// Form field carrying the password.
31pub const PASSWORD_FIELD: &str = "password";
32/// Hidden form field carrying the validated post-login destination.
33pub const NEXT_FIELD: &str = "next";

35/// The one error every failed login renders, so accounts cannot be

37pub(super) const GENERIC_ERROR: &str = "Invalid email or password.";

52/// The login URL with a validated `next` back to the requested page.
53pub(super) fn login_url_with_next(cx: &Cx) -> String {
54    let mut url = login_url(cx);
55    let request = uri(cx);
56    let requested = match request.query() {
57        Some(query) => format!("{}?{query}", request.path()),
58        None => request.path().to_string(),
59    };
60    if let Some(next) = safe_next(&requested) {
61        let mut serializer = form_urlencoded::Serializer::new(String::new());
62        serializer.append_pair(NEXT_FIELD, next);
63        url.push('?');
64        url.push_str(&serializer.finish());
65    }
66    url
67}

69/// Accept only same-origin relative paths as a post-login destination
70/// (ADR-0013): absolute URLs, scheme-relative `//host` targets, backslash
71/// tricks, and control characters are rejected.
72pub(crate) fn safe_next(next: &str) -> Option<&str> {
73    let next = next.trim();
74    if !next.starts_with('/') || next.starts_with("//") {
75        return None;
76    }
77    if next.contains('\\') || next.chars().any(|c| c.is_control()) {
78        return None;
79    }
80    Some(next)
81}

83/// The validated `?next=` the login page embeds as a hidden field.
84fn next_from_query(cx: &Cx) -> Option<String> {
85    let query = uri(cx).query()?;
86    form_urlencoded::parse(query.as_bytes())
87        .find(|(key, _)| key == NEXT_FIELD)
88        .map(|(_, value)| value.into_owned())
89        .filter(|value| safe_next(value).is_some())
90}

92/// What a failed login attempt renders: the generic credential rejection, or
93/// the sign-in outage.
94///
95/// Two variants and no more, each with its own copy and status, so a failed
96/// attempt is always a deliberate answer: a driver failure is never rendered
97/// as a credential verdict, and a credential verdict never borrows the outage
98/// copy. Neither variant carries driver text.
99#[derive(Debug, Clone, Copy)]
100enum LoginError {
101    /// Wrong password, unknown account, empty fields, or valid credentials
102    /// without panel access: one 403 with one message (ADR-0013).
103    Credentials,
104    /// The database behind sign-in could not answer: a 503 that says so,
105    /// because the user must not read an outage as a rejected password.
106    Unavailable,
107}
108
109impl LoginError {
110    /// The message the login page's alert carries.
111    fn message(self) -> &'static str {
112        match self {
113            Self::Credentials => GENERIC_ERROR,
114            Self::Unavailable => UNAVAILABLE_ERROR,
115        }
116    }
117
118    /// The status a failed attempt answers with.
119    fn status(self) -> http::StatusCode {
120        match self {
121            Self::Credentials => http::StatusCode::FORBIDDEN,
122            Self::Unavailable => http::StatusCode::SERVICE_UNAVAILABLE,
123        }
124    }
125}

127/// The status and route a login attempt renders: a [`LoginError`] page or a
128/// redirect back to `next`.
129///
130/// The login page is a settled view, so [`ViewExt::single`] resolves it into
131/// an owned handle before the response is built — no borrowed view escapes.
132async fn login_response(
133    cx: &Cx,
134    error: Option<LoginError>,
135    next: String,
136) -> topcoat::Result<topcoat::router::response::Response> {
137    let page = render_login_page(cx, error, next).await?;
138    page.single().await?.into_response(cx)
139}

141/// `GET {prefix}/login` — the standalone login page.
142pub(crate) fn login_page(cx: &Cx, _body: Body) -> RouteFuture<'_> {
143    let next = next_from_query(cx).unwrap_or_default();
144    Box::pin(login_response(cx, None, next))
145}

147/// `POST {prefix}/login` — verify, rotate the session, redirect to `next`.
148pub(crate) fn login_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
149    Box::pin(async move {
150        // Login/logout carry no file parts: the values half is all they read.
151        let values = crate::panel::parse_form_body(cx, body).await?.values;
152        crate::csrf::verify(cx, &values)?;
153        // Keep a validated destination across a failed attempt so the retry
154        // form still returns where the visitor was headed (US6).
155        let next = values
156            .get(NEXT_FIELD)
157            .and_then(|value| safe_next(value))
158            .map(str::to_string)
159            .or_else(|| next_from_query(cx))
160            .unwrap_or_default();
161        let login = values.get(LOGIN_FIELD).map(|value| value.trim());
162        let password = values.get(PASSWORD_FIELD).map(String::as_str);
163        let panel = current(cx).ok_or_else(topcoat::router::error::not_found)?;
164        let verified = match (panel.auth.authenticator(), login, password) {
165            (Some(authenticator), Some(login), Some(password))
166                if !login.is_empty() && !password.is_empty() =>
167            {
168                match authenticator.verify(cx, login, password).await {
169                    Ok(user) => user,
170                    // A driver failure is not a credential verdict: the page
171                    // says sign-in is unavailable instead of rendering a
172                    // rejection.
173                    Err(error) => return failed(cx, error, next).await,
174                }
175            }
176            _ => None,
177        };
178        // One path for every failure: wrong password, unknown account, empty
179        // fields, or valid credentials without panel access (ADR-0013).
180        let Some(user) = verified.filter(|user| user.can_access_panel()) else {
181            return login_response(cx, Some(LoginError::Credentials), next).await;
182        };
183        // Rotate on login: a token this request presented cannot be replayed.
184        if let Some(hash) = session::token_hash(cx).await? {
185            delete_session(cx, &hash).await?;
186        }
187        // Bounded housekeeping: the sweep goes with the rotation. A failure is
188        // logged rather than fatal — a credential that verified must not become
189        // a 503 because cleanup could not run.
190        if let Err(error) = sweep_expired_sessions(cx).await {
191            tracing::error!(error = %error, "expired-session sweep failed");
192        }
193        let session = session::start(cx).await?;
194        // The credentials were right but the row could not be recorded: the
195        // same outage page as a failed verification.
196        if let Err(error) = record(cx, &session, &*user, panel).await {
197            return failed(cx, error, next).await;
198        }
199        let target = if next.is_empty() {
200            panel_root(cx)
201        } else {
202            next
203        };
204        // Success stays on the `Ok` path so `Set-Cookie` flushes
205        // (upstream topcoat#126).
206        topcoat::router::error::see_other(target).into_response(cx)
207    })
208}

210/// A login attempt that failed on `error`: the outage page for a driver
211/// failure, whose text goes to the log; an app-authored error keeps its
212/// mapping.
213async fn failed(
214    cx: &Cx,
215    error: topcoat::Error,
216    next: String,
217) -> topcoat::Result<topcoat::router::response::Response> {
218    let error = infrastructure_failure(error);
219    if crate::error::TabloError::is_infrastructure(&error) {
220        return login_response(cx, Some(LoginError::Unavailable), next).await;
221    }
222    Err(error)
223}

225/// `POST {prefix}/logout` — delete the session row and clear the cookie.
226pub(crate) fn logout_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
227    Box::pin(async move {
228        // Defense in depth: the route only exists on gated panels, but it
229        // re-checks so a missing layer cannot leave logout ungated. Any
230        // resolved user may log out — the gate answers this route for a user
231        // without panel access too, so demanding panel access here would
232        // strand their session row + cookie to expiry.
233        if resolved(cx).is_none() {
234            return Err(unauthenticated_error(cx));
235        }
236        // Login/logout carry no file parts: the values half is all they read.
237        let values = crate::panel::parse_form_body(cx, body).await?.values;
238        crate::csrf::verify(cx, &values)?;
239        if let Some(hash) = session::stop(cx).await? {
240            delete_session(cx, &hash).await?;
241        }
242        let target = login_url(cx);
243        topcoat::router::error::see_other(target).into_response(cx)
244    })
245}

247/// `POST {prefix}/tenant` — act for another of the user's tenants, then land
248/// on the panel root: a page of the old tenant's records has nothing to show
249/// under the new one.
250///
251/// A tenant that is not one of the user's [`tenants`](super::PanelUser::tenants)
252/// answers 403. The selection is stored on the session and checked against
253/// the memberships on every request, so a membership removed later stops
254/// applying at once.
255pub(crate) fn tenant_post(cx: &Cx, body: Body) -> RouteFuture<'_> {
256    Box::pin(async move {
257        let Some(signed) = signed(cx) else {
258            return Err(unauthenticated_error(cx));
259        };
260        let values = crate::panel::parse_form_body(cx, body).await?.values;
261        crate::csrf::verify(cx, &values)?;
262        let tenant = values
263            .get(TENANT_FIELD)
264            .and_then(|value| uuid::Uuid::parse_str(value).ok())
265            .filter(|tenant| {
266                signed
267                    .user
268                    .tenants()
269                    .iter()
270                    .any(|membership| membership.tenant == *tenant)
271            })
272            .ok_or_else(topcoat::router::error::forbidden)?;
273        select_tenant(cx, tenant).await?;
274        topcoat::router::error::see_other(panel_root(cx)).into_response(cx)
275    })
276}

283/// The standalone login document: brand and dark mode honored, CSRF hidden
284/// field, one error slot carrying a [`LoginError`]'s deliberate copy, no
285/// sidebar (ADR-0013).
286async fn render_login_page<'a>(
287    cx: &'a Cx,
288    error: Option<LoginError>,
289    next: String,
290) -> topcoat::Result<BoxView<'a>> {
291    let csrf = crate::csrf::ensure_token(cx);
292    let action = login_url(cx);
293    let brand = Panel::render_brand(cx).await?;
294    let hint = current(cx).and_then(|panel| panel.login_hint.clone());
295    let body = topcoat::view::view! {
296        cx =>
297        <div class="flex min-h-svh items-center justify-center bg-muted p-6">
298            <div
299                class="flex w-full max-w-sm flex-col gap-6 rounded-xl border border-border bg-card p-6 text-card-foreground shadow-sm"
300            >
301                if let Some(error) = error {
302                    (error.status())
303                }
304                <div class="flex flex-col items-center gap-2">
305                    (brand)
306                    <h1 class="text-lg font-semibold text-foreground">"Sign in"</h1>
307                </div>
308                <form method="post" action=(action) class="flex flex-col gap-4">
309                    (crate::csrf::field(cx, &csrf))
310                    <input type="hidden" name=(NEXT_FIELD) value=(next)>
311                    if let Some(error) = error {
312                        tablo_ui::alert(
313                            variant: tablo_ui::AlertVariant::Destructive,
314                            attrs: topcoat::view::attributes! { role="alert" },
315                            tablo_ui::alert_title((error.message()))
316                        )
317                    }
318                    tablo_ui::field(
319                        tablo_ui::field_label(
320                            attrs: topcoat::view::attributes! { for="email" },
321                            "Email or username"
322                        )
323                        tablo_ui::input(
324                            attrs: topcoat::view::attributes! {
325                                id="email"
326                                name=(LOGIN_FIELD)
327                                type="text"
328                                required=""
329                                autocomplete="username"
330                                autofocus=""
331                            }
332                        )
333                    )
334                    tablo_ui::field(
335                        tablo_ui::field_label(
336                            attrs: topcoat::view::attributes! { for="password" },
337                            "Password"
338                        )
339                        tablo_ui::input(
340                            attrs: topcoat::view::attributes! {
341                                id="password"
342                                name=(PASSWORD_FIELD)
343                                type="password"
344                                required=""
345                                autocomplete="current-password"
346                            }
347                        )
348                    )
349                    tablo_ui::button(
350                        variant: tablo_ui::ButtonVariant::Primary,
351                        attrs: topcoat::view::attributes! { type="submit" class="w-full" },
352                        "Sign in"
353                    )
354                </form>
355                if let Some(hint) = hint {
356                    <p class="text-center text-xs text-muted-foreground">(hint)</p>
357                }
358            </div>
359        </div>
360    }
361    .boxed();
362    Panel::render_document(cx, "Sign in".to_string(), body).await
363}
