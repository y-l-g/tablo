10/// A dummy Argon2id PHC string verified against when the account does not
11/// exist, so unknown logins pay the same work as known ones (ADR-0013).
12/// Generated with `Argon2::default()` parameters (`m=19456,t=2,p=1`).
13const DUMMY_PASSWORD_HASH: &str = "$argon2id$v=19$m=19456,t=2,p=1$h3oXdPBVwhcgZ1OTO/PuzQ$zLrHLgIkwhqu4ZlLTfSyB8mPuL6mAtaswv/eXJ5ADO8";

15/// The shipped credential model (ADR-0013): unique email, Argon2id PHC hash,
16/// display name, and an active flag. It belongs to no tenant; an app whose
17/// users belong to tenants implements [`PanelUser`] for its own model.
18///
19/// Register it (plus [`AuthSession`](super::AuthSession)) in the app's `Db`
20/// model list, seed one row, and the default [`PasswordAuth`] works with no
21/// further wiring:
22/// `toasty::models!(…, tablo_core::auth::AdminUser, tablo_core::auth::AuthSession)`.
23#[derive(Debug, Clone, toasty::Model)]
24pub struct AdminUser {
25    #[key]
26    #[auto]
27    pub id: Uuid,
28    #[unique]
29    pub email: String,
30    /// Argon2id password hash in PHC string format.
31    pub password_hash: String,
32    pub display_name: String,
33    /// `false` denies login and panel access immediately.
34    pub active: bool,
35    pub created_at: Timestamp,
36}
37
38impl PanelUser for AdminUser {
39    fn user_id(&self) -> String {
40        self.id.to_string()
41    }
42
43    fn display_name(&self) -> &str {
44        &self.display_name
45    }
46
47    fn can_access_panel(&self) -> bool {
48        self.active
49    }
50}

52/// The shipped default authenticator: Argon2id verification against
53/// [`AdminUser`]. An inactive user does not authenticate.
54#[derive(Debug, Default, Clone, Copy)]
55pub struct PasswordAuth;
56
57impl Authenticator for PasswordAuth {
58    type User = AdminUser;
59
60    async fn verify(
61        &self,
62        cx: &Cx,
63        login: &str,
64        password: &str,
65    ) -> topcoat::Result<Option<AdminUser>> {
66        let mut db = crate::db::db(cx);
67        let user = AdminUser::filter(AdminUser::fields().email().eq(login.to_string()))
68            .first()
69            .exec(&mut db)
70            .await
71            .map_err(infrastructure_failure)?;
72        let hash = user.as_ref().map(|user| user.password_hash.as_str());
73        if !verify_password(password, hash) {
74            return Ok(None);
75        }
76        Ok(user)
77    }
78
79    async fn find_by_id(&self, cx: &Cx, id: &str) -> topcoat::Result<Option<AdminUser>> {
80        let Ok(id) = Uuid::parse_str(id) else {
81            return Ok(None);
82        };
83        let mut db = crate::db::db(cx);
84        let user = AdminUser::filter(AdminUser::fields().id().eq(id))
85            .first()
86            .exec(&mut db)
87            .await
88            .map_err(infrastructure_failure)?;
89        // A deactivated account stops resolving: its live sessions are
90        // purged and the next request redirects to login (spec #127 US11).
91        Ok(user.filter(|user| user.active))
92    }
93}

95/// Hash a password with Argon2id into a PHC string (the shipped storage
96/// format). Seeds and record fns call this; it never stores the plaintext.
97///
98/// # Errors
99///
100/// When Argon2 cannot hash, which the default parameters do not cause.
101pub fn hash_password(password: &str) -> topcoat::Result<String> {
102    use argon2::password_hash::PasswordHasher;
103
104    argon2::Argon2::default()
105        .hash_password(password.as_bytes())
106        .map(|hash| hash.to_string())
107        .map_err(topcoat::Error::from)
108}

110/// Verify a password against the account's Argon2id PHC hash, or `None` for
111/// an account that does not exist.
112///
113/// An unknown account is verified against a dummy hash and answers `false`,
114/// so it costs the same work as a wrong password and response times do not
115/// reveal which accounts exist. A malformed hash answers `false`. An
116/// [`Authenticator`] over its own user table calls it with the hash it loaded:
117///
118/// ```ignore
119/// let staff = Staff::filter_by_email(login).first().exec(&mut db).await?;
120/// if !verify_password(password, staff.as_ref().map(|s| s.password_hash.as_str())) {
121///     return Ok(None);
122/// }
123/// ```
124#[must_use]
125pub fn verify_password(password: &str, hash: Option<&str>) -> bool {
126    use argon2::password_hash::{PasswordVerifier, phc::PasswordHash};
127
128    let Ok(parsed) = PasswordHash::new(hash.unwrap_or(DUMMY_PASSWORD_HASH)) else {
129        return false;
130    };
131    let verified = argon2::Argon2::default()
132        .verify_password(password.as_bytes(), &parsed)
133        .is_ok();
134    verified && hash.is_some()
135}
