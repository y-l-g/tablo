1//! The crate's own failures, and the one mapping of a driver failure.
2//!
3//! Every error Tablo raises itself is a [`TabloError`]; `?` converts it into
4//! [`topcoat::Error`], and the seams that answer differently by kind read it
5//! back with [`TabloError::of`]. An app's errors and Topcoat's own (a 404, a
6//! 403) pass through untouched.
7
8use std::fmt;
9
10/// A failure Tablo raises itself.
11#[derive(Debug)]
12pub(crate) enum TabloError {
13    /// A malformed `?after=`/`?before=` token. Retrying the identical URL can
14    /// never succeed, so the list's retry link drops the cursor.
15    Cursor(String),
16    /// A well-formed cursor the query's ordering refuses: it was cut from a
17    /// different `ORDER BY`. Same retry contract as [`Self::Cursor`].
18    CursorRejected(String),
19    /// A panel or resource declaration that cannot work. The message names
20    /// what to change; it is the app author's, so it may name types.
21    Declaration(String),
22    /// The database, or another dependency, failed. The message is the only
23    /// text the response carries; the cause went to the log.
24    Infrastructure(&'static str),
25}
26
27impl TabloError {
28    /// The `TabloError` inside `error`, if Tablo raised it.
29    pub(crate) fn of(error: &topcoat::Error) -> Option<&Self> {
30        error.downcast_ref::<Self>()
31    }
32
33    /// Whether `error` is the request's cursor's fault: a malformed token or
34    /// a token the ordering rejects. The list page drops the cursor from its
35    /// retry link, and the live retry resets it.
36    pub(crate) fn is_cursor(error: &topcoat::Error) -> bool {
37        matches!(
38            Self::of(error),
39            Some(Self::Cursor(_) | Self::CursorRejected(_))
40        )
41    }
42
43    /// Whether `error` is an infrastructure failure.
44    pub(crate) fn is_infrastructure(error: &topcoat::Error) -> bool {
45        matches!(Self::of(error), Some(Self::Infrastructure(_)))
46    }
47}
48
49impl fmt::Display for TabloError {
50    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
51        let message = match self {
52            Self::Cursor(message) | Self::CursorRejected(message) | Self::Declaration(message) => {
53                message.as_str()
54            }
55            Self::Infrastructure(message) => message,
56        };
57        f.write_str(message)
58    }
59}
60
61impl std::error::Error for TabloError {}
62
63/// The message a database failure carries.
64pub(crate) const DATABASE_UNAVAILABLE: &str = "database unavailable";
65
66/// Map a database infrastructure failure (pool or transaction open, probe,
67/// statement, commit) to an opaque 500: `source` is logged for operators and
68/// never reaches the page.
69pub(crate) fn unavailable(source: impl fmt::Display) -> topcoat::Error {
70    tracing::error!(error = %source, "database unavailable");
71    TabloError::Infrastructure(DATABASE_UNAVAILABLE).into()
72}
73
74/// Map an error that may be the driver's: a [`toasty::Error`] is an
75/// infrastructure failure, logged and answered with `message`; anything else is
76/// the app's own (a record hook's guard, a custom authenticator's error) and
77/// keeps its mapping, so a guard's 404 stays a 404.
78///
79/// The error's type settles which it is, never its text.
80pub(crate) fn driver_failure(
81    error: impl Into<topcoat::Error>,
82    message: &'static str,
83) -> topcoat::Error {
84    let error = error.into();
85    if error.is::<toasty::Error>() {
86        tracing::error!(error = %error, answer = message, "infrastructure failure");
87        TabloError::Infrastructure(message).into()
88    } else {
89        error
90    }
91}
