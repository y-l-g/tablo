1//! Read-only relation rendering for detail pages.
2//!
3//! A detail page shows a record's related rows — a post's comments — from the
4//! rows `Resource::query`'s `include` already loaded. This renders them, and
5//! deliberately renders *only* them: no pager, no search, no row actions, no
6//! bulk column. A relation on a record page is a fixed, already-loaded set,
7//! and the list's chrome exists to narrow a query this page never runs.
8//!
9//! Two bounds apply to that set: the related resource's `can_view`
10//! decides which rows the reader may see, and [`MAX_RELATION_ROWS`] caps how
11//! many render. Both are in-memory decisions over the loaded rows, so the
12//! relation still issues no query.
13//!
14//! Everything here is owned: a column's projection returns a `String`, so the
15//! rendered view borrows the request context and nothing else. That is what
16//! lets `Resource::view_relations(cx, record)` return a view that outlives the
17//! record it read — the page is rendered before the handler's `record` binding
18//! drops, and the borrow checker says so if it is not.
19

33/// The most rows a relation table renders.
34///
35/// The cap bounds the rendered page, not a query: the related rows are already
36/// loaded, so what it removes is cell projection and DOM size. 50 covers the
37/// one-to-many sets a detail page summarises while keeping a runaway relation
38/// from making the page unusable; the table prints an overflow line when it
39/// truncates, so a capped relation never reads as a complete one.
40pub const MAX_RELATION_ROWS: usize = 50;
41
42/// One column of a relation's read-only table.
43///
44/// The projection is the same shape a list column uses — a typed closure over
45/// the related record — minus everything that only makes sense against a
46/// query: no `sortable`, no `searchable`, no key.
47pub struct RelationColumn<R> {
48    label: String,
49    display: Box<dyn Fn(&R) -> String + Send + Sync>,
50    /// The width this column claims in the table's fixed layout.
51    width: ColumnWidth,
52}
53
54impl<R> RelationColumn<R> {
55    /// Declare a column by label and projection.
56    ///
57    /// Named for the shape it is, as
58    /// [`TextColumn::computed`](crate::resource::TextColumn::computed) is on the list's side: a
59    /// relation column has no lens to bind, because the related rows arrive as values rather
60    /// than as a query. What a column *is* is its projection.
61    pub fn computed(
62        label: impl Into<String>,
63        display: impl Fn(&R) -> String + Send + Sync + 'static,
64    ) -> Self {
65        Self {
66            label: label.into(),
67            display: Box::new(display),
68            width: ColumnWidth::Wide,
69        }
70    }
71
72    /// Declare this column's width in the table's fixed layout:
73    /// `.width(ColumnWidth::Percent(20))` for a column that knows its own
74    /// measure.
75    ///
76    /// The default is [`Wide`](ColumnWidth::Wide): a relation column holds a
77    /// related row's own text — a comment body, a name — so it takes a share
78    /// of what the declared columns leave. Override it when the content
79    /// disagrees: a status, date, or count is
80    /// [`Narrow`](ColumnWidth::Narrow).
81    pub fn width(mut self, width: ColumnWidth) -> Self {
82        self.width = width;
83        self
84    }
85
86    /// The width this column declares, which the renderer emits on its `th`
87    /// and on every `td` of its column.
88    pub fn column_width(&self) -> ColumnWidth {
89        self.width
90    }
91
92    /// The column's heading.
93    pub fn label(&self) -> &str {
94        &self.label
95    }
96
97    /// This column's cell for `row`.
98    pub fn render_cell(&self, row: &R) -> String {
99        (self.display)(row)
100    }
101}
102
103impl<R> std::fmt::Debug for RelationColumn<R> {
104    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
105        f.debug_struct("RelationColumn")
106            .field("label", &self.label)
107            .field("width", &self.width)
108            .finish_non_exhaustive()
109    }
110}

112/// The columns of a relation's table, declared as a tuple or a single column.
113///
114/// A separate collection from the list's `IntoColumns` because the two tables
115/// answer different questions: one is queryable, this one is not.
116pub struct RelationColumns<R> {
117    pub(crate) columns: Vec<RelationColumn<R>>,
118}
119
120impl<R> RelationColumns<R> {
121    /// Collect one or more columns.
122    pub fn columns(columns: impl IntoRelationColumns<R>) -> Self {
123        columns.into_relation_columns()
124    }
125}

148/// The width every column of one relation render declares: one `style` value
149/// per declared column, in column order, plus the table-level `min-width`.
150/// The arithmetic is the list table's, without its chrome columns: kind
151/// defaults share one budget, explicit widths are verbatim, and a wide column
152/// takes what the declared ones leave.
153fn relation_widths<R>(
154    columns: &[RelationColumn<R>],
155) -> (Vec<Option<Cow<'static, str>>>, Option<Cow<'static, str>>) {
156    let total: u32 = columns
157        .iter()
158        .filter_map(|col| col.column_width().default_percent())
159        .map(u32::from)
160        .sum();
161    let cells = columns
162        .iter()
163        .map(|col| column_width_style(col.column_width(), total))
164        .collect();
165    let mut min_width = MinWidth::default();
166    for col in columns {
167        min_width.column(col.column_width(), total);
168    }
169    (cells, min_width.style())
170}

172/// Render `rows` as a titled, read-only table.
173///
174/// `R` is the related resource. Its [`can_view`](Resource::can_view) is applied
175/// to every row before that row is projected, so a relation declared through the
176/// framework cannot render a row the reader may not see; naming the resource at
177/// the call (`render_relation::<CommentResource>(cx, …)`) is what makes the
178/// policy the resource's, rather than a filter the caller writes and a later
179/// edit drops. Only the per-row predicate runs: `can_view_any` gates the related
180/// resource's own list page, which this table does not render.
181///
182/// `rows` is the related records the record already carries — the caller passes
183/// `record.comments.get().iter().cloned().collect()` or the equivalent — so this
184/// runs no query, and `can_view` is an in-memory predicate over those rows for
185/// the same reason. At most [`MAX_RELATION_ROWS`] rows render, with a line
186/// naming the truncation below the table. An empty set renders the title with an
187/// honest "none" line rather than an empty table, which would read as a failure
188/// to load.
189///
190/// The rows are rendered in the order given; a detail page shows what the query
191/// loaded, and `Resource::query` owns that order.
192pub fn render_relation<'a, R: Resource>(
193    cx: &'a Cx,
194    title: &str,
195    columns: RelationColumns<R::Model>,
196    rows: &[R::Model],
197) -> BoxView<'a> {
198    let title = title.to_string();
199    // Own everything before the `view!` block: the emitted view must borrow
200    // the request context and nothing else, or the caller's `record` (which
201    // owns these rows) would have to outlive the page.
202    let heads: Vec<String> = columns
203        .columns
204        .iter()
205        .map(|c| c.label().to_string())
206        .collect();
207    // Policy first, then the cap: a row the reader may not see is not a row
208    // this table shows, and it must not consume a slot a visible row needs.
209    let visible: Vec<&R::Model> = rows.iter().filter(|row| R::can_view(cx, row)).collect();
210    let total = visible.len();
211    let cells: Vec<Vec<String>> = visible
212        .iter()
213        .take(MAX_RELATION_ROWS)
214        .map(|row| {
215            columns
216                .columns
217                .iter()
218                .map(|column| column.render_cell(row))
219                .collect()
220        })
221        .collect();
222    let has_rows = !cells.is_empty();
223    let overflow = (total > MAX_RELATION_ROWS).then(|| {
224        format!("Showing the first {MAX_RELATION_ROWS} of {total} related rows you can view.")
225    });
226    // The declared widths are a property of the columns, not of the row, so
227    // they are resolved once here: the same CSS for every row.
228    let (cell_widths, table_min_width) = relation_widths(&columns.columns);
229    // Row ids are positional: this table does not reorder or swap, so it needs
230    // no record key — the list's table-key contract exists for keyed diffs
231    // and action URLs, and neither exists here.
232    view! {
233        cx =>
234        <section class="flex flex-col gap-3">
235            <h2 class="text-base font-semibold">(title)</h2>
236            if !has_rows {
237                <p class="text-sm text-muted-foreground">"None."</p>
238            } else {
239                table(
240                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
241                    table_header(
242                        table_row(
243                            for (head, width) in heads.iter().zip(&cell_widths) {
244                                table_head(
245                                    attrs: attributes! { style=(width.as_deref()) },
246                                    (head.clone())
247                                )
248                            }
249                        )
250                    )
251                    table_body(
252                        for row in &cells {
253                            table_row(
254                                for (cell, width) in row.iter().zip(&cell_widths) {
255                                    table_cell(
256                                        attrs: attributes! { class="truncate" style=(width.as_deref()) },
257                                        (cell.clone())
258                                    )
259                                }
260                            )
261                        }
262                    )
263                )
264                if let Some(overflow) = overflow {
265                    <p class="text-sm text-muted-foreground">(overflow)</p>
266                }
267            }
268        </section>
269    }
270    .boxed()
271}
