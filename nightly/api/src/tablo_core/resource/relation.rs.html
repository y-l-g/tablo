1//! Relation tables: a record page's related rows, through the related
2//! resource's own list table.
3//!
4//! A [`Relation`] names a child [`Resource`] and the key that ties its rows to
5//! the owner record. The panel renders it on the owner's detail and edit pages
6//! as the child's list table — its columns, search, sort, filters, pager and
7//! row actions — with the rows narrowed to the owner. The same binding seeds
8//! the child's create form, so a related record is created from the page that
9//! shows it.
10
11use std::sync::Arc;
12
13use toasty::stmt::{Expr, IntoExpr};
14use topcoat::{context::Cx, view::BoxView};
15
16use super::Resource;
17use crate::{
18    form::FormScalar,
19    schema::{FieldLens, LensBinding},
20};
21
22/// One relation of a parent resource's records: the child resource whose rows
23/// belong to a record, and the key that says which.
24///
25/// Declared by [`Resource::relations`]:
26///
27/// ```ignore
28/// fn relations() -> Vec<Relation<Post>> {
29///     vec![Relation::has_many::<CommentResource, _>(
30///         Comment::fields().post_id(),
31///         |post: &Post| post.id,
32///     )]
33/// }
34/// ```
35///
36/// Every row goes through the child's own tenant-scoped query and policies —
37/// `ViewAny` decides whether the table renders at all, and the row
38/// actions are gated per row as on its list. Writes started from the table
39/// return to the page that shows it.
40pub struct Relation<P> {
41    key: String,
42    label: String,
43    foreign_key: String,
44    bind: BindFn<P>,
45    render: RenderFn,
46    search: crate::panel::RelationSearchFn,
47    /// Why `foreign_key` binds no column, when it does not.
48    misdeclared: Option<String>,
49}
50
51/// An owner's rows filter on the child, and the owner's key as the child's
52/// form value.
53type BindFn<P> = Arc<dyn Fn(&P) -> (Expr<bool>, String) + Send + Sync>;
54
55/// The monomorphized renderer of one child resource, over what a relation
56/// resolved against its owner.
57pub(crate) type RenderFn = for<'a> fn(&'a Cx, BoundRelation) -> BoxView<'a>;
58
59/// A relation resolved against one owner record, for the page at `page`.
60pub(crate) struct BoundRelation {
61    /// The slug of the resource that owns the record.
62    pub(crate) parent: String,
63    /// The prefix of the table's URL parameters.
64    pub(crate) key: String,
65    /// The section title.
66    pub(crate) label: String,
67    /// The child's rows that belong to the owner.
68    pub(crate) scope: Expr<bool>,
69    /// The child's form key for the owner, and the owner's value for it.
70    pub(crate) seed: (String, String),
71    /// The path of the page that renders the table.
72    pub(crate) page: String,
73    /// Whether the page only shows the rows — the detail page — so the table
74    /// carries no write action and no create link.
75    pub(crate) read_only: bool,
76}
77
78impl<P> Relation<P> {
79    /// The child resource `C`, whose `foreign_key` column holds the owner's
80    /// `owner_key` — a post's comments:
81    ///
82    /// ```ignore
83    /// Relation::has_many::<CommentResource, _>(Comment::fields().post_id(), |post: &Post| post.id)
84    /// ```
85    ///
86    /// A nullable foreign key binds the same way, with the owner's key in
87    /// `Some`. The key's form spelling ([`FormScalar::to_form`]) seeds `C`'s
88    /// create form.
89    ///
90    /// The section is titled with `C`'s navigation label, and its table's URL
91    /// parameters are prefixed with `C`'s slug (`comments.q=`).
92    ///
93    /// A `foreign_key` that is not a single column of `C`'s model is a
94    /// misdeclaration [`RouterBuilderPanelExt::panel`](crate::RouterBuilderPanelExt::panel)
95    /// reports, like every lens a declaration binds.
96    pub fn has_many<C, T>(
97        foreign_key: FieldLens<C::Model, T>,
98        owner_key: impl Fn(&P) -> T + Send + Sync + 'static,
99    ) -> Self
100    where
101        C: Resource,
102        T: IntoExpr<T> + FormScalar + Send + Sync + 'static,
103    {
104        let binding = LensBinding::of(foreign_key.clone());
105        let search = crate::panel::relation_search_handler_for::<C, T>(foreign_key.clone());
106        Self {
107            key: C::slug(),
108            label: C::navigation_label(),
109            foreign_key: binding.name,
110            misdeclared: binding.misdeclared,
111            bind: Arc::new(move |owner| {
112                let value = owner_key(owner);
113                let seed = value.to_form();
114                (foreign_key.clone().eq(value), seed)
115            }),
116            render: crate::panel::relation_table::<C>,
117            search,
118        }
119    }
120
121    /// Title the section `label` instead of the child's navigation label.
122    pub fn label(mut self, label: impl Into<String>) -> Self {
123        self.label = label.into();
124        self
125    }
126
127    /// Why the relation's foreign key binds no column, when it does not.
128    pub(crate) fn misdeclared(&self) -> Option<&str> {
129        self.misdeclared.as_deref()
130    }
131
132    /// The prefix of this relation's URL parameters: the child's slug.
133    pub(crate) fn key(&self) -> &str {
134        &self.key
135    }
136
137    /// The live-search loader of this relation's table, for the panel's
138    /// relation registry.
139    pub(crate) fn search_handler(&self) -> crate::panel::RelationSearchFn {
140        self.search.clone()
141    }
142
143    /// Render this relation's table for `owner` on the page at `page`,
144    /// without write actions when `read_only`. `parent` is the slug of the
145    /// resource that owns the record, keying the relation's live-search
146    /// handler alongside `key`.
147    pub(crate) fn render<'a>(
148        &self,
149        cx: &'a Cx,
150        owner: &P,
151        page: &str,
152        read_only: bool,
153        parent: &str,
154    ) -> BoxView<'a> {
155        let (scope, value) = (self.bind)(owner);
156        (self.render)(
157            cx,
158            BoundRelation {
159                parent: parent.to_string(),
160                key: self.key.clone(),
161                label: self.label.clone(),
162                scope,
163                seed: (self.foreign_key.clone(), value),
164                page: page.to_string(),
165                read_only,
166            },
167        )
168    }
169}
170
171impl<P> std::fmt::Debug for Relation<P> {
172    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
173        f.debug_struct("Relation")
174            .field("key", &self.key)
175            .field("label", &self.label)
176            .field("foreign_key", &self.foreign_key)
177            .finish_non_exhaustive()
178    }
