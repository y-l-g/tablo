7/// The `href` and runtime-navigation attributes of a link to another panel
8/// page (GH #395).
9///
10/// The runtime swaps the page in without a document load. The mode is the
11/// context's [`prefetch_mode`](topcoat::runtime::prefetch_mode), which the
12/// panel's router sets to never.
13pub(crate) fn runtime_link(cx: &Cx, href: &str) -> Attributes {
14    topcoat::runtime::link_attrs(cx, href.to_string(), topcoat::runtime::prefetch_mode(cx))
15}

17/// Where a sidebar entry points.
18///
19/// A [`Resource`] cannot name its own URL: [`Resource::navigation`] takes no
20/// `Cx` and no prefix, so the entry it declares by default carries no URL at
21/// all — [`NavTarget::Derived`] — and the [`Panel`](crate::panel::Panel) that
22/// owns the item resolves it from its own mount prefix plus the resource's
23/// [`slug`](Resource::slug). [`NavTarget::Url`] is a URL its author wrote out,
24/// and a Panel passes it through untouched.
25#[derive(Clone, Default, PartialEq, Eq)]
26pub enum NavTarget {
27    /// No URL yet: the owning Panel resolves it to `{prefix}/{slug}` of the
28    /// resource whose `navigation()` declared this item. What
29    /// [`NavigationItem::for_resource`] — and so the default
30    /// [`Resource::navigation`] — returns.
31    #[default]
32    Derived,
33    /// An explicit URL: a custom path, a query view, another panel's mount.
34    /// Active state is string matching (exact, or a slash-boundary prefix).
35    Url(String),
36}
37
38impl NavTarget {
39    /// The URL this target names, or `None` while it is still
40    /// [`Self::Derived`] — i.e. before the owning Panel has resolved it.
41    pub fn url(&self) -> Option<&str> {
42        match self {
43            Self::Derived => None,
44            Self::Url(url) => Some(url),
45        }
46    }
47}
48
49impl std::fmt::Debug for NavTarget {
50    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
51        match self {
52            Self::Derived => f.write_str("Derived"),
53            Self::Url(url) => f.debug_tuple("Url").field(url).finish(),
54        }
55    }
56}

58/// Sidebar entry derived from a `Resource` (see `CONTEXT.md`).
59#[derive(Clone, Debug, Default, PartialEq, Eq)]
60pub struct NavigationItem {
61    pub label: String,
62    /// Where this entry points. [`NavTarget::Derived`] until the owning Panel
63    /// resolves it — see [`NavTarget`]. Build items with
64    /// [`NavigationItem::for_resource`] or [`NavigationItem::at`] rather than
65    /// spelling the variant out.
66    pub target: NavTarget,
67    /// Sort key for the sidebar: items render in stable `order`
68    /// order, so declaration order breaks ties. Resources declare in
69    /// `Panel::resource` order (all default `0`); a
70    /// [`Resource::navigation`] override interleaves by setting a lower value,
71    /// e.g. `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`
72    /// pins above the resources.
73    pub order: i32,
74}
75
76impl NavigationItem {
77    /// The default sidebar entry for `R`: its
78    /// [`navigation_label`](Resource::navigation_label), and no URL yet.
79    ///
80    /// This is what [`Resource::navigation`] returns unless overridden — and
81    /// what an override decorates, e.g.
82    /// `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`.
83    /// The resource cannot know where it is mounted, so the URL stays
84    /// [`NavTarget::Derived`] until the owning [`Panel`](crate::panel::Panel)
85    /// resolves it; an entry declared here can therefore never link at a mount
86    /// the resource guessed.
87    pub fn for_resource<R: Resource>() -> Self {
88        Self {
89            label: R::navigation_label(),
90            target: NavTarget::Derived,
91            order: 0,
92        }
93    }
94
95    /// A sidebar entry at an explicit `url`.
96    ///
97    /// Active state is string matching: exact path, or a slash-boundary prefix,
98    /// so `/admin/users` is current on `/admin/users/create`.
99    pub fn at(label: impl Into<String>, url: impl Into<String>) -> Self {
100        Self {
101            label: label.into(),
102            target: NavTarget::Url(url.into()),
103            order: 0,
104        }
105    }
106
107    /// Resolve a [`NavTarget::Derived`] entry against the Panel that owns it
108    /// leaving an explicit target untouched.
109    ///
110    /// [`Resource::navigation`] cannot know its panel — it takes no `Cx` and no
111    /// prefix — so the entry it declares carries no URL. The Panel consumes it
112    /// through `Panel::resource`, which passes its own prefix and the
113    /// resource's mount slug; there is no panel-root case, since nothing
114    /// generates an entry without a resource behind it.
115    ///
116    /// There is no guessing here: a URL an author wrote out — including one
117    /// that happens to look like `/admin/{slug}` — is a different
118    /// [`NavTarget`] variant and is never rewritten.
119    pub(crate) fn resolved(mut self, prefix: &str, slug: &str) -> Self {
120        if matches!(self.target, NavTarget::Derived) {
121            self.target = NavTarget::Url(format!("{}/{slug}", mount(prefix)));
122        }
123        self
124    }
125
126    /// The URL this entry points at, or `None` while it is unresolved — i.e.
127    /// still [`NavTarget::Derived`], not yet handed to a Panel. Panel-owned
128    /// items are always resolved (`Panel::resource`).
129    pub fn url(&self) -> Option<&str> {
130        self.target.url()
131    }
132
133    /// Whether this item is current for the given request path (without query):
134    /// an exact match, or a prefix match on a slash boundary (so
135    /// `/admin/users` is active on `/admin/users/create` but not on
136    /// `/admin/userships`). Uniform for every item — since resources mount at
137    /// `{prefix}/{slug}`, no generated item points at the bare panel
138    /// prefix.
139    ///
140    /// This is the whole active-state contract: `Panel::render_shell` takes the
141    /// request path as a parameter, so it can judge an item without a `Cx` (and
142    /// the shell stays testable without a full `http::request::Parts` in `Cx`).
143    /// It is the only entry point to that judgment.
144    pub fn is_current_path(&self, current_path: &str) -> bool {
145        let Some(url) = self.url() else {
146            // Unresolved: no URL to be current for.
147            return false;
148        };
149        if current_path == url {
150            return true;
151        }
152        current_path
153            .strip_prefix(url)
154            .is_some_and(|rest| rest.starts_with('/'))
155    }
156}
157
158/// The mount a panel prefix normalises to: `/admin` when it is empty — the same
159/// rule [`Panel::new`](crate::panel::Panel::new) applies.
160fn mount(prefix: &str) -> String {
161    let trimmed = prefix.trim_matches('/').trim();
162    if trimmed.is_empty() {
163        "/admin".to_string()
164    } else {
165        format!("/{trimmed}")
166    }
167}
