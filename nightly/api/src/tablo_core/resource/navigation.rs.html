7/// The `href` and runtime-navigation attributes of a link to another panel
8/// page (GH #395).
9///
10/// The runtime swaps the page in without a document load. The mode is the
11/// context's [`prefetch_mode`](topcoat::runtime::prefetch_mode), which the
12/// panel's router sets to never.
13pub(crate) fn runtime_link(cx: &Cx, href: &str) -> Attributes {
14    topcoat::runtime::link_attrs(cx, href.to_string(), topcoat::runtime::prefetch_mode(cx))
15}

17/// Where a sidebar entry points.
18///
19/// A [`Resource`] or a [`Page`](crate::Page) cannot name its own URL: its
20/// `navigation()` takes no `Cx` and no prefix, so the entry it declares by
21/// default carries no URL at all — [`NavTarget::Derived`] — and the
22/// [`Panel`](crate::panel::Panel) that owns the item resolves it to the URL it
23/// mounts the resource or page at. [`NavTarget::Url`] is a URL its author wrote
24/// out, and a Panel passes it through untouched.
25#[derive(Clone, Default, PartialEq, Eq)]
26pub enum NavTarget {
27    /// No URL yet: the owning Panel resolves it to `{prefix}/{slug}` of the
28    /// resource or page whose `navigation()` declared this item (the prefix
29    /// itself for the home page). What [`NavigationItem::for_resource`] and
30    /// [`NavigationItem::for_page`] — and so the default `navigation()` —
31    /// return.
32    #[default]
33    Derived,
34    /// An explicit URL: a custom path, a query view, another panel's mount.
35    /// Active state is string matching (exact, or a slash-boundary prefix).
36    Url(String),
37}
38
39impl NavTarget {
40    /// The URL this target names, or `None` while it is still
41    /// [`Self::Derived`] — i.e. before the owning Panel has resolved it.
42    pub fn url(&self) -> Option<&str> {
43        match self {
44            Self::Derived => None,
45            Self::Url(url) => Some(url),
46        }
47    }
48}
49
50impl std::fmt::Debug for NavTarget {
51    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
52        match self {
53            Self::Derived => f.write_str("Derived"),
54            Self::Url(url) => f.debug_tuple("Url").field(url).finish(),
55        }
56    }
57}

59/// Sidebar entry derived from a `Resource` or a `Page` (see `CONTEXT.md`).
60#[derive(Clone, Debug, Default)]
61pub struct NavigationItem {
62    pub label: String,
63    /// Where this entry points. [`NavTarget::Derived`] until the owning Panel
64    /// resolves it — see [`NavTarget`]. Build items with
65    /// [`NavigationItem::for_resource`], [`NavigationItem::for_page`] or
66    /// [`NavigationItem::at`] rather than spelling the variant out.
67    pub target: NavTarget,
68    /// Sort key for the sidebar: items render in stable `order`
69    /// order, so declaration order breaks ties. Resources and pages declare in
70    /// registration order (all default `0`); a
71    /// [`Resource::navigation`] override interleaves by setting a lower value,
72    /// e.g. `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`
73    /// pins above the resources.
74    pub order: i32,
75    /// The icon before the label, if any; set it with [`Self::icon`].
76    /// `tablo_ui::icons` carries a set of navigation icons an app can use
77    /// without staging an icon set of its own.
78    pub icon: Option<IconData>,
79}
80
81impl NavigationItem {
82    /// The default sidebar entry for `R`: its
83    /// [`navigation_label`](Resource::navigation_label), and no URL yet.
84    ///
85    /// This is what [`Resource::navigation`] returns unless overridden — and
86    /// what an override decorates, e.g.
87    /// `NavigationItem { order: -1, ..NavigationItem::for_resource::<Self>() }`.
88    /// The resource cannot know where it is mounted, so the URL stays
89    /// [`NavTarget::Derived`] until the owning [`Panel`](crate::panel::Panel)
90    /// resolves it; an entry declared here can therefore never link at a mount
91    /// the resource guessed.
92    pub fn for_resource<R: Resource>() -> Self {
93        Self {
94            label: R::navigation_label(),
95            target: NavTarget::Derived,
96            order: 0,
97            icon: None,
98        }
99    }
100
101    /// The default sidebar entry for the [`Page`](crate::Page) `P`: its
102    /// [`navigation_label`](crate::Page::navigation_label), and no URL yet —
103    /// the owning [`Panel`](crate::panel::Panel) resolves it where it mounts
104    /// the page.
105    pub fn for_page<P: crate::Page>() -> Self {
106        Self {
107            label: P::navigation_label(),
108            target: NavTarget::Derived,
109            order: 0,
110            icon: None,
111        }
112    }
113
114    /// A sidebar entry at an explicit `url`.
115    ///
116    /// Active state is string matching: exact path, or a slash-boundary prefix,
117    /// so `/admin/users` is current on `/admin/users/create`.
118    pub fn at(label: impl Into<String>, url: impl Into<String>) -> Self {
119        Self {
120            label: label.into(),
121            target: NavTarget::Url(url.into()),
122            order: 0,
123            icon: None,
124        }
125    }
126
127    /// This entry with `icon` before its label, e.g.
128    /// `NavigationItem::for_resource::<Self>().icon(tablo_ui::icons::USERS)`
129    /// in a [`Resource::navigation`] override.
130    #[must_use]
131    pub fn icon(mut self, icon: IconData) -> Self {
132        self.icon = Some(icon);
133        self
134    }
135
136    /// Resolve a [`NavTarget::Derived`] entry to `url`, the route its Panel
137    /// mounts the resource or page at, leaving an explicit target untouched.
138    ///
139    /// [`Resource::navigation`] and [`Page::navigation`](crate::Page::navigation)
140    /// cannot know their panel — they take no `Cx` and no prefix — so the entry
141    /// they declare carries no URL. The Panel passes the same URL it registers
142    /// the route at, so the sidebar and the router cannot disagree.
143    ///
144    /// There is no guessing here: a URL an author wrote out — including one
145    /// that happens to look like `/admin/{slug}` — is a different
146    /// [`NavTarget`] variant and is never rewritten.
147    pub(crate) fn resolved(mut self, url: &str) -> Self {
148        if matches!(self.target, NavTarget::Derived) {
149            self.target = NavTarget::Url(url.to_string());
150        }
151        self
152    }
153
154    /// The URL this entry points at, or `None` while it is unresolved — i.e.
155    /// still [`NavTarget::Derived`], not yet handed to a Panel. Panel-owned
156    /// items are always resolved (`Panel::resource`, `Panel::page`,
157    /// `Panel::home`).
158    pub fn url(&self) -> Option<&str> {
159        self.target.url()
160    }
161
162    /// Whether this item is current for the given request path (without query):
163    /// an exact match, or a prefix match on a slash boundary (so
164    /// `/admin/users` is active on `/admin/users/create` but not on
165    /// `/admin/userships`).
166    ///
167    /// Several items can match one path — a home entry at the bare prefix
168    /// matches every page under it — so the sidebar marks one item active: the
169    /// first, in sidebar order, of the matches with the longest URL. `Panel::render_shell` takes
170    /// the request path as a parameter, so it can judge an item without a `Cx` (and the
171    /// shell stays testable without a full `http::request::Parts` in `Cx`).
172    pub fn is_current_path(&self, current_path: &str) -> bool {
173        let Some(url) = self.url() else {
174            // Unresolved: no URL to be current for.
175            return false;
176        };
177        if current_path == url {
178            return true;
179        }
180        current_path
181            .strip_prefix(url)
182            .is_some_and(|rest| rest.starts_with('/'))
183    }
184}
