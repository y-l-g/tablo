63/// A table link's attributes: `href` always, and on a live table (`signals`) a
64/// click that writes the link's own query to the `query` signal instead of
65/// navigating, so the shard re-renders the table in place. `href` stays the
66/// no-JS fallback and spells the same state.
67pub(super) fn live_link(cx: &Cx, url: String, signals: Option<&TableSignals>) -> Attributes {
68    match signals {
69        Some(signals) => {
70            let query = signals.query.clone();
71            let next = query_of(&url).to_string();
72            attributes! {
73                cx =>
74                href=(url)
75                @click=$(|e: Event| {
76                    e.prevent_default();
77                    query.set(next.clone());
78                })
79            }
80        }
81        None => attributes! { cx => href=(url) },
82    }
83}
