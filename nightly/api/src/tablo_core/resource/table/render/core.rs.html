18impl<M> Table<M> {
19    /// Render the table for the given loaded page.
20    ///
21    /// Real chrome, no fake affordances: the header renders sort **links**
22    /// driving `?sort=`/`?dir=` and a search toolbar driving `?q=` (shown by
23    /// default when any column is `searchable()`), rows are keyed by the
24    /// projection declared via [`Table::new`](super::super::Table::new) per
25    /// `CONTEXT.md` and rendered via
26    /// each column's typed projection, pagination shows Previous/Next links
27    /// built from the executed page's **real** cursors (never invented page
28    /// numbers), and the empty state reflects whether a search was active.
29    ///
30    /// Composes the synced `tablo-ui` primitives and Token classes
31    /// (`border-border` on the chrome, `bg-background`/`shadow-xs` on the
32    /// toolbar controls, `text-muted-foreground` on the captions) — no raw
33    /// colors, no `ac-*`.
34    pub async fn render<'a>(&self, cx: &'a Cx, page: TablePage<M>) -> Result<BoxView<'a>>
35    where
36        M: toasty::schema::Model + Send + Sync + 'static,
37    {
38        let state = TableState::from_cx(cx);
39        let path = topcoat::context::try_request_context::<http::request::Parts>(cx)
40            .map(|parts| parts.uri.path().to_string())
41            .unwrap_or_default();
42        self.render_with_state(cx, page, &state, &path).await
43    }
44
45    /// Render with explicit list state and path instead of reading them from
46    /// `cx`: a page that owns its table passes the state it parsed and the
47    /// list URL its links should target.
48    ///
49    /// Normalizes the state it is handed, so an unknown `?group_by=` never
50    /// echoes through a link.
51    pub async fn render_with_state<'a>(
52        &self,
53        cx: &'a Cx,
54        page: TablePage<M>,
55        state: &TableState,
56        path: &str,
57    ) -> Result<BoxView<'a>>
58    where
59        M: toasty::schema::Model + Send + Sync + 'static,
60    {
61        self.render_inner(cx, page, &self.normalize_state(state), path, None)
62            .await
63    }
64
65    /// Render the interactive body for a live table: the same
66    /// presentation as [`Self::render_with_state`], with the sort links, the
67    /// pager, the filter bar, and the empty-state clear links writing the
68    /// `query` signal — each interaction writes it and the browser morphs the
69    /// shard's new output in place, without a navigation or a scroll jump.
70    /// Every bound control keeps its real `href`/form, so a page without JS
71    /// still navigates as before.
72    ///
73    /// The row-delete dialog is not part of this output: it lives
74    /// outside the region a rerun swaps, rendered once by the page that owns
75    /// the signals, so a caller rendering only through this method renders
76    /// [`Self::render_delete_dialog`] itself to keep the `?delete=` fallback.
77    pub(crate) async fn render_live<'a>(
78        &self,
79        cx: &'a Cx,
80        page: TablePage<M>,
81        state: &TableState,
82        path: &str,
83        signals: TableSignals,
84    ) -> Result<BoxView<'a>>
85    where
86        M: toasty::schema::Model + Send + Sync + 'static,
87    {
88        self.render_inner(cx, page, state, path, Some(signals))
89            .await
90    }
91
92    async fn render_inner<'a>(
93        &self,
94        cx: &'a Cx,
95        page: TablePage<M>,
96        state: &TableState,
97        path: &str,
98        signals: Option<TableSignals>,
99    ) -> Result<BoxView<'a>>
100    where
101        M: toasty::schema::Model + Send + Sync + 'static,
102    {
103        let row_key = self.row_key.clone();
104        let delete_prefix = self.delete_prefix.clone();
105        let with_actions = self.with_actions();
106        let with_bulk = self.bulk_enabled();
107        let record_key = self.record_key.clone();
108        let head = self
109            .render_thead(cx, state, path, with_actions, with_bulk, signals.as_ref())
110            .await?;
111        let show_search = self.search_enabled();
112        let search_bar = if show_search {
113            Some(self.render_search_bar(cx, state, path).await?)
114        } else {
115            None
116        };
117        let show_filters = self.filter_bar_enabled();
118        let filter_bar = if show_filters {
119            Some(
120                self.render_filter_bar(cx, state, path, signals.as_ref())
121                    .await?,
122            )
123        } else {
124            None
125        };
126        let bulk_bar_view = self.render_bulk_bar(cx, signals.as_ref());
127        let pager = self
128            .render_pager(cx, state, path, &page, signals.as_ref())
129            .await?;
130        let filter_warning = self.render_filter_warning(cx, state, path);
131        // The declared grouping, when `?group_by=` names it. Read
132        // before the row projection so each row can carry its group label,
133        // which the page-local shim orders by.
134        let group_key = self.effective_group_key(state);
135        let row_data = self.row_views(
136            state,
137            path,
138            &page,
139            &row_key,
140            &record_key,
141            group_key.as_ref(),
142        );
143        // The confirmation dialog lives with the delete chrome and
144        // ships closed, so a row control opens it in place. A live
145        // output carries none: the page that owns the signals renders it once,
146        // outside the region a rerun swaps (`panel::resource_list_live`). The
147        // dialog renders on every page with delete chrome, not only under
148        // `?delete=`, so without this gate every shard rerun would morph a
149        // second copy — and its ids — into the page.
150        let delete_dialog = if signals.is_none() {
151            self.render_delete_dialog(cx, state).await?
152        } else {
153            None
154        };
155
156        // Body-only branch: the empty and rows pages share the one
157        // chrome wrapper built below — only the table body differs. Group
158        // headers and the pager exist solely on rows pages: an empty page
159        // renders the honest empty cell instead (its pager would be empty
160        // anyway, and grouping an empty page yields no headers).
161        //
162        // The declared widths are a property of the columns, not of the row,
163        // so they are resolved once here: the same CSS for every row, and a
164        // `for` whose expression names `self.columns` would carry the table's
165        // borrow into the view.
166        let ColumnWidths {
167            cells: cell_widths,
168            actions_min,
169            table_min_width,
170            ..
171        } = self.column_widths();
172        let mut pager_views: Vec<BoxView<'_>> = Vec::new();
173        let body: BoxView<'_> = if page.rows.is_empty() {
174            let empty_cell = self
175                .render_empty_cell(cx, state, path, with_actions, with_bulk, signals.as_ref())
176                .await?;
177            view! {
178                cx =>
179                table(
180                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
181                    (head)
182                    (empty_cell)
183                )
184            }
185            .boxed()
186        } else {
187            pager_views = pager;
188            // The one dialog every row control on this table opens; empty
189            // without delete chrome, where no control renders one.
190            let chrome = RowChrome {
191                with_bulk,
192                with_actions,
193                header_colspan: self.columns.len()
194                    + usize::from(with_bulk)
195                    + usize::from(with_actions),
196                cell_widths,
197                actions_min,
198                delete_dialog_id: delete_prefix
199                    .as_deref()
200                    .map(Self::delete_dialog_dom_id)
201                    .unwrap_or_default(),
202            };
203            let rows = render_rows(cx, row_data, &chrome);
204            view! {
205                cx =>
206                // The table-level layout is a static class: Tailwind sees the
207                // literal, and the decision carries no per-column value. Each
208                // column's width, which does, rides the `th`/`td` inline
209                // `style`. Fixed layout is what stops a filter or a page change
210                // from re-measuring the columns. The `min-width` is the sum of
211                // those declared widths: with `w-full` the table never exceeds
212                // its container on its own, so without the floor the wrapper's
213                // `overflow-x-auto` never scrolls.
214                table(
215                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
216                    (head)
217                    table_body(
218                        #[key(row.key.as_str())]
219                        for row in rows {
220                            (row.view)
221                        }
222                    )
223                )
224            }
225            .boxed()
226        };
227
228        // The refresh control: a live table's region re-renders when
229        // a signal its shard tracked changes, and a mutation changes rows the
230        // tracked inputs do not describe — the query is the same, the data is
231        // not. One write that means "re-read the table" is therefore the
232        // mutation's only honest in-place effect: the client bumps this
233        // revision token and the shard re-runs the query, morphing and
234        // re-hydrating the region through the seam that already exists.
235        //
236        // The signal is declared here, inside the shard's own output, so it
237        // belongs to the shard's content scope: its id derives from the shard
238        // invocation's identity, this call site, and the list path, so it is
239        // stable across reruns and distinct per list (every resource's list
240        // runs the same shard), and the runtime keeps its value when the
241        // declaration renders again. Reading it for the input's initial
242        // value is what declares the dependency; the token itself is opaque.
243        //
244        // A static table renders no control at all: it has no shard to re-run,
245        // its region is inert markup, and the client replaces it wholesale
246        // So the control's presence *is* the page's answer to
247        // "can this table refresh in place?".
248        let revision_attrs = signals.as_ref().map(|_| {
249            let revision = topcoat::runtime::signal(&cx.keyed(path), || "0".to_string());
250            attributes! {
251                cx =>
252                type="hidden"
253                value=(revision.get())
254                :value=$(revision.get())
255                @change=$(|e: Event| revision.set(e.target.value))
256                data-table-revision=""
257            }
258        });
259
260        // One chrome for both branches: search bar, filter bar, bulk bar,
261        // warning, table body, pager, dialog, inside the frame the morph swaps.
262        let content = view! {
263            cx =>
264            if show_search {
265                (search_bar.expect("search bar built when enabled"))
266            }
267            if show_filters {
268                (filter_bar.expect("filter bar built when enabled"))
269            }
270            (bulk_bar_view)
271            if let Some(attrs) = revision_attrs {
272                <input (attrs)>
273            }
274            if let Some(warning) = filter_warning {
275                (warning)
276            }
277            (body)
278            for p in pager_views {
279                (p)
280            }
281            if let Some(dialog) = delete_dialog {
282                (dialog)
283            }
284        };
285        Ok(table_frame(cx, false, content.boxed()))
286    }
287}

289/// The table's two wrappers, shared by the loaded table and its skeleton so
290/// the swap lands on the same shape: the `data-boundary` region the morph
291/// swaps, and the bordered root inside it. A loading skeleton marks both
292/// `aria-busy`: the boundary so assistive tech sees the live region, the root
293/// so the busy state reads on the table itself.
294pub(super) fn table_frame<'a>(cx: &'a Cx, busy: bool, content: BoxView<'a>) -> BoxView<'a> {
295    let busy = busy.then_some("true");
296    view! {
297        cx =>
298        <div data-boundary="table" aria-busy=(busy)>
299            <div
300                class="rounded-xl border border-border overflow-hidden"
301                data-table-root=""
302                aria-busy=(busy)
303            >
304                (content)
305            </div>
306        </div>
307    }
308    .boxed()
309}
