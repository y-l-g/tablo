23/// The readability floor one [`ColumnWidth::Wide`](super::super::ColumnWidth::Wide)
24/// column contributes to the table's `min-width`, in whole rem.
25///
26/// A wide column declares no width — it takes what the declared columns leave —
27/// so a sum of declared widths alone would let it crush to zero on a narrow
28/// viewport (the measured 38px cells at 480px). Six rem keeps body text
29/// readable and, summed across the wide columns, trips the wrapper's
30/// horizontal scroll before the fixed layout crushes them.
31const WIDE_COLUMN_MIN_REM: u8 = 6;
32
33/// The share of the table the bulk-selection column claims: one
34/// checkbox plus the cell's `p-3` padding at the widths a list is read at. A
35/// percentage, not a length: the column keeps its share as the table narrows,
36/// and the columns that declare none keep theirs.
37const BULK_COLUMN_PERCENT: u8 = 5;
38
39/// The most of the table the kind defaults claim together.
40///
41/// The defaults are shares of the table, and the columns that declare none
42/// take what they leave: a total over 100% gives those columns no space at
43/// all, and `table-fixed` renders a column with no space at zero width, header
44/// text included. The budget keeps the rest of the table for them whatever the
45/// column set.
46const DEFAULT_WIDTH_BUDGET_PERCENT: u8 = 60;
47
48/// The share a kind default claims, scaled down when the table's defaults
49/// together exceed [`DEFAULT_WIDTH_BUDGET_PERCENT`].
50fn scaled_default_percent(nominal: u8, total: u32) -> u8 {
51    if total <= u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) {
52        return nominal;
53    }
54    let scaled = u32::from(nominal) * u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) / total;
55    // `scaled` is at most the budget, so the conversion cannot fail.
56    u8::try_from(scaled).unwrap_or(DEFAULT_WIDTH_BUDGET_PERCENT)
57}
58
59/// The `style` value a kind default emits.
60fn default_width_style(percent: u8) -> Cow<'static, str> {
61    Cow::Owned(format!("width: {percent}%"))
62}

64/// The width every column of one render declares: one `style` value
65/// per declared column, in column order, plus the two chrome columns.
66/// `None` is a column that declares no width — a wide column, which takes a
67/// share of what the declared ones leave.
68///
69/// `actions_min` is the actions column's content floor for its body cells
70/// (the header carries the share *and* the floor); `table_min_width` is the
71/// table-level floor — the sum of the declared widths — that lets the
72/// wrapper's `overflow-x-auto` scroll on a narrow viewport instead of
73/// crushing the cells.
74pub(super) struct ColumnWidths {
75    cells: Vec<Option<Cow<'static, str>>>,
76    bulk: Option<Cow<'static, str>>,
77    actions: Option<Cow<'static, str>>,
78    actions_min: Option<Cow<'static, str>>,
79    pub(super) table_min_width: Option<Cow<'static, str>>,
80}

83    /// Render the table for the given loaded page.
84    ///
85    /// Real chrome, no fake affordances: the header renders sort **links**
86    /// driving `?sort=`/`?dir=` and a search toolbar driving `?q=` (shown by
87    /// default when any column is `searchable()`), rows are keyed by the
88    /// projection declared via [`Table::new`](super::super::Table::new) per
89    /// `CONTEXT.md` and rendered via
90    /// each column's typed projection, pagination shows Previous/Next links
91    /// built from the executed page's **real** cursors (never invented page
92    /// numbers), and the empty state reflects whether a search was active.
93    ///
94    /// Composes the synced `tablo-ui` primitives and Token classes
95    /// (`border-border` on the chrome, `bg-background`/`shadow-xs` on the
96    /// toolbar controls, `text-muted-foreground` on the captions) — no raw
97    /// colors, no `ac-*`.
98    ///
99    /// # Errors
100    ///
101    /// Errors when pagination declares `per_page = 0`.
102    pub async fn render<'a>(&self, cx: &'a Cx, page: TablePage<M>) -> Result<BoxView<'a>>
103    where
104        M: toasty::schema::Model + Send + Sync + 'static,
105    {
106        let state = TableState::from_cx(cx);
107        let path = topcoat::context::try_request_context::<http::request::Parts>(cx)
108            .map(|parts| parts.uri.path().to_string())
109            .unwrap_or_default();
110        self.render_with_state(cx, page, &state, &path).await
111    }
112
113    /// Render with explicit list state and path instead of reading them from
114    /// `cx` — the seam a live-search shard needs: shard requests hit
115    /// the `table_search` shard's own endpoint, so `TableState::from_cx` would
116    /// see the endpoint URI, not the list page's `?q=/filters/sort`. Callers
117    /// pass the page's state (or shard args rebuilt via
118    /// [`TableSignals::to_state`]) and the list URL explicitly.
119    ///
120    /// Normalizes the state it is handed, so a page calling this
121    /// directly needs no knowledge of `NormalizedState`; a caller that
122    /// already normalized once per request goes through
123    /// `Self::render_normalized` instead.
124    pub async fn render_with_state<'a>(
125        &self,
126        cx: &'a Cx,
127        page: TablePage<M>,
128        state: &TableState,
129        path: &str,
130    ) -> Result<BoxView<'a>>
131    where
132        M: toasty::schema::Model + Send + Sync + 'static,
133    {
134        self.render_normalized(cx, page, &self.normalize_state(state), path)
135            .await
136    }

138    /// [`Self::render_with_state`] with the state already normalized
139    /// the request entry normalizes once and every seam below takes
140    /// the proof, so a live list request never normalizes the same state
141    /// twice.
142    pub(crate) async fn render_normalized<'a>(
143        &self,
144        cx: &'a Cx,
145        page: TablePage<M>,
146        state: &NormalizedState,
147        path: &str,
148    ) -> Result<BoxView<'a>>
149    where
150        M: toasty::schema::Model + Send + Sync + 'static,
151    {
152        self.render_inner(cx, page, state, path, None).await
153    }

155    /// Render the interactive body for a live table: the same
156    /// presentation as [`Self::render_with_state`], with the sort links, the
157    /// pager, the filter transport, and the empty-state clear links bound to
158    /// `signals` — each interaction writes a signal and the browser morphs the
159    /// shard's new output in place, without a navigation or a scroll jump.
160    /// Every bound control keeps its real `href`/form, so a page without JS
161    /// still navigates as before.
162    ///
163    /// The row-delete dialog is not part of this output: it lives
164    /// outside the region a rerun swaps, rendered once by the page that owns
165    /// the signals, so a caller rendering only through this method renders
166    /// [`Self::render_delete_dialog_normalized`] itself to keep the `?delete=` fallback.
167    pub(crate) async fn render_live_normalized<'a>(
168        &self,
169        cx: &'a Cx,
170        page: TablePage<M>,
171        state: &NormalizedState,
172        path: &str,
173        signals: TableSignals,
174    ) -> Result<BoxView<'a>>
175    where
176        M: toasty::schema::Model + Send + Sync + 'static,
177    {
178        self.render_inner(cx, page, state, path, Some(signals))
179            .await
180    }

182    async fn render_inner<'a>(
183        &self,
184        cx: &'a Cx,
185        page: TablePage<M>,
186        state: &NormalizedState,
187        path: &str,
188        signals: Option<TableSignals>,
189    ) -> Result<BoxView<'a>>
190    where
191        M: toasty::schema::Model + Send + Sync + 'static,
192    {
193        let row_key = self.row_key.clone();
194        let delete_prefix = self.delete_prefix.clone();
195        let with_actions = self.with_actions();
196        let with_bulk = self.bulk_enabled();
197        let record_key = self.record_key.clone();
198        let head = self
199            .render_thead(cx, state, path, with_actions, with_bulk, signals.as_ref())
200            .await?;
201        let show_search = self.search_enabled();
202        let search_bar = if show_search {
203            Some(self.render_search_bar(cx, state, path).await?)
204        } else {
205            None
206        };
207        let show_filters = self.filter_bar_enabled();
208        let filter_bar = if show_filters {
209            Some(
210                self.render_filter_bar(cx, state, path, signals.as_ref())
211                    .await?,
212            )
213        } else {
214            None
215        };
216        let bulk_bar_view = self.render_bulk_bar(cx, signals.as_ref());
217        let pager = self
218            .render_pager(cx, state, path, &page, signals.as_ref())
219            .await?;
220        let filter_warning = self.render_filter_warning(cx, state, path);
221        // The declared grouping, when `?group_by=` names it. Read
222        // before the row projection so each row can carry its group label,
223        // which the page-local shim orders by.
224        let group_key = self.effective_group_key(state);
225        let row_data = self.row_views(
226            state,
227            path,
228            &page,
229            &row_key,
230            &record_key,
231            group_key.as_ref(),
232        );
233        // The confirmation dialog lives with the delete chrome and
234        // ships closed, so a row control opens it in place. A live
235        // output carries none: the page that owns the signals renders it once,
236        // outside the region a rerun swaps (`panel::resource_list_live`). The
237        // dialog renders on every page with delete chrome, not only under
238        // `?delete=`, so without this gate every shard rerun would morph a
239        // second copy — and its ids — into the page.
240        let delete_dialog = if signals.is_none() {
241            self.render_delete_dialog_normalized(cx, state).await?
242        } else {
243            None
244        };
245
246        // Body-only branch: the empty and rows pages share the one
247        // chrome wrapper built below — only the table body differs. Group
248        // headers and the pager exist solely on rows pages: an empty page
249        // renders the honest empty cell instead (its pager would be empty
250        // anyway, and grouping an empty page yields no headers).
251        //
252        // The declared widths are a property of the columns, not of the row,
253        // so they are resolved once here: the same CSS for every row, and a
254        // `for` whose expression names `self.columns` would carry the table's
255        // borrow into the view.
256        let ColumnWidths {
257            cells: cell_widths,
258            actions_min,
259            table_min_width,
260            ..
261        } = self.column_widths();
262        let mut pager_views: Vec<BoxView<'_>> = Vec::new();
263        let body: BoxView<'_> = if page.rows.is_empty() {
264            let empty_cell = self
265                .render_empty_cell(cx, state, path, with_actions, with_bulk, signals.as_ref())
266                .await?;
267            view! {
268                cx =>
269                table(
270                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
271                    (head)
272                    (empty_cell)
273                )
274            }
275            .boxed()
276        } else {
277            pager_views = pager;
278            // One table body for grouped and ungrouped pages: each grouped
279            // row carries the header its group's first row owns, so
280            // the header lands inside the table immediately above its own
281            // rows instead of a count legend stacked over an ungrouped table.
282            let header_colspan =
283                self.columns.len() + usize::from(with_bulk) + usize::from(with_actions);
284            // The one dialog every row control on this table opens;
285            // empty without delete chrome, where no control renders one.
286            let delete_dialog_id = delete_prefix
287                .as_deref()
288                .map(Self::delete_dialog_dom_id)
289                .unwrap_or_default();
290            view! {
291                cx =>
292                // The table-level layout is a static class: Tailwind sees the
293                // literal, and the decision carries no per-column value. Each
294                // column's width, which does, rides the `th`/`td` inline
295                // `style`. Fixed layout is what stops a filter or a
296                // page change from re-measuring the columns. The `min-width`
297                // is the sum of those declared widths: with `w-full` the
298                // table never exceeds its container on its own, so without
299                // the floor the wrapper's `overflow-x-auto` never scrolls.
300                table(
301                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
302                    (head)
303                    table_body(
304                        #[key(row.key.as_str())]
305                        for row in &row_data {
306                            let key_for_row = row.key.clone();
307                            let key_for_select = row.record_id.clone();
308                            let view_for_row = row.view_url.clone();
309                            let edit_for_row = row.edit_url.clone();
310                            let open_for_row = row.delete_url.clone();
311                            let delete_action_for_row = row.delete_action.clone();
312                            let delete_dialog_for_row = delete_dialog_id.clone();
313                            let selectable_for_row = row.selectable;
314                            let row_dom_id = row_dom_id(&key_for_row);
315                            if let Some(header) = row.group_header.clone() {
316                                table_row(
317                                    attrs: attributes! { id=(header.dom_id) },
318                                    table_cell(
319                                        attrs: attributes! {
320                                            colspan=(header_colspan)
321                                            class="px-4 py-2 bg-muted text-sm font-medium"
322                                        },
323                                        (header.text)
324                                    )
325                                )
326                            }
327                            table_row(
328                                attrs: attributes! { id=(row_dom_id) },
329                                if with_bulk {
330                                    if selectable_for_row {
331                                        table_cell(
332                                            <input
333                                                type="checkbox"
334                                                value=(key_for_select)
335                                                aria-label="Select row"
336                                                data-row-select=""
337                                            >
338                                        )
339                                    } else {
340                                        // A refused row renders no checkbox:
341                                        // selecting it could only produce a
342                                        // batch the handler refuses. The cell
343                                        // stays so the row keeps its shape.
344                                        table_cell()
345                                    }
346                                }
347                                // `row.cells` is built column-for-column, so the
348                                // zip pairs each cell with the column that owns
349                                // its width. The cell repeats the
350                                // width its header declares and truncates:
351                                // under the table's fixed layout a value wider
352                                // than the column clips to an ellipsis instead
353                                // of stretching the column. `truncate` is a
354                                // static class, which Tailwind does generate —
355                                // only the per-column width has to be data.
356                                for (cell, width) in row.cells.iter().zip(&cell_widths) {
357                                    table_cell(
358                                        attrs: attributes! { class="truncate" style=(width.as_deref()) },
359                                        (cell.clone())
360                                    )
361                                }
362                                // Every row carries the actions cell its header
363                                // declares; a row refused every link keeps an
364                                // empty cell so the row keeps its shape.
365                                if with_actions {
366                                    table_cell(
367                                        attrs: attributes! { style=(actions_min.as_deref()) },
368                                        <div class="flex gap-2">
369                                            if let Some(url) = view_for_row {
370                                                <a
371                                                    (crate::resource::runtime_link(cx, &url))
372                                                    class=(button_variants(
373                                                        ButtonVariant::Outline,
374                                                        ButtonSize::Md,
375                                                    ))
376                                                >
377                                                    "View"
378                                                </a>
379                                            }
380                                            if let Some(url) = edit_for_row {
381                                                <a
382                                                    (crate::resource::runtime_link(cx, &url))
383                                                    class=(button_variants(
384                                                        ButtonVariant::Outline,
385                                                        ButtonSize::Md,
386                                                    ))
387                                                >
388                                                    "Edit"
389                                                </a>
390                                            }
391                                            if let (Some(url), Some(action)) = (
392                                                open_for_row,
393                                                delete_action_for_row,
394                                            ) {
395                                                <a
396                                                    href=(url)
397                                                    data-row-delete-trigger=(delete_dialog_for_row)
398                                                    data-row-delete-action=(action)
399                                                    class=(button_variants(
400                                                        ButtonVariant::Destructive,
401                                                        ButtonSize::Md,
402                                                    ))
403                                                >
404                                                    "Delete"
405                                                </a>
406                                            }
407                                        </div>
408                                    )
409                                }
410                            )
411                        }
412                    )
413                )
414            }
415            .boxed()
416        };
417
418        // The refresh control: a live table's region re-renders when
419        // a signal its shard tracked changes, and a mutation changes rows the
420        // tracked inputs do not describe — the query is the same, the data is
421        // not. One write that means "re-read the table" is therefore the
422        // mutation's only honest in-place effect: the client bumps this
423        // revision token and the shard re-runs the query, morphing and
424        // re-hydrating the region through the seam that already exists.
425        //
426        // The signal is declared here, inside the shard's own output, so it
427        // belongs to the shard's content scope: its id derives from the shard
428        // invocation's identity, this call site, and the list path, so it is
429        // stable across reruns and distinct per list (every resource's list
430        // runs the same shard), and the runtime keeps its value when the
431        // declaration renders again. Reading it for the input's initial
432        // value is what declares the dependency; the token itself is opaque.
433        //
434        // A static table renders no control at all: it has no shard to re-run,
435        // its region is inert markup, and the client replaces it wholesale
436        // So the control's presence *is* the page's answer to
437        // "can this table refresh in place?".
438        let revision_attrs = signals.as_ref().map(|_| {
439            let revision = topcoat::runtime::signal(&cx.keyed(path), || "0".to_string());
440            attributes! {
441                cx =>
442                type="hidden"
443                value=(revision.get())
444                :value=$(revision.get())
445                @change=$(|e: Event| revision.set(e.target.value))
446                data-table-revision=""
447            }
448        });
449
450        // One chrome wrapper for both branches: search bar, filter bar, bulk
451        // bar, warning, table body, pager, dialog, inside the
452        // `data-boundary` region the morph swaps.
453        let inner = view! {
454            cx =>
455            <div
456                class="rounded-xl border border-border overflow-hidden"
457                data-table-root=""
458            >
459                if show_search {
460                    (search_bar.expect("search bar built when enabled"))
461                }
462                if show_filters {
463                    (filter_bar.expect("filter bar built when enabled"))
464                }
465                (bulk_bar_view)
466                if let Some(attrs) = revision_attrs {
467                    <input (attrs)>
468                }
469                if let Some(warning) = filter_warning {
470                    (warning)
471                }
472                (body)
473                for p in pager_views {
474                    (p)
475                }
476                if let Some(dialog) = delete_dialog {
477                    (dialog)
478                }
479            </div>
480        };
481        Ok(view! { cx => <div data-boundary="table">(inner)</div> }.boxed())
482    }

484    /// Project the loaded page into the row presentation the template renders.
485    ///
486    /// Precomputed so template bodies capture only owned data — the lazy view
487    /// outlives the render call, so it must never borrow `self` or `page`.
488    ///
489    /// The per-row delete URL opens the confirmation dialog on the list page
490    /// (`?delete=<key>`); the per-row edit URL links to `{prefix}/{key}/edit`.
491    /// Both — and the bulk checkbox values — carry the *record* key, resolved by
492    /// handlers as the model's typed PK; the display `key` stays on keyed diffs
493    /// and DOM ids.
494    ///
495    /// The chrome prefixes say which links the table *can* render; the
496    /// [`Table::row_actions`] policy says which of them *this* record may use.
497    /// A denied action emits no URL, and a row denied `delete` renders no bulk
498    /// checkbox. The policy is consulted only when a prefix is wired.
499    ///
500    /// The delete URL's shared parameters are encoded once for the whole page,
501    /// because the filter transport is the expensive half and rebuilding it per
502    /// row is work a client can inflate with one oversized `?filters=`.
503    ///
504    /// Page-local grouping is display-only: `group_by` is a bare key closure with
505    /// no lens, so no `ORDER BY` is derivable and a group cannot span pages. The
506    /// shim therefore reorders *this page's* rows by the group label — a stable
507    /// sort, so rows keep the query's order inside their group — and hangs each
508    /// group's header off its first row. The query, its cursors and the export
509    /// keep the declared ordering.
510    ///
511    /// Row keys must be injective within a page: duplicates corrupt keyed diffs
512    /// and bulk selection.
513    fn row_views(
514        &self,
515        state: &NormalizedState,
516        path: &str,
517        page: &TablePage<M>,
518        row_key: &RowKey<M>,
519        record_key: &RowKey<M>,
520        group_key: Option<&GroupKey<M>>,
521    ) -> Vec<RowView>
522    where
523        M: toasty::schema::Model,
524    {
525        let delete_url_base = self
526            .delete_prefix
527            .as_ref()
528            .map(|_| state.row_url_base(path));
529        let gated = self.delete_prefix.is_some()
530            || self.edit_prefix.is_some()
531            || self.view_prefix.is_some();
532        let mut row_data: Vec<RowView> = page
533            .rows
534            .iter()
535            .map(|row| {
536                let key = row_key(row);
537                let record_id = record_key(row);
538                let actions = if gated {
539                    self.actions_for(row)
540                } else {
541                    RowActions::ALL
542                };
543                let cells: Vec<String> = self
544                    .columns
545                    .iter()
546                    .map(|col| col.render_cell(row))
547                    .collect();
548                let edit_url = self
549                    .edit_prefix
550                    .as_ref()
551                    .filter(|_| actions.edit)
552                    .map(|prefix| row_edit_url(prefix, &record_id));
553                let view_url = self
554                    .view_prefix
555                    .as_ref()
556                    .filter(|_| actions.view)
557                    .map(|prefix| row_view_url(prefix, &record_id));
558                let delete_url = delete_url_base
559                    .as_ref()
560                    .filter(|_| actions.delete)
561                    .map(|base| base.delete_dialog(&record_id));
562                // The shared dialog's POST target for this row: the
563                // row control hands it over before opening the dialog, so the
564                // action and the control come from the one policy decision.
565                let delete_action = self
566                    .delete_prefix
567                    .as_ref()
568                    .filter(|_| actions.delete)
569                    .map(|prefix| delete_action_url(prefix, &record_id));
570                RowView {
571                    key,
572                    record_id,
573                    cells,
574                    view_url,
575                    edit_url,
576                    delete_url,
577                    delete_action,
578                    selectable: actions.delete,
579                    group: group_key.map(|group| group(row)),
580                    group_header: None,
581                }
582            })
583            .collect();
584        if group_key.is_some() {
585            row_data.sort_by(|a, b| a.group.cmp(&b.group));
586            let mut start = 0;
587            while start < row_data.len() {
588                let label = row_data[start].group.clone().unwrap_or_default();
589                let mut end = start;
590                while end < row_data.len() && row_data[end].group.as_deref() == Some(label.as_str())
591                {
592                    end += 1;
593                }
594                // The count is page-local, and says so: a group split across
595                // pages must not read as a table total. The header
596                // carries an id derived from its label — never from its
597                // position — so the in-place morph can follow it.
598                row_data[start].group_header = Some(GroupHeader {
599                    dom_id: group_header_dom_id(&label),
600                    text: format!("{label} ({} on this page)", end - start),
601                });
602                start = end;
603            }
604        }
605        debug_assert!(
606            {
607                let mut seen = std::collections::HashSet::new();
608                row_data.iter().all(|row| seen.insert(row.key.clone()))
609            },
610            "duplicate table keys in one page: the key projection must be injective"
611        );
612        row_data
613    }

615    /// The zero-rows cell — one honest message, not two: "no records yet"
616    /// when unfiltered, "no results" with a Clear link when a search is
617    /// active. The dead Create button is gone (create pages are not wired
618    /// yet). Wrapped in a single cell spanning the table so it sits inside
619    /// the table. For live tables (`signals`) the clear/back links write the
620    /// signals instead of navigating; `href` stays the fallback.
621    async fn render_empty_cell<'a>(
622        &self,
623        cx: &'a Cx,
624        state: &TableState,
625        path: &str,
626        with_actions: bool,
627        with_bulk: bool,
628        signals: Option<&TableSignals>,
629    ) -> Result<BoxView<'a>>
630    where
631        M: toasty::schema::Model,
632    {
633        let mut colspan = self.columns.len();
634        if with_bulk {
635            colspan += 1;
636        }
637        if with_actions {
638            colspan += 1;
639        }
640        let filtered = state.search.is_some() || !state.filters.is_empty();
641        // Clear only the dimension the link names and keep the rest of the
642        // state (follow-up): the URL is rebuilt from the full state, so
643        // `group_by` survives, and a "Clear search" link leaves the filters
644        // alone.
645        let clear_url = filtered.then(|| {
646            if state.search.is_some() {
647                state.without_search(path)
648            } else {
649                state.without_filters(path)
650            }
651        });
652        // Search matches anywhere in the value, so the empty copy
653        // says "matches", not "prefix matches".
654        let message = match &state.search {
655            Some(term) => format!("No matches for \u{201c}{term}\u{201d}"),
656            None if !state.filters.is_empty() => "No results for these filters".to_string(),
657            None => "No records yet".to_string(),
658        };
659        let clear_label = if state.search.is_some() {
660            "Clear search"
661        } else {
662            "Clear filters"
663        };
664        // Void window: a cursor that lands past the last row (e.g.
665        // rows deleted under pagination) leaves an empty page with no pager —
666        // link back to the first page instead of a dead end. State is
667        // preserved, only the cursor is dropped.
668        let first_page_url =
669            (state.after.is_some() || state.before.is_some()).then(|| state.without_cursor(path));
670        // Live links write the signals in place (keeping the state the link
671        // does not name); `href` stays the no-JS fallback.
672        let clear_link: Option<BoxView<'a>> = clear_url.map(|url| {
673            let attrs = match signals {
674                Some(signals) => {
675                    let none = crate::resource::cursor_none();
676                    let (q, filters, cursor) = (
677                        signals.q.clone(),
678                        signals.filters.clone(),
679                        signals.cursor.clone(),
680                    );
681                    let clearing_search = state.search.is_some();
682                    attributes! {
683                        cx =>
684                        href=(url)
685                        @click=$(|e: Event| {
686                            e.prevent_default();
687                            if clearing_search {
688                                q.set("".to_owned());
689                            } else {
690                                filters.set("".to_owned());
691                            }
692                            cursor.set(none.clone());
693                        })
694                    }
695                }
696                None => attributes! { cx => href=(url) },
697            };
698            view! {
699                cx =>
700                <a class="text-sm text-primary hover:underline" (attrs)>
701                    (clear_label)
702                </a>
703            }
704            .boxed()
705        });
706        let first_page_link: Option<BoxView<'a>> = first_page_url.map(|url| {
707            let attrs = match signals {
708                Some(signals) => {
709                    let cursor = signals.cursor.clone();
710                    let none = crate::resource::cursor_none();
711                    attributes! {
712                        cx =>
713                        href=(url)
714                        @click=$(|e: Event| {
715                            e.prevent_default();
716                            cursor.set(none.clone());
717                        })
718                    }
719                }
720                None => attributes! { cx => href=(url) },
721            };
722            view! {
723                cx =>
724                <a class="text-sm text-primary hover:underline" (attrs)>
725                    "Back to first page"
726                </a>
727            }
728            .boxed()
729        });
730        Ok(view! {
731            cx =>
732            table_body(
733                table_row(
734                    table_cell(
735                        attrs: attributes! { colspan=(colspan) class="px-6 py-16 text-center" },
736                        <div class="flex flex-col items-center gap-4">
737                            <p class="text-sm text-muted-foreground">(message)</p>
738                            if let Some(link) = clear_link {
739                                (link)
740                            }
741                            if let Some(link) = first_page_link {
742                                (link)
743                            }
744                        </div>
745                    )
746                )
747            )
748        }
749        .boxed())
750    }

752    /// Whether the table renders a row-actions column.
753    pub(super) fn with_actions(&self) -> bool {
754        self.delete_prefix.is_some() || self.edit_prefix.is_some() || self.view_prefix.is_some()
755    }

757    /// How many row links sit side by side in the actions column.
758    fn action_link_count(&self) -> usize {
759        usize::from(self.view_prefix.is_some())
760            + usize::from(self.edit_prefix.is_some())
761            + usize::from(self.delete_prefix.is_some())
762    }

764    /// The share of the table the row-actions column claims: the row
765    /// links sit side by side and each is a fixed-size control, so the share
766    /// grows with the number of links the table renders. The values hold the
767    /// widest set at a 1280px window and the narrower sets inside it.
768    fn actions_percent(&self) -> u8 {
769        match self.action_link_count() {
770            2 => 18,
771            3.. => 25,
772            _ => 12,
773        }
774    }
775
776    /// The content floor of the row-actions column, in whole rem: one row of
777    /// `Md` buttons plus the cell's `p-3` padding, by link count. The share
778    /// above is a fraction of the table and shrinks with it, so on a narrow
779    /// viewport the buttons would spill past the table and clip against the
780    /// chrome's `overflow-hidden`; the floor keeps the column as wide as its
781    /// buttons, and the table's `min-width` keeps the table as wide as its
782    /// columns, so the wrapper scrolls instead.
783    fn actions_min_rem(&self) -> u8 {
784        match self.action_link_count() {
785            2 => 11,
786            3.. => 15,
787            _ => 7,
788        }
789    }

791    /// The width every column of this table declares.
792    ///
793    /// The kind defaults — a [`ColumnWidth::Narrow`] column, the bulk
794    /// checkbox, the row actions — are shares of the table, scaled down
795    /// together when their nominal total exceeds
796    /// [`DEFAULT_WIDTH_BUDGET_PERCENT`]: the wide columns take what the
797    /// declared ones leave, and a table that spends every percent on declared
798    /// columns leaves them none. An explicit `Rem`/`Percent` is emitted as
799    /// declared.
800    ///
801    /// The table-level `min-width` is the sum of those declarations: every
802    /// share as emitted, every `Rem` verbatim, the actions column's content
803    /// floor, and one [`WIDE_COLUMN_MIN_REM`] per wide column (which declares
804    /// nothing and would otherwise crush to zero). With `w-full` the table
805    /// never exceeds its container on its own, so without the floor the
806    /// wrapper's `overflow-x-auto` never scrolls; with it the table keeps its
807    /// measure on a narrow viewport and the wrapper scrolls. Emitted only
808    /// when the sum carries a length — shares alone are a fraction of the
809    /// container and can never overflow it.
810    pub(super) fn column_widths(&self) -> ColumnWidths
811    where
812        M: toasty::schema::Model,
813    {
814        let bulk = self.bulk_enabled().then_some(BULK_COLUMN_PERCENT);
815        let actions = self.with_actions().then(|| self.actions_percent());
816        let actions_floor = self.with_actions().then(|| self.actions_min_rem());
817        let total: u32 = bulk
818            .into_iter()
819            .chain(
820                self.columns
821                    .iter()
822                    .filter_map(|col| col.column_width().default_percent()),
823            )
824            .chain(actions)
825            .map(u32::from)
826            .sum();
827        let default_style =
828            |nominal: u8| default_width_style(scaled_default_percent(nominal, total));
829
830        let cells = self
831            .columns
832            .iter()
833            .map(|col| {
834                let width = col.column_width();
835                // `Wide` declares nothing; a kind default is resolved against
836                // the rest of the table; `Rem`/`Percent` are verbatim.
837                width
838                    .explicit_css()
839                    .or_else(|| width.default_percent().map(default_style))
840            })
841            .collect();
842        // The `min-width` terms, in layout order: the shares first, then the
843        // lengths as one rem total. A scaled share is the emitted one, so the
844        // floor and the column agree.
845        let mut percent_terms: Vec<u8> = Vec::new();
846        if let Some(share) = bulk {
847            percent_terms.push(scaled_default_percent(share, total));
848        }
849        let mut rem_total: u32 = 0;
850        for col in &self.columns {
851            match col.column_width() {
852                ColumnWidth::Wide => rem_total += u32::from(WIDE_COLUMN_MIN_REM),
853                ColumnWidth::Narrow => {
854                    percent_terms.push(scaled_default_percent(NARROW_DEFAULT_PERCENT, total));
855                }
856                ColumnWidth::Rem(rem) => rem_total += u32::from(rem),
857                ColumnWidth::Percent(share) => percent_terms.push(share),
858            }
859        }
860        let mut actions_style = None;
861        if let (Some(share), Some(floor)) = (actions, actions_floor) {
862            let scaled = scaled_default_percent(share, total);
863            percent_terms.push(scaled);
864            rem_total += u32::from(floor);
865            actions_style = Some(Cow::Owned(format!(
866                "width: {scaled}%; min-width: {floor}rem"
867            )));
868        }
869        let table_min_width = (rem_total > 0).then(|| {
870            let mut parts: Vec<String> = percent_terms
871                .iter()
872                .map(|share| format!("{share}%"))
873                .collect();
874            parts.push(format!("{rem_total}rem"));
875            if parts.len() == 1 {
876                Cow::Owned(format!("min-width: {}", parts[0]))
877            } else {
878                Cow::Owned(format!("min-width: calc({})", parts.join(" + ")))
879            }
880        });
881        ColumnWidths {
882            cells,
883            bulk: bulk.map(default_style),
884            actions: actions_style,
885            actions_min: actions_floor.map(|floor| Cow::Owned(format!("min-width: {floor}rem"))),
886            table_min_width,
887        }
888    }

890    /// The shared column-header row — the single source of the `<thead>`
891    /// markup: labels and **links** on sortable columns that toggle
892    /// `?sort=`/`?dir=` (a Lucide arrow with `aria-sort` when active,
893    /// `arrow-up-down` when inactive). Every render branch (skeleton / empty
894    /// / rows) composes it, so an a11y or styling change happens once.
895    ///
896    /// With `signals` (a live table) the link also writes the sort signals and
897    /// clears the cursors; its `href` stays the no-JS fallback.
898    pub(super) async fn render_thead<'a>(
899        &self,
900        cx: &'a Cx,
901        state: &TableState,
902        path: &str,
903        with_actions: bool,
904        with_bulk: bool,
905        signals: Option<&TableSignals>,
906    ) -> Result<BoxView<'a>>
907    where
908        M: toasty::schema::Model,
909    {
910        // The active sort only counts when it names a declared sortable column.
911        let active = state.sort.as_ref().filter(|s| {
912            self.columns
913                .iter()
914                .any(|c| c.is_sortable() && c.name() == s.column)
915        });
916        let widths = self.column_widths();
917        let mut heads: Vec<BoxView<'_>> = Vec::with_capacity(self.columns.len());
918        for (index, col) in self.columns.iter().enumerate() {
919            // Owned per iteration: the view must not borrow the resolved list.
920            let width = widths.cells[index].clone();
921            let label = col.label().to_string();
922            // The declared width rides the header cell's inline `style`
923            // a Tailwind class assembled at render would emit no
924            // CSS, because Tailwind only generates the literals it finds in
925            // source. A wide column declares nothing and takes a share of what
926            // the declared columns leave.
927            // A static preview renders plain labels: no link to an interaction
928            // the page does not honor.
929            let sortable = col.is_sortable();
930            let (head_class, aria_sort, header) = if sortable {
931                let (aria, sort_icon, next_desc) = match active {
932                    Some(s) if s.column == col.name() => (
933                        if s.descending {
934                            "descending"
935                        } else {
936                            "ascending"
937                        },
938                        if s.descending {
939                            icons::ARROW_DOWN
940                        } else {
941                            icons::ARROW_UP
942                        },
943                        // toggling the active column flips the direction
944                        !s.descending,
945                    ),
946                    _ => ("none", icons::ARROW_UP_DOWN, false),
947                };
948                let href = state.sorted_by(path, col.name(), next_desc);
949                let aria_label = format!(
950                    "Sort by {} {}",
951                    label,
952                    if next_desc { "descending" } else { "ascending" }
953                );
954                let link_attrs = if let Some(signals) = signals {
955                    let none = crate::resource::cursor_none();
956                    let (sort, dir, cursor) = (
957                        signals.sort.clone(),
958                        signals.dir.clone(),
959                        signals.cursor.clone(),
960                    );
961                    let column = col.name().to_string();
962                    let next_dir = if next_desc { "desc" } else { "asc" }.to_owned();
963                    attributes! {
964                        cx =>
965                        href=(href)
966                        aria-label=(aria_label)
967                        @click=$(|e: Event| {
968                            e.prevent_default();
969                            sort.set(column.clone());
970                            dir.set(next_dir.clone());
971                            cursor.set(none.clone());
972                        })
973                    }
974                } else {
975                    attributes! { cx => href=(href) aria-label=(aria_label) }
976                };
977                (
978                    "cursor-pointer hover:bg-foreground/5",
979                    Some(aria),
980                    view! {
981                        cx =>
982                        <a
983                            class="inline-flex items-center gap-1 hover:text-foreground"
984                            (link_attrs)
985                        >
986                            (label.clone())
987                            icon(
988                                data: sort_icon,
989                                attrs: attributes! { class="size-4 shrink-0 text-muted-foreground" }
990                            )
991                        </a>
992                    }
993                    .boxed(),
994                )
995            } else {
996                ("", None, view! { cx => (label.clone()) }.boxed())
997            };
998            heads.push(
999                view! {
1000                    cx =>
1001                    table_head(
1002                        attrs: attributes! {
1003                            class=(head_class)
1004                            aria-sort=(aria_sort)
1005                            style=(width.as_deref())
1006                        },
1007                        (header)
1008                    )
1009                }
1010                .boxed(),
1011            );
1012        }
1013        if with_actions {
1014            heads.push(
1015                view! {
1016                    cx =>
1017                    // The share sizes the column at wide viewports; the floor
1018                    // sizes it to its buttons at narrow ones, where the share
1019                    // alone would let them spill past the table. Each row's
1020                    // `td` repeats only the floor.
1021                    table_head(
1022                        attrs: attributes! { style=(widths.actions.as_deref()) },
1023                        "Actions"
1024                    )
1025                }
1026                .boxed(),
1027            );
1028        }
1029        Ok(view! {
1030            cx =>
1031            table_header(
1032                table_row(
1033                    if with_bulk {
1034                        // The header row is the row `table-fixed` measures, so
1035                        // the chrome columns declare their width here and
1036                        // their `td`s declare none.
1037                        table_head(
1038                            attrs: attributes! { style=(widths.bulk.as_deref()) },
1039                            <input
1040                                type="checkbox"
1041                                aria-label="Select all rows"
1042                                data-bulk-select-all=""
1043                            >
1044                        )
1045                    }
1046                    for h in heads {
1047                        (h)
1048                    }
1049                )
1050            )
1051        }
1052        .boxed())
1053    }

1056/// Precomputed per-row presentation for the table body: the display row key,
1057/// the record key, the rendered cells, and the optional Edit / delete-dialog
1058/// action URLs. A struct (not a tuple): five anonymous positions would
1059/// mislead readers and trip `clippy::type_complexity`.
1060///
1061/// `key` is the display projection (keyed diffs, DOM ids);
1062/// `record_id` is the record projection (URLs, bulk values), resolved
1063/// by handlers as the typed PK.
1064struct RowView {
1065    key: String,
1066    record_id: String,
1067    cells: Vec<String>,
1068    view_url: Option<String>,
1069    edit_url: Option<String>,
1070    delete_url: Option<String>,
1071    /// The row's delete POST target (`{prefix}/{key}/delete`): the
1072    /// Delete control hands it to the shared dialog before opening it, so the
1073    /// confirmed POST keeps the route the `?delete=` fallback uses.
1074    delete_action: Option<String>,
1075    /// Whether the row renders a bulk checkbox: a row the
1076    /// [`Table::row_actions`] policy denies `delete` renders none, so
1077    /// `bulk.js` never sees its key and select-all cannot submit a batch the
1078    /// handler refuses wholesale.
1079    selectable: bool,
1080    /// The row's group label, when `?group_by=` named the declared group.
1081    /// Carried on every row so the page-local shim can order by it.
1082    group: Option<String>,
1083    /// The header this row renders above itself, `Some` only on the first row
1084    /// of its group.
1085    group_header: Option<GroupHeader>,
1086}

1088/// One page-local group header: the label with its page-local count,
1089/// and the stable DOM id the injected header row carries so the in-place morph
1090/// can follow it (`row_dom_id`'s contract).

1095    /// [`group_header_dom_id`] of the label.
