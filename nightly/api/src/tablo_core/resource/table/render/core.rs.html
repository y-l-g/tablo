23/// The share of the table the bulk-selection column claims: one
24/// checkbox plus the cell's `p-3` padding at the widths a list is read at. A
25/// percentage, not a length: the column keeps its share as the table narrows,
26/// and the columns that declare none keep theirs.
27const BULK_COLUMN_PERCENT: u8 = 5;
28
29/// The width every column of one render declares: one `style` value
30/// per declared column, in column order, plus the two chrome columns.
31/// `None` is a column that declares no width — a wide column, which takes a
32/// share of what the declared ones leave.
33///
34/// `actions_min` is the actions column's content floor for its body cells
35/// (the header carries the share *and* the floor); `table_min_width` is the
36/// table-level floor — the sum of the declared widths — that lets the
37/// wrapper's `overflow-x-auto` scroll on a narrow viewport instead of
38/// crushing the cells.
39pub(super) struct ColumnWidths {
40    cells: Vec<Option<Cow<'static, str>>>,
41    bulk: Option<Cow<'static, str>>,
42    actions: Option<Cow<'static, str>>,
43    actions_min: Option<Cow<'static, str>>,
44    pub(super) table_min_width: Option<Cow<'static, str>>,
45}

48    /// Render the table for the given loaded page.
49    ///
50    /// Real chrome, no fake affordances: the header renders sort **links**
51    /// driving `?sort=`/`?dir=` and a search toolbar driving `?q=` (shown by
52    /// default when any column is `searchable()`), rows are keyed by the
53    /// projection declared via [`Table::new`](super::super::Table::new) per
54    /// `CONTEXT.md` and rendered via
55    /// each column's typed projection, pagination shows Previous/Next links
56    /// built from the executed page's **real** cursors (never invented page
57    /// numbers), and the empty state reflects whether a search was active.
58    ///
59    /// Composes the synced `tablo-ui` primitives and Token classes
60    /// (`border-border` on the chrome, `bg-background`/`shadow-xs` on the
61    /// toolbar controls, `text-muted-foreground` on the captions) — no raw
62    /// colors, no `ac-*`.
63    ///
64    /// # Errors
65    ///
66    /// Errors when pagination declares `per_page = 0`.
67    pub async fn render<'a>(&self, cx: &'a Cx, page: TablePage<M>) -> Result<BoxView<'a>>
68    where
69        M: toasty::schema::Model + Send + Sync + 'static,
70    {
71        let state = TableState::from_cx(cx);
72        let path = topcoat::context::try_request_context::<http::request::Parts>(cx)
73            .map(|parts| parts.uri.path().to_string())
74            .unwrap_or_default();
75        self.render_with_state(cx, page, &state, &path).await
76    }

78    /// Render with explicit list state and path instead of reading them from
79    /// `cx` — the seam a live-search shard needs: shard requests hit
80    /// the `table_search` shard's own endpoint, so `TableState::from_cx` would
81    /// see the endpoint URI, not the list page's `?q=/filters/sort`. Callers
82    /// pass the page's state (or shard args rebuilt via
83    /// [`TableSignals::to_state`]) and the list URL explicitly.
84    ///
85    /// Normalizes the state it is handed, so a page calling this
86    /// directly needs no knowledge of `NormalizedState`; a caller that
87    /// already normalized once per request goes through
88    /// `Self::render_normalized` instead.
89    pub async fn render_with_state<'a>(
90        &self,
91        cx: &'a Cx,
92        page: TablePage<M>,
93        state: &TableState,
94        path: &str,
95    ) -> Result<BoxView<'a>>
96    where
97        M: toasty::schema::Model + Send + Sync + 'static,
98    {
99        self.render_normalized(cx, page, &self.normalize_state(state), path)
100            .await
101    }

103    /// [`Self::render_with_state`] with the state already normalized
104    /// the request entry normalizes once and every seam below takes
105    /// the proof, so a live list request never normalizes the same state
106    /// twice.
107    pub(crate) async fn render_normalized<'a>(
108        &self,
109        cx: &'a Cx,
110        page: TablePage<M>,
111        state: &NormalizedState,
112        path: &str,
113    ) -> Result<BoxView<'a>>
114    where
115        M: toasty::schema::Model + Send + Sync + 'static,
116    {
117        self.render_inner(cx, page, state, path, None).await
118    }

120    /// Render the interactive body for a live table: the same
121    /// presentation as [`Self::render_with_state`], with the sort links, the
122    /// pager, the filter transport, and the empty-state clear links bound to
123    /// `signals` — each interaction writes a signal and the browser morphs the
124    /// shard's new output in place, without a navigation or a scroll jump.
125    /// Every bound control keeps its real `href`/form, so a page without JS
126    /// still navigates as before.
127    ///
128    /// The row-delete dialog is not part of this output: it lives
129    /// outside the region a rerun swaps, rendered once by the page that owns
130    /// the signals, so a caller rendering only through this method renders
131    /// [`Self::render_delete_dialog_normalized`] itself to keep the `?delete=` fallback.
132    pub(crate) async fn render_live_normalized<'a>(
133        &self,
134        cx: &'a Cx,
135        page: TablePage<M>,
136        state: &NormalizedState,
137        path: &str,
138        signals: TableSignals,
139    ) -> Result<BoxView<'a>>
140    where
141        M: toasty::schema::Model + Send + Sync + 'static,
142    {
143        self.render_inner(cx, page, state, path, Some(signals))
144            .await
145    }

147    async fn render_inner<'a>(
148        &self,
149        cx: &'a Cx,
150        page: TablePage<M>,
151        state: &NormalizedState,
152        path: &str,
153        signals: Option<TableSignals>,
154    ) -> Result<BoxView<'a>>
155    where
156        M: toasty::schema::Model + Send + Sync + 'static,
157    {
158        let row_key = self.row_key.clone();
159        let delete_prefix = self.delete_prefix.clone();
160        let with_actions = self.with_actions();
161        let with_bulk = self.bulk_enabled();
162        let record_key = self.record_key.clone();
163        let head = self
164            .render_thead(cx, state, path, with_actions, with_bulk, signals.as_ref())
165            .await?;
166        let show_search = self.search_enabled();
167        let search_bar = if show_search {
168            Some(self.render_search_bar(cx, state, path).await?)
169        } else {
170            None
171        };
172        let show_filters = self.filter_bar_enabled();
173        let filter_bar = if show_filters {
174            Some(
175                self.render_filter_bar(cx, state, path, signals.as_ref())
176                    .await?,
177            )
178        } else {
179            None
180        };
181        let bulk_bar_view = self.render_bulk_bar(cx, signals.as_ref());
182        let pager = self
183            .render_pager(cx, state, path, &page, signals.as_ref())
184            .await?;
185        let filter_warning = self.render_filter_warning(cx, state, path);
186        // The declared grouping, when `?group_by=` names it. Read
187        // before the row projection so each row can carry its group label,
188        // which the page-local shim orders by.
189        let group_key = self.effective_group_key(state);
190        let row_data = self.row_views(
191            state,
192            path,
193            &page,
194            &row_key,
195            &record_key,
196            group_key.as_ref(),
197        );
198        // The confirmation dialog lives with the delete chrome and
199        // ships closed, so a row control opens it in place. A live
200        // output carries none: the page that owns the signals renders it once,
201        // outside the region a rerun swaps (`panel::resource_list_live`). The
202        // dialog renders on every page with delete chrome, not only under
203        // `?delete=`, so without this gate every shard rerun would morph a
204        // second copy — and its ids — into the page.
205        let delete_dialog = if signals.is_none() {
206            self.render_delete_dialog_normalized(cx, state).await?
207        } else {
208            None
209        };
210
211        // Body-only branch: the empty and rows pages share the one
212        // chrome wrapper built below — only the table body differs. Group
213        // headers and the pager exist solely on rows pages: an empty page
214        // renders the honest empty cell instead (its pager would be empty
215        // anyway, and grouping an empty page yields no headers).
216        //
217        // The declared widths are a property of the columns, not of the row,
218        // so they are resolved once here: the same CSS for every row, and a
219        // `for` whose expression names `self.columns` would carry the table's
220        // borrow into the view.
221        let ColumnWidths {
222            cells: cell_widths,
223            actions_min,
224            table_min_width,
225            ..
226        } = self.column_widths();
227        let mut pager_views: Vec<BoxView<'_>> = Vec::new();
228        let body: BoxView<'_> = if page.rows.is_empty() {
229            let empty_cell = self
230                .render_empty_cell(cx, state, path, with_actions, with_bulk, signals.as_ref())
231                .await?;
232            view! {
233                cx =>
234                table(
235                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
236                    (head)
237                    (empty_cell)
238                )
239            }
240            .boxed()
241        } else {
242            pager_views = pager;
243            // One table body for grouped and ungrouped pages: each grouped
244            // row carries the header its group's first row owns, so
245            // the header lands inside the table immediately above its own
246            // rows instead of a count legend stacked over an ungrouped table.
247            let header_colspan =
248                self.columns.len() + usize::from(with_bulk) + usize::from(with_actions);
249            // The one dialog every row control on this table opens;
250            // empty without delete chrome, where no control renders one.
251            let delete_dialog_id = delete_prefix
252                .as_deref()
253                .map(Self::delete_dialog_dom_id)
254                .unwrap_or_default();
255            view! {
256                cx =>
257                // The table-level layout is a static class: Tailwind sees the
258                // literal, and the decision carries no per-column value. Each
259                // column's width, which does, rides the `th`/`td` inline
260                // `style`. Fixed layout is what stops a filter or a
261                // page change from re-measuring the columns. The `min-width`
262                // is the sum of those declared widths: with `w-full` the
263                // table never exceeds its container on its own, so without
264                // the floor the wrapper's `overflow-x-auto` never scrolls.
265                table(
266                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
267                    (head)
268                    table_body(
269                        #[key(row.key.as_str())]
270                        for row in &row_data {
271                            let key_for_row = row.key.clone();
272                            let key_for_select = row.record_id.clone();
273                            let view_for_row = row.view_url.clone();
274                            let edit_for_row = row.edit_url.clone();
275                            let open_for_row = row.delete_url.clone();
276                            let delete_action_for_row = row.delete_action.clone();
277                            let delete_dialog_for_row = delete_dialog_id.clone();
278                            let selectable_for_row = row.selectable;
279                            let row_dom_id = row_dom_id(&key_for_row);
280                            if let Some(header) = row.group_header.clone() {
281                                table_row(
282                                    attrs: attributes! { id=(header.dom_id) },
283                                    table_cell(
284                                        attrs: attributes! {
285                                            colspan=(header_colspan)
286                                            class="px-4 py-2 bg-muted text-sm font-medium"
287                                        },
288                                        (header.text)
289                                    )
290                                )
291                            }
292                            table_row(
293                                attrs: attributes! { id=(row_dom_id) },
294                                if with_bulk {
295                                    if selectable_for_row {
296                                        table_cell(
297                                            <input
298                                                type="checkbox"
299                                                value=(key_for_select)
300                                                aria-label="Select row"
301                                                data-row-select=""
302                                            >
303                                        )
304                                    } else {
305                                        // A refused row renders no checkbox:
306                                        // selecting it could only produce a
307                                        // batch the handler refuses. The cell
308                                        // stays so the row keeps its shape.
309                                        table_cell()
310                                    }
311                                }
312                                // `row.cells` is built column-for-column, so the
313                                // zip pairs each cell with the column that owns
314                                // its width. The cell repeats the
315                                // width its header declares and truncates:
316                                // under the table's fixed layout a value wider
317                                // than the column clips to an ellipsis instead
318                                // of stretching the column. `truncate` is a
319                                // static class, which Tailwind does generate —
320                                // only the per-column width has to be data.
321                                for (cell, width) in row.cells.iter().zip(&cell_widths) {
322                                    table_cell(
323                                        attrs: attributes! { class="truncate" style=(width.as_deref()) },
324                                        (cell.clone())
325                                    )
326                                }
327                                // Every row carries the actions cell its header
328                                // declares; a row refused every link keeps an
329                                // empty cell so the row keeps its shape.
330                                if with_actions {
331                                    table_cell(
332                                        attrs: attributes! { style=(actions_min.as_deref()) },
333                                        <div class="flex gap-2">
334                                            if let Some(url) = view_for_row {
335                                                <a
336                                                    (crate::resource::runtime_link(cx, &url))
337                                                    class=(button_variants(
338                                                        ButtonVariant::Outline,
339                                                        ButtonSize::Md,
340                                                    ))
341                                                >
342                                                    "View"
343                                                </a>
344                                            }
345                                            if let Some(url) = edit_for_row {
346                                                <a
347                                                    (crate::resource::runtime_link(cx, &url))
348                                                    class=(button_variants(
349                                                        ButtonVariant::Outline,
350                                                        ButtonSize::Md,
351                                                    ))
352                                                >
353                                                    "Edit"
354                                                </a>
355                                            }
356                                            if let (Some(url), Some(action)) = (
357                                                open_for_row,
358                                                delete_action_for_row,
359                                            ) {
360                                                <a
361                                                    href=(url)
362                                                    data-row-delete-trigger=(delete_dialog_for_row)
363                                                    data-row-delete-action=(action)
364                                                    class=(button_variants(
365                                                        ButtonVariant::Destructive,
366                                                        ButtonSize::Md,
367                                                    ))
368                                                >
369                                                    "Delete"
370                                                </a>
371                                            }
372                                        </div>
373                                    )
374                                }
375                            )
376                        }
377                    )
378                )
379            }
380            .boxed()
381        };
382
383        // The refresh control: a live table's region re-renders when
384        // a signal its shard tracked changes, and a mutation changes rows the
385        // tracked inputs do not describe — the query is the same, the data is
386        // not. One write that means "re-read the table" is therefore the
387        // mutation's only honest in-place effect: the client bumps this
388        // revision token and the shard re-runs the query, morphing and
389        // re-hydrating the region through the seam that already exists.
390        //
391        // The signal is declared here, inside the shard's own output, so it
392        // belongs to the shard's content scope: its id derives from the shard
393        // invocation's identity, this call site, and the list path, so it is
394        // stable across reruns and distinct per list (every resource's list
395        // runs the same shard), and the runtime keeps its value when the
396        // declaration renders again. Reading it for the input's initial
397        // value is what declares the dependency; the token itself is opaque.
398        //
399        // A static table renders no control at all: it has no shard to re-run,
400        // its region is inert markup, and the client replaces it wholesale
401        // So the control's presence *is* the page's answer to
402        // "can this table refresh in place?".
403        let revision_attrs = signals.as_ref().map(|_| {
404            let revision = topcoat::runtime::signal(&cx.keyed(path), || "0".to_string());
405            attributes! {
406                cx =>
407                type="hidden"
408                value=(revision.get())
409                :value=$(revision.get())
410                @change=$(|e: Event| revision.set(e.target.value))
411                data-table-revision=""
412            }
413        });
414
415        // One chrome wrapper for both branches: search bar, filter bar, bulk
416        // bar, warning, table body, pager, dialog, inside the
417        // `data-boundary` region the morph swaps.
418        let inner = view! {
419            cx =>
420            <div
421                class="rounded-xl border border-border overflow-hidden"
422                data-table-root=""
423            >
424                if show_search {
425                    (search_bar.expect("search bar built when enabled"))
426                }
427                if show_filters {
428                    (filter_bar.expect("filter bar built when enabled"))
429                }
430                (bulk_bar_view)
431                if let Some(attrs) = revision_attrs {
432                    <input (attrs)>
433                }
434                if let Some(warning) = filter_warning {
435                    (warning)
436                }
437                (body)
438                for p in pager_views {
439                    (p)
440                }
441                if let Some(dialog) = delete_dialog {
442                    (dialog)
443                }
444            </div>
445        };
446        Ok(view! { cx => <div data-boundary="table">(inner)</div> }.boxed())
447    }

449    /// Project the loaded page into the row presentation the template renders.
450    ///
451    /// Precomputed so template bodies capture only owned data — the lazy view
452    /// outlives the render call, so it must never borrow `self` or `page`.
453    ///
454    /// The per-row delete URL opens the confirmation dialog on the list page
455    /// (`?delete=<key>`); the per-row edit URL links to `{prefix}/{key}/edit`.
456    /// Both — and the bulk checkbox values — carry the *record* key, resolved by
457    /// handlers as the model's typed PK; the display `key` stays on keyed diffs
458    /// and DOM ids.
459    ///
460    /// The chrome prefixes say which links the table *can* render; the
461    /// [`Table::row_actions`] policy says which of them *this* record may use.
462    /// A denied action emits no URL, and a row denied `delete` renders no bulk
463    /// checkbox. The policy is consulted only when a prefix is wired.
464    ///
465    /// The delete URL's shared parameters are encoded once for the whole page,
466    /// because the filter transport is the expensive half and rebuilding it per
467    /// row is work a client can inflate with one oversized `?filters=`.
468    ///
469    /// Page-local grouping is display-only: `group_by` is a bare key closure with
470    /// no lens, so no `ORDER BY` is derivable and a group cannot span pages. The
471    /// shim therefore reorders *this page's* rows by the group label — a stable
472    /// sort, so rows keep the query's order inside their group — and hangs each
473    /// group's header off its first row. The query, its cursors and the export
474    /// keep the declared ordering.
475    ///
476    /// Row keys must be injective within a page: duplicates corrupt keyed diffs
477    /// and bulk selection.
478    fn row_views(
479        &self,
480        state: &NormalizedState,
481        path: &str,
482        page: &TablePage<M>,
483        row_key: &RowKey<M>,
484        record_key: &RowKey<M>,
485        group_key: Option<&GroupKey<M>>,
486    ) -> Vec<RowView>
487    where
488        M: toasty::schema::Model,
489    {
490        let delete_url_base = self
491            .delete_prefix
492            .as_ref()
493            .map(|_| state.row_url_base(path));
494        let gated = self.delete_prefix.is_some()
495            || self.edit_prefix.is_some()
496            || self.view_prefix.is_some();
497        let mut row_data: Vec<RowView> = page
498            .rows
499            .iter()
500            .map(|row| {
501                let key = row_key(row);
502                let record_id = record_key(row);
503                let actions = if gated {
504                    self.actions_for(row)
505                } else {
506                    RowActions::ALL
507                };
508                let cells: Vec<String> = self
509                    .columns
510                    .iter()
511                    .map(|col| col.render_cell(row))
512                    .collect();
513                let edit_url = self
514                    .edit_prefix
515                    .as_ref()
516                    .filter(|_| actions.edit)
517                    .map(|prefix| row_edit_url(prefix, &record_id));
518                let view_url = self
519                    .view_prefix
520                    .as_ref()
521                    .filter(|_| actions.view)
522                    .map(|prefix| row_view_url(prefix, &record_id));
523                let delete_url = delete_url_base
524                    .as_ref()
525                    .filter(|_| actions.delete)
526                    .map(|base| base.delete_dialog(&record_id));
527                // The shared dialog's POST target for this row: the
528                // row control hands it over before opening the dialog, so the
529                // action and the control come from the one policy decision.
530                let delete_action = self
531                    .delete_prefix
532                    .as_ref()
533                    .filter(|_| actions.delete)
534                    .map(|prefix| delete_action_url(prefix, &record_id));
535                RowView {
536                    key,
537                    record_id,
538                    cells,
539                    view_url,
540                    edit_url,
541                    delete_url,
542                    delete_action,
543                    selectable: actions.delete,
544                    group: group_key.map(|group| group(row)),
545                    group_header: None,
546                }
547            })
548            .collect();
549        if group_key.is_some() {
550            row_data.sort_by(|a, b| a.group.cmp(&b.group));
551            let mut start = 0;
552            while start < row_data.len() {
553                let label = row_data[start].group.clone().unwrap_or_default();
554                let mut end = start;
555                while end < row_data.len() && row_data[end].group.as_deref() == Some(label.as_str())
556                {
557                    end += 1;
558                }
559                // The count is page-local, and says so: a group split across
560                // pages must not read as a table total. The header
561                // carries an id derived from its label — never from its
562                // position — so the in-place morph can follow it.
563                row_data[start].group_header = Some(GroupHeader {
564                    dom_id: group_header_dom_id(&label),
565                    text: format!("{label} ({} on this page)", end - start),
566                });
567                start = end;
568            }
569        }
570        debug_assert!(
571            {
572                let mut seen = std::collections::HashSet::new();
573                row_data.iter().all(|row| seen.insert(row.key.clone()))
574            },
575            "duplicate table keys in one page: the key projection must be injective"
576        );
577        row_data
578    }

580    /// The zero-rows cell — one honest message, not two: "no records yet"
581    /// when unfiltered, "no results" with a Clear link when a search is
582    /// active. The dead Create button is gone (create pages are not wired
583    /// yet). Wrapped in a single cell spanning the table so it sits inside
584    /// the table. For live tables (`signals`) the clear/back links write the
585    /// signals instead of navigating; `href` stays the fallback.
586    async fn render_empty_cell<'a>(
587        &self,
588        cx: &'a Cx,
589        state: &TableState,
590        path: &str,
591        with_actions: bool,
592        with_bulk: bool,
593        signals: Option<&TableSignals>,
594    ) -> Result<BoxView<'a>>
595    where
596        M: toasty::schema::Model,
597    {
598        let mut colspan = self.columns.len();
599        if with_bulk {
600            colspan += 1;
601        }
602        if with_actions {
603            colspan += 1;
604        }
605        let filtered = state.search.is_some() || !state.filters.is_empty();
606        // Clear only the dimension the link names and keep the rest of the
607        // state (follow-up): the URL is rebuilt from the full state, so
608        // `group_by` survives, and a "Clear search" link leaves the filters
609        // alone.
610        let clear_url = filtered.then(|| {
611            if state.search.is_some() {
612                state.without_search(path)
613            } else {
614                state.without_filters(path)
615            }
616        });
617        // Search matches anywhere in the value, so the empty copy
618        // says "matches", not "prefix matches".
619        let message = match &state.search {
620            Some(term) => format!("No matches for \u{201c}{term}\u{201d}"),
621            None if !state.filters.is_empty() => "No results for these filters".to_string(),
622            None => "No records yet".to_string(),
623        };
624        let clear_label = if state.search.is_some() {
625            "Clear search"
626        } else {
627            "Clear filters"
628        };
629        // Void window: a cursor that lands past the last row (e.g.
630        // rows deleted under pagination) leaves an empty page with no pager —
631        // link back to the first page instead of a dead end. State is
632        // preserved, only the cursor is dropped.
633        let first_page_url =
634            (state.after.is_some() || state.before.is_some()).then(|| state.without_cursor(path));
635        // Live links write the signals in place (keeping the state the link
636        // does not name); `href` stays the no-JS fallback.
637        let clear_link: Option<BoxView<'a>> = clear_url.map(|url| {
638            let attrs = match signals {
639                Some(signals) => {
640                    let none = crate::resource::cursor_none();
641                    let (q, filters, cursor) = (
642                        signals.q.clone(),
643                        signals.filters.clone(),
644                        signals.cursor.clone(),
645                    );
646                    let clearing_search = state.search.is_some();
647                    attributes! {
648                        cx =>
649                        href=(url)
650                        @click=$(|e: Event| {
651                            e.prevent_default();
652                            if clearing_search {
653                                q.set("".to_owned());
654                            } else {
655                                filters.set("".to_owned());
656                            }
657                            cursor.set(none.clone());
658                        })
659                    }
660                }
661                None => attributes! { cx => href=(url) },
662            };
663            view! {
664                cx =>
665                <a class="text-sm text-primary hover:underline" (attrs)>
666                    (clear_label)
667                </a>
668            }
669            .boxed()
670        });
671        let first_page_link: Option<BoxView<'a>> = first_page_url.map(|url| {
672            let attrs = match signals {
673                Some(signals) => {
674                    let cursor = signals.cursor.clone();
675                    let none = crate::resource::cursor_none();
676                    attributes! {
677                        cx =>
678                        href=(url)
679                        @click=$(|e: Event| {
680                            e.prevent_default();
681                            cursor.set(none.clone());
682                        })
683                    }
684                }
685                None => attributes! { cx => href=(url) },
686            };
687            view! {
688                cx =>
689                <a class="text-sm text-primary hover:underline" (attrs)>
690                    "Back to first page"
691                </a>
692            }
693            .boxed()
694        });
695        Ok(view! {
696            cx =>
697            table_body(
698                table_row(
699                    table_cell(
700                        attrs: attributes! { colspan=(colspan) class="px-6 py-16 text-center" },
701                        <div class="flex flex-col items-center gap-4">
702                            <p class="text-sm text-muted-foreground">(message)</p>
703                            if let Some(link) = clear_link {
704                                (link)
705                            }
706                            if let Some(link) = first_page_link {
707                                (link)
708                            }
709                        </div>
710                    )
711                )
712            )
713        }
714        .boxed())
715    }

717    /// Whether the table renders a row-actions column.
718    pub(super) fn with_actions(&self) -> bool {
719        self.delete_prefix.is_some() || self.edit_prefix.is_some() || self.view_prefix.is_some()
720    }
721
722    /// How many row links sit side by side in the actions column.
723    fn action_link_count(&self) -> usize {
724        usize::from(self.view_prefix.is_some())
725            + usize::from(self.edit_prefix.is_some())
726            + usize::from(self.delete_prefix.is_some())
727    }
728
729    /// The share of the table the row-actions column claims: the row
730    /// links sit side by side and each is a fixed-size control, so the share
731    /// grows with the number of links the table renders. The values hold the
732    /// widest set at a 1280px window and the narrower sets inside it.
733    fn actions_percent(&self) -> u8 {
734        match self.action_link_count() {
735            2 => 18,
736            3.. => 25,
737            _ => 12,
738        }
739    }
740
741    /// The content floor of the row-actions column, in whole rem: one row of
742    /// `Md` buttons plus the cell's `p-3` padding, by link count. The share
743    /// above is a fraction of the table and shrinks with it, so on a narrow
744    /// viewport the buttons would spill past the table and clip against the
745    /// chrome's `overflow-hidden`; the floor keeps the column as wide as its
746    /// buttons, and the table's `min-width` keeps the table as wide as its
747    /// columns, so the wrapper scrolls instead.
748    fn actions_min_rem(&self) -> u8 {
749        match self.action_link_count() {
750            2 => 11,
751            3.. => 15,
752            _ => 7,
753        }
754    }

756    /// The width every column of this table declares.
757    ///
758    /// The kind defaults — a [`ColumnWidth::Narrow`] column, the bulk
759    /// checkbox, the row actions — are shares of the table, scaled down
760    /// together when their nominal total exceeds
761    /// `DEFAULT_WIDTH_BUDGET_PERCENT`: the wide columns take what the
762    /// declared ones leave, and a table that spends every percent on declared
763    /// columns leaves them none. An explicit `Rem`/`Percent` is emitted as
764    /// declared.
765    ///
766    /// The table-level `min-width` is the sum of those declarations: every
767    /// share as emitted, every `Rem` verbatim, the actions column's content
768    /// floor, and one `WIDE_COLUMN_MIN_REM` per wide column (which declares
769    /// nothing and would otherwise crush to zero). With `w-full` the table
770    /// never exceeds its container on its own, so without the floor the
771    /// wrapper's `overflow-x-auto` never scrolls; with it the table keeps its
772    /// measure on a narrow viewport and the wrapper scrolls. Emitted only
773    /// when the sum carries a length — shares alone are a fraction of the
774    /// container and can never overflow it.
775    pub(super) fn column_widths(&self) -> ColumnWidths
776    where
777        M: toasty::schema::Model,
778    {
779        let bulk = self.bulk_enabled().then_some(BULK_COLUMN_PERCENT);
780        let actions = self.with_actions().then(|| self.actions_percent());
781        let actions_floor = self.with_actions().then(|| self.actions_min_rem());
782        let total: u32 = bulk
783            .into_iter()
784            .chain(
785                self.columns
786                    .iter()
787                    .filter_map(|col| col.column_width().default_percent()),
788            )
789            .chain(actions)
790            .map(u32::from)
791            .sum();
792        let cells = self
793            .columns
794            .iter()
795            .map(|col| column_width_style(col.column_width(), total))
796            .collect();
797        // The `min-width` terms, in layout order. A scaled share is the
798        // emitted one, so the floor and the column agree.
799        let mut min_width = MinWidth::default();
800        if let Some(share) = bulk {
801            min_width.share(scaled_default_percent(share, total));
802        }
803        for col in &self.columns {
804            min_width.column(col.column_width(), total);
805        }
806        let mut actions_style = None;
807        if let (Some(share), Some(floor)) = (actions, actions_floor) {
808            let scaled = scaled_default_percent(share, total);
809            min_width.share(scaled);
810            min_width.rem(floor);
811            actions_style = Some(Cow::Owned(format!(
812                "width: {scaled}%; min-width: {floor}rem"
813            )));
814        }
815        ColumnWidths {
816            cells,
817            bulk: bulk.map(|share| default_width_style(scaled_default_percent(share, total))),
818            actions: actions_style,
819            actions_min: actions_floor.map(|floor| Cow::Owned(format!("min-width: {floor}rem"))),
820            table_min_width: min_width.style(),
821        }
822    }

824    /// The shared column-header row — the single source of the `<thead>`
825    /// markup: labels and **links** on sortable columns that toggle
826    /// `?sort=`/`?dir=` (a Lucide arrow with `aria-sort` when active,
827    /// `arrow-up-down` when inactive). Every render branch (skeleton / empty
828    /// / rows) composes it, so an a11y or styling change happens once.
829    ///
830    /// With `signals` (a live table) the link also writes the sort signals and
831    /// clears the cursors; its `href` stays the no-JS fallback.
832    pub(super) async fn render_thead<'a>(
833        &self,
834        cx: &'a Cx,
835        state: &TableState,
836        path: &str,
837        with_actions: bool,
838        with_bulk: bool,
839        signals: Option<&TableSignals>,
840    ) -> Result<BoxView<'a>>
841    where
842        M: toasty::schema::Model,
843    {
844        // The active sort only counts when it names a declared sortable column.
845        let active = state.sort.as_ref().filter(|s| {
846            self.columns
847                .iter()
848                .any(|c| c.is_sortable() && c.name() == s.column)
849        });
850        let widths = self.column_widths();
851        let mut heads: Vec<BoxView<'_>> = Vec::with_capacity(self.columns.len());
852        for (index, col) in self.columns.iter().enumerate() {
853            // Owned per iteration: the view must not borrow the resolved list.
854            let width = widths.cells[index].clone();
855            let label = col.label().to_string();
856            // The declared width rides the header cell's inline `style`
857            // a Tailwind class assembled at render would emit no
858            // CSS, because Tailwind only generates the literals it finds in
859            // source. A wide column declares nothing and takes a share of what
860            // the declared columns leave.
861            // A static preview renders plain labels: no link to an interaction
862            // the page does not honor.
863            let sortable = col.is_sortable();
864            let (head_class, aria_sort, header) = if sortable {
865                let (aria, sort_icon, next_desc) = match active {
866                    Some(s) if s.column == col.name() => (
867                        if s.descending {
868                            "descending"
869                        } else {
870                            "ascending"
871                        },
872                        if s.descending {
873                            icons::ARROW_DOWN
874                        } else {
875                            icons::ARROW_UP
876                        },
877                        // toggling the active column flips the direction
878                        !s.descending,
879                    ),
880                    _ => ("none", icons::ARROW_UP_DOWN, false),
881                };
882                let href = state.sorted_by(path, col.name(), next_desc);
883                let aria_label = format!(
884                    "Sort by {} {}",
885                    label,
886                    if next_desc { "descending" } else { "ascending" }
887                );
888                let link_attrs = if let Some(signals) = signals {
889                    let none = crate::resource::cursor_none();
890                    let (sort, dir, cursor) = (
891                        signals.sort.clone(),
892                        signals.dir.clone(),
893                        signals.cursor.clone(),
894                    );
895                    let column = col.name().to_string();
896                    let next_dir = if next_desc { "desc" } else { "asc" }.to_owned();
897                    attributes! {
898                        cx =>
899                        href=(href)
900                        aria-label=(aria_label)
901                        @click=$(|e: Event| {
902                            e.prevent_default();
903                            sort.set(column.clone());
904                            dir.set(next_dir.clone());
905                            cursor.set(none.clone());
906                        })
907                    }
908                } else {
909                    attributes! { cx => href=(href) aria-label=(aria_label) }
910                };
911                (
912                    "cursor-pointer hover:bg-foreground/5",
913                    Some(aria),
914                    view! {
915                        cx =>
916                        <a
917                            class="inline-flex items-center gap-1 hover:text-foreground"
918                            (link_attrs)
919                        >
920                            (label.clone())
921                            icon(
922                                data: sort_icon,
923                                attrs: attributes! { class="size-4 shrink-0 text-muted-foreground" }
924                            )
925                        </a>
926                    }
927                    .boxed(),
928                )
929            } else {
930                ("", None, view! { cx => (label.clone()) }.boxed())
931            };
932            heads.push(
933                view! {
934                    cx =>
935                    table_head(
936                        attrs: attributes! {
937                            class=(head_class)
938                            aria-sort=(aria_sort)
939                            style=(width.as_deref())
940                        },
941                        (header)
942                    )
943                }
944                .boxed(),
945            );
946        }
947        if with_actions {
948            heads.push(
949                view! {
950                    cx =>
951                    // The share sizes the column at wide viewports; the floor
952                    // sizes it to its buttons at narrow ones, where the share
953                    // alone would let them spill past the table. Each row's
954                    // `td` repeats only the floor.
955                    table_head(
956                        attrs: attributes! { style=(widths.actions.as_deref()) },
957                        "Actions"
958                    )
959                }
960                .boxed(),
961            );
962        }
963        Ok(view! {
964            cx =>
965            table_header(
966                table_row(
967                    if with_bulk {
968                        // The header row is the row `table-fixed` measures, so
969                        // the chrome columns declare their width here and
970                        // their `td`s declare none.
971                        table_head(
972                            attrs: attributes! { style=(widths.bulk.as_deref()) },
973                            <input
974                                type="checkbox"
975                                aria-label="Select all rows"
976                                data-bulk-select-all=""
977                            >
978                        )
979                    }
980                    for h in heads {
981                        (h)
982                    }
983                )
984            )
985        }
986        .boxed())
987    }

990/// Precomputed per-row presentation for the table body: the display row key,
991/// the record key, the rendered cells, and the optional Edit / delete-dialog
992/// action URLs. A struct (not a tuple): five anonymous positions would
993/// mislead readers and trip `clippy::type_complexity`.
994///
995/// `key` is the display projection (keyed diffs, DOM ids);
996/// `record_id` is the record projection (URLs, bulk values), resolved
997/// by handlers as the typed PK.
998struct RowView {
999    key: String,
1000    record_id: String,
1001    cells: Vec<String>,
1002    view_url: Option<String>,
1003    edit_url: Option<String>,
1004    delete_url: Option<String>,
1005    /// The row's delete POST target (`{prefix}/{key}/delete`): the
1006    /// Delete control hands it to the shared dialog before opening it, so the
1007    /// confirmed POST keeps the route the `?delete=` fallback uses.
1008    delete_action: Option<String>,
1009    /// Whether the row renders a bulk checkbox: a row the
1010    /// [`Table::row_actions`] policy denies `delete` renders none, so
1011    /// `bulk.js` never sees its key and select-all cannot submit a batch the
1012    /// handler refuses wholesale.
1013    selectable: bool,
1014    /// The row's group label, when `?group_by=` named the declared group.
1015    /// Carried on every row so the page-local shim can order by it.
1016    group: Option<String>,
1017    /// The header this row renders above itself, `Some` only on the first row
1018    /// of its group.
1019    group_header: Option<GroupHeader>,
1020}
