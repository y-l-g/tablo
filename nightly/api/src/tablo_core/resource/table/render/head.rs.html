1//! The column-header row and its sort links.
2
3use tablo_ui::{icons, table_head, table_header, table_row};
4use topcoat::{Result, context::Cx, icon::icon, view::*};
5
6use super::{
7    super::{
8        super::state::{TableSignals, TableState},
9        Table,
10    },
11    live_link,
12};
13
14impl<M> Table<M> {
15    /// The shared column-header row — the single source of the `<thead>`
16    /// markup: labels and **links** on sortable columns that toggle
17    /// `?sort=`/`?dir=` (a Lucide arrow with `aria-sort` when active,
18    /// `arrow-up-down` when inactive). Every render branch (skeleton / empty
19    /// / rows) composes it, so an a11y or styling change happens once.
20    ///
21    /// With `signals` (a live table) the link writes its own query, which drops
22    /// the cursor, to the `query` signal; its `href` stays the no-JS fallback.
23    pub(super) async fn render_thead<'a>(
24        &self,
25        cx: &'a Cx,
26        state: &TableState,
27        path: &str,
28        with_actions: bool,
29        with_bulk: bool,
30        signals: Option<&TableSignals>,
31    ) -> Result<BoxView<'a>>
32    where
33        M: toasty::schema::Model,
34    {
35        // The active sort only counts when it names a declared sortable column.
36        let active = state.sort.as_ref().filter(|s| {
37            self.columns
38                .iter()
39                .any(|c| c.is_sortable() && c.name() == s.column)
40        });
41        let widths = self.column_widths();
42        let mut heads: Vec<BoxView<'_>> = Vec::with_capacity(self.columns.len());
43        for (index, col) in self.columns.iter().enumerate() {
44            // Owned per iteration: the view must not borrow the resolved list.
45            let width = widths.cells[index].clone();
46            let label = col.label().to_string();
47            // The declared width rides the header cell's inline `style`
48            // a Tailwind class assembled at render would emit no
49            // CSS, because Tailwind only generates the literals it finds in
50            // source. A wide column declares nothing and takes a share of what
51            // the declared columns leave.
52            // A static preview renders plain labels: no link to an interaction
53            // the page does not honor.
54            let sortable = col.is_sortable();
55            let (head_class, aria_sort, header) = if sortable {
56                let (aria, sort_icon, next_desc) = match active {
57                    Some(s) if s.column == col.name() => (
58                        if s.descending {
59                            "descending"
60                        } else {
61                            "ascending"
62                        },
63                        if s.descending {
64                            icons::ARROW_DOWN
65                        } else {
66                            icons::ARROW_UP
67                        },
68                        // toggling the active column flips the direction
69                        !s.descending,
70                    ),
71                    _ => ("none", icons::ARROW_UP_DOWN, false),
72                };
73                let href = state.sorted_by(path, col.name(), next_desc);
74                let aria_label = format!(
75                    "Sort by {} {}",
76                    label,
77                    if next_desc { "descending" } else { "ascending" }
78                );
79                let link_attrs = live_link(cx, href, signals);
80                (
81                    "cursor-pointer hover:bg-foreground/5",
82                    Some(aria),
83                    view! {
84                        cx =>
85                        <a
86                            class="inline-flex items-center gap-1 hover:text-foreground"
87                            aria-label=(aria_label)
88                            (link_attrs)
89                        >
90                            (label.clone())
91                            icon(
92                                data: sort_icon,
93                                attrs: attributes! { class="size-4 shrink-0 text-muted-foreground" }
94                            )
95                        </a>
96                    }
97                    .boxed(),
98                )
99            } else {
100                ("", None, view! { cx => (label.clone()) }.boxed())
101            };
102            heads.push(
103                view! {
104                    cx =>
105                    table_head(
106                        attrs: attributes! {
107                            class=(head_class)
108                            aria-sort=(aria_sort)
109                            style=(width.as_deref())
110                        },
111                        (header)
112                    )
113                }
114                .boxed(),
115            );
116        }
117        if with_actions {
118            heads.push(
119                view! {
120                    cx =>
121                    // The share sizes the column at wide viewports; the floor
122                    // sizes it to its buttons at narrow ones, where the share
123                    // alone would let them spill past the table. Each row's
124                    // `td` repeats only the floor.
125                    // The controls are icons, so the heading names the column
126                    // for assistive tech only.
127                    table_head(
128                        attrs: attributes! { style=(widths.actions.as_deref()) },
129                        <span class="sr-only">"Actions"</span>
130                    )
131                }
132                .boxed(),
133            );
134        }
135        Ok(view! {
136            cx =>
137            table_header(
138                table_row(
139                    if with_bulk {
140                        // The header row is the row `table-fixed` measures, so
141                        // the chrome columns declare their width here and
142                        // their `td`s declare none.
143                        table_head(
144                            attrs: attributes! { style=(widths.bulk.as_deref()) },
145                            <input
146                                type="checkbox"
147                                aria-label="Select all rows"
148                                data-bulk-select-all=""
149                            >
150                        )
151                    }
152                    for h in heads {
153                        (h)
154                    }
155                )
156            )
157        }
158        .boxed())
159    }
