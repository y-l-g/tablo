1//! The streamed placeholder the list shell swaps the table into.
2
3use tablo_ui::{table, table_body, table_cell, table_row};
4use topcoat::{Result, context::Cx, view::*};
5
6use super::super::{NormalizedState, Table};
7
8impl<M> Table<M> {
9    /// The skeleton placeholder table — three pulsing rows under the real
10    /// column header. This is the [`suspense`] fallback for tables whose rows
11    /// stream in. Wrapped in the same `data-boundary` region as the real table
12    /// so the markup shape matches when the swap arrives.
13    /// Carries `aria-busy` while loading plus toolbar/pager pulse placeholders
14    /// so the streamed chrome lands without a layout shift.
15    ///
16    /// Takes the state already normalized: the panel parses and normalizes
17    /// once per request and renders the streamed placeholder from that same
18    /// state, so the placeholder header links never echo an unknown
19    /// `?group_by=`.
20    pub(crate) async fn render_skeleton_normalized<'a>(
21        &self,
22        cx: &'a Cx,
23        state: &NormalizedState,
24    ) -> Result<BoxView<'a>>
25    where
26        M: toasty::schema::Model,
27    {
28        let path = topcoat::context::try_request_context::<http::request::Parts>(cx)
29            .map(|parts| parts.uri.path().to_string())
30            .unwrap_or_default();
31        // The action column exists for any of the three row links, matching
32        // `render_inner` — a `with_view`-only table must not swap a
33        // narrower skeleton for a wider table.
34        let with_actions = self.with_actions();
35        let with_bulk = self.bulk_enabled();
36        let head = self
37            .render_thead(cx, state, &path, with_actions, with_bulk, None)
38            .await?;
39        // The same floor the loaded table carries, so the swap lands without
40        // a layout shift.
41        let table_min_width = self.column_widths().table_min_width;
42        let column_count = self.columns.len();
43        let inner = view! {
44            cx =>
45            <div
46                class="rounded-xl border border-border overflow-hidden"
47                data-table-root=""
48                aria-busy="true"
49            >
50                <div class="border-b border-border p-3" aria-hidden="true">
51                    <div class="animate-pulse rounded-md bg-foreground/10 h-9 w-64"></div>
52                </div>
53                table(
54                    attrs: attributes! { class="table-fixed" style=(table_min_width.as_deref()) },
55                    (head)
56                    table_body(
57                        #[key(i)]
58                        for i in 0..3 {
59                            table_row(
60                                if with_bulk {
61                                    table_cell(
62                                        <div
63                                            class="animate-pulse rounded-md bg-foreground/10 h-4 w-4"
64                                        ></div>
65                                    )
66                                }
67                                for _ in 0..column_count {
68                                    table_cell(
69                                        <div
70                                            class="animate-pulse rounded-md bg-foreground/10 h-4 w-full"
71                                        ></div>
72                                    )
73                                }
74                                if with_actions {
75                                    table_cell(
76                                        <div
77                                            class="animate-pulse rounded-md bg-foreground/10 h-4 w-12"
78                                        ></div>
79                                    )
80                                }
81                            )
82                        }
83                    )
84                )
85                <div class="border-t border-border p-3" aria-hidden="true">
86                    <div class="animate-pulse rounded-md bg-foreground/10 h-9 w-40"></div>
87                </div>
88            </div>
89        };
90        // The busy state rides on the morph boundary so assistive
91        // tech sees the live region, not just the swapped root below it.
92        Ok(view! { cx => <div data-boundary="table" aria-busy="true">(inner)</div> }.boxed())
93    }
94}
