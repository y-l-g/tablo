14impl<M> Table<M> {
15    /// The row-delete confirmation dialog, rendered with the table the panel
16    /// wired the delete route into.
17    ///
18    /// One dialog per table: the row Delete controls name it
19    /// (`data-row-delete-trigger`) and carry the record's POST target
20    /// (`data-row-delete-action`), which `assets/dialog.js` writes to the form
21    /// before opening it in place. The control keeps its `?delete=<row key>`
22    /// href, so a page without the script opens the dialog through the URL —
23    /// and that render ships it open. `?open=false`, the mirror `dialog.js`
24    /// writes on dismissal ([`TableState::open`](crate::resource::TableState::open)), leaves it
25    /// closed. Cancel is a `data-dialog-close` button on both paths, so dismissal never
26    /// navigates.
27    ///
28    /// [`Self::render_with_state`] renders it with the table; the live-search
29    /// page (`panel::resource_list_live`) calls this separately because the
30    /// shard swaps the table per keystroke and must not carry dialog state.
31    ///
32    /// Escape/backdrop dismissal, the `data-dialog-close` cancel hook and the
33    /// trigger wiring need `assets/dialog.js` (`tablo_ui::DIALOG_JS`),
34    /// emitted by `Panel::render_document` on every document with shell assets
35    /// (ADR-0014). Without the document scripts Cancel is inert and Delete still
36    /// POSTs; the dialog primitives are vendored under the ADR-0007 sync guard,
37    /// so they carry no note themselves.
38    pub(crate) async fn render_delete_dialog<'a>(
39        &self,
40        cx: &'a Cx,
41        state: &TableState,
42    ) -> Result<Option<BoxView<'a>>> {
43        let Some(prefix) = self.delete_prefix.as_deref() else {
44            return Ok(None);
45        };
46        let key = state
47            .delete
48            .as_deref()
49            .filter(|_| state.open != Some(false));
50        let server_open = key.is_some();
51        let action = key.map(|key| self.action_url(delete_action_url(prefix, key)));
52        // Only the URL-driven dialog mirrors its dismissal into the URL: a
53        // dialog a row control opens client-side has no `?delete=` to close,
54        // so dismissing it leaves the URL alone (GH #154 §3).
55        let open_param = server_open.then(|| state.param("open"));
56        let csrf = crate::csrf::current_token(cx);
57        let footer = view! {
58            cx =>
59            <form
60                method="post"
61                action=(action)
62                class="contents"
63                data-row-delete-form=""
64                data-mutation-submit=""
65            >
66                (crate::csrf::field(cx, &csrf))
67                (confirm_controls(cx))
68            </form>
69        }
70        .boxed();
71        Ok(Some(confirm_dialog(
72            cx,
73            ConfirmDialog {
74                id: Self::delete_dialog_dom_id(prefix),
75                open: server_open,
76                title: "Delete this record?",
77                attrs: attributes! { cx => data-dialog-open-param=(open_param) },
78                description_attrs: Attributes::default(),
79                footer,
80            },
81        )))
82    }
83
84    /// The DOM id of a table's row-delete dialog: the delete prefix
85    /// with its slashes flattened, so two tables with different delete prefixes
86    /// never share an id. Two tables over one prefix (a page rendering the same
87    /// resource twice) derive the same ids; the panel renders one list table
88    /// per page — the list parameters are shared — so its own routes cannot
89    /// reach that. The row controls name the dialog they open, and its
90    /// `aria-labelledby`/`aria-describedby` ids derive from it.
91    pub(super) fn delete_dialog_dom_id(prefix: &str) -> String {
92        chrome_dom_id(prefix, "delete-dialog")
93    }
94}
95
96/// The DOM id of one piece of a table's chrome: the delete prefix with its
97/// slashes flattened, then `suffix`, so two tables with different delete
98/// prefixes never share an id.
99pub(super) fn chrome_dom_id(prefix: &str, suffix: &str) -> String {
100    format!("{}-{suffix}", prefix.replace('/', "-"))
101}

103/// A destructive confirmation dialog.
104pub(super) struct ConfirmDialog<'a> {
105    /// The dialog's DOM id; the title and description ids derive from it.
106    pub(super) id: String,
107    /// Whether it renders open (a URL-driven dialog) or closed.
108    pub(super) open: bool,
109    pub(super) title: &'static str,
110    /// Extra attributes on the dialog element.
111    pub(super) attrs: Attributes,
112    /// Extra attributes on the description.
113    pub(super) description_attrs: Attributes,
114    /// The footer: [`confirm_controls`], wrapped in a form when the dialog is
115    /// not already inside the one it submits.
116    pub(super) footer: BoxView<'a>,
117}
118
119/// An alert dialog asking to confirm a destructive action, labelled by its
120/// title and described by "This action cannot be undone.".
121pub(super) fn confirm_dialog<'a>(cx: &'a Cx, dialog: ConfirmDialog<'a>) -> BoxView<'a> {
122    let ConfirmDialog {
123        id,
124        open,
125        title,
126        attrs: extra,
127        description_attrs: extra_description,
128        footer,
129    } = dialog;
130    let title_id = format!("{id}-title");
131    let description_id = format!("{id}-description");
132    let mut attrs = attributes! {
133        cx =>
134        id=(id)
135        aria-labelledby=(title_id.clone())
136        aria-describedby=(description_id.clone())
137    };
138    attrs.extend(extra);
139    let mut description_attrs = attributes! { cx => id=(description_id) };
140    description_attrs.extend(extra_description);
141    view! {
142        cx =>
143        alert_dialog(
144            open: open,
145            attrs: attrs,
146            dialog_content(
147                dialog_header(
148                    dialog_title(attrs: attributes! { id=(title_id) }, (title))
149                    dialog_description(
150                        attrs: description_attrs,
151                        "This action cannot be undone."
152                    )
153                )
154                dialog_footer((footer))
155            )
156        )
157    }
158    .boxed()
159}

161/// The controls every delete confirmation submits: Cancel (closes the dialog
162/// through `dialog.js`), the `confirm=1` marker the handler requires, and the
163/// destructive submit.
164pub(super) fn confirm_controls<'a>(cx: &'a Cx) -> BoxView<'a> {
165    view! {
166        cx =>
167        button(
168            variant: ButtonVariant::Outline,
169            size: ButtonSize::Md,
170            attrs: attributes! { type="button" data-dialog-close="" },
171            "Cancel"
172        )
173        <input type="hidden" name="confirm" value="1">
174        button(
175            variant: ButtonVariant::Destructive,
176            size: ButtonSize::Md,
177            attrs: attributes! { type="submit" },
178            "Delete"
179        )
180    }
181    .boxed()
