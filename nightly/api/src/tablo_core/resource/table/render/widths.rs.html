3use std::borrow::Cow;
4
5use super::super::{
6    super::column::{ColumnWidth, NARROW_DEFAULT_PERCENT},
7    Table,
8};

10/// The share of the table the bulk-selection column claims.
11pub(super) const BULK_COLUMN_PERCENT: u8 = 5;
12
13/// The width every column of one render declares, in column order, plus the chrome columns.
14pub(super) struct ColumnWidths {
15    pub(super) cells: Vec<Option<Cow<'static, str>>>,
16    pub(super) bulk: Option<Cow<'static, str>>,
17    pub(super) actions: Option<Cow<'static, str>>,
18    pub(super) actions_min: Option<Cow<'static, str>>,
19    pub(super) table_min_width: Option<Cow<'static, str>>,
20}
21
22impl<M> Table<M> {
23    /// Whether the table renders a row-actions column.
24    pub(super) fn with_actions(&self) -> bool {
25        self.delete_prefix.is_some()
26            || self.edit_prefix.is_some()
27            || self.view_prefix.is_some()
28            || self.row_custom_actions().next().is_some()
29    }
30
31    /// Count the row links side by side in the actions column, counting a labeled custom action as
32    /// two.
33    fn action_link_count(&self) -> usize {
34        usize::from(self.view_prefix.is_some())
35            + usize::from(self.edit_prefix.is_some())
36            + usize::from(self.delete_prefix.is_some())
37            + 2 * self.row_custom_actions().count()
38    }
39
40    /// Claim the share of the table the row-actions column takes by link count.
41    fn actions_percent(&self) -> u8 {
42        match self.action_link_count() {
43            2 => 12,
44            3.. => 15,
45            _ => 8,
46        }
47    }
48
49    /// Floor the row-actions column to its buttons by link count.
50    fn actions_min_rem(&self) -> u8 {
51        match self.action_link_count() {
52            2 => 7,
53            3.. => 9,
54            _ => 4,
55        }
56    }
57
58    /// Resolve the width every column of this table declares.
59    pub(super) fn column_widths(&self) -> ColumnWidths
60    where
61        M: toasty::schema::Model,
62    {
63        let bulk = self.bulk_enabled().then_some(BULK_COLUMN_PERCENT);
64        let actions = self.with_actions().then(|| self.actions_percent());
65        let actions_floor = self.with_actions().then(|| self.actions_min_rem());
66        let total: u32 = bulk
67            .into_iter()
68            .chain(
69                self.columns
70                    .iter()
71                    .filter_map(|col| col.column_width().default_percent()),
72            )
73            .chain(actions)
74            .map(u32::from)
75            .sum();
76        let cells = self
77            .columns
78            .iter()
79            .map(|col| column_width_style(col.column_width(), total))
80            .collect();
81        let mut min_width = MinWidth::default();
82        if let Some(share) = bulk {
83            min_width.share(scaled_default_percent(share, total));
84        }
85        for col in &self.columns {
86            min_width.column(col.column_width(), total);
87        }
88        let mut actions_style = None;
89        if let (Some(share), Some(floor)) = (actions, actions_floor) {
90            let scaled = scaled_default_percent(share, total);
91            min_width.share(scaled);
92            min_width.rem(floor);
93            actions_style = Some(Cow::Owned(format!(
94                "width: {scaled}%; min-width: {floor}rem"
95            )));
96        }
97        ColumnWidths {
98            cells,
99            bulk: bulk.map(|share| default_width_style(scaled_default_percent(share, total))),
100            actions: actions_style,
101            actions_min: actions_floor.map(|floor| Cow::Owned(format!("min-width: {floor}rem"))),
102            table_min_width: min_width.style(),
103        }
104    }
105}

107/// Contribute six rem per wide column to the table's `min-width`.
108const WIDE_COLUMN_MIN_REM: u8 = 6;
109
110/// Cap the kind defaults' combined share of the table so undeclared columns keep space.
111pub(super) const DEFAULT_WIDTH_BUDGET_PERCENT: u8 = 60;
112
113/// The share a kind default claims, scaled down when the table's defaults
114/// together (`total`) exceed [`DEFAULT_WIDTH_BUDGET_PERCENT`].
115fn scaled_default_percent(nominal: u8, total: u32) -> u8 {
116    if total <= u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) {
117        return nominal;
118    }
119    let scaled = u32::from(nominal) * u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) / total;
120    // `scaled` is at most the budget, so the conversion cannot fail.
121    u8::try_from(scaled).unwrap_or(DEFAULT_WIDTH_BUDGET_PERCENT)
122}
123
124/// The `style` value a kind default emits.
125fn default_width_style(percent: u8) -> Cow<'static, str> {
126    Cow::Owned(format!("width: {percent}%"))
127}
128
129/// Resolve a data column's cell `style` from its explicit CSS or scaled kind default.
130fn column_width_style(width: ColumnWidth, total: u32) -> Option<Cow<'static, str>> {
131    width.explicit_css().or_else(|| {
132        width
133            .default_percent()
134            .map(|nominal| default_width_style(scaled_default_percent(nominal, total)))
135    })
136}

138/// Accumulate a fixed-layout table's `min-width` terms.
139#[derive(Default)]
140struct MinWidth {
141    percent: Vec<u8>,
142    rem: u32,
143}
144
145impl MinWidth {
146    /// A share of the table, as the column emits it.
147    fn share(&mut self, percent: u8) {
148        self.percent.push(percent);
149    }
150
151    /// A length, in whole rem.
152    fn rem(&mut self, rem: u8) {
153        self.rem += u32::from(rem);
154    }
155
156    /// A data column's term: its scaled share or its length, and
157    /// [`WIDE_COLUMN_MIN_REM`] for a wide column, which declares nothing.
158    fn column(&mut self, width: ColumnWidth, total: u32) {
159        match width {
160            ColumnWidth::Wide => self.rem(WIDE_COLUMN_MIN_REM),
161            ColumnWidth::Narrow => {
162                self.share(scaled_default_percent(NARROW_DEFAULT_PERCENT, total))
163            }
164            ColumnWidth::Rem(rem) => self.rem(rem),
165            ColumnWidth::Percent(share) => self.share(share),
166        }
167    }
168
169    /// The `min-width` style, emitted only when the sum carries a length:
170    /// shares alone are a fraction of the container and can never overflow it.
171    fn style(&self) -> Option<Cow<'static, str>> {
172        (self.rem > 0).then(|| {
173            let mut parts: Vec<String> = self
174                .percent
175                .iter()
176                .map(|share| format!("{share}%"))
177                .collect();
178            parts.push(format!("{}rem", self.rem));
179            if parts.len() == 1 {
180                Cow::Owned(format!("min-width: {}", parts[0]))
181            } else {
182                Cow::Owned(format!("min-width: calc({})", parts.join(" + ")))
183            }
184        })
185    }
186}
