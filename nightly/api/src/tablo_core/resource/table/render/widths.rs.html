1//! Column widths: the declared shares, the chrome columns, and the table floor.
2
3use std::borrow::Cow;
4
5use super::super::{
6    super::column::{MinWidth, column_width_style, default_width_style, scaled_default_percent},
7    Table,
8};
9
10/// The share of the table the bulk-selection column claims: one
11/// checkbox plus the cell's `p-3` padding at the widths a list is read at. A
12/// percentage, not a length: the column keeps its share as the table narrows,
13/// and the columns that declare none keep theirs.
14pub(super) const BULK_COLUMN_PERCENT: u8 = 5;
15
16/// The width every column of one render declares: one `style` value
17/// per declared column, in column order, plus the two chrome columns.
18/// `None` is a column that declares no width — a wide column, which takes a
19/// share of what the declared ones leave.
20///
21/// `actions_min` is the actions column's content floor for its body cells
22/// (the header carries the share *and* the floor); `table_min_width` is the
23/// table-level floor — the sum of the declared widths — that lets the
24/// wrapper's `overflow-x-auto` scroll on a narrow viewport instead of
25/// crushing the cells.
26pub(super) struct ColumnWidths {
27    pub(super) cells: Vec<Option<Cow<'static, str>>>,
28    pub(super) bulk: Option<Cow<'static, str>>,
29    pub(super) actions: Option<Cow<'static, str>>,
30    pub(super) actions_min: Option<Cow<'static, str>>,
31    pub(super) table_min_width: Option<Cow<'static, str>>,
32}
33
34impl<M> Table<M> {
35    /// Whether the table renders a row-actions column.
36    pub(super) fn with_actions(&self) -> bool {
37        self.delete_prefix.is_some() || self.edit_prefix.is_some() || self.view_prefix.is_some()
38    }
39
40    /// How many row links sit side by side in the actions column.
41    fn action_link_count(&self) -> usize {
42        usize::from(self.view_prefix.is_some())
43            + usize::from(self.edit_prefix.is_some())
44            + usize::from(self.delete_prefix.is_some())
45    }
46
47    /// The share of the table the row-actions column claims: the row
48    /// links sit side by side and each is a fixed-size control, so the share
49    /// grows with the number of links the table renders. The values hold the
50    /// widest set at a 1280px window and the narrower sets inside it.
51    fn actions_percent(&self) -> u8 {
52        match self.action_link_count() {
53            2 => 18,
54            3.. => 25,
55            _ => 12,
56        }
57    }
58
59    /// The content floor of the row-actions column, in whole rem: one row of
60    /// `Md` buttons plus the cell's `p-3` padding, by link count. The share
61    /// above is a fraction of the table and shrinks with it, so on a narrow
62    /// viewport the buttons would spill past the table and clip against the
63    /// chrome's `overflow-hidden`; the floor keeps the column as wide as its
64    /// buttons, and the table's `min-width` keeps the table as wide as its
65    /// columns, so the wrapper scrolls instead.
66    fn actions_min_rem(&self) -> u8 {
67        match self.action_link_count() {
68            2 => 11,
69            3.. => 15,
70            _ => 7,
71        }
72    }
73
74    /// The width every column of this table declares.
75    ///
76    /// The kind defaults — a [`ColumnWidth::Narrow`] column, the bulk
77    /// checkbox, the row actions — are shares of the table, scaled down
78    /// together when their nominal total exceeds
79    /// `DEFAULT_WIDTH_BUDGET_PERCENT`: the wide columns take what the
80    /// declared ones leave, and a table that spends every percent on declared
81    /// columns leaves them none. An explicit `Rem`/`Percent` is emitted as
82    /// declared.
83    ///
84    /// The table-level `min-width` is the sum of those declarations: every
85    /// share as emitted, every `Rem` verbatim, the actions column's content
86    /// floor, and one `WIDE_COLUMN_MIN_REM` per wide column (which declares
87    /// nothing and would otherwise crush to zero). With `w-full` the table
88    /// never exceeds its container on its own, so without the floor the
89    /// wrapper's `overflow-x-auto` never scrolls; with it the table keeps its
90    /// measure on a narrow viewport and the wrapper scrolls. Emitted only
91    /// when the sum carries a length — shares alone are a fraction of the
92    /// container and can never overflow it.
93    pub(super) fn column_widths(&self) -> ColumnWidths
94    where
95        M: toasty::schema::Model,
96    {
97        let bulk = self.bulk_enabled().then_some(BULK_COLUMN_PERCENT);
98        let actions = self.with_actions().then(|| self.actions_percent());
99        let actions_floor = self.with_actions().then(|| self.actions_min_rem());
100        let total: u32 = bulk
101            .into_iter()
102            .chain(
103                self.columns
104                    .iter()
105                    .filter_map(|col| col.column_width().default_percent()),
106            )
107            .chain(actions)
108            .map(u32::from)
109            .sum();
110        let cells = self
111            .columns
112            .iter()
113            .map(|col| column_width_style(col.column_width(), total))
114            .collect();
115        // The `min-width` terms, in layout order. A scaled share is the
116        // emitted one, so the floor and the column agree.
117        let mut min_width = MinWidth::default();
118        if let Some(share) = bulk {
119            min_width.share(scaled_default_percent(share, total));
120        }
121        for col in &self.columns {
122            min_width.column(col.column_width(), total);
123        }
124        let mut actions_style = None;
125        if let (Some(share), Some(floor)) = (actions, actions_floor) {
126            let scaled = scaled_default_percent(share, total);
127            min_width.share(scaled);
128            min_width.rem(floor);
129            actions_style = Some(Cow::Owned(format!(
130                "width: {scaled}%; min-width: {floor}rem"
131            )));
132        }
133        ColumnWidths {
134            cells,
135            bulk: bulk.map(|share| default_width_style(scaled_default_percent(share, total))),
136            actions: actions_style,
137            actions_min: actions_floor.map(|floor| Cow::Owned(format!("min-width: {floor}rem"))),
138            table_min_width: min_width.style(),
139        }
140    }
141}
