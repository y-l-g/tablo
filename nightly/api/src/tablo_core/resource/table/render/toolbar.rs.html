15/// Keystroke-quiet delay before a live search input reloads the table
16/// (~150-250ms): `assets/live-search.js` waits this long after the
17/// last keystroke, then forwards the value through the bound transport below,
18/// so typing "published" triggers one reload instead of nine. The forwarded
19/// write is an ordinary signal write, so Topcoat's abort-in-flight
20/// coalescing still applies to the resulting rerun.
21pub(crate) const LIVE_SEARCH_DEBOUNCE_MS: u32 = 200;

23/// The hidden inputs a table toolbar carries across its submit, in the order
24/// given; an input whose state holds no value renders nothing.
25pub(super) fn hidden_state_inputs<'a>(
26    cx: &'a Cx,
27    inputs: Vec<(String, Option<String>)>,
28) -> BoxView<'a> {
29    let fields: Vec<BoxView<'a>> = inputs
30        .into_iter()
31        .filter_map(|(name, value)| {
32            value.map(|value| {
33                view! { cx => <input type="hidden" name=(name) value=(value)> }.boxed()
34            })
35        })
36        .collect();
37    view! {
38        cx =>
39        for field in fields {
40            (field)
41        }
42    }
43    .boxed()
44}

47    /// The bulk-delete bar and its confirmation dialog, or the
48    /// placeholder that keeps the chrome's node order stable when
49    /// [`Self::bulk_enabled`] is off.
50    ///
51    /// One destructive form for the whole page: the transport is fed by the
52    /// row checkboxes (`bulk.js`) and ships `,a,b,`-delimited, while on a live
53    /// table the selection lives in a signal instead, so a shard
54    /// rerun re-renders the transport from the selection rather than dropping
55    /// it. The trigger ships enabled: the confirmation dialog gates
56    /// the write and reads the selection when it opens, so an empty selection
57    /// is answered by the dialog rather than by a disabled control whose state
58    /// has to be kept in step across a live swap.
59    ///
60    /// The dialog lives *inside* the form so its `confirm` marker ships with
61    /// the same payload as the selection — the confirm button is an ordinary
62    /// submit of that form, and the handler refuses a POST without the marker.
63    /// It renders closed and opens client-side (`showModal`) rather than
64    /// through a runtime signal: the trigger is `type="button"`, so opening
65    /// the dialog is not a result-set change and must not reload the table.
66    pub(super) fn render_bulk_bar<'a>(
67        &self,
68        cx: &'a Cx,
69        signals: Option<&TableSignals>,
70    ) -> BoxView<'a> {
71        if !self.bulk_enabled() {
72            return view! { cx => <span></span> }.boxed();
73        }
74        let prefix = self
75            .delete_prefix
76            .clone()
77            .expect("bulk chrome rides the delete prefix (see bulk_enabled)");
78        let bulk_action = self.action_url(bulk_delete_url(&prefix));
79        let csrf = crate::csrf::current_token(cx);
80        // Stable ids so the dialog's confirm button can submit this form
81        // from inside the dialog.
82        let bulk_form_id = chrome_dom_id(&prefix, "bulk-form");
83        // Destructive confirm: a batch is the one place a misclick costs many
84        // rows, so it asks first — the same alert dialog the row delete uses.
85        // It sits inside the bulk form, so its controls submit that form.
86        let confirm = confirm_dialog(
87            cx,
88            ConfirmDialog {
89                id: chrome_dom_id(&prefix, "bulk-form-confirm"),
90                open: false,
91                title: "Delete the selected records?",
92                attrs: attributes! { cx => data-bulk-confirm-dialog="" },
93                description_attrs: attributes! { cx => data-bulk-confirm-description="" },
94                footer: confirm_controls(cx),
95            },
96        );
97        // No visible `ids` field: the transport is fed by the row
98        // checkboxes (`bulk.js`) and ships `,a,b,`-delimited. On a live
99        // table the selection lives in a signal instead, so a
100        // shard rerun re-renders the transport from the selection rather
101        // than dropping it.
102        let transport_attrs = match signals {
103            Some(signals) => {
104                let bulk = signals.bulk.clone();
105                attributes! {
106                    cx =>
107                    type="hidden"
108                    name="ids"
109                    :value=$(bulk.get())
110                    @change=$(|e: Event| bulk.set(e.target.value))
111                }
112            }
113            None => attributes! { cx => type="hidden" name="ids" value="" },
114        };
115        view! {
116            cx =>
117            <form
118                method="post"
119                action=(bulk_action)
120                class="ml-auto flex items-center gap-2"
121                data-bulk-form=""
122                data-mutation-submit=""
123                id=(bulk_form_id.clone())
124            >
125                (crate::csrf::field(cx, &csrf))
126                <input (transport_attrs)>
127                button(
128                    variant: ButtonVariant::Outline,
129                    size: ButtonSize::Md,
130                    attrs: attributes! { type="button" data-bulk-confirm-trigger="" },
131                    icon(
132                        data: tablo_ui::icons::TRASH,
133                        attrs: attributes! { class="text-destructive" }
134                    )
135                    "Delete selected"
136                )
137                (confirm)
138            </form>
139        }
140        .boxed()
141    }

143    /// The search toolbar (GET form); live tables instead render the host
144    /// input eagerly and the shard invocation in the streamed region (see
145    /// [`Self::render_live_search_bar`] / [`Self::render_live_invocation`]).
146    pub(super) async fn render_search_bar<'a>(
147        &self,
148        cx: &'a Cx,
149        state: &TableState,
150        path: &str,
151    ) -> Result<BoxView<'a>> {
152        let action = path.to_string();
153        let q_display = state.search.clone().unwrap_or_default();
154        let sort_hidden = state.sort.as_ref().map(|s| s.column.clone());
155        let dir_hidden = state.sort.as_ref().map(|s| {
156            if s.descending {
157                "desc".to_string()
158            } else {
159                "asc".to_string()
160            }
161        });
162        // Pre-normalized by the render seams: `state.group_by` is
163        // the declared name or `None`, never an unknown value.
164        let group_hidden = state.group_by.clone();
165        // Clear only renders when something survives the search term; every
166        // branch below projects the same URL, so one intent serves all three.
167        let clear_url =
168            (state.sort.is_some() || !state.filters.is_empty() || group_hidden.is_some())
169                .then(|| state.without_search(path));
170        let mut inputs = vec![
171            (state.param("sort"), sort_hidden),
172            (state.param("dir"), dir_hidden),
173        ];
174        inputs.extend(
175            state
176                .filters
177                .iter()
178                .map(|(name, value)| (state.filter_param(name), Some(value.clone()))),
179        );
180        inputs.push((state.param("group_by"), group_hidden));
181        let hidden = hidden_state_inputs(cx, inputs);
182        let search_name = state.param("q");
183        Ok(view! {
184            cx =>
185            <form method="get" action=(action) class=(SEARCH_FORM_CLASS)>
186                (hidden)
187                <div class=(SEARCH_FIELD_CLASS)>
188                    icon(
189                        data: tablo_ui::icons::SEARCH,
190                        attrs: attributes! { class=(SEARCH_ICON_CLASS) }
191                    )
192                    ui_input(
193                        attrs: attributes! {
194                            type="search"
195                            name=(search_name)
196                            value=(q_display)
197                            placeholder="Search…"
198                            aria-label="Search table"
199                            class="pl-8"
200                        }
201                    )
202                </div>
203                button(
204                    variant: ButtonVariant::Outline,
205                    size: ButtonSize::Md,
206                    attrs: attributes! { type="submit" },
207                    "Search"
208                )
209                if let Some(url) = clear_url {
210                    <a href=(url) class=(QUIET_LINK_CLASS)>"Clear"</a>
211                }
212            </form>
213        }
214        .boxed())
215    }

217    /// Eager live-search input for live tables: the signal-backed
218    /// input plus the GET form as `<noscript>` fallback. Rendered eagerly
219    /// above the streamed region; the shard invocation that fills the table
220    /// lives in the streamed region (`Self::render_live_invocation`) so the
221    /// table can only ever render once per response.
222    ///
223    /// The visible input is deliberately unbound: typing stays
224    /// local until it pauses for `LIVE_SEARCH_DEBOUNCE_MS`, then
225    /// `assets/live-search.js` rewrites `q` in the hidden transport bound to
226    /// the `query` signal and drops the cursor (a new term is a new result
227    /// set). The shard re-renders in place.
228    ///
229    /// The panel's live list page (`panel::resource_list_live`) renders it
230    /// from the request's one normalized state.
231    pub(crate) async fn render_live_search_bar<'a>(
232        &self,
233        cx: &'a Cx,
234        state: &TableState,
235        path: &str,
236        signals: &TableSignals,
237    ) -> Result<BoxView<'a>> {
238        let fallback = self.render_search_bar(cx, state, path).await?;
239        let q_display = state.search.clone().unwrap_or_default();
240        let query = signals.query.clone();
241        // The query prefix a relation's parameters carry (`comments.`), so
242        // `live-search.js` rewrites the table's own `q` and cursor instead of
243        // the list's. Absent on a page-owned list, whose parameters are bare.
244        let query_prefix = state.prefix.as_deref().map(|prefix| format!("{prefix}."));
245        Ok(view! {
246            cx =>
247            <div class=(BAR_CLASS) data-live-search="" data-query-prefix=(query_prefix)>
248                <div class=(SEARCH_FIELD_CLASS)>
249                    icon(
250                        data: tablo_ui::icons::SEARCH,
251                        attrs: attributes! { class=(SEARCH_ICON_CLASS) }
252                    )
253                    ui_input(
254                        attrs: attributes! {
255                            type="search"
256                            value=(q_display)
257                            placeholder="Search…"
258                            aria-label="Live search table"
259                            class="pl-8"
260                            data-live-search-input=""
261                            data-debounce-ms=(LIVE_SEARCH_DEBOUNCE_MS)
262                        }
263                    )
264                </div>
265                <input
266                    type="hidden"
267                    :value=$(query.get())
268                    @change=$(|e: Event| query.set(e.target.value))
269                    data-live-search-transport=""
270                >
271                <noscript>(fallback)</noscript>
272            </div>
273        }
274        .boxed())
275    }

277    /// The `table_search` shard invocation filling a live table's streamed
278    /// region. The signal handles travel as arguments; every
279    /// tracked read inside the shard becomes a `dep` marker the browser
280    /// watches, so sort/filter/pager/search changes re-render the table in
281    /// place.
282    pub(crate) async fn render_live_invocation<'a>(
283        &self,
284        cx: &'a Cx,
285        path: &str,
286        signals: TableSignals,
287    ) -> Result<BoxView<'a>> {
288        use crate::panel::table_search;
289
290        // No snapshot here: grouping travels in the query (seeded from the
291        // request's query by the caller) and the shard normalizes on read.
292        let live_path = path.to_string();
293        let TableSignals { query, bulk } = signals;
294        Ok(view! {
295            cx =>
296            table_search(path: $(live_path.clone()), query: $(query), bulk: $(bulk))
297        }
298        .boxed())
299    }

301    /// The `table_relation_search` shard invocation filling a live relation
302    /// table's streamed region. The signal handles travel as arguments; every
303    /// tracked read inside the shard becomes a `dep` marker the browser
304    /// watches, so sort/filter/pager/search changes re-render the table in
305    /// place.
306    pub(crate) async fn render_live_relation_invocation<'a>(
307        &self,
308        cx: &'a Cx,
309        parent: &str,
310        child: &str,
311        ctx: crate::panel::RelationRequest,
312        signals: TableSignals,
313    ) -> Result<BoxView<'a>> {
314        use crate::panel::table_relation_search;
315
316        // No snapshot here: grouping travels in the query (seeded from the
317        // request's query by the caller) and the shard normalizes on read.
318        // Slugs never carry `/`, so the pair and the seed travel as one wire
319        // arg, split off the front by the shard.
320        let crate::panel::RelationRequest {
321            seed,
322            page: live_page,
323            read_only,
324        } = ctx;
325        let scope = format!("{parent}/{child}/{seed}");
326        let TableSignals { query, bulk } = signals;
327        Ok(view! {
328            cx =>
329            table_relation_search(
330                scope: $(scope.clone()),
331                page: $(live_page.clone()),
332                read_only: $(read_only),
333                query: $(query),
334                bulk: $(bulk)
335            )
336        }
337        .boxed())
338    }
