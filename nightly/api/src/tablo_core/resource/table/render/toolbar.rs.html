15/// Keystroke-quiet delay before a live search input reloads the table
16/// (~150-250ms): `assets/live-search.js` waits this long after the
17/// last keystroke, then forwards the value through the bound transport below,
18/// so typing "published" triggers one reload instead of nine. The forwarded
19/// write is an ordinary signal write, so Topcoat's abort-in-flight
20/// coalescing still applies to the resulting rerun.
21pub(crate) const LIVE_SEARCH_DEBOUNCE_MS: u32 = 200;

23/// The hidden inputs a table toolbar carries across its submit, in the order
24/// given; an input whose state holds no value renders nothing.
25pub(super) fn hidden_state_inputs<'a>(
26    cx: &'a Cx,
27    inputs: Vec<(String, Option<String>)>,
28) -> BoxView<'a> {
29    let fields: Vec<BoxView<'a>> = inputs
30        .into_iter()
31        .filter_map(|(name, value)| {
32            value.map(|value| {
33                view! { cx => <input type="hidden" name=(name) value=(value)> }.boxed()
34            })
35        })
36        .collect();
37    view! {
38        cx =>
39        for field in fields {
40            (field)
41        }
42    }
43    .boxed()
44}

47    /// The bulk bar — the custom bulk actions, then bulk delete and its
48    /// confirmation dialog — or the
49    /// placeholder that keeps the chrome's node order stable when
50    /// [`Self::bulk_enabled`] is off.
51    ///
52    /// One destructive form for the whole page: the transport is fed by the
53    /// row checkboxes (`bulk.js`) and ships `,a,b,`-delimited, while on a live
54    /// table the selection lives in a signal instead, so a shard
55    /// rerun re-renders the transport from the selection rather than dropping
56    /// it. The trigger ships enabled: the confirmation dialog gates
57    /// the write and reads the selection when it opens, so an empty selection
58    /// is answered by the dialog rather than by a disabled control whose state
59    /// has to be kept in step across a live swap.
60    ///
61    /// The dialog lives *inside* the form so its `confirm` marker ships with
62    /// the same payload as the selection — the confirm button is an ordinary
63    /// submit of that form, and the handler refuses a POST without the marker.
64    /// It renders closed and opens client-side (`showModal`) rather than
65    /// through a runtime signal: the trigger is `type="button"`, so opening
66    /// the dialog is not a result-set change and must not reload the table.
67    pub(super) fn render_bulk_bar<'a>(
68        &self,
69        cx: &'a Cx,
70        signals: Option<&TableSignals>,
71    ) -> BoxView<'a> {
72        if !self.bulk_enabled() {
73            return view! { cx => <span></span> }.boxed();
74        }
75        let delete_prefix = self
76            .delete_prefix
77            .clone()
78            .filter(|_| self.bulk_delete_enabled());
79        // The list URL every bulk route hangs off: bulk delete's, else the
80        // custom actions'. `bulk_enabled` holds one or the other.
81        let prefix = delete_prefix
82            .clone()
83            .or_else(|| self.actions_prefix.clone())
84            .expect("bulk chrome rides the delete or the actions prefix (see bulk_enabled)");
85        // Each custom bulk action is a submit of this form to its own route
86        // (`formaction`), so it carries the same selection transport.
87        let custom: Vec<(String, String)> = self
88            .bulk_custom_actions()
89            .map(|action| {
90                (
91                    action.label.clone(),
92                    self.action_url(bulk_action_url(&prefix, action.name)),
93                )
94            })
95            .collect();
96        // The form's own target is bulk delete's when it renders, so the
97        // confirm dialog's submit reaches it; otherwise the first action's.
98        let bulk_action = match &delete_prefix {
99            Some(prefix) => self.action_url(bulk_delete_url(prefix)),
100            None => custom
101                .first()
102                .map(|(_, url)| url.clone())
103                .unwrap_or_default(),
104        };
105        let csrf = crate::csrf::current_token(cx);
106        // Stable ids so the dialog's confirm button can submit this form
107        // from inside the dialog.
108        let bulk_form_id = chrome_dom_id(&prefix, "bulk-form");
109        // Destructive confirm: a batch is the one place a misclick costs many
110        // rows, so it asks first — the same alert dialog the row delete uses.
111        // It sits inside the bulk form, so its controls submit that form.
112        let confirm = delete_prefix.as_ref().map(|_| {
113            confirm_dialog(
114                cx,
115                ConfirmDialog {
116                    id: chrome_dom_id(&prefix, "bulk-form-confirm"),
117                    open: false,
118                    title: "Delete the selected records?",
119                    attrs: attributes! { cx => data-bulk-confirm-dialog="" },
120                    description_attrs: attributes! { cx => data-bulk-confirm-description="" },
121                    footer: confirm_controls(cx),
122                },
123            )
124        });
125        let with_delete = confirm.is_some();
126        let custom_buttons: Vec<BoxView<'a>> = custom
127            .into_iter()
128            .map(|(label, url)| {
129                view! {
130                    cx =>
131                    button(
132                        variant: ButtonVariant::Outline,
133                        size: ButtonSize::Md,
134                        attrs: attributes! { type="submit" formaction=(url) data-bulk-action="" },
135                        (label)
136                    )
137                }
138                .boxed()
139            })
140            .collect();
141        // No visible `ids` field: the transport is fed by the row
142        // checkboxes (`bulk.js`) and ships `,a,b,`-delimited. On a live
143        // table the selection lives in a signal instead, so a
144        // shard rerun re-renders the transport from the selection rather
145        // than dropping it.
146        let transport_attrs = match signals {
147            Some(signals) => {
148                let bulk = signals.bulk.clone();
149                attributes! {
150                    cx =>
151                    type="hidden"
152                    name="ids"
153                    :value=$(bulk.get())
154                    @change=$(|e: Event| bulk.set(e.target.value))
155                }
156            }
157            None => attributes! { cx => type="hidden" name="ids" value="" },
158        };
159        view! {
160            cx =>
161            <form
162                method="post"
163                action=(bulk_action)
164                class="ml-auto flex items-center gap-2"
165                data-bulk-form=""
166                data-mutation-submit=""
167                id=(bulk_form_id.clone())
168            >
169                (crate::csrf::field(cx, &csrf))
170                <input (transport_attrs)>
171                for custom_button in custom_buttons {
172                    (custom_button)
173                }
174                if with_delete {
175                    button(
176                        variant: ButtonVariant::Outline,
177                        size: ButtonSize::Md,
178                        attrs: attributes! { type="button" data-bulk-confirm-trigger="" },
179                        icon(
180                            data: tablo_ui::icons::TRASH,
181                            attrs: attributes! { class="text-destructive" }
182                        )
183                        "Delete selected"
184                    )
185                }
186                if let Some(confirm) = confirm {
187                    (confirm)
188                }
189            </form>
190        }
191        .boxed()
192    }

194    /// The search toolbar (GET form); live tables instead render the host
195    /// input eagerly and the shard invocation in the streamed region (see
196    /// [`Self::render_live_search_bar`] / [`Self::render_live_invocation`]).
197    pub(super) async fn render_search_bar<'a>(
198        &self,
199        cx: &'a Cx,
200        state: &TableState,
201        path: &str,
202    ) -> Result<BoxView<'a>> {
203        let action = path.to_string();
204        let q_display = state.search.clone().unwrap_or_default();
205        let sort_hidden = state.sort.as_ref().map(|s| s.column.clone());
206        let dir_hidden = state.sort.as_ref().map(|s| {
207            if s.descending {
208                "desc".to_string()
209            } else {
210                "asc".to_string()
211            }
212        });
213        // Pre-normalized by the render seams: `state.group_by` is
214        // the declared name or `None`, never an unknown value.
215        let group_hidden = state.group_by.clone();
216        // Clear only renders when something survives the search term; every
217        // branch below projects the same URL, so one intent serves all three.
218        let clear_url =
219            (state.sort.is_some() || !state.filters.is_empty() || group_hidden.is_some())
220                .then(|| state.without_search(path));
221        let mut inputs = vec![
222            (state.param("sort"), sort_hidden),
223            (state.param("dir"), dir_hidden),
224        ];
225        inputs.extend(
226            state
227                .filters
228                .iter()
229                .map(|(name, value)| (state.filter_param(name), Some(value.clone()))),
230        );
231        inputs.push((state.param("group_by"), group_hidden));
232        let hidden = hidden_state_inputs(cx, inputs);
233        let search_name = state.param("q");
234        Ok(view! {
235            cx =>
236            <form method="get" action=(action) class=(SEARCH_FORM_CLASS)>
237                (hidden)
238                <div class=(SEARCH_FIELD_CLASS)>
239                    icon(
240                        data: tablo_ui::icons::SEARCH,
241                        attrs: attributes! { class=(SEARCH_ICON_CLASS) }
242                    )
243                    ui_input(
244                        attrs: attributes! {
245                            type="search"
246                            name=(search_name)
247                            value=(q_display)
248                            placeholder="Search…"
249                            aria-label="Search table"
250                            class="pl-8"
251                        }
252                    )
253                </div>
254                button(
255                    variant: ButtonVariant::Outline,
256                    size: ButtonSize::Md,
257                    attrs: attributes! { type="submit" },
258                    "Search"
259                )
260                if let Some(url) = clear_url {
261                    <a href=(url) class=(QUIET_LINK_CLASS)>"Clear"</a>
262                }
263            </form>
264        }
265        .boxed())
266    }

268    /// Eager live-search input for live tables: the signal-backed
269    /// input plus the GET form as `<noscript>` fallback. Rendered eagerly
270    /// above the streamed region; the shard invocation that fills the table
271    /// lives in the streamed region (`Self::render_live_invocation`) so the
272    /// table can only ever render once per response.
273    ///
274    /// The visible input is deliberately unbound: typing stays
275    /// local until it pauses for `LIVE_SEARCH_DEBOUNCE_MS`, then
276    /// `assets/live-search.js` rewrites `q` in the hidden transport bound to
277    /// the `query` signal and drops the cursor (a new term is a new result
278    /// set). The shard re-renders in place.
279    ///
280    /// The panel's live list page (`panel::resource_list_live`) renders it
281    /// from the request's one normalized state.
282    pub(crate) async fn render_live_search_bar<'a>(
283        &self,
284        cx: &'a Cx,
285        state: &TableState,
286        path: &str,
287        signals: &TableSignals,
288    ) -> Result<BoxView<'a>> {
289        let fallback = self.render_search_bar(cx, state, path).await?;
290        let q_display = state.search.clone().unwrap_or_default();
291        let query = signals.query.clone();
292        // The query prefix a relation's parameters carry (`comments.`), so
293        // `live-search.js` rewrites the table's own `q` and cursor instead of
294        // the list's. Absent on a page-owned list, whose parameters are bare.
295        let query_prefix = state.prefix.as_deref().map(|prefix| format!("{prefix}."));
296        Ok(view! {
297            cx =>
298            <div class=(BAR_CLASS) data-live-search="" data-query-prefix=(query_prefix)>
299                <div class=(SEARCH_FIELD_CLASS)>
300                    icon(
301                        data: tablo_ui::icons::SEARCH,
302                        attrs: attributes! { class=(SEARCH_ICON_CLASS) }
303                    )
304                    ui_input(
305                        attrs: attributes! {
306                            type="search"
307                            value=(q_display)
308                            placeholder="Search…"
309                            aria-label="Live search table"
310                            class="pl-8"
311                            data-live-search-input=""
312                            data-debounce-ms=(LIVE_SEARCH_DEBOUNCE_MS)
313                        }
314                    )
315                </div>
316                <input
317                    type="hidden"
318                    :value=$(query.get())
319                    @change=$(|e: Event| query.set(e.target.value))
320                    data-live-search-transport=""
321                >
322                <noscript>(fallback)</noscript>
323            </div>
324        }
325        .boxed())
326    }

328    /// The `table_search` shard invocation filling a live table's streamed
329    /// region. The signal handles travel as arguments; every
330    /// tracked read inside the shard becomes a `dep` marker the browser
331    /// watches, so sort/filter/pager/search changes re-render the table in
332    /// place.
333    pub(crate) async fn render_live_invocation<'a>(
334        &self,
335        cx: &'a Cx,
336        path: &str,
337        signals: TableSignals,
338    ) -> Result<BoxView<'a>> {
339        use crate::panel::table_search;
340
341        // No snapshot here: grouping travels in the query (seeded from the
342        // request's query by the caller) and the shard normalizes on read.
343        let live_path = path.to_string();
344        let TableSignals { query, bulk } = signals;
345        Ok(view! {
346            cx =>
347            table_search(path: $(live_path.clone()), query: $(query), bulk: $(bulk))
348        }
349        .boxed())
350    }

352    /// The `table_relation_search` shard invocation filling a live relation
353    /// table's streamed region. The signal handles travel as arguments; every
354    /// tracked read inside the shard becomes a `dep` marker the browser
355    /// watches, so sort/filter/pager/search changes re-render the table in
356    /// place.
357    pub(crate) async fn render_live_relation_invocation<'a>(
358        &self,
359        cx: &'a Cx,
360        parent: &str,
361        child: &str,
362        ctx: crate::panel::RelationRequest,
363        signals: TableSignals,
364    ) -> Result<BoxView<'a>> {
365        use crate::panel::table_relation_search;
366
367        // No snapshot here: grouping travels in the query (seeded from the
368        // request's query by the caller) and the shard normalizes on read.
369        // Slugs never carry `/`, so the pair and the seed travel as one wire
370        // arg, split off the front by the shard.
371        let crate::panel::RelationRequest {
372            seed,
373            page: live_page,
374            read_only,
375        } = ctx;
376        let scope = format!("{parent}/{child}/{seed}");
377        let TableSignals { query, bulk } = signals;
378        Ok(view! {
379            cx =>
380            table_relation_search(
381                scope: $(scope.clone()),
382                page: $(live_page.clone()),
383                read_only: $(read_only),
384                query: $(query),
385                bulk: $(bulk)
386            )
387        }
388        .boxed())
389    }
