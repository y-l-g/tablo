17/// One filter control: a labelled `<select data-filter-name=…>` carrying the
18/// `value`/`label` pairs, with the leading empty "All" option that clears the
19/// filter.
20///
21/// The empty value is reserved for that clear-filter option. Every pair in
22/// `options` renders verbatim, so a caller whose declared options can include
23/// `""` supplies the label that option shows: [`Filter::Select`] passes
24/// `"All"`, the label the empty value already carries, while
25/// [`Filter::Variant`] passes the key itself.
26///
27/// The control has no `name`, so it never submits on its own; `filters.js`
28/// composes it into the hidden `filters` transport.
29fn filter_select<'a>(
30    cx: &'a Cx,
31    label: &str,
32    name: &str,
33    options: Vec<(String, String)>,
34    current: &str,
35) -> BoxView<'a> {
36    let label = label.to_string();
37    let name = name.to_string();
38    let aria = label.clone();
39    let current = current.to_string();
40    view! {
41        cx =>
42        <label class="flex items-center gap-2 text-sm text-muted-foreground">
43            (label)
44            <select
45                data-filter-name=(name)
46                aria-label=(aria)
47                class="flex h-9 rounded-md border border-border bg-background px-3 py-1 text-sm shadow-xs"
48            >
49                <option value="" selected=(current.is_empty())>"All"</option>
50                for (value, text) in options {
51                    <option value=(value.clone()) selected=(current == value)>
52                        (text)
53                    </option>
54                }
55            </select>
56        </label>
57    }
58    .boxed()
59}

61impl<M> Table<M> {
62    /// The fail-visible filter banner: requested filters that produced
63    /// no predicate render as a `role=alert` banner; the list keeps a 200 while
64    /// the export refuses with 400 (see `resource_export`).
65    ///
66    /// No false tail: when other filters still apply, "unfiltered" would be a
67    /// lie — a malformed segment can ride alongside valid ones.
68    /// Conversely an invalid-only request applies nothing, so "other filter(s)"
69    /// would be the lie — the consequence keys off applied
70    /// predicates, not raw entries.
71    pub(super) fn render_filter_warning<'a>(
72        &self,
73        cx: &'a Cx,
74        state: &NormalizedState,
75        path: &str,
76    ) -> Option<BoxView<'a>>
77    where
78        M: toasty::schema::Model,
79    {
80        let unapplied = self.unapplied_filters(state);
81        if unapplied.is_empty() {
82            return None;
83        }
84        let detail = unapplied
85            .iter()
86            .map(|(pair, reason)| format!("{pair} ({reason})"))
87            .collect::<Vec<_>>()
88            .join(", ");
89        let consequence = if self.filter_expr(state).is_none() {
90            "showing unfiltered results"
91        } else {
92            "other filter(s) still apply"
93        };
94        let text = format!("Ignored filter(s): {detail} — {consequence}.");
95        let clear = state.without_filters(path);
96        Some(
97            view! {
98                cx =>
99                <div
100                    class="border-b border-destructive/30 bg-muted px-4 py-2 text-sm"
101                    role="alert"
102                >
103                    (text)
104                    " "
105                    <a href=(clear) class="underline">"Clear filters"</a>
106                </div>
107            }
108            .boxed(),
109        )
110    }

112    /// The filter bar for a live table, rendered eagerly by the page that owns
113    /// the signals — the counterpart of [`Self::render_live_search_bar`].
114    ///
115    /// Hoisting matters for focus: a `<select>` change writes the `filters`
116    /// signal, and a bar rebuilt by that rerun would collapse the native popup
117    /// and drop keyboard context. The table renders without the bar
118    /// (`Table::filter_bar(false)`), so the control the user touched is never
119    /// replaced.
120    pub async fn render_live_filter_bar<'a>(
121        &self,
122        cx: &'a Cx,
123        state: &TableState,
124        path: &str,
125        signals: &TableSignals,
126    ) -> Result<BoxView<'a>>
127    where
128        M: toasty::schema::Model,
129    {
130        // Called with raw page state: normalize so the no-JS
131        // fallback form carries the same normalized values the GET path would.
132        self.render_live_filter_bar_normalized(cx, &self.normalize_state(state), path, signals)
133            .await
134    }

136    /// [`Self::render_live_filter_bar`] with the state already normalized
137    /// the panel's live page renders the hoisted bar from the
138    /// request's one normalized state.
139    pub(crate) async fn render_live_filter_bar_normalized<'a>(
140        &self,
141        cx: &'a Cx,
142        state: &NormalizedState,
143        path: &str,
144        signals: &TableSignals,
145    ) -> Result<BoxView<'a>>
146    where
147        M: toasty::schema::Model,
148    {
149        self.render_filter_bar(cx, state, path, Some(signals)).await
150    }

152    /// The typed filter bar. For live tables (`signals`) the hidden `filters`
153    /// transport is bound to the `filters` signal and `filters.js` dispatches
154    /// a `change` into it instead of submitting, so the shard re-renders the
155    /// table in place; the GET form stays as the no-JS fallback and `href`s
156    /// remain real.
157    pub(super) async fn render_filter_bar<'a>(
158        &self,
159        cx: &'a Cx,
160        state: &TableState,
161        path: &str,
162        signals: Option<&TableSignals>,
163    ) -> Result<BoxView<'a>>
164    where
165        M: toasty::schema::Model,
166    {
167        // No `filters.is_empty()` early return: the caller (`render_inner`)
168        // already guards on `show_filters`, so an empty bar is unreachable.
169        let action = path.to_string();
170        let filters_display = state.filters_param().unwrap_or_default();
171        let sort_hidden = state.sort.as_ref().map(|s| s.column.clone());
172        let dir_hidden = state.sort.as_ref().map(|s| {
173            if s.descending {
174                "desc".to_string()
175            } else {
176                "asc".to_string()
177            }
178        });
179        let q_hidden = state.search.clone();
180        let group_hidden = state.group_by.clone();
181        let hidden = hidden_state_inputs(
182            cx,
183            vec![
184                ("q", q_hidden),
185                ("sort", sort_hidden),
186                ("dir", dir_hidden),
187                ("group_by", group_hidden),
188            ],
189        );
190        let clear_url = if !state.filters.is_empty() {
191            Some(state.without_filters(path))
192        } else {
193            None
194        };
195        // One typed control per declared filter. Controls carry only
196        // `data-filter-name` (no `name`, so they never submit on their own);
197        // `filters.js` composes them into the hidden `filters` transport and
198        // submits on change, rewriting it even when every control is
199        // "All" so the stale value can never be resubmitted. The free-text
200        // input and Apply button survive only inside `<noscript>` as the
201        // no-JS fallback.
202        let mut controls: Vec<BoxView<'_>> = Vec::with_capacity(self.filters.len());
203        for f in &self.filters {
204            let current = state.filters.get(f.name()).cloned().unwrap_or_default();
205            match f {
206                Filter::Select(s) => {
207                    // A declared empty option is the clear-filter value, so it
208                    // renders as the "All" option: value `""`, label "All".
209                    let options = s
210                        .options()
211                        .iter()
212                        .map(|opt| {
213                            let label = if opt.is_empty() { "All" } else { opt.as_str() };
214                            (opt.clone(), label.to_string())
215                        })
216                        .collect();
217                    controls.push(filter_select(
218                        cx,
219                        s.label_str(),
220                        s.name(),
221                        options,
222                        &current,
223                    ));
224                }
225                Filter::Ternary(t) => {
226                    let options = vec![
227                        ("true".to_string(), "True".to_string()),
228                        ("false".to_string(), "False".to_string()),
229                    ];
230                    controls.push(filter_select(
231                        cx,
232                        t.label_str(),
233                        t.name(),
234                        options,
235                        &current,
236                    ));
237                }
238                Filter::Date(d) => {
239                    let name = d.name().to_string();
240                    let label = d.label_str().to_string();
241                    let aria = label.clone();
242                    // `<input type=date>` needs YYYY-MM-DD; truncate RFC3339.
243                    let date_value = current.split('T').next().unwrap_or(&current).to_string();
244                    controls.push(
245                        view! {
246                            cx =>
247                            <label
248                                class="flex items-center gap-2 text-sm text-muted-foreground"
249                            >
250                                (label)
251                                <input
252                                    type="date"
253                                    data-filter-name=(name)
254                                    value=(date_value)
255                                    aria-label=(aria)
256                                    class="flex h-9 rounded-md border border-border bg-background px-3 py-1 text-sm shadow-xs"
257                                >
258                            </label>
259                        }
260                        .boxed(),
261                    );
262                }
263                Filter::Variant(v) => {
264                    let options = v
265                        .options()
266                        .iter()
267                        .map(|(key, _)| (key.clone(), key.clone()))
268                        .collect();
269                    controls.push(filter_select(
270                        cx,
271                        v.label_str(),
272                        v.name(),
273                        options,
274                        &current,
275                    ));
276                }
277            }
278        }
279        let form_attrs = attributes! {
280            cx =>
281            method="get"
282            action=(action)
283            class="flex flex-wrap items-center gap-2 border-b border-border p-3"
284            data-filters-form=""
285            if signals.is_some() {
286                data-filters-live=""
287            }
288        };
289        // Live tables bind the transport to the `filters` signal: `filters.js`
290        // composes and dispatches, the shard re-renders in place. Static
291        // tables keep the server-rendered value the GET form submits.
292        let transport_attrs = if let Some(signals) = signals {
293            let (filters, cursor) = (signals.filters.clone(), signals.cursor.clone());
294            let none = crate::resource::cursor_none();
295            attributes! {
296                cx =>
297                name="filters"
298                :value=$(filters.get())
299                @change=$(|e: Event| {
300                    filters.set(e.target.value);
301                    cursor.set(none.clone());
302                })
303                data-filters-transport=""
304            }
305        } else {
306            attributes! {
307                cx =>
308                name="filters"
309                value=(filters_display.clone())
310                data-filters-transport=""
311            }
312        };
313        let clear_link: Option<BoxView<'a>> = clear_url.map(|url| {
314            let attrs = match signals {
315                Some(signals) => {
316                    let (filters, cursor) = (signals.filters.clone(), signals.cursor.clone());
317                    let none = crate::resource::cursor_none();
318                    attributes! {
319                        cx =>
320                        href=(url.clone())
321                        @click=$(|e: Event| {
322                            e.prevent_default();
323                            filters.set("".to_owned());
324                            cursor.set(none.clone());
325                        })
326                    }
327                }
328                None => attributes! { cx => href=(url) },
329            };
330            view! {
331                cx =>
332                <a class="text-sm text-muted-foreground hover:text-foreground" (attrs)>
333                    "Clear filters"
334                </a>
335            }
336            .boxed()
337        });
338        Ok(view! {
339            cx =>
340            <form (form_attrs)>
341                (hidden)
342                for ctl in controls {
343                    (ctl)
344                }
345                <noscript>
346                    ui_input(
347                        attrs: attributes! {
348                            type="text"
349                            name="filters"
350                            value=(filters_display)
351                            placeholder="filters e.g. status:published"
352                            aria-label="Filter table (free text)"
353                            class="w-64"
354                        }
355                    )
356                    button(
357                        variant: ButtonVariant::Secondary,
358                        size: ButtonSize::Md,
359                        attrs: attributes! { type="submit" },
360                        "Apply filters"
361                    )
362                </noscript>
363                <input type="hidden" (transport_attrs)>
364                if let Some(link) = clear_link {
365                    (link)
366                }
367            </form>
368        }
369        .boxed())
370    }

390    async fn filter_widgets_render_typed_controls() {
391        let cx = CxTestBuilder::new().build();
392        let table_task1 = Table::<Task>::r#for(&cx)
393            .key(|t| t.id.to_string())
394            .columns(TextColumn::r#for(Task::fields().title(), |t| {
395                t.title.clone()
396            }))
397            .filters((
398                SelectFilter::r#for(
399                    Task::fields().status(),
400                    vec!["draft".to_string(), "published".to_string()],
401                ),
402                TernaryFilter::r#for(Task::fields().featured()),
403                DateFilter::r#for(Task::fields().created_at()),
404            ));
405        let page: TablePage<Task> = Vec::new().into();
406        // State with an active select value pre-selects it.
407        let mut filters = HashMap::new();
408        filters.insert("status".to_string(), "published".to_string());
409        let state = TableState {
410            filters,
411            ..TableState::default()
412        };
413        let html = table_task1
414            .render_with_state(&cx, page, &state, "/admin/tasks")
415            .await
416            .unwrap()
417            .single()
418            .await
419            .unwrap()
420            .render(&cx);
421        assert!(
422            html.contains("data-filters-form"),
423            "missing filters form in {html}"
424        );
425        for name in ["status", "featured", "created_at"] {
426            assert!(
427                html.contains(&format!("data-filter-name=\"{name}\"")),
428                "missing control for {name} in {html}"
429            );
430        }
431        // Select options + current selection.
432        assert!(
433            html.contains("draft") && html.contains("published"),
434            "missing select options in {html}"
435        );
436        assert!(
437            html.contains("value=\"published\" selected")
438                || html.contains("value=\"published\" selected=\"\""),
439            "published should be selected in {html}"
440        );
441        // Ternary + date controls.
442        assert!(
443            html.contains("value=\"true\"") && html.contains("value=\"false\""),
444            "missing ternary options in {html}"
445        );
446        assert!(
447            html.contains("type=\"date\""),
448            "missing date input in {html}"
449        );
450        // The hidden transport carries the composed value for auto-apply; the
451        // free-text input + Apply button survive only as the `<noscript>`
452        // fallback.
453        assert!(
454            html.contains("data-filters-transport")
455                && html.contains("name=\"filters\"")
456                && html.contains("status:published"),
457            "missing hidden filters transport in {html}"
458        );
459        assert!(
460            html.contains("<noscript>") && html.contains("Apply filters"),
461            "missing no-JS filter fallback in {html}"
462        );
