18/// One filter control: a labelled `<select name="f.<name>">` (prefixed like the
19/// table's other parameters, `param`) carrying the
20/// `value`/`label` pairs, with the leading empty "All" option that clears the
21/// filter.
22///
23/// The empty value is reserved for that clear-filter option. Every pair in
24/// `options` renders verbatim, so a caller whose declared options can include
25/// `""` supplies the label that option shows: [`Filter::Select`] passes
26/// `"All"`, the label the empty value already carries, while
27/// [`Filter::Variant`] passes the key itself.
28///
29/// The control is a real form field, so the GET form submits it as
30/// `?f.<name>=<value>`; an empty value is no filter. `data-filter-name` is the
31/// hook `filters.js` reads on a live table.
32fn filter_select<'a>(
33    cx: &'a Cx,
34    label: &str,
35    name: &str,
36    param: String,
37    options: Vec<(String, String)>,
38    current: &str,
39) -> BoxView<'a> {
40    let label = label.to_string();
41    let name = name.to_string();
42    let aria = label.clone();
43    let option_views: Vec<BoxView<'a>> = std::iter::once((String::new(), "All".to_string()))
44        .chain(options)
45        .map(|(value, text)| {
46            let selected = current == value;
47            crate::schema::option_view(cx, value, text, selected)
48        })
49        .collect();
50    view! {
51        cx =>
52        <label class="flex items-center gap-2 text-sm text-muted-foreground">
53            (label)
54            <select
55                name=(param)
56                data-filter-name=(name)
57                aria-label=(aria)
58                class="flex h-9 rounded-md border border-border bg-background px-3 py-1 text-sm shadow-xs"
59            >
60                for option in option_views {
61                    (option)
62                }
63            </select>
64        </label>
65    }
66    .boxed()
67}

69impl<M> Table<M> {
70    /// The fail-visible filter banner: requested filters that produced
71    /// no predicate render as a `role=alert` banner; the list keeps a 200 while
72    /// the export refuses with 400 (see `resource_export`).
73    ///
74    /// No false tail: when other filters still apply, "unfiltered" would be a
75    /// lie — a malformed segment can ride alongside valid ones.
76    /// Conversely an invalid-only request applies nothing, so "other filter(s)"
77    /// would be the lie — the consequence keys off applied
78    /// predicates, not raw entries.
79    pub(super) fn render_filter_warning<'a>(
80        &self,
81        cx: &'a Cx,
82        state: &TableState,
83        path: &str,
84    ) -> Option<BoxView<'a>>
85    where
86        M: toasty::schema::Model,
87    {
88        let unapplied = self.unapplied_filters(state);
89        if unapplied.is_empty() {
90            return None;
91        }
92        let detail = unapplied
93            .iter()
94            .map(|(pair, reason)| format!("{pair} ({reason})"))
95            .collect::<Vec<_>>()
96            .join(", ");
97        let consequence = if self.filter_expr(state).is_none() {
98            "showing unfiltered results"
99        } else {
100            "other filter(s) still apply"
101        };
102        let text = format!("Ignored filter(s): {detail} — {consequence}.");
103        let clear = state.without_filters(path);
104        Some(
105            view! {
106                cx =>
107                <div
108                    class="border-b border-destructive/30 bg-muted px-4 py-2 text-sm"
109                    role="alert"
110                >
111                    (text)
112                    " "
113                    <a href=(clear) class="underline">"Clear filters"</a>
114                </div>
115            }
116            .boxed(),
117        )
118    }

120    /// The filter bar for a live table, rendered eagerly by the page that owns
121    /// the signals — the counterpart of [`Self::render_live_search_bar`].
122    ///
123    /// Hoisting matters for focus: a `<select>` change writes the `query`
124    /// signal, and a bar rebuilt by that rerun would collapse the native popup
125    /// and drop keyboard context. The table renders without the bar
126    /// (`Table::hide_filter_bar`), so the control the user touched is never
127    /// replaced.
128    pub(crate) async fn render_live_filter_bar<'a>(
129        &self,
130        cx: &'a Cx,
131        state: &TableState,
132        path: &str,
133        signals: &TableSignals,
134    ) -> Result<BoxView<'a>>
135    where
136        M: toasty::schema::Model,
137    {
138        self.render_filter_bar(cx, state, path, Some(signals)).await
139    }
140
141    /// The typed filter bar: a GET form whose controls are the `f.<name>`
142    /// parameters. `filters.js` submits it on change. For live tables
143    /// (`signals`) a hidden transport is bound to the `query` signal instead,
144    /// and `filters.js` rewrites the query's filter parameters in it, so the
145    /// shard re-renders the table in place; the form stays the no-JS fallback
146    /// and `href`s remain real.
147    pub(super) async fn render_filter_bar<'a>(
148        &self,
149        cx: &'a Cx,
150        state: &TableState,
151        path: &str,
152        signals: Option<&TableSignals>,
153    ) -> Result<BoxView<'a>>
154    where
155        M: toasty::schema::Model,
156    {
157        // No `filters.is_empty()` early return: the caller (`render_inner`)
158        // already guards on `show_filters`, so an empty bar is unreachable.
159        let action = path.to_string();
160        let sort_hidden = state.sort.as_ref().map(|s| s.column.clone());
161        let dir_hidden = state.sort.as_ref().map(|s| {
162            if s.descending {
163                "desc".to_string()
164            } else {
165                "asc".to_string()
166            }
167        });
168        let q_hidden = state.search.clone();
169        let group_hidden = state.group_by.clone();
170        let hidden = hidden_state_inputs(
171            cx,
172            vec![
173                (state.param("q"), q_hidden),
174                (state.param("sort"), sort_hidden),
175                (state.param("dir"), dir_hidden),
176                (state.param("group_by"), group_hidden),
177            ],
178        );
179        let clear_url = if !state.filters.is_empty() {
180            Some(state.without_filters(path))
181        } else {
182            None
183        };
184        // One typed control per declared filter, each a real `f.<name>` field.
185        // `filters.js` submits on change; the Apply button survives only
186        // inside `<noscript>` as the no-JS path.
187        let mut controls: Vec<BoxView<'_>> = Vec::with_capacity(self.filters.len());
188        for f in &self.filters {
189            let current = state.filters.get(f.name()).cloned().unwrap_or_default();
190            match f {
191                Filter::Select(s) => {
192                    // A declared empty option is the clear-filter value, so it
193                    // renders as the "All" option: value `""`, label "All".
194                    let options = s
195                        .options()
196                        .iter()
197                        .map(|opt| {
198                            let label = if opt.is_empty() { "All" } else { opt.as_str() };
199                            (opt.clone(), label.to_string())
200                        })
201                        .collect();
202                    controls.push(filter_select(
203                        cx,
204                        s.label_str(),
205                        s.name(),
206                        state.filter_param(s.name()),
207                        options,
208                        &current,
209                    ));
210                }
211                Filter::Ternary(t) => {
212                    let options = vec![
213                        ("true".to_string(), "True".to_string()),
214                        ("false".to_string(), "False".to_string()),
215                    ];
216                    controls.push(filter_select(
217                        cx,
218                        t.label_str(),
219                        t.name(),
220                        state.filter_param(t.name()),
221                        options,
222                        &current,
223                    ));
224                }
225                Filter::Date(d) => {
226                    let name = d.name().to_string();
227                    let param = state.filter_param(&name);
228                    let label = d.label_str().to_string();
229                    let aria = label.clone();
230                    // `<input type=date>` needs YYYY-MM-DD; truncate RFC3339.
231                    let date_value = current.split('T').next().unwrap_or(&current).to_string();
232                    controls.push(
233                        view! {
234                            cx =>
235                            <label
236                                class="flex items-center gap-2 text-sm text-muted-foreground"
237                            >
238                                (label)
239                                <input
240                                    type="date"
241                                    name=(param)
242                                    data-filter-name=(name)
243                                    value=(date_value)
244                                    aria-label=(aria)
245                                    class="flex h-9 rounded-md border border-border bg-background px-3 py-1 text-sm shadow-xs"
246                                >
247                            </label>
248                        }
249                        .boxed(),
250                    );
251                }
252                Filter::Variant(v) => {
253                    let options = v
254                        .options()
255                        .iter()
256                        .map(|(key, _)| (key.clone(), key.clone()))
257                        .collect();
258                    controls.push(filter_select(
259                        cx,
260                        v.label_str(),
261                        v.name(),
262                        state.filter_param(v.name()),
263                        options,
264                        &current,
265                    ));
266                }
267            }
268        }
269        let form_attrs = attributes! {
270            cx =>
271            method="get"
272            action=(action)
273            class=(BAR_CLASS)
274            data-filters-form=""
275            if signals.is_some() {
276                data-filters-live=""
277            }
278        };
279        // Live tables bind a transport to the `query` signal: `filters.js`
280        // rewrites the query's filter parameters in it and dispatches, and the
281        // shard re-renders in place. It has no `name`, so the GET form never
282        // submits it.
283        let transport: Option<BoxView<'a>> = signals.map(|signals| {
284            let query = signals.query.clone();
285            view! {
286                cx =>
287                <input
288                    type="hidden"
289                    :value=$(query.get())
290                    @change=$(|e: Event| query.set(e.target.value))
291                    data-filters-transport=""
292                >
293            }
294            .boxed()
295        });
296        // A plain link: the bar is rendered once, so on a live table
297        // `filters.js` clears the filters from the transport's current query
298        // rather than writing this page-load URL over newer state.
299        let clear_link: Option<BoxView<'a>> = clear_url.map(|url| {
300            view! {
301                cx =>
302                <a class=(QUIET_LINK_CLASS) href=(url) data-filters-clear="">
303                    "Clear filters"
304                </a>
305            }
306            .boxed()
307        });
308        Ok(view! {
309            cx =>
310            <form (form_attrs)>
311                (hidden)
312                for ctl in controls {
313                    (ctl)
314                }
315                <noscript>
316                    button(
317                        variant: ButtonVariant::Secondary,
318                        size: ButtonSize::Md,
319                        attrs: attributes! { type="submit" },
320                        "Apply filters"
321                    )
322                </noscript>
323                if let Some(transport) = transport {
324                    (transport)
325                }
326                if let Some(link) = clear_link {
327                    (link)
328                }
329            </form>
330        }
331        .boxed())
332    }
