18impl<M> Table<M> {
19    /// The fail-visible filter banner: requested filters that produced
20    /// no predicate render as a `role=alert` banner; the list keeps a 200 while
21    /// the export refuses with 400 (see `resource_export`).
22    ///
23    /// No false tail: when other filters still apply, "unfiltered" would be a
24    /// lie — a malformed segment can ride alongside valid ones.
25    /// Conversely an invalid-only request applies nothing, so "other filter(s)"
26    /// would be the lie — the consequence keys off applied
27    /// predicates, not raw entries.
28    pub(super) fn render_filter_warning<'a>(
29        &self,
30        cx: &'a Cx,
31        state: &TableState,
32        path: &str,
33    ) -> Option<BoxView<'a>>
34    where
35        M: toasty::schema::Model,
36    {
37        let unapplied = self.unapplied_filters(state);
38        if unapplied.is_empty() {
39            return None;
40        }
41        let detail = unapplied
42            .iter()
43            .map(|(pair, reason)| format!("{pair} ({reason})"))
44            .collect::<Vec<_>>()
45            .join(", ");
46        let consequence = if self.filter_expr(state).is_none() {
47            "showing unfiltered results"
48        } else {
49            "other filter(s) still apply"
50        };
51        let text = format!("Ignored filter(s): {detail} — {consequence}.");
52        let clear = state.without_filters(path);
53        Some(
54            view! {
55                cx =>
56                <div
57                    class="border-b border-border bg-destructive/10 px-4 py-2 text-sm text-destructive"
58                    role="alert"
59                >
60                    (text)
61                    " "
62                    <a href=(clear) class="font-medium underline underline-offset-4">
63                        "Clear filters"
64                    </a>
65                </div>
66            }
67            .boxed(),
68        )
69    }

71    /// The filter bar for a live table, rendered eagerly by the page that owns
72    /// the signals — the counterpart of [`Self::render_live_search_bar`].
73    ///
74    /// Hoisting matters for focus: a `<select>` change writes the `query`
75    /// signal, and a bar rebuilt by that rerun would collapse the native popup
76    /// and drop keyboard context. The table renders without the bar
77    /// (`Table::hide_filter_bar`), so the control the user touched is never
78    /// replaced.
79    pub(crate) async fn render_live_filter_bar<'a>(
80        &self,
81        cx: &'a Cx,
82        state: &TableState,
83        path: &str,
84        signals: &TableSignals,
85    ) -> Result<BoxView<'a>>
86    where
87        M: toasty::schema::Model,
88    {
89        self.render_filter_bar(cx, state, path, Some(signals)).await
90    }
91
92    /// The typed filter bar: a GET form whose controls are the `f.<name>`
93    /// parameters. `filters.js` submits it on change. For live tables
94    /// (`signals`) a hidden transport is bound to the `query` signal instead,
95    /// and `filters.js` rewrites the query's filter parameters in it, so the
96    /// shard re-renders the table in place; the form stays the no-JS fallback
97    /// and `href`s remain real.
98    pub(super) async fn render_filter_bar<'a>(
99        &self,
100        cx: &'a Cx,
101        state: &TableState,
102        path: &str,
103        signals: Option<&TableSignals>,
104    ) -> Result<BoxView<'a>>
105    where
106        M: toasty::schema::Model,
107    {
108        // No `filters.is_empty()` early return: the caller (`render_inner`)
109        // already guards on `show_filters`, so an empty bar is unreachable.
110        let action = path.to_string();
111        let sort_hidden = state.sort.as_ref().map(|s| s.column.clone());
112        let dir_hidden = state.sort.as_ref().map(|s| {
113            if s.descending {
114                "desc".to_string()
115            } else {
116                "asc".to_string()
117            }
118        });
119        let q_hidden = state.search.clone();
120        let group_hidden = state.group_by.clone();
121        let hidden = hidden_state_inputs(
122            cx,
123            vec![
124                (state.param("q"), q_hidden),
125                (state.param("sort"), sort_hidden),
126                (state.param("dir"), dir_hidden),
127                (state.param("group_by"), group_hidden),
128            ],
129        );
130        let clear_url = if !state.filters.is_empty() {
131            Some(state.without_filters(path))
132        } else {
133            None
134        };
135        // One typed control per declared filter, each a real `f.<name>` field.
136        // `filters.js` submits on change; the Apply button survives only
137        // inside `<noscript>` as the no-JS path.
138        // The query prefix a relation's parameters carry (`comments.`), so
139        // `filters.js` rewrites the table's own filters and cursor instead of
140        // the list's. Absent on a page-owned list, whose parameters are bare.
141        let query_prefix = state.prefix.as_deref().map(|prefix| format!("{prefix}."));
142        let mut controls: Vec<BoxView<'_>> = Vec::with_capacity(self.filters.len());
143        for f in &self.filters {
144            let current = state.filters.get(f.name()).cloned().unwrap_or_default();
145            let input =
146                FilterInput::new(f.name(), state.filter_param(f.name()), f.label(), current);
147            controls.push(f.control(cx, input));
148        }
149        let form_attrs = attributes! {
150            cx =>
151            method="get"
152            action=(action)
153            class=(BAR_CLASS)
154            data-filters-form=""
155            data-query-prefix=(query_prefix)
156            if signals.is_some() {
157                data-filters-live=""
158            }
159        };
160        // Live tables bind a transport to the `query` signal: `filters.js`
161        // rewrites the query's filter parameters in it and dispatches, and the
162        // shard re-renders in place. It has no `name`, so the GET form never
163        // submits it.
164        let transport: Option<BoxView<'a>> = signals.map(|signals| {
165            let query = signals.query.clone();
166            view! {
167                cx =>
168                <input
169                    type="hidden"
170                    :value=$(query.get())
171                    @change=$(|e: Event| query.set(e.target.value))
172                    data-filters-transport=""
173                >
174            }
175            .boxed()
176        });
177        // A plain link: the bar is rendered once, so on a live table
178        // `filters.js` clears the filters from the transport's current query
179        // rather than writing this page-load URL over newer state.
180        let clear_link: Option<BoxView<'a>> = clear_url.map(|url| {
181            view! {
182                cx =>
183                <a class=(QUIET_LINK_CLASS) href=(url) data-filters-clear="">
184                    "Clear filters"
185                </a>
186            }
187            .boxed()
188        });
189        Ok(view! {
190            cx =>
191            <form (form_attrs)>
192                (hidden)
193                for ctl in controls {
194                    (ctl)
195                }
196                <noscript>
197                    button(
198                        variant: ButtonVariant::Outline,
199                        size: ButtonSize::Md,
200                        attrs: attributes! { type="submit" },
201                        "Apply filters"
202                    )
203                </noscript>
204                if let Some(transport) = transport {
205                    (transport)
206                }
207                if let Some(link) = clear_link {
208                    (link)
209                }
210            </form>
211        }
212        .boxed())
213    }
