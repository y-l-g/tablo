19/// What every row of one render shares: the chrome columns, the declared
20/// widths (resolved once per render, the same for every row), and the one
21/// delete dialog every row control opens.
22pub(super) struct RowChrome {
23    pub(super) with_bulk: bool,
24    pub(super) with_actions: bool,
25    /// Columns a group header spans: the data columns plus the chrome.
26    pub(super) header_colspan: usize,
27    pub(super) cell_widths: Vec<Option<Cow<'static, str>>>,
28    pub(super) actions_min: Option<Cow<'static, str>>,
29    pub(super) delete_dialog_id: String,
30}

32/// One rendered row, keyed for the body's diff.
33pub(super) struct RenderedRow<'a> {
34    /// The row's display key: stable for the record, never a loop index.
35    pub(super) key: String,
36    /// The group header the row opens, if any, then the row itself.
37    pub(super) view: BoxView<'a>,
38}

40/// Render the page's rows for the table body. One body serves grouped and
41/// ungrouped pages: a grouped row carries the header its group's first row
42/// owns, so the header lands immediately above its own rows.
43pub(super) fn render_rows<'a>(
44    cx: &'a Cx,
45    rows: Vec<RowView>,
46    chrome: &RowChrome,
47) -> Vec<RenderedRow<'a>> {
48    rows.into_iter()
49        .map(|row| RenderedRow {
50            key: row.key.clone(),
51            view: render_row(cx, row, chrome),
52        })
53        .collect()
54}

56/// One row: its group header when it opens a group, the bulk checkbox, the
57/// cells, and the actions cell.
58fn render_row<'a>(cx: &'a Cx, row: RowView, chrome: &RowChrome) -> BoxView<'a> {
59    let header = row.group_header.clone().map(|header| {
60        let colspan = chrome.header_colspan;
61        view! {
62            cx =>
63            table_row(
64                attrs: attributes! { id=(header.dom_id) },
65                table_cell(
66                    attrs: attributes! {
67                        colspan=(colspan)
68                        class="px-4 py-2 bg-muted text-sm font-medium"
69                    },
70                    (header.text)
71                )
72            )
73        }
74        .boxed()
75    });
76    let bulk_cell = chrome.with_bulk.then(|| {
77        if row.selectable {
78            let value = row.record_id.clone();
79            view! {
80                cx =>
81                table_cell(
82                    <input
83                        type="checkbox"
84                        value=(value)
85                        aria-label="Select row"
86                        data-row-select=""
87                    >
88                )
89            }
90            .boxed()
91        } else {
92            // A refused row renders no checkbox: selecting it could only
93            // produce a batch the handler refuses. The cell stays so the row
94            // keeps its shape.
95            view! { cx => table_cell() }.boxed()
96        }
97    });
98    // `row.cells` is built column-for-column, so the zip pairs each cell with
99    // the column that owns its width. The cell repeats the width its header
100    // declares and truncates: under the table's fixed layout a value wider than
101    // the column clips to an ellipsis instead of stretching the column.
102    let cells: Vec<(String, Option<Cow<'static, str>>)> = row
103        .cells
104        .iter()
105        .cloned()
106        .zip(chrome.cell_widths.iter().cloned())
107        .collect();
108    // Every row carries the actions cell its header declares; a row refused
109    // every link keeps an empty cell so the row keeps its shape.
110    let actions = chrome
111        .with_actions
112        .then(|| render_actions(cx, &row, chrome));
113    let dom_id = row_dom_id(&row.key);
114    view! {
115        cx =>
116        if let Some(header) = header {
117            (header)
118        }
119        table_row(
120            attrs: attributes! { id=(dom_id) },
121            if let Some(cell) = bulk_cell {
122                (cell)
123            }
124            for (cell, width) in cells {
125                table_cell(
126                    attrs: attributes! { class="truncate" style=(width.as_deref()) },
127                    (cell)
128                )
129            }
130            if let Some(actions) = actions {
131                (actions)
132            }
133        )
134    }
135    .boxed()
136}

138/// The row's actions cell: View, Edit and Delete, each only when the row's
139/// policy allows it.
140fn render_actions<'a>(cx: &'a Cx, row: &RowView, chrome: &RowChrome) -> BoxView<'a> {
141    let actions_min = chrome.actions_min.clone();
142    let view_url = row.view_url.clone();
143    let edit_url = row.edit_url.clone();
144    let delete = row
145        .delete_url
146        .clone()
147        .zip(row.delete_action.clone())
148        .map(|(url, action)| (url, action, chrome.delete_dialog_id.clone()));
149    let link_class = button_variants(ButtonVariant::Outline, ButtonSize::Md);
150    let edit_class = link_class.clone();
151    let delete_class = button_variants(ButtonVariant::Destructive, ButtonSize::Md);
152    view! {
153        cx =>
154        table_cell(
155            attrs: attributes! { style=(actions_min.as_deref()) },
156            <div class="flex gap-2">
157                if let Some(url) = view_url {
158                    <a (crate::resource::runtime_link(cx, &url)) class=(link_class)>
159                        "View"
160                    </a>
161                }
162                if let Some(url) = edit_url {
163                    <a (crate::resource::runtime_link(cx, &url)) class=(edit_class)>
164                        "Edit"
165                    </a>
166                }
167                if let Some((url, action, dialog)) = delete {
168                    <a
169                        href=(url)
170                        data-row-delete-trigger=(dialog)
171                        data-row-delete-action=(action)
172                        class=(delete_class)
173                    >
174                        "Delete"
175                    </a>
176                }
177            </div>
178        )
179    }
180    .boxed()
181}

183/// Precomputed per-row presentation for the table body: the display row key,
184/// the record key, the rendered cells, and the optional Edit / delete-dialog
185/// action URLs. A struct (not a tuple): five anonymous positions would
186/// mislead readers and trip `clippy::type_complexity`.
187///
188/// `key` is the display projection (keyed diffs, DOM ids);
189/// `record_id` is the record projection (URLs, bulk values), resolved
190/// by handlers as the typed PK.
191pub(super) struct RowView {
192    key: String,
193    record_id: String,
194    cells: Vec<String>,
195    view_url: Option<String>,
196    edit_url: Option<String>,
197    delete_url: Option<String>,
198    /// The row's delete POST target (`{prefix}/{key}/delete`): the
199    /// Delete control hands it to the shared dialog before opening it, so the
200    /// confirmed POST keeps the route the `?delete=` fallback uses.
201    delete_action: Option<String>,
202    /// Whether the row renders a bulk checkbox: a row the
203    /// [`Table::row_actions`] policy denies `delete` renders none, so
204    /// `bulk.js` never sees its key and select-all cannot submit a batch the
205    /// handler refuses wholesale.
206    selectable: bool,
207    /// The row's group label, when `?group_by=` named the declared group.
208    /// Carried on every row so the page-local shim can order by it.
209    group: Option<String>,
210    /// The header this row renders above itself, `Some` only on the first row
211    /// of its group.
212    group_header: Option<GroupHeader>,
213}

215/// One page-local group header: the label with its page-local count,
216/// and the stable DOM id the injected header row carries so the in-place morph
217/// can follow it (`row_dom_id`'s contract).
218#[derive(Clone)]
219struct GroupHeader {
220    /// `"{label} ({n} on this page)"`.
221    text: String,
222    /// [`group_header_dom_id`] of the label.
223    dom_id: String,
224}

226impl<M> Table<M> {
227    /// Project the loaded page into the row presentation the template renders.
228    ///
229    /// Precomputed so template bodies capture only owned data — the lazy view
230    /// outlives the render call, so it must never borrow `self` or `page`.
231    ///
232    /// The per-row delete URL opens the confirmation dialog on the list page
233    /// (`?delete=<key>`); the per-row edit URL links to `{prefix}/{key}/edit`.
234    /// Both — and the bulk checkbox values — carry the *record* key, resolved by
235    /// handlers as the model's typed PK; the display `key` stays on keyed diffs
236    /// and DOM ids.
237    ///
238    /// The chrome prefixes say which links the table *can* render; the
239    /// [`Table::row_actions`] policy says which of them *this* record may use.
240    /// A denied action emits no URL, and a row denied `delete` renders no bulk
241    /// checkbox. The policy is consulted only when a prefix is wired.
242    ///
243    /// The delete URL's shared parameters are encoded once for the whole page:
244    /// rebuilding them per row is work a client can inflate with an oversized
245    /// query.
246    ///
247    /// Page-local grouping is display-only: `group_by` is a bare key closure with
248    /// no lens, so no `ORDER BY` is derivable and a group cannot span pages. The
249    /// shim therefore reorders *this page's* rows by the group label — a stable
250    /// sort, so rows keep the query's order inside their group — and hangs each
251    /// group's header off its first row. The query, its cursors and the export
252    /// keep the declared ordering.
253    ///
254    /// Row keys must be injective within a page: duplicates corrupt keyed diffs
255    /// and bulk selection.
256    pub(super) fn row_views(
257        &self,
258        state: &TableState,
259        path: &str,
260        page: &TablePage<M>,
261        row_key: &RowKey<M>,
262        record_key: &RowKey<M>,
263        group_key: Option<&GroupKey<M>>,
264    ) -> Vec<RowView>
265    where
266        M: toasty::schema::Model,
267    {
268        let delete_url_base = self
269            .delete_prefix
270            .as_ref()
271            .map(|_| state.row_url_base(path));
272        let gated = self.delete_prefix.is_some()
273            || self.edit_prefix.is_some()
274            || self.view_prefix.is_some();
275        let mut row_data: Vec<RowView> = page
276            .rows
277            .iter()
278            .map(|row| {
279                let key = row_key(row);
280                let record_id = record_key(row);
281                let actions = if gated {
282                    self.actions_for(row)
283                } else {
284                    RowActions::ALL
285                };
286                let cells: Vec<String> = self
287                    .columns
288                    .iter()
289                    .map(|col| col.render_cell(row))
290                    .collect();
291                let edit_url = self
292                    .edit_prefix
293                    .as_ref()
294                    .filter(|_| actions.edit)
295                    .map(|prefix| self.action_url(row_edit_url(prefix, &record_id)));
296                let view_url = self
297                    .view_prefix
298                    .as_ref()
299                    .filter(|_| actions.view)
300                    .map(|prefix| row_view_url(prefix, &record_id));
301                let delete_url = delete_url_base
302                    .as_ref()
303                    .filter(|_| actions.delete)
304                    .map(|base| base.delete_dialog(&record_id));
305                // The shared dialog's POST target for this row: the
306                // row control hands it over before opening the dialog, so the
307                // action and the control come from the one policy decision.
308                let delete_action = self
309                    .delete_prefix
310                    .as_ref()
311                    .filter(|_| actions.delete)
312                    .map(|prefix| self.action_url(delete_action_url(prefix, &record_id)));
313                RowView {
314                    key,
315                    record_id,
316                    cells,
317                    view_url,
318                    edit_url,
319                    delete_url,
320                    delete_action,
321                    selectable: actions.delete,
322                    group: group_key.map(|group| group(row)),
323                    group_header: None,
324                }
325            })
326            .collect();
327        if group_key.is_some() {
328            row_data.sort_by(|a, b| a.group.cmp(&b.group));
329            let mut start = 0;
330            while start < row_data.len() {
331                let label = row_data[start].group.clone().unwrap_or_default();
332                let mut end = start;
333                while end < row_data.len() && row_data[end].group.as_deref() == Some(label.as_str())
334                {
335                    end += 1;
336                }
337                // The count is page-local, and says so: a group split across
338                // pages must not read as a table total. The header
339                // carries an id derived from its label — never from its
340                // position — so the in-place morph can follow it.
341                row_data[start].group_header = Some(GroupHeader {
342                    dom_id: group_header_dom_id(&label),
343                    text: format!("{label} ({} on this page)", end - start),
344                });
345                start = end;
346            }
347        }
348        debug_assert!(
349            {
350                let mut seen = std::collections::HashSet::new();
351                row_data.iter().all(|row| seen.insert(row.key.clone()))
352            },
353            "duplicate table keys in one page: the key projection must be injective"
354        );
355        row_data
356    }
