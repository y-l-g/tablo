19/// What every row of one render shares: the chrome columns, the declared
20/// widths (resolved once per render, the same for every row), and the one
21/// delete dialog every row control opens.
22pub(super) struct RowChrome {
23    pub(super) with_bulk: bool,
24    pub(super) with_actions: bool,
25    /// Columns a group header spans: the data columns plus the chrome.
26    pub(super) header_colspan: usize,
27    pub(super) cell_widths: Vec<Option<Cow<'static, str>>>,
28    pub(super) actions_min: Option<Cow<'static, str>>,
29    pub(super) delete_dialog_id: String,
30}

32/// One rendered row, keyed for the body's diff.
33pub(super) struct RenderedRow<'a> {
34    /// The row's display key: stable for the record, never a loop index.
35    pub(super) key: String,
36    /// The group header the row opens, if any, then the row itself.
37    pub(super) view: BoxView<'a>,
38}

40/// Render the page's rows for the table body. One body serves grouped and
41/// ungrouped pages: a grouped row carries the header its group's first row
42/// owns, so the header lands immediately above its own rows.
43pub(super) fn render_rows<'a>(
44    cx: &'a Cx,
45    rows: Vec<RowView<'a>>,
46    chrome: &RowChrome,
47) -> Vec<RenderedRow<'a>> {
48    rows.into_iter()
49        .map(|row| RenderedRow {
50            key: row.key.clone(),
51            view: render_row(cx, row, chrome),
52        })
53        .collect()
54}

56/// One row: its group header when it opens a group, the bulk checkbox, the
57/// cells, and the actions cell.
58fn render_row<'a>(cx: &'a Cx, mut row: RowView<'a>, chrome: &RowChrome) -> BoxView<'a> {
59    let header = row.group_header.clone().map(|header| {
60        let colspan = chrome.header_colspan;
61        view! {
62            cx =>
63            table_row(
64                attrs: attributes! { id=(header.dom_id) },
65                table_cell(
66                    attrs: attributes! {
67                        colspan=(colspan)
68                        class="bg-muted/60 px-3 py-2 text-sm font-medium text-foreground"
69                    },
70                    (header.text)
71                )
72            )
73        }
74        .boxed()
75    });
76    let bulk_cell = chrome.with_bulk.then(|| {
77        if row.selectable {
78            let value = row.record_id.clone();
79            view! {
80                cx =>
81                table_cell(
82                    <input
83                        type="checkbox"
84                        value=(value)
85                        aria-label="Select row"
86                        data-row-select=""
87                    >
88                )
89            }
90            .boxed()
91        } else {
92            // A refused row renders no checkbox: selecting it could only
93            // produce a batch the handler refuses. The cell stays so the row
94            // keeps its shape.
95            view! { cx => table_cell() }.boxed()
96        }
97    });
98    // `row.cells` is built column-for-column, so the zip pairs each cell with
99    // the column that owns its width. The cell repeats the width its header
100    // declares and truncates: under the table's fixed layout a value wider than
101    // the column clips to an ellipsis instead of stretching the column.
102    let cells: Vec<(BoxView<'a>, Option<Cow<'static, str>>)> = std::mem::take(&mut row.cells)
103        .into_iter()
104        .zip(chrome.cell_widths.iter().cloned())
105        .collect();
106    // Every row carries the actions cell its header declares; a row refused
107    // every link keeps an empty cell so the row keeps its shape.
108    let actions = chrome
109        .with_actions
110        .then(|| render_actions(cx, &row, chrome));
111    let dom_id = row_dom_id(&row.key);
112    view! {
113        cx =>
114        if let Some(header) = header {
115            (header)
116        }
117        table_row(
118            attrs: attributes! { id=(dom_id) },
119            if let Some(cell) = bulk_cell {
120                (cell)
121            }
122            for (cell, width) in cells {
123                table_cell(
124                    attrs: attributes! { class="truncate" style=(width.as_deref()) },
125                    (cell)
126                )
127            }
128            if let Some(actions) = actions {
129                (actions)
130            }
131        )
132    }
133    .boxed()
134}

136/// The row's actions cell: View, Edit and Delete, each only when the row's
137/// policy allows it.
138fn render_actions<'a>(cx: &'a Cx, row: &RowView<'a>, chrome: &RowChrome) -> BoxView<'a> {
139    let actions_min = chrome.actions_min.clone();
140    let view_url = row.view_url.clone();
141    let edit_url = row.edit_url.clone();
142    let delete = row
143        .delete_url
144        .clone()
145        .zip(row.delete_action.clone())
146        .map(|(url, action)| (url, action, chrome.delete_dialog_id.clone()));
147    // Icon-only controls: the label rides `aria-label` for assistive tech and
148    // `title` for a pointer, and the icon carries it on screen.
149    let link_class = button_variants(ButtonVariant::Ghost, ButtonSize::Icon);
150    let edit_class = link_class.clone();
151    let delete_class = link_class.clone();
152    // Each custom action is its own POST form, marked like the delete forms:
153    // `mutation-submit.js` posts it and refreshes the table in place, keeping
154    // the list's query; without the script it POSTs and 303s. Its button
155    // carries the label, not an icon.
156    let csrf = (!row.custom.is_empty()).then(|| crate::csrf::current_token(cx));
157    let custom: Vec<BoxView<'a>> = row
158        .custom
159        .iter()
160        .map(|(label, url)| {
161            let label = label.clone();
162            let url = url.clone();
163            let token = csrf.clone().unwrap_or_default();
164            view! {
165                cx =>
166                <form
167                    method="post"
168                    action=(url)
169                    class="contents"
170                    data-mutation-submit=""
171                >
172                    (crate::csrf::field(cx, &token))
173                    button(
174                        variant: ButtonVariant::Ghost,
175                        size: ButtonSize::Sm,
176                        attrs: attributes! { type="submit" },
177                        (label)
178                    )
179                </form>
180            }
181            .boxed()
182        })
183        .collect();
184    view! {
185        cx =>
186        table_cell(
187            attrs: attributes! { style=(actions_min.as_deref()) },
188            <div class="flex items-center justify-end gap-1">
189                for form in custom {
190                    (form)
191                }
192                if let Some(url) = view_url {
193                    <a
194                        (crate::resource::runtime_link(cx, &url))
195                        class=(link_class)
196                        aria-label="View"
197                        title="View"
198                    >
199                        icon(data: tablo_ui::icons::EYE)
200                    </a>
201                }
202                if let Some(url) = edit_url {
203                    <a
204                        (crate::resource::runtime_link(cx, &url))
205                        class=(edit_class)
206                        aria-label="Edit"
207                        title="Edit"
208                    >
209                        icon(data: tablo_ui::icons::PENCIL)
210                    </a>
211                }
212                if let Some((url, action, dialog)) = delete {
213                    <a
214                        href=(url)
215                        data-row-delete-trigger=(dialog)
216                        data-row-delete-action=(action)
217                        class=(delete_class)
218                        aria-label="Delete"
219                        title="Delete"
220                    >
221                        // The glyph carries the destructive color: the ghost
222                        // variant already sets the control's text color.
223                        icon(
224                            data: tablo_ui::icons::TRASH,
225                            attrs: attributes! { class="text-destructive" }
226                        )
227                    </a>
228                }
229            </div>
230        )
231    }
232    .boxed()
233}

235/// Precomputed per-row presentation for the table body: the display row key,
236/// the record key, the rendered cells, and the optional Edit / delete-dialog
237/// action URLs. A struct (not a tuple): five anonymous positions would
238/// mislead readers and trip `clippy::type_complexity`.
239///
240/// `key` is the display projection (keyed diffs, DOM ids);
241/// `record_id` is the record projection (URLs, bulk values), resolved
242/// by handlers as the typed PK.
243pub(super) struct RowView<'a> {
244    key: String,
245    record_id: String,
246    /// One rendered cell per column, in column order.
247    cells: Vec<BoxView<'a>>,
248    view_url: Option<String>,
249    edit_url: Option<String>,
250    delete_url: Option<String>,
251    /// The row's delete POST target (`{prefix}/{key}/delete`): the
252    /// Delete control hands it to the shared dialog before opening it, so the
253    /// confirmed POST keeps the route the `?delete=` fallback uses.
254    delete_action: Option<String>,
255    /// The custom row actions this record allows: each button's label and
256    /// its POST target.
257    custom: Vec<(String, String)>,
258    /// Whether the row renders a bulk checkbox: a row that neither bulk
259    /// delete nor any bulk custom action allows renders none, so `bulk.js`
260    /// never sees its key.
261    selectable: bool,
262    /// The row's group label, when `?group_by=` named the declared group.
263    /// Carried on every row so the page-local shim can order by it.
264    group: Option<String>,
265    /// The header this row renders above itself, `Some` only on the first row
266    /// of its group.
267    group_header: Option<GroupHeader>,
268}

270/// One page-local group header: the label with its page-local count,
271/// and the stable DOM id the injected header row carries so the in-place morph
272/// can follow it (`row_dom_id`'s contract).
273#[derive(Clone)]
274struct GroupHeader {
275    /// `"{label} ({n} on this page)"`.
276    text: String,
277    /// [`group_header_dom_id`] of the label.
278    dom_id: String,
279}

281impl<M> Table<M> {
282    /// Project the loaded page into the row presentation the template renders.
283    ///
284    /// Precomputed so template bodies capture only owned data — the lazy view
285    /// outlives the render call, so it must never borrow `self` or `page`.
286    ///
287    /// The per-row delete URL opens the confirmation dialog on the list page
288    /// (`?delete=<key>`); the per-row edit URL links to `{prefix}/{key}/edit`.
289    /// Both — and the bulk checkbox values — carry the *record* key, resolved by
290    /// handlers as the model's typed PK; the display `key` stays on keyed diffs
291    /// and DOM ids.
292    ///
293    /// The chrome prefixes say which links the table *can* render; the
294    /// [`Table::row_actions`] policy says which of them *this* record may use.
295    /// A denied action emits no URL, and a row denied `delete` renders no bulk
296    /// checkbox. The policy is consulted only when a prefix is wired.
297    ///
298    /// The delete URL's shared parameters are encoded once for the whole page:
299    /// rebuilding them per row is work a client can inflate with an oversized
300    /// query.
301    ///
302    /// Page-local grouping is display-only: `group_by` is a bare key closure with
303    /// no lens, so no `ORDER BY` is derivable and a group cannot span pages. The
304    /// shim therefore reorders *this page's* rows by the group label — a stable
305    /// sort, so rows keep the query's order inside their group — and hangs each
306    /// group's header off its first row. The query, its cursors and the export
307    /// keep the declared ordering.
308    ///
309    /// Row keys must be injective within a page: duplicates corrupt keyed diffs
310    /// and bulk selection.
311    pub(super) fn row_views<'a>(
312        &self,
313        cx: &'a Cx,
314        state: &TableState,
315        path: &str,
316        page: &TablePage<M>,
317        group_key: Option<&GroupKey<M>>,
318    ) -> Vec<RowView<'a>>
319    where
320        M: toasty::schema::Model,
321    {
322        let delete_url_base = self
323            .delete_prefix
324            .as_ref()
325            .map(|_| state.row_url_base(path));
326        let gated = self.delete_prefix.is_some()
327            || self.edit_prefix.is_some()
328            || self.view_prefix.is_some();
329        let bulk_delete = self.bulk_delete_enabled();
330        let mut row_data: Vec<RowView<'a>> = page
331            .rows
332            .iter()
333            .map(|row| {
334                let key = (self.row_key)(row);
335                let record_id = (self.record_key)(row);
336                let actions = if gated {
337                    self.actions_for(row)
338                } else {
339                    RowActions::ALL
340                };
341                let cells: Vec<BoxView<'a>> =
342                    self.columns.iter().map(|col| col.cell(cx, row)).collect();
343                let edit_url = self
344                    .edit_prefix
345                    .as_ref()
346                    .filter(|_| actions.edit)
347                    .map(|prefix| self.action_url(row_edit_url(prefix, &record_id)));
348                let view_url = self
349                    .view_prefix
350                    .as_ref()
351                    .filter(|_| actions.view)
352                    .map(|prefix| row_view_url(prefix, &record_id));
353                let delete_url = delete_url_base
354                    .as_ref()
355                    .filter(|_| actions.delete)
356                    .map(|base| base.delete_dialog(&record_id));
357                // The shared dialog's POST target for this row: the
358                // row control hands it over before opening the dialog, so the
359                // action and the control come from the one policy decision.
360                let delete_action = self
361                    .delete_prefix
362                    .as_ref()
363                    .filter(|_| actions.delete)
364                    .map(|prefix| self.action_url(delete_action_url(prefix, &record_id)));
365                let custom = self
366                    .actions_prefix
367                    .as_ref()
368                    .map(|prefix| {
369                        self.row_custom_actions()
370                            .filter(|action| (action.allowed)(row))
371                            .map(|action| {
372                                let url = row_action_url(prefix, &record_id, action.name);
373                                (action.label.clone(), self.action_url(url))
374                            })
375                            .collect()
376                    })
377                    .unwrap_or_default();
378                let selectable = (bulk_delete && actions.delete)
379                    || self
380                        .bulk_custom_actions()
381                        .any(|action| (action.allowed)(row));
382                RowView {
383                    key,
384                    record_id,
385                    cells,
386                    view_url,
387                    edit_url,
388                    delete_url,
389                    delete_action,
390                    custom,
391                    selectable,
392                    group: group_key.map(|group| group(row)),
393                    group_header: None,
394                }
395            })
396            .collect();
397        if group_key.is_some() {
398            row_data.sort_by(|a, b| a.group.cmp(&b.group));
399            let mut start = 0;
400            while start < row_data.len() {
401                let label = row_data[start].group.clone().unwrap_or_default();
402                let mut end = start;
403                while end < row_data.len() && row_data[end].group.as_deref() == Some(label.as_str())
404                {
405                    end += 1;
406                }
407                // The count is page-local, and says so: a group split across
408                // pages must not read as a table total. The header
409                // carries an id derived from its label — never from its
410                // position — so the in-place morph can follow it.
411                row_data[start].group_header = Some(GroupHeader {
412                    dom_id: group_header_dom_id(&label),
413                    text: format!("{label} ({} on this page)", end - start),
414                });
415                start = end;
416            }
417        }
418        debug_assert!(
419            {
420                let mut seen = std::collections::HashSet::new();
421                row_data.iter().all(|row| seen.insert(row.key.clone()))
422            },
423            "duplicate table keys in one page: the key projection must be injective"
424        );
425        row_data
426    }
