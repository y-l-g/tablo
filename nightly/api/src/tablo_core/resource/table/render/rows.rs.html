19/// What every row of one render shares: the chrome columns, the declared
20/// widths (resolved once per render, the same for every row), and the one
21/// delete dialog every row control opens.
22pub(super) struct RowChrome {
23    pub(super) with_bulk: bool,
24    pub(super) with_actions: bool,
25    /// Columns a group header spans: the data columns plus the chrome.
26    pub(super) header_colspan: usize,
27    pub(super) cell_widths: Vec<Option<Cow<'static, str>>>,
28    pub(super) actions_min: Option<Cow<'static, str>>,
29    pub(super) delete_dialog_id: String,
30}

32/// One rendered row, keyed for the body's diff.
33pub(super) struct RenderedRow<'a> {
34    /// The row's display key: stable for the record, never a loop index.
35    pub(super) key: String,
36    /// The group header the row opens, if any, then the row itself.
37    pub(super) view: BoxView<'a>,
38}

40/// Render the page's rows for the table body, carrying each group's header on its first row.
41pub(super) fn render_rows<'a>(
42    cx: &'a Cx,
43    rows: Vec<RowView<'a>>,
44    chrome: &RowChrome,
45) -> Vec<RenderedRow<'a>> {
46    rows.into_iter()
47        .map(|row| RenderedRow {
48            key: row.key.clone(),
49            view: render_row(cx, row, chrome),
50        })
51        .collect()
52}

54/// One row: its group header when it opens a group, the bulk checkbox, the
55/// cells, and the actions cell.
56fn render_row<'a>(cx: &'a Cx, mut row: RowView<'a>, chrome: &RowChrome) -> BoxView<'a> {
57    let header = row.group_header.clone().map(|header| {
58        let colspan = chrome.header_colspan;
59        view! {
60            cx =>
61            table_row(
62                attrs: attributes! { id=(header.dom_id) },
63                table_cell(
64                    attrs: attributes! {
65                        colspan=(colspan)
66                        class="bg-muted/60 px-3 py-2 text-sm font-medium text-foreground"
67                    },
68                    (header.text)
69                )
70            )
71        }
72        .boxed()
73    });
74    let bulk_cell = chrome.with_bulk.then(|| {
75        if row.selectable {
76            let value = row.record_id.clone();
77            view! {
78                cx =>
79                table_cell(
80                    <input
81                        type="checkbox"
82                        value=(value)
83                        aria-label="Select row"
84                        data-row-select=""
85                    >
86                )
87            }
88            .boxed()
89        } else {
90            view! { cx => table_cell() }.boxed()
91        }
92    });
93    let cells: Vec<(BoxView<'a>, Option<Cow<'static, str>>)> = std::mem::take(&mut row.cells)
94        .into_iter()
95        .zip(chrome.cell_widths.iter().cloned())
96        .collect();
97    let actions = chrome
98        .with_actions
99        .then(|| render_actions(cx, &row, chrome));
100    let dom_id = row_dom_id(&row.key);
101    view! {
102        cx =>
103        if let Some(header) = header {
104            (header)
105        }
106        table_row(
107            attrs: attributes! { id=(dom_id) },
108            if let Some(cell) = bulk_cell {
109                (cell)
110            }
111            for (cell, width) in cells {
112                table_cell(
113                    attrs: attributes! { class="truncate" style=(width.as_deref()) },
114                    (cell)
115                )
116            }
117            if let Some(actions) = actions {
118                (actions)
119            }
120        )
121    }
122    .boxed()
123}

125/// The row's actions cell: View, Edit and Delete, each only when the row's
126/// policy allows it.
127fn render_actions<'a>(cx: &'a Cx, row: &RowView<'a>, chrome: &RowChrome) -> BoxView<'a> {
128    let actions_min = chrome.actions_min.clone();
129    let view_url = row.view_url.clone();
130    let edit_url = row.edit_url.clone();
131    let delete = row
132        .delete_url
133        .clone()
134        .zip(row.delete_action.clone())
135        .map(|(url, action)| (url, action, chrome.delete_dialog_id.clone()));
136    let link_class = button_variants(ButtonVariant::Ghost, ButtonSize::Icon);
137    let edit_class = link_class.clone();
138    let delete_class = link_class.clone();
139    let csrf = (!row.custom.is_empty()).then(|| crate::csrf::current_token(cx));
140    let custom: Vec<BoxView<'a>> = row
141        .custom
142        .iter()
143        .map(|(label, url)| {
144            let label = label.clone();
145            let url = url.clone();
146            let token = csrf.clone().unwrap_or_default();
147            view! {
148                cx =>
149                <form
150                    method="post"
151                    action=(url)
152                    class="contents"
153                    data-mutation-submit=""
154                >
155                    (crate::csrf::field(cx, &token))
156                    button(
157                        variant: ButtonVariant::Ghost,
158                        size: ButtonSize::Sm,
159                        attrs: attributes! { type="submit" },
160                        (label)
161                    )
162                </form>
163            }
164            .boxed()
165        })
166        .collect();
167    view! {
168        cx =>
169        table_cell(
170            attrs: attributes! { style=(actions_min.as_deref()) },
171            <div class="flex items-center justify-end gap-1">
172                for form in custom {
173                    (form)
174                }
175                if let Some(url) = view_url {
176                    <a
177                        (crate::resource::runtime_link(cx, &url))
178                        class=(link_class)
179                        aria-label="View"
180                        title="View"
181                    >
182                        icon(data: tablo_ui::icons::EYE)
183                    </a>
184                }
185                if let Some(url) = edit_url {
186                    <a
187                        (crate::resource::runtime_link(cx, &url))
188                        class=(edit_class)
189                        aria-label="Edit"
190                        title="Edit"
191                    >
192                        icon(data: tablo_ui::icons::PENCIL)
193                    </a>
194                }
195                if let Some((url, action, dialog)) = delete {
196                    <a
197                        href=(url)
198                        data-row-delete-trigger=(dialog)
199                        data-row-delete-action=(action)
200                        class=(delete_class)
201                        aria-label="Delete"
202                        title="Delete"
203                    >
204                        icon(
205                            data: tablo_ui::icons::TRASH,
206                            attrs: attributes! { class="text-destructive" }
207                        )
208                    </a>
209                }
210            </div>
211        )
212    }
213    .boxed()
214}

216/// Precompute per-row presentation for the table body in owned data the lazy view captures.
217pub(super) struct RowView<'a> {
218    /// The display projection driving keyed diffs and DOM ids.
219    key: String,
220    /// The record projection driving URLs and bulk values, resolved by handlers as the typed PK.
221    record_id: String,
222    /// One rendered cell per column, in column order.
223    cells: Vec<BoxView<'a>>,
224    view_url: Option<String>,
225    edit_url: Option<String>,
226    delete_url: Option<String>,
227    /// The row's delete POST target handed to the shared dialog.
228    delete_action: Option<String>,
229    /// The custom row actions this record allows: each button's label and
230    /// its POST target.
231    custom: Vec<(String, String)>,
232    /// Whether the row renders a bulk checkbox.
233    selectable: bool,
234    /// The row's group label, when `?group_by=` names the declared group.
235    group: Option<String>,
236    /// The header this row renders above itself, `Some` only on the first row
237    /// of its group.
238    group_header: Option<GroupHeader>,
239}

241/// One page-local group header: the label with its page-local count and the stable DOM id the
242/// injected header row carries.
243#[derive(Clone)]
244struct GroupHeader {
245    /// `"{label} ({n} on this page)"`.
246    text: String,
247    dom_id: String,
248}

250impl<M> Table<M> {
251    /// Project the loaded page into the row presentation the template renders, requiring injective
252    /// row keys within a page.
253    pub(super) fn row_views<'a>(
254        &self,
255        cx: &'a Cx,
256        state: &TableState,
257        path: &str,
258        page: &TablePage<M>,
259        group_key: Option<&GroupKey<M>>,
260    ) -> Vec<RowView<'a>>
261    where
262        M: toasty::schema::Model,
263    {
264        let delete_url_base = self
265            .delete_prefix
266            .as_ref()
267            .map(|_| state.row_url_base(path));
268        let gated = self.delete_prefix.is_some()
269            || self.edit_prefix.is_some()
270            || self.view_prefix.is_some();
271        let bulk_delete = self.bulk_delete_enabled();
272        let mut row_data: Vec<RowView<'a>> = page
273            .rows
274            .iter()
275            .map(|row| {
276                let key = (self.row_key)(row);
277                let record_id = (self.record_key)(row);
278                let actions = if gated {
279                    self.actions_for(row)
280                } else {
281                    RowActions::ALL
282                };
283                let cells: Vec<BoxView<'a>> =
284                    self.columns.iter().map(|col| col.cell(cx, row)).collect();
285                let edit_url = self
286                    .edit_prefix
287                    .as_ref()
288                    .filter(|_| actions.edit)
289                    .map(|prefix| self.action_url(row_edit_url(prefix, &record_id)));
290                let view_url = self
291                    .view_prefix
292                    .as_ref()
293                    .filter(|_| actions.view)
294                    .map(|prefix| row_view_url(prefix, &record_id));
295                let delete_url = delete_url_base
296                    .as_ref()
297                    .filter(|_| actions.delete)
298                    .map(|base| base.delete_dialog(&record_id));
299                let delete_action = self
300                    .delete_prefix
301                    .as_ref()
302                    .filter(|_| actions.delete)
303                    .map(|prefix| self.action_url(delete_action_url(prefix, &record_id)));
304                let custom = self
305                    .actions_prefix
306                    .as_ref()
307                    .map(|prefix| {
308                        self.row_custom_actions()
309                            .filter(|action| (action.allowed)(row))
310                            .map(|action| {
311                                let url = row_action_url(prefix, &record_id, action.name);
312                                (action.label.clone(), self.action_url(url))
313                            })
314                            .collect()
315                    })
316                    .unwrap_or_default();
317                let selectable = (bulk_delete && actions.delete)
318                    || self
319                        .bulk_custom_actions()
320                        .any(|action| (action.allowed)(row));
321                RowView {
322                    key,
323                    record_id,
324                    cells,
325                    view_url,
326                    edit_url,
327                    delete_url,
328                    delete_action,
329                    custom,
330                    selectable,
331                    group: group_key.map(|group| group(row)),
332                    group_header: None,
333                }
334            })
335            .collect();
336        if group_key.is_some() {
337            row_data.sort_by(|a, b| a.group.cmp(&b.group));
338            let mut start = 0;
339            while start < row_data.len() {
340                let label = row_data[start].group.clone().unwrap_or_default();
341                let mut end = start;
342                while end < row_data.len() && row_data[end].group.as_deref() == Some(label.as_str())
343                {
344                    end += 1;
345                }
346                row_data[start].group_header = Some(GroupHeader {
347                    dom_id: group_header_dom_id(&label),
348                    text: format!("{label} ({} on this page)", end - start),
349                });
350                start = end;
351            }
352        }
353        debug_assert!(
354            {
355                let mut seen = std::collections::HashSet::new();
356                row_data.iter().all(|row| seen.insert(row.key.clone()))
357            },
358            "duplicate table keys in one page: the key projection must be injective"
359        );
360        row_data
361    }
