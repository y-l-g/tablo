20/// Row-key projection: reads the row identity off one model instance
21/// (typically `|u| u.id.to_string()`). Toasty models are plain structs with
22/// no instance→field reflection, so the key cannot be extracted generically
23/// (upstream gap #119).
24pub type RowKey<M> = Arc<dyn Fn(&M) -> String + Send + Sync>;
25
26/// Group-label projection: reads a row's group off one model instance
27/// (typically `|u| u.status.clone()`).
28pub type GroupKey<M> = Arc<dyn Fn(&M) -> String + Send + Sync>;
29
30/// Per-record action policy: reads which row actions one model instance allows.
31/// See [`RowActions`].
32pub(crate) type RowPolicy<M> = Arc<dyn Fn(&M) -> RowActions + Send + Sync>;
33
34/// Which row actions one record may use.
35///
36/// The action prefixes say which affordances a resource declares; this says
37/// which of them the caller may use
38/// on one loaded row. The panel wires the projection into the table and the
39/// renderer consults it per row — a denied action emits no link, and a row
40/// denied `delete` renders no bulk checkbox, so the row can never enter the
41/// selection transport.
42///
43/// The panel derives one from the resource's
44/// [`can_view`](crate::resource::Resource::can_view) /
45/// [`can_update`](crate::resource::Resource::can_update) /
46/// [`can_delete`](crate::resource::Resource::can_delete), pairing each action
47/// with the same predicates its route checks, so a rendered affordance and the
48/// route that answers it cannot disagree.
49#[derive(Debug, Clone, Copy, PartialEq, Eq)]
50pub(crate) struct RowActions {
51    /// Whether the row renders its `View` link — the detail route's
52    /// `can_view`.
53    pub view: bool,
54    /// Whether the row renders its `Edit` link — the edit route's `can_view`
55    /// **and** `can_update`.
56    pub edit: bool,
57    /// Whether the row renders its `Delete` link and an enabled bulk checkbox —
58    /// the delete route's `can_view` **and** `can_delete`.
59    pub delete: bool,
60}
61
62impl RowActions {
63    /// Every action allowed: what a table without a panel-wired row policy
64    /// renders.
65    pub const ALL: Self = Self {
66        view: true,
67        edit: true,
68        delete: true,
69    };
70}

82/// [`Resource::table`](crate::resource::Resource::table) returns a table
83/// carrying no delete/edit/view prefix: the panel attaches them from the

101/// Table description of a `Resource`'s list view. Declares columns and how they
102/// map to queries.
103///
104/// Row identity is mandatory and typed: [`Table::new`] takes the key
105/// projection driving both key halves, and cells render via
106/// [`TextColumn`]'s lens-bound closure where typos fail at compile time instead
107/// of panicking at render.
108pub struct Table<M> {
109    columns: Vec<TextColumn<M>>,
110    filters: Vec<Filter<M>>,
111    group_by: Option<GroupDef<M>>,
112    row_key: RowKey<M>,
113    record_key: RowKey<M>,
114    row_policy: Option<RowPolicy<M>>,
115    page_size: NonZeroUsize,
116    hide_search: bool,
117    hide_filter_bar: bool,
118    delete_prefix: Option<String>,
119    edit_prefix: Option<String>,
120    view_prefix: Option<String>,
121    bulk_delete: bool,
122    live_search: bool,
123    _marker: PhantomData<M>,
124}

145impl<M> Table<M> {
146    /// Declare a table with its row key and columns.
147    ///
148    /// One primary-key projection (typically `|u| u.id.to_string()`) drives
149    /// both halves: keyed diffs and DOM ids, and the action URLs and bulk
150    /// values handlers resolve as the model's typed PK. The projection must be
151    /// injective within a page: duplicate keys corrupt keyed diffs and bulk
152    /// selection, and are debug-asserted at render time. A table whose display
153    /// projects a non-PK value uses [`Self::new_split`].
154    ///
155    /// # Panics
156    ///
157    /// Panics on duplicate [`TextColumn::name`], the guard every constructor
158    /// applies: sort resolution is
159    /// first-sortable-`name()`-match, so duplicate sortable names would
160    /// silently misresolve `?sort=`. The guard covers computed names too
161    /// (`TextColumn::computed("Status", ..)` derives `name = "status"`) for
162    /// namespace consistency and future-proofing. Same fail-loud policy as
163    /// the GH #101 searchable/sortable panics and the Schema GH #100 guard.
164    ///
165    /// Also panics on an empty column set: a table with no columns renders a
166    /// headers-only list, which no resource declares.
167    pub fn new(
168        key: impl Fn(&M) -> String + Send + Sync + 'static,
169        cols: impl IntoColumns<M>,
170    ) -> Self
171    where
172        M: toasty::schema::Model,
173    {
174        let key = Arc::new(key);
175        Self::from_keys(key.clone(), key, cols)
176    }
177
178    /// Declare a table whose display projects a non-PK value.
179    ///
180    /// `display` drives keyed diffs and DOM ids; `record` drives action URLs
181    /// and bulk checkbox values handlers resolve as the model's typed PK
182    /// (`pk_eq_expr` / `pk_in_expr` — an unparseable value 404s), so emitting
183    /// a display key there 404s every delete and bulk submit.
184    ///
185    /// # Panics
186    ///
187    /// Panics on duplicate [`TextColumn::name`] and on an empty column set,
188    /// like [`Self::new`].
189    pub fn new_split(
190        display: impl Fn(&M) -> String + Send + Sync + 'static,
191        record: impl Fn(&M) -> String + Send + Sync + 'static,
192        cols: impl IntoColumns<M>,
193    ) -> Self
194    where
195        M: toasty::schema::Model,
196    {
197        Self::from_keys(Arc::new(display), Arc::new(record), cols)
198    }

200    /// The record key of `record`: the model's primary key as the action URLs
201    /// carry it.
202    pub(crate) fn record_key_of(&self, record: &M) -> String {
203        (self.record_key)(record)

206    /// The one constructor body: rejects duplicate column names, then builds
207    /// the table around the two declared projections.
208    fn from_keys(row_key: RowKey<M>, record_key: RowKey<M>, cols: impl IntoColumns<M>) -> Self
209    where
210        M: toasty::schema::Model,
211    {
212        let cols = cols.into_columns();
213        assert!(
214            !cols.is_empty(),
215            "a Table needs at least one column: declare columns with Table::new(key, columns)"
216        );
217        let mut seen = std::collections::HashSet::with_capacity(cols.len());
218        for c in &cols {
219            let name = c.name();
220            assert!(
221                seen.insert(name),
222                "duplicate column name '{name}': each Table column needs a distinct name (GH #156)"
223            );
224        }
225        Self {
226            columns: cols,
227            filters: Vec::new(),
228            group_by: None,
229            row_key,
230            record_key,
231            row_policy: None,
232            page_size: DEFAULT_PAGE_SIZE,
233            hide_search: false,
234            hide_filter_bar: false,
235            delete_prefix: None,
236            edit_prefix: None,
237            view_prefix: None,
238            bulk_delete: false,
239            live_search: false,
240            _marker: PhantomData,
241        }
242    }

244    /// Declare the per-record action policy: which of the wired row
245    /// actions each record may use.
246    ///
247    /// The `with_*` methods say which affordances the table declares; this says
248    /// which of them a row may use. The renderer consults it per row — a denied
249    /// action emits no link, and a row denied `delete` renders no bulk
250    /// checkbox, so the selection transport never carries a key the handler
251    /// refuses.
252    ///
253    /// Defaults to [`RowActions::ALL`], so a table that declares no policy
254    /// renders exactly the chrome its `with_*` calls attached. The policy is
255    /// consulted only where chrome is wired: a table with no action prefix
256    /// never calls it, and a resource that declares no chrome keeps its list
257    /// page free of per-record predicate calls.
258    ///
259    /// The panel wires this from
260    /// [`can_view`](crate::resource::Resource::can_view) /
261    /// [`can_update`](crate::resource::Resource::can_update) /
262    /// [`can_delete`](crate::resource::Resource::can_delete), each action
263    /// mirroring the predicates its route checks.
264    pub(crate) fn row_actions(
265        mut self,
266        policy: impl Fn(&M) -> RowActions + Send + Sync + 'static,
267    ) -> Self {
268        self.row_policy = Some(Arc::new(policy));
269        self
270    }

272    /// Declare filters. Accepts a single filter or tuple of filters.
273    ///
274    /// Panics on duplicate [`Filter::name`], the same fail-loud
275    /// policy as [`Self::new`]: a filter travels as one `?f.<name>=` parameter,
276    /// and the parser keeps the first value for a repeated name, so two filters
277    /// sharing a name would silently drop one of them.
278    pub fn filters(mut self, filters: impl IntoFilters<M>) -> Self
279    where
280        M: toasty::schema::Model,
281    {
282        let filters = filters.into_filters();
283        let mut seen = std::collections::HashSet::with_capacity(filters.len());
284        for f in &filters {
285            let name = f.name();
286            assert!(
287                seen.insert(name),
288                "duplicate filter name '{name}': each Table filter needs a distinct name (GH #294)"
289            );
290        }
291        self.filters = filters;
292        self
293    }

295    /// `query` with every relation this table's columns declared
296    /// ([`TextColumn::include`](super::TextColumn::include)) included, once
297    /// each. The list and the export load through this; every column renders,
298    /// so the set is the union over all of them.
299    pub(crate) fn include_relations(
300        &self,
301        mut query: toasty::stmt::Query<List<M>>,
302    ) -> toasty::stmt::Query<List<M>>
303    where
304        M: toasty::schema::Model,
305    {
306        let mut seen: Vec<&toasty_core::stmt::Include> = Vec::new();
307        for include in self.columns.iter().flat_map(|c| c.includes()) {
308            if !seen.contains(&include) {
309                seen.push(include);
310                query = query.include(include.clone());
311            }
312        }
313        query
314    }

316    /// Filter predicate for the current `TableState` — `AND` of active filter exprs.
317    pub fn filter_expr(&self, state: &TableState) -> Option<Expr<bool>>
318    where
319        M: toasty::schema::Model,
320    {
321        let mut exprs = Vec::new();
322        for f in &self.filters {
323            if let Some(v) = state.filters.get(f.name())
324                && let Some(e) = f.to_expr(v)
325            {
326                exprs.push(e);
327            }
328        }
329        if exprs.is_empty() {
330            None
331        } else {
332            // `Expr::and` chain: first and all.
333            let mut iter = exprs.into_iter();
334            let first = iter.next().unwrap();
335            Some(iter.fold(first, |acc, e| acc.and(e)))
336        }
337    }

339    /// Requested filters that produced no predicate: `(key:value, reason)`
340    /// where reason is `"unknown filter"` (no declared filter owns the key)
341    /// or `"invalid value"` (the declared filter rejected the value).
342    ///
343    /// Documented no-op values are exempt: `TernaryFilter`'s `all`
344    /// selects no predicate by contract, so it is never flagged.
345    ///
346    /// Filter parameters the parse dropped ([`TableState::filters_dropped`]:
347    /// too many, too long, or the retired `?filters=` spelling) are reported
348    /// as one entry with their own reason.
349    ///
350    /// The list view renders these as a `role=alert` banner and keeps a 200;
351    /// the export refuses the request with 400 instead of silently
352    /// over-sharing an effectively-unfiltered CSV.
353    pub fn unapplied_filters(&self, state: &TableState) -> Vec<(String, String)>
354    where
355        M: toasty::schema::Model,
356    {
357        let mut out = Vec::new();
358        for (key, value) in &state.filters {
359            match self.filters.iter().find(|f| f.name() == key) {
360                None => out.push((format!("{key}:{value}"), "unknown filter".to_string())),
361                Some(f) if f.to_expr(value).is_none() && !f.is_noop_value(value) => {
362                    out.push((format!("{key}:{value}"), "invalid value".to_string()))
363                }
364                Some(_) => {}
365            }
366        }
367        if state.filters_dropped {
368            out.push((
369                "dropped filters".to_string(),
370                format!(
371                    "more than {}, over {} bytes, or the retired filters= form",
372                    super::state::MAX_FILTERS,
373                    super::state::MAX_FILTER_LEN
374                ),
375            ));
376        }
377        out.sort();
378        out
379    }

381    /// Group rows in-memory by a named key (count summarizer). No GROUP BY SQL.
382    ///
383    /// `name` declares the `?group_by=` value this table accepts
384    /// (e.g. `"status"`); any other value renders no group headers and is
385    /// dropped from pager/sort/filter links instead of silently
386    /// grouping by the single declared key. Counts are page-local.
387    ///
388    /// The key closure reads the loaded row, so a relation it reads must be
389    /// loaded: include it on a column ([`TextColumn::include`](super::TextColumn::include))
390    /// or in [`Resource::query`](crate::resource::Resource::query).
391    ///
392    /// In live tables `group_by` travels in the query signal and persists
393    /// across in-place reruns; changing it is still a navigation
394    /// (`?group_by=` links) until a live control ships.
395    pub fn group_by(
396        mut self,
397        name: impl Into<String>,
398        key: impl Fn(&M) -> String + Send + Sync + 'static,
399    ) -> Self {
400        self.group_by = Some(GroupDef {
401            name: name.into(),
402            key: Arc::new(key),
403        });
404        self
405    }

407    /// The declared grouping iff `state.group_by` names it.
408    fn effective_group_key(&self, state: &TableState) -> Option<GroupKey<M>> {
409        match (&self.group_by, &state.group_by) {
410            (Some(def), Some(want)) if def.name == *want => Some(def.key.clone()),
411            _ => None,
412        }
413    }

415    /// Normalize `state.group_by` against the declared grouping (GH #92, GH
416    /// #153): an unknown `?group_by=` value renders no group headers and is
417    /// dropped from every link instead of round-tripping.
418    ///
419    /// The panel's list page and the `table_search` shard normalize where they
420    /// parse the state and render from the result; the public
421    /// [`render_with_state`](Self::render_with_state) normalizes the state it
422    /// is handed, so a page calling it directly keeps the GH #153 guarantee.
423    /// Normalizing twice is a no-op.
424    pub(crate) fn normalize_state(&self, state: &TableState) -> TableState {
425        let mut out = state.clone();
426        if self.group_by.as_ref().map(|def| def.name.as_str()) != out.group_by.as_deref() {
427            out.group_by = None;
428        }
429        out
430    }

566    /// Global search predicate — OR across searchable columns.
567    ///
568    /// Substring match (`?q=` anywhere in the value), escaped so a term
569    /// containing `%` or `_` stays literal; see
570    /// [`TextColumn::to_search_expr`](crate::resource::TextColumn::to_search_expr)
571    /// for the driver case-sensitivity caveat.
572    pub fn search_expr(&self, term: &str) -> Option<Expr<bool>>
573    where
574        M: toasty::schema::Model,
575    {
576        let t = term.trim();
577        if t.is_empty() {
578            return None;
579        }
580        let mut exprs = self.columns.iter().filter_map(|c| c.to_search_expr(t));
581        let first = exprs.next()?;
582        Some(exprs.fold(first, |acc, e| acc.or(e)))
583    }

585    /// First sortable column's order_by. Cursor determinism needs no
586    /// app-level tie-breaker: toasty's engine appends the physical PK columns
587    /// to ambiguous cursor orderings internally (`normalize_cursor_order`,
588    /// tokio-rs/toasty#1142), so page contents are deterministic on SQL
589    /// backends without Tablo's help.
590    pub fn order_by(&self, descending: bool) -> Option<OrderByExpr>
591    where
592        M: toasty::schema::Model,
593    {
594        self.columns.iter().find_map(|c| c.to_order_by(descending))
595    }

597    /// Order-bys over the model's primary key (asc, in declared order) —
598    /// built through the public facade (`Model::path_field` + `Path::asc`),
599    /// no `toasty_core` needed.
600    ///
601    /// Used only when a table declares no sortable column at all: toasty's
602    /// planner requires an `ORDER BY` for cursor pagination and its
603    /// normalization only extends a non-empty ordering, so the PK order must
604    /// be declared app-side in that one case.
605    ///
606    /// # Panics
607    ///
608    /// Never panics: a non-root model has no primary key, so this returns
609    /// empty (and debug-asserts) instead of panicking per request —
610    /// the engine then reports its descriptive "requires an ORDER BY" error
611    /// at load.
612    fn pk_order_bys() -> Vec<OrderByExpr>
613    where
614        M: toasty::schema::Model,
615    {
616        let app_model = M::schema();
617        let Some(root) = app_model.as_root() else {
618            debug_assert!(
619                false,
620                "pk_order_bys: {} is not a root model; deterministic pagination needs its primary key",
621                std::any::type_name::<M>()
622            );
623            return Vec::new();
624        };
625        root.primary_key
626            .fields
627            .iter()
628            .map(|fid| M::path_field::<toasty::stmt::Value>(fid.index).asc())
629            .collect()
630    }

632    /// Resolve the full query ordering for a request.
633    ///
634    /// Single source of truth for loaders, render and the export:
635    /// 1. `?sort=<column>&dir=asc|desc` when `<column>` names a declared sortable column — that
636    ///    column's direction (toasty appends the PK tie-breakers internally, see
637    ///    [`Self::order_by`]);
638    /// 2. otherwise the declared default (first sortable column asc);
639    /// 3. otherwise the PK alone — cursor pagination requires a deterministic order even with no
640    ///    sortable column, and toasty only *extends* an existing non-empty ordering.
641    ///
642    /// Loaders that also need the search term parse the state once with
643    /// [`TableState::from_cx`] and apply the declaration through
644    /// `Self::apply_declaration` (see `crate::panel::Panel`'s generic
645    /// resource list handler).
646    pub fn order_bys_for(&self, state: &TableState) -> Vec<OrderByExpr>
647    where
648        M: toasty::schema::Model,
649    {
650        if let Some(sort) = &state.sort
651            && let Some(col) = self
652                .columns
653                .iter()
654                .find(|c| c.is_sortable() && c.name() == sort.column)
655            && let Some(ord) = col.to_order_by(sort.descending)
656        {
657            return vec![ord];
658        }
659        let out: Vec<OrderByExpr> = self.order_by(false).into_iter().collect();
660        if out.is_empty() {
661            return Self::pk_order_bys();
662        }
663        out
664    }
665
666    /// Apply this table's declaration to `query` — the one routine that turns
667    /// the search term, the filters and the ordering into a query.
668    ///
669    /// `query` is the caller's seed: the list and the export both pass the
670    /// tenant-scoped [`Resource::query`](crate::resource::Resource::query) (the
671    /// row-scoping seam, ADR-0002), and a page-owned table passes its own. One
672    /// routine for both, so a new search or filter dimension cannot reach the
673    /// list and miss the CSV — the drift class GH #172 fixed.
674    pub(crate) fn apply_declaration(
675        &self,
676        mut query: toasty::stmt::Query<List<M>>,
677        state: &TableState,
678    ) -> toasty::stmt::Query<List<M>>
679    where
680        M: toasty::schema::Model,
681    {
682        if let Some(term) = &state.search
683            && let Some(expr) = self.search_expr(term)
684        {
685            query = query.filter(expr);
686        }
687        if let Some(expr) = self.filter_expr(state) {
688            query = query.filter(expr);
689        }
690        for ord in self.order_bys_for(state) {
691            query = query.order_by(ord);
692        }
693        query
694    }

696    /// Whether the search toolbar renders: at least one `searchable()` column,
697    /// unless [`Self::hide_search`] hid it.
698    pub(crate) fn search_enabled(&self) -> bool
699    where
700        M: toasty::schema::Model,
701    {
702        !self.hide_search && self.columns.iter().any(|c| c.is_searchable())
703    }
