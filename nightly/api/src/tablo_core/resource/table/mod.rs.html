22/// Row-key projection: reads the row identity off one model instance
23/// (typically `|u| u.id.to_string()`). Toasty models are plain structs with
24/// no instance→field reflection, so the key cannot be extracted generically
25/// (upstream gap #119).
26pub type RowKey<M> = Arc<dyn Fn(&M) -> String + Send + Sync>;
27
28/// Group-label projection: reads a row's group off one model instance
29/// (typically `|u| u.status.clone()`).
30pub type GroupKey<M> = Arc<dyn Fn(&M) -> String + Send + Sync>;
31
32/// Per-record action policy: reads which row actions one model instance allows.
33/// See [`RowActions`].
34pub(crate) type RowPolicy<M> = Arc<dyn Fn(&M) -> RowActions + Send + Sync>;

36/// Which row actions one record may use.
37///
38/// The action prefixes say which affordances a resource declares; this says
39/// which of them the caller may use
40/// on one loaded row. The panel wires the projection into the table and the
41/// renderer consults it per row — a denied action emits no link, and a row
42/// denied `delete` renders no bulk checkbox, so the row can never enter the
43/// selection transport.
44///
45/// The panel derives one from the resource's
46/// [`can_view`](crate::resource::Resource::can_view) /
47/// [`can_update`](crate::resource::Resource::can_update) /
48/// [`can_delete`](crate::resource::Resource::can_delete), pairing each action
49/// with the same predicates its route checks, so a rendered affordance and the
50/// route that answers it cannot disagree.
51#[derive(Debug, Clone, Copy, PartialEq, Eq)]
52pub(crate) struct RowActions {
53    /// Whether the row renders its `View` link — the detail route's
54    /// `can_view`.
55    pub view: bool,
56    /// Whether the row renders its `Edit` link — the edit route's `can_view`
57    /// **and** `can_update`.
58    pub edit: bool,
59    /// Whether the row renders its `Delete` link and an enabled bulk checkbox —
60    /// the delete route's `can_view` **and** `can_delete`.
61    pub delete: bool,
62}
63
64impl RowActions {
65    /// Every action allowed: what a table without a panel-wired row policy
66    /// renders.
67    pub const ALL: Self = Self {
68        view: true,
69        edit: true,
70        delete: true,
71    };
72}

237impl<M> Table<M> {
238    /// Declare a table with its row key and columns.
239    ///
240    /// One primary-key projection (typically `|u| u.id.to_string()`) drives
241    /// both halves: keyed diffs and DOM ids, and the action URLs and bulk
242    /// values handlers resolve as the model's typed PK. The projection must be
243    /// injective within a page: duplicate keys corrupt keyed diffs and bulk
244    /// selection, and are debug-asserted at render time. A table whose display
245    /// projects a non-PK value uses [`Self::new_split`].
246    ///
247    /// Two columns sharing a [`Column::name`] are a misdeclaration
248    /// ([`Self::declaration_errors`]): sort resolution is
249    /// first-sortable-`name()`-match, so duplicate sortable names would
250    /// silently misresolve `?sort=`. The rule covers computed names too
251    /// (`TextColumn::computed("Status", ..)` derives `name = "status"`). So is
252    /// an empty column set: a table with no columns renders a headers-only
253    /// list, which no resource declares.
254    pub fn new(
255        key: impl Fn(&M) -> String + Send + Sync + 'static,
256        cols: impl IntoColumns<M>,
257    ) -> Self
258    where
259        M: toasty::schema::Model,
260    {
261        let key = Arc::new(key);
262        Self::from_keys(key.clone(), key, cols)
263    }
264
265    /// Declare a table whose display projects a non-PK value.
266    ///
267    /// `display` drives keyed diffs and DOM ids; `record` drives action URLs
268    /// and bulk checkbox values handlers resolve as the model's typed PK
269    /// (`pk_eq_expr` / `pk_in_expr` — an unparseable value 404s), so emitting
270    /// a display key there 404s every delete and bulk submit.
271    ///
272    /// Duplicate column names and an empty column set are misdeclarations,
273    /// as for [`Self::new`].
274    pub fn new_split(
275        display: impl Fn(&M) -> String + Send + Sync + 'static,
276        record: impl Fn(&M) -> String + Send + Sync + 'static,
277        cols: impl IntoColumns<M>,
278    ) -> Self
279    where
280        M: toasty::schema::Model,
281    {
282        Self::from_keys(Arc::new(display), Arc::new(record), cols)
283    }

332    /// Declare the per-record action policy: which of the wired row
333    /// actions each record may use.
334    ///
335    /// The `with_*` methods say which affordances the table declares; this says
336    /// which of them a row may use. The renderer consults it per row — a denied
337    /// action emits no link, and a row denied `delete` renders no bulk

342    /// renders exactly the chrome its `with_*` calls attached. The policy is
343    /// consulted only where chrome is wired: a table with no action prefix
344    /// never calls it, and a resource that declares no chrome keeps its list
345    /// page free of per-record predicate calls.
346    ///
347    /// The panel wires this from
348    /// [`can_view`](crate::resource::Resource::can_view) /
349    /// [`can_update`](crate::resource::Resource::can_update) /
350    /// [`can_delete`](crate::resource::Resource::can_delete), each action
351    /// mirroring the predicates its route checks.
352    pub(crate) fn row_actions(
353        mut self,
354        policy: impl Fn(&M) -> RowActions + Send + Sync + 'static,
355    ) -> Self {
356        self.row_policy = Some(Arc::new(policy));
357        self
358    }

360    /// Declare filters. Accepts a single filter or tuple of filters.
361    ///
362    /// Two filters sharing a [`Filter::name`](super::Filter::name) are a
363    /// misdeclaration, as for [`Self::new`]: a filter travels as one
364    /// `?f.<name>=` parameter, and the parser keeps the first value for a
365    /// repeated name, so two filters sharing a name would silently drop one of
366    /// them.
367    pub fn filters(mut self, filters: impl IntoFilters<M>) -> Self
368    where
369        M: toasty::schema::Model,
370    {
371        let filters = filters.into_filters();
372        self.filters = filters;
373        self
374    }

376    /// `query` with every relation this table's columns declared
377    /// ([`Column::includes`]) included, once
378    /// each. The list and the export load through this; every column renders,
379    /// so the set is the union over all of them.
380    pub(crate) fn include_relations(
381        &self,
382        mut query: toasty::stmt::Query<List<M>>,
383    ) -> toasty::stmt::Query<List<M>>
384    where
385        M: toasty::schema::Model,
386    {
387        let mut seen: Vec<toasty_core::stmt::Include> = Vec::new();
388        for include in self.columns.iter().flat_map(|c| c.includes().into_vec()) {
389            if !seen.contains(&include) {
390                query = query.include(include.clone());
391                seen.push(include);
392            }
393        }
394        query
395    }

397    /// Filter predicate for the current `TableState` — `AND` of active filter exprs.
398    pub fn filter_expr(&self, state: &TableState) -> Option<Expr<bool>>
399    where
400        M: toasty::schema::Model,
401    {
402        let mut exprs = Vec::new();
403        for f in &self.filters {
404            if let Some(v) = state.filters.get(f.name())
405                && let Some(e) = f.to_expr(v)
406            {
407                exprs.push(e);
408            }
409        }
410        if exprs.is_empty() {
411            None
412        } else {
413            // `Expr::and` chain: first and all.
414            let mut iter = exprs.into_iter();
415            let first = iter.next().unwrap();
416            Some(iter.fold(first, |acc, e| acc.and(e)))
417        }
418    }

420    /// Requested filters that produced no predicate: `(key:value, reason)`
421    /// where reason is `"unknown filter"` (no declared filter owns the key)
422    /// or `"invalid value"` (the declared filter rejected the value).
423    ///
424    /// Documented no-op values are exempt: `TernaryFilter`'s `all`
425    /// selects no predicate by contract, so it is never flagged.
426    ///
427    /// Filter parameters the parse dropped ([`TableState::filters_dropped`]:
428    /// too many, too long, or the retired `?filters=` spelling) are reported
429    /// as one entry with their own reason.
430    ///
431    /// The list view renders these as a `role=alert` banner and keeps a 200;
432    /// the export refuses the request with 400 instead of silently
433    /// over-sharing an effectively-unfiltered CSV.
434    pub fn unapplied_filters(&self, state: &TableState) -> Vec<(String, String)>
435    where
436        M: toasty::schema::Model,
437    {
438        let mut out = Vec::new();
439        for (key, value) in &state.filters {
440            match self.filters.iter().find(|f| f.name() == key) {
441                None => out.push((format!("{key}:{value}"), "unknown filter".to_string())),
442                Some(f) if f.to_expr(value).is_none() && !f.is_noop_value(value) => {
443                    out.push((format!("{key}:{value}"), "invalid value".to_string()))
444                }
445                Some(_) => {}
446            }
447        }
448        if state.filters_dropped {
449            out.push((
450                "dropped filters".to_string(),
451                format!(
452                    "more than {}, over {} bytes, or the retired filters= form",
453                    super::state::MAX_FILTERS,
454                    super::state::MAX_FILTER_LEN
455                ),
456            ));
457        }
458        out.sort();
459        out
460    }

462    /// Group rows in-memory by a named key (count summarizer). No GROUP BY SQL.
463    ///
464    /// `name` declares the `?group_by=` value this table accepts
465    /// (e.g. `"status"`); any other value renders no group headers and is
466    /// dropped from pager/sort/filter links instead of silently
467    /// grouping by the single declared key. Counts are page-local.
468    ///
469    /// The key closure reads the loaded row, so a relation it reads must be
470    /// loaded: include it on a column ([`TextColumn::include`](super::TextColumn::include),
471    /// [`Column::includes`]) or in [`Resource::query`](crate::resource::Resource::query).
472    ///
473    /// In live tables `group_by` travels in the query signal and persists
474    /// across in-place reruns; changing it is still a navigation
475    /// (`?group_by=` links) until a live control ships.

477        mut self,
478        name: impl Into<String>,
479        key: impl Fn(&M) -> String + Send + Sync + 'static,
480    ) -> Self {
481        self.group_by = Some(GroupDef {
482            name: name.into(),
483            key: Arc::new(key),

488    /// The declared grouping iff `state.group_by` names it.
489    fn effective_group_key(&self, state: &TableState) -> Option<GroupKey<M>> {
490        match (&self.group_by, &state.group_by) {
491            (Some(def), Some(want)) if def.name == *want => Some(def.key.clone()),
492            _ => None,
493        }

496    /// Normalize `state.group_by` against the declared grouping (GH #92, GH
497    /// #153): an unknown `?group_by=` value renders no group headers and is
498    /// dropped from every link instead of round-tripping.
499    ///
500    /// The panel's list page and the `table_search` shard normalize where they
501    /// parse the state and render from the result; the public
502    /// [`render_with_state`](Self::render_with_state) normalizes the state it
503    /// is handed, so a page calling it directly keeps the GH #153 guarantee.
504    /// Normalizing twice is a no-op.
505    pub(crate) fn normalize_state(&self, state: &TableState) -> TableState {
506        let mut out = state.clone();
507        if self.group_by.as_ref().map(|def| def.name.as_str()) != out.group_by.as_deref() {
508            out.group_by = None;
509        }
510        out
511    }

513    /// Set the page size. Every table paginates with cursor pagination, at
514    /// [`DEFAULT_PAGE_SIZE`] rows unless it declares otherwise; the render
515    /// shows Previous/Next links built from the executed page's cursors, never
516    /// page numbers.
517    ///
518    /// Zero is a misdeclaration ([`Self::declaration_errors`]): a page of no
519    /// rows is a programmer error. The table keeps its page size.
520    pub fn paginate(mut self, per_page: usize) -> Self {
521        match NonZeroUsize::new(per_page) {
522            Some(size) => self.page_size = size,
523            None => self
524                .misdeclared
525                .push("Table::paginate: a page size must be at least 1".to_string()),
526        }
527        self
528    }

530    /// What is wrong with this declaration: an empty column set, two columns
531    /// or two filters sharing a name, a zero page size, and what each column
532    /// and filter reports of itself (a lens that binds no single field, a
533    /// search or sort asked of a computed column).
534    ///
535    /// [`RouterBuilderPanelExt::panel`](crate::RouterBuilderPanelExt::panel) refuses a resource
536    /// whose table reports any, and rendering one fails with them.
537    pub fn declaration_errors(&self) -> Vec<String> {
538        let mut errors = self.misdeclared.clone();
539        if self.columns.is_empty() {
540            errors.push(
541                "a Table needs at least one column: declare columns with Table::new(key, columns)"
542                    .to_string(),
543            );
544        }
545        let mut seen = std::collections::HashSet::with_capacity(self.columns.len());
546        for column in &self.columns {
547            match column.misdeclared() {
548                Some(error) => errors.push(error),
549                None if !seen.insert(column.name()) => errors.push(format!(
550                    "duplicate column name '{}': each Table column needs a distinct name",
551                    column.name()
552                )),
553                None => {}
554            }
555        }
556        let mut seen = std::collections::HashSet::with_capacity(self.filters.len());
557        for filter in &self.filters {
558            match filter.misdeclared() {
559                Some(error) => errors.push(error),
560                None if !seen.insert(filter.name()) => errors.push(format!(
561                    "duplicate filter name '{}': each Table filter needs a distinct name",
562                    filter.name()
563                )),
564                None => {}
565            }
566        }
567        errors
568    }

742    /// Global search predicate — OR across searchable columns.
743    ///
744    /// Substring match (`?q=` anywhere in the value), escaped so a term
745    /// containing `%` or `_` stays literal; see
746    /// [`TextColumn`](crate::resource::TextColumn)'s search for the driver
747    /// case-sensitivity caveat.
748    pub fn search_expr(&self, term: &str) -> Option<Expr<bool>>
749    where
750        M: toasty::schema::Model,
751    {
752        let t = term.trim();
753        if t.is_empty() {
754            return None;
755        }
756        let mut exprs = self.columns.iter().filter_map(|c| c.search_expr(t));
757        let first = exprs.next()?;
758        Some(exprs.fold(first, |acc, e| acc.or(e)))
759    }

761    /// First sortable column's order_by. Cursor determinism needs no
762    /// app-level tie-breaker: toasty's engine appends the physical PK columns
763    /// to ambiguous cursor orderings internally (`normalize_cursor_order`,
764    /// tokio-rs/toasty#1142), so page contents are deterministic on SQL
765    /// backends without Tablo's help.
766    pub fn order_by(&self, descending: bool) -> Option<OrderByExpr>
767    where
768        M: toasty::schema::Model,
769    {
770        self.columns.iter().find_map(|c| c.order_by(descending))
771    }

773    /// Order-bys over the model's primary key (asc, in declared order) —
774    /// built through the public facade (`Model::path_field` + `Path::asc`),
775    /// no `toasty_core` needed.
776    ///
777    /// Used only when a table declares no sortable column at all: toasty's
778    /// planner requires an `ORDER BY` for cursor pagination and its
779    /// normalization only extends a non-empty ordering, so the PK order must
780    /// be declared app-side in that one case.
781    ///
782    /// # Panics
783    ///
784    /// Never panics: a non-root model has no primary key, so this returns
785    /// empty (and debug-asserts) instead of panicking per request —
786    /// the engine then reports its descriptive "requires an ORDER BY" error
787    /// at load.
788    fn pk_order_bys() -> Vec<OrderByExpr>
789    where
790        M: toasty::schema::Model,
791    {
792        let app_model = M::schema();
793        let Some(root) = app_model.as_root() else {
794            debug_assert!(
795                false,
796                "pk_order_bys: {} is not a root model; deterministic pagination needs its primary key",
797                std::any::type_name::<M>()
798            );
799            return Vec::new();
800        };
801        root.primary_key
802            .fields
803            .iter()
804            .map(|fid| M::path_field::<toasty::stmt::Value>(fid.index).asc())
805            .collect()
806    }

808    /// Resolve the full query ordering for a request.
809    ///
810    /// Single source of truth for loaders, render and the export:
811    /// 1. `?sort=<column>&dir=asc|desc` when `<column>` names a declared sortable column — that
812    ///    column's direction (toasty appends the PK tie-breakers internally, see
813    ///    [`Self::order_by`]);
814    /// 2. otherwise the declared default (first sortable column asc);
815    /// 3. otherwise the PK alone — cursor pagination requires a deterministic order even with no
816    ///    sortable column, and toasty only *extends* an existing non-empty ordering.
817    ///
818    /// Loaders that also need the search term parse the state once with
819    /// [`TableState::from_cx`] and apply the declaration through
820    /// `Self::apply_declaration` (see `crate::panel::Panel`'s generic
821    /// resource list handler).
822    pub fn order_bys_for(&self, state: &TableState) -> Vec<OrderByExpr>
823    where
824        M: toasty::schema::Model,
825    {
826        if let Some(sort) = &state.sort
827            && let Some(col) = self
828                .columns
829                .iter()
830                .find(|c| c.is_sortable() && c.name() == sort.column)
831            && let Some(ord) = col.order_by(sort.descending)
832        {
833            return vec![ord];
834        }
835        let out: Vec<OrderByExpr> = self.order_by(false).into_iter().collect();
836        if out.is_empty() {
837            return Self::pk_order_bys();
838        }
839        out
840    }

842    /// Apply this table's declaration to `query` — the one routine that turns
843    /// the search term, the filters and the ordering into a query.
844    ///
845    /// `query` is the caller's seed: the list and the export both pass the
846    /// tenant-scoped [`Resource::query`](crate::resource::Resource::query) (the
847    /// row-scoping seam, ADR-0002), and a page-owned table passes its own. One
848    /// routine for both, so a new search or filter dimension cannot reach the
849    /// list and miss the CSV — the drift class GH #172 fixed.
850    pub(crate) fn apply_declaration(
851        &self,
852        mut query: toasty::stmt::Query<List<M>>,
853        state: &TableState,
854    ) -> toasty::stmt::Query<List<M>>
855    where
856        M: toasty::schema::Model,
857    {
858        if let Some(term) = &state.search
859            && let Some(expr) = self.search_expr(term)
860        {
861            query = query.filter(expr);
862        }
863        if let Some(expr) = self.filter_expr(state) {
864            query = query.filter(expr);
865        }
866        for ord in self.order_bys_for(state) {
867            query = query.order_by(ord);
868        }
869        query
870    }
