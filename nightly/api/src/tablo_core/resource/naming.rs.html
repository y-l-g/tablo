6/// The last segment of a type's full path, e.g.
7/// `tablo_core::resource::tests::UserResource` → `UserResource`.
8pub(crate) fn type_short_name<T: ?Sized>() -> &'static str {
9    let name = std::any::type_name::<T>();
10    name.rsplit("::").next().unwrap_or(name)
11}

13/// Pluralize a capitalized English word with a compact ruleset (Filament
14/// pluralizes via Laravel's `Str::plural`; this is the admin-grade subset):
15/// a small irregular table (`person` → `people`, …), consonant-`y` → `ies`
16/// (`Category` → `Categories`), sibilant endings → `es` (`Box` → `Boxes`),
17/// `f`/`fe` → `ves` (`Knife` → `Knives`) with a few `+s` exceptions, and the
18/// default `+s`.
19pub(crate) fn pluralize(word: &str) -> String {
20    if word.is_empty() {
21        return word.to_string();
22    }
23    let lower = word.to_lowercase();
24    const IRREGULAR: &[(&str, &str)] = &[
25        ("person", "people"),
26        ("man", "men"),
27        ("woman", "women"),
28        ("child", "children"),
29        ("mouse", "mice"),
30        ("goose", "geese"),
31        ("foot", "feet"),
32        ("tooth", "teeth"),
33        ("datum", "data"),
34        ("criterion", "criteria"),
35        ("index", "indices"),
36        ("matrix", "matrices"),
37        ("vertex", "vertices"),
38        ("axis", "axes"),
39        ("crisis", "crises"),
40        ("analysis", "analyses"),
41    ];
42    if let Some((_, plural)) = IRREGULAR.iter().find(|(singular, _)| *singular == lower) {
43        return match word.chars().next() {
44            Some(first) if first.is_uppercase() => capitalize(plural),
45            _ => (*plural).to_string(),
46        };
47    }
48    // `f`/`fe` → `ves`, except the words that simply take `s`.
49    const F_EXCEPTIONS: &[&str] = &["roof", "chief", "belief", "chef", "cliff", "cuff"];
50    const UNCOUNTABLE: &[&str] = &[
51        "fish",
52        "sheep",
53        "deer",
54        "moose",
55        "series",
56        "species",
57        "news",
58        "equipment",
59        "information",
60        "rice",
61    ];
62    if UNCOUNTABLE.contains(&lower.as_str()) {
63        return word.to_string();
64    }
65    if F_EXCEPTIONS.contains(&lower.as_str()) {
66        format!("{word}s")
67    } else if lower.ends_with('f') {
68        format!("{}ves", &word[..word.len() - 1])
69    } else if lower.ends_with("fe") {
70        format!("{}ves", &word[..word.len() - 2])
71    } else if lower.ends_with('y')
72        && word.len() > 1
73        && !"aeiou".contains(word.chars().nth(word.len() - 2).unwrap_or(' '))
74    {
75        format!("{}ies", &word[..word.len() - 1])
76    } else if ["s", "ss", "sh", "ch", "x", "z"]
77        .iter()
78        .any(|suffix| lower.ends_with(suffix))
79    {
80        format!("{word}es")
81    } else {
82        format!("{word}s")
83    }
84}

86/// Convert a CamelCase identifier to kebab-case: `BlogPost` → `blog-post`,
87/// `APIKey` → `api-key`.
88///
89/// Delegates to `heck::ToKebabCase`: digits split words
90/// (`User2FA` → `user2-fa`) and so do underscores (`Audit_Log` → `audit-log`).
91/// Name resources without underscores or override
92/// [`Resource::slug`](crate::Resource::slug).
93pub(crate) fn kebab_case(name: &str) -> String {
94    use heck::ToKebabCase;
95    name.to_kebab_case()
96}
