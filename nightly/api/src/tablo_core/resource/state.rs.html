17/// The live table's browser state.
18///
19/// The page owns these signals and hands their handles to the `table_search`
20/// shard through `Table::render_live_with_state`; each tracked read inside
21/// the shard becomes a `dep` marker the browser watches, so writing any signal
22/// re-renders the table in place — no navigation, no scroll jump. Sort links,
23/// the pager, the filter transport, the bulk selection, and the clear links
24/// rendered by the table write them.
25///
26/// The two carriers meet in exactly two methods:
27/// [`TableState::to_signals`] seeds these handles from a parsed state, and
28/// [`TableSignals::to_state`] rebuilds the state from their current values.
29/// A new interaction dimension is a field here plus one arm in each, instead
30/// of a hand-written conversion at every seam.
31///
32/// `q`/`filters`/`sort`/`dir`/`group_by` reset the cursor when they change;
33/// [`Self::cursor`] pages within the current result set. All values are
34/// untrusted by the time the shard reads them back (the client owns the
35/// signal).
36#[derive(Clone)]
37pub struct TableSignals {
38    /// `?q=` — the search term (escaped substring match).
39    pub q: Signal<String>,
40    /// `?filters=` — the composed `key:value,key2:value2` transport.
41    pub filters: Signal<String>,
42    /// `?sort=` — the active sort column name (`""` = the table default).
43    pub sort: Signal<String>,
44    /// `?dir=` — `asc`/`desc` for [`Self::sort`].
45    pub dir: Signal<String>,
46    /// The live cursor, as one signal: `""` (no cursor),
47    /// `after:<token>` or `before:<token>`. One signal makes the
48    /// `after`+`before` pair Toasty rejects unrepresentable in the browser — no
49    /// cross-write interleaving can produce it — and lets every result-set
50    /// transition clear pagination with a single write. Written through
51    /// `cursor_after` / `cursor_before` / `cursor_none`, read through
52    /// `split_cursor`.
53    pub cursor: Signal<String>,
54    /// `?group_by=` — the active grouping (`""` = ungrouped).
55    /// Seeded from the page-load state and changed via navigation
56    /// (`?group_by=` links); no live control writes it yet, so it persists
57    /// across in-place reruns. A future control writing it must clear the
58    /// cursor like the other result-set dimensions.
59    pub group_by: Signal<String>,
60    /// The bulk selection: comma-separated record keys, `""` when nothing is
61    /// selected. Row checkboxes carry no `checked` attribute —
62    /// `bulk.js` sets `checked` from the transport after every swap and change
63    /// — and the script writes the transport, whose bound `change` handler
64    /// writes this signal, so a live rerun re-renders the boxes from the
65    /// selection instead of dropping it. The shard carries the handle without
66    /// reading it: the table needs it to bind the boxes, but a checkbox click
67    /// must not reload rows.
68    pub bulk: Signal<String>,
69}

71/// The live cursor wire format: `after:<token>` / `before:<token>`,
72/// with the empty string meaning "no cursor". One signal carries it, so the
73/// browser can never hold both cursors at once.
74const CURSOR_AFTER: &str = "after:";
75const CURSOR_BEFORE: &str = "before:";
76
77/// Wire value for a forward cursor (`?after=<token>`).
78pub(crate) fn cursor_after(token: &str) -> String {
79    format!("{CURSOR_AFTER}{token}")
80}
81
82/// Wire value for a backward cursor (`?before=<token>`).
83pub(crate) fn cursor_before(token: &str) -> String {
84    format!("{CURSOR_BEFORE}{token}")
85}
86
87/// Wire value for no cursor — what every result-set transition writes, and
88/// what a fresh page seeds.
89pub(crate) fn cursor_none() -> String {
90    String::new()
91}

93/// Split the live cursor wire value into the `(after, before)` pair the loader
94/// consumes. At most one side is ever `Some`: a value naming neither direction
95/// (a tampered signal, or one the client never sent) degrades to "no cursor" —
96/// the drop-pagination retry contract of GH #110 — rather than the pair error
97/// of GH #155 that a URL carrying both cursors reaches at load time.
98pub(crate) fn split_cursor(wire: &str) -> (Option<String>, Option<String>) {
99    let wire = wire.trim();
100    for (prefix, forward) in [(CURSOR_AFTER, true), (CURSOR_BEFORE, false)] {
101        if let Some(token) = wire.strip_prefix(prefix) {
102            let token = token.trim();
103            if token.is_empty() {
104                break;
105            }
106            return if forward {
107                (Some(token.to_string()), None)
108            } else {
109                (None, Some(token.to_string()))
110            };
111        }
112    }
113    (None, None)
114}

116/// Whether `key` is selected in the live bulk wire.
117///
118/// The wire is comma-delimited on both ends — `,a,b,`, empty when nothing is
119/// selected — so the client-side `checked` binding tests membership with a
120/// plain `contains(",<key>,")` instead of a substring test that would confuse
121/// `b` with `ab`. [`parse_bulk_ids`](crate::panel) already ignores the empty
122/// segments the delimiters produce, so the same wire is the form transport.
123#[cfg(test)]
124pub(crate) fn bulk_wire_contains(wire: &str, key: &str) -> bool {
125    wire.split(',').any(|segment| segment == key)
126}

128/// One executed page of rows for `Table::render`.
129///
130/// For paginated tables build it from toasty's `Page` via
131/// [`Self::from_toasty_page`] (which URL-encodes the engine cursors); for
132/// unpaginated tables `Vec<M>` converts directly. An absent cursor simply
133/// means no Previous/Next link is rendered — the chrome never invents pages.
134#[derive(Debug, Clone)]
135pub struct TablePage<M> {
136    /// The rows of this page.
137    pub rows: Vec<M>,
138    /// Encoded cursor for the next page (`?after=`), when one exists.
139    pub next_cursor: Option<String>,
140    /// Encoded cursor for the previous page (`?before=`), when one exists.
141    pub prev_cursor: Option<String>,
142}
143
144impl<M> From<Vec<M>> for TablePage<M> {
145    fn from(rows: Vec<M>) -> Self {
146        Self {
147            rows,
148            next_cursor: None,
149            prev_cursor: None,
150        }
151    }
152}

154impl<M: toasty::schema::Model> TablePage<M> {
155    /// Wrap a toasty cursor-pagination result, encoding its cursors for URLs.
156    ///
157    /// # Errors
158    ///
159    /// Errors when a cursor contains a value the URL codec cannot represent
160    /// (see `crate::cursor`).
161    pub fn from_toasty_page(page: toasty::stmt::Page<M>) -> Result<Self> {
162        Ok(Self {
163            rows: page.items,
164            next_cursor: page
165                .next_cursor
166                .as_ref()
167                .map(crate::cursor::encode)
168                .transpose()?,
169            prev_cursor: page
170                .prev_cursor
171                .as_ref()
172                .map(crate::cursor::encode)
173                .transpose()?,
174        })
175    }
176}

188/// Request-scoped table state, parsed from the current URL query.
189///
190/// The single parse point shared by loaders (the search term, ordering via
191/// `Table::order_bys_for`) and render (active sort, toolbar values,
192/// pagination links), so the URL is the one truth for list state. The fixed parameter
193/// names assume one table per page — per-table prefixes are deferred until a
194/// real page needs two tables.
195#[derive(Debug, Clone, Default, PartialEq, Eq)]
196pub struct TableState {
197    /// `?q=` — trimmed and clamped to `MAX_QUERY_TERM` chars; `None` when
198    /// absent or blank.
199    pub search: Option<String>,
200    /// `?sort=` + `?dir=` — `None` when absent or blank.
201    pub sort: Option<Sort>,
202    /// `?after=` — encoded forward cursor.
203    pub after: Option<String>,
204    /// `?before=` — encoded backward cursor.
205    pub before: Option<String>,
206    /// `?filters=` — `key:value,key2:value2` (comma-separated, colon-delimited),
207    /// bounded at parse time by `MAX_FILTERS_PARAM`/`MAX_FILTER_SEGMENTS`.
208    pub filters: HashMap<String, String>,
209    /// `?filters=` segments that carry no `key:value` pair: kept so
210    /// `Table::unapplied_filters` can flag them (list banner, export 400)
211    /// instead of silently dropping them, and so [`Self::filters_param`]
212    /// round-trips them — a link built from this state keeps the warning
213    /// until a valid `?filters=` replaces it.
214    pub malformed_filters: Vec<String>,
215    /// `?group_by=` — field name to group by (in-memory, `count` summarizer).
216    pub group_by: Option<String>,
217    /// `?delete=` — the row key whose delete confirmation dialog opens on the
218    /// list page. The dialog's confirmed POST re-enters the delete
219    /// route; the parameter itself is never a write.
220    pub delete: Option<String>,
221    /// `?open=false` — set by `dialog.js` when Escape/backdrop dismisses the
222    /// delete dialog, so the next render stays closed. Absent (or `true`)
223    /// renders it open.
224    pub open: Option<bool>,
225}

227/// The URL parameters one table link projects, named so a call site reads
228/// which intent drops what.
229///
230/// `Default` is the projection that drops everything: a caller names only the
231/// parameters it keeps.
232#[derive(Default)]
233struct UrlProjection<'a> {
234    /// `?q=` search term.
235    search: Option<&'a str>,
236    /// `?sort=` column and `?dir=` value.
237    sort: Option<(&'a str, &'a str)>,
238    /// `?filters=` transport.
239    filters: Option<&'a str>,
240    /// `?group_by=` column.
241    group_by: Option<&'a str>,
242    /// `?after=` forward cursor.
243    after: Option<&'a str>,
244    /// `?before=` backward cursor.
245    before: Option<&'a str>,
246    /// `?delete=` row key for the confirmation dialog.
247    delete: Option<&'a str>,
248}

250/// Longest `?filters=` transport parsed: the live shard hands this
251/// the client-owned `filters` signal, and the router buffers shard bodies up
252/// to megabytes — so the same bounded-echoed-state posture as
253/// [`MAX_QUERY_TERM`](crate::query_term::MAX_QUERY_TERM) has to hold here,
254/// where the transport is parsed, rather than at the shard that happens to read
255/// it.
256pub(crate) const MAX_FILTERS_PARAM: usize = 1024;
257
258/// Most segments one `?filters=` transport may carry: the byte cap
259/// alone still admits a thousand one-character segments, and each surviving
260/// segment becomes a map entry every rebuilt URL echoes.
261pub(crate) const MAX_FILTER_SEGMENTS: usize = 32;
262
263/// The one segment an over-long or over-full `?filters=` collapses to.
264/// It carries no `key:value` pair, so it rides the GH #148 malformed channel:
265/// [`Table::unapplied_filters`](crate::resource::Table::unapplied_filters)
266/// flags it (the list warns, the export refuses with 400 instead of exporting
267/// an over-broad CSV) and [`TableState::filters_param`] re-emits it, so the
268/// warning survives pagination and sort links.
269pub(crate) const FILTERS_OVERFLOW_SEGMENT: &str = "filters=overflow";

271impl TableState {
272    /// Parse the state from the request in `cx`.
273    ///
274    /// A blank or unknown query parses as neutral state rather than failing
275    /// the request. Duplicate keys (`?filters=a&filters=b`) resolve to the
276    /// first occurrence, so a repeated filter never vanishes: rejecting the
277    /// duplicate would fail the whole decode, and answering empty state instead
278    /// would drop every filter — including export's fail-closed guard.
279    /// Cursor errors still surface later, at decode time, where they are
280    /// precise — including the conflicting `after` + `before` pair, which fails
281    /// at load time. Renders without a request context (e.g. unit
282    /// tests) get neutral state.
283    pub fn from_cx(cx: &Cx) -> Self {
284        let Some(parts) = topcoat::context::try_request_context::<http::request::Parts>(cx) else {
285            return Self::default();
286        };
287        let params = first_wins_query_params(parts.uri.query().unwrap_or(""));
288        Self::from_parts(|key| params.get(key).map(String::as_str))
289    }
290
291    /// One shared constructor behind [`Self::from_cx`] and
292    /// [`Self::from_live_args`]: every query-state parse
293    /// funnels through one contract, so the public live-args entry point — the
294    /// documented seam for a page owning its own signals — cannot be the
295    /// looser one. `q` is trimmed and clamped to
296    /// [`MAX_QUERY_TERM`](crate::query_term::MAX_QUERY_TERM), `dir` is
297    /// trimmed before comparing, and the `filters` transport is bounded at
298    /// [`MAX_FILTERS_PARAM`]/[`MAX_FILTER_SEGMENTS`].
299    ///
300    /// The live args arrive named at the single `match` below, where a
301    /// transposed positional pair would not compile silently.
302    fn from_parts<'a>(get: impl Fn(&str) -> Option<&'a str>) -> Self {
303        let non_empty = |v: Option<&str>| {
304            v.map(str::trim)
305                .filter(|t| !t.is_empty())
306                .map(str::to_string)
307        };
308        let (filters, malformed_filters) = parse_filters_param(get("filters").unwrap_or_default());
309        Self {
310            search: get("q").map(clamp_query_term).filter(|t| !t.is_empty()),
311            sort: non_empty(get("sort")).map(|column| Sort {
312                column,
313                descending: get("dir").map(str::trim) == Some("desc"),
314            }),
315            after: non_empty(get("after")),
316            before: non_empty(get("before")),
317            filters,
318            malformed_filters,
319            group_by: non_empty(get("group_by")),
320            // The delete dialog is opt-in through `?delete=`; `?open=false`
321            // is the dismissal mirror `dialog.js` writes. Any other
322            // `open` value stays neutral (open).
323            delete: non_empty(get("delete")),
324            open: match get("open") {
325                Some("false") => Some(false),
326                Some("true") => Some(true),
327                _ => None,
328            },
329        }
330    }

332    /// Serialized `filters` for URL (`key:value,key2:value2`), or `None` when empty.
333    ///
334    /// Keys/values escape `%`, `:`, `,` (`%25`/`%3A`/`%2C`) so a
335    /// free-text value like `a,b` round-trips instead of splitting.
336    ///
337    /// This is the expensive half of a URL projection (a `format!` per pair,
338    /// a sort, a join, and a percent-encode per byte), so a table render must
339    /// encode it a bounded number of times — never once per row.
340    /// `TableState::row_url_base` exists to make that structural.
341    pub fn filters_param(&self) -> Option<String> {
342        if self.filters.is_empty() && self.malformed_filters.is_empty() {
343            return None;
344        }
345        let mut pairs: Vec<String> = self
346            .filters
347            .iter()
348            .map(|(k, v)| {
349                format!(
350                    "{}:{}",
351                    encode_filter_component(k),
352                    encode_filter_component(v)
353                )
354            })
355            .collect();
356        pairs.sort();
357        // Malformed segments ride along verbatim: they have no
358        // colon to protect and re-enter `parse_filters_param` as malformed on
359        // the next request, keeping the banner (and export's fail-closed 400)
360        // alive across pagination.
361        pairs.extend(self.malformed_filters.iter().cloned());
362        Some(pairs.join(","))
363    }

365    /// URL projection: `TableState` owns the table's URL vocabulary.
366    /// Callers ask for a user intent, never a parameter list, so adding a
367    /// parameter cannot silently drop it from half the links.
368    ///
369    /// One private encoder ([`Self::project_url`]) holds the vocabulary in
370    /// canonical order `q, sort, dir, filters, group_by, after, before`
371    /// (`delete` appended by its intent). The parser is first-wins with unique
372    /// keys, so order is semantically irrelevant.
373    ///
374    /// Expects `group_by` pre-normalized: render seams normalize through
375    /// [`Table::normalize_state`], so the projection echoes `state.group_by`
376    /// as-is. `open` is never emitted by any link; `delete` only by
377    /// [`Self::row_url_base`]'s dialog intent.
378    ///
379    /// Full state, including cursors; never `delete`/`open`. The streamed
380    /// retry link for failures that keep their evidence.
381    pub(crate) fn list_url(&self, path: &str) -> String {
382        let filters = self.filters_param();
383        self.project_url(
384            path,
385            UrlProjection {
386                search: self.search.as_deref(),
387                sort: self.sort_pair(),
388                filters: filters.as_deref(),
389                group_by: self.group_by.as_deref(),
390                after: self.after.as_deref(),
391                before: self.before.as_deref(),
392                ..Default::default()
393            },
394        )
395    }

541    /// Rebuild list state from live-search shard args.
542    ///
543    /// Shard requests hit the `table_search` shard's own endpoint, so
544    /// [`Self::from_cx`] would see the endpoint URI — not the list page's
545    /// query. The page hands over the current values of its interaction
546    /// signals instead: the search term, the filter transport, the sort column
547    /// and its direction, and the page's grouping. Live search resets
548    /// pagination (`after`/`before` are always `None` — a new search is a new
549    /// result set, same as the GET toolbar) and keeps the page's `group_by`.
550    ///
551    /// Every argument is client-owned by the time the shard reads it back, so
552    /// this applies [`Self::from_cx`]'s bounds through the shared
553    /// `Self::from_parts` — the public constructor is not the
554    /// looser one.
555    ///
556    /// [`TableSignals::to_state`] is the shard's call site for this:
557    /// it reads the signals and passes their values here, so the shard never
558    /// rebuilds state by hand.
559    pub fn from_live_args(
560        q: &str,
561        filters_param: &str,
562        sort: &str,
563        dir: &str,
564        group_by: &str,
565    ) -> Self {
566        // `after`/`before`/`delete`/`open` stay `None`: live search resets
567        // pagination and the delete dialog with every keystroke (a new search
568        // is a new result set, same as the GET toolbar), and the panel
569        // renders the dialog outside the shard region for live tables.
570        // Missing keys read as absent through `from_parts`.
571        Self::from_parts(|key| match key {
572            "q" => Some(q),
573            "filters" => Some(filters_param),
574            "sort" => Some(sort),
575            "dir" => Some(dir),
576            "group_by" => Some(group_by),
577            _ => None,
578        })
579    }

581    /// Seed the page's interaction signals from this state.
582    ///
583    /// The one state→signal conversion: the page owns the handles
584    /// ([`TableSignals`]) and every control it renders writes them, so the
585    /// values the page loaded with are what the signals start from. The
586    /// cursor is one signal seeded from the `after`/`before` pair
587    /// in this state; the bulk selection starts empty (the page never loads a
588    /// selection); `group_by` seeds from the page-load value and persists
589    /// across in-place reruns.
590    ///
591    /// Call it with the *parsed* state, before normalizing: an unknown
592    /// `?group_by=` seeds the signal as written, and the shard's
593    /// [`TableSignals::to_state`] + `Table::normalize_state` drop it on the
594    /// way back in, exactly as the GET path does.
595    ///
596    /// Creates the signals, so it carries [`topcoat::runtime::signal`]'s
597    /// contract: call it while a view is collecting signal declarations — the
598    /// panel calls it from the live page's render, and the declarations ride
599    /// that page's hoisted parts.
600    pub fn to_signals(&self, cx: &Cx) -> TableSignals {
601        TableSignals {
602            q: signal(cx, || self.search.clone().unwrap_or_default()),
603            filters: signal(cx, || self.filters_param().unwrap_or_default()),
604            // The projection's own spelling of `?sort=`/`?dir=`, so
605            // the signal and every link agree on the direction word.
606            sort: signal(cx, || {
607                self.sort_pair()
608                    .map(|(column, _)| column.to_string())
609                    .unwrap_or_default()
610            }),
611            dir: signal(cx, || {
612                self.sort_pair()
613                    .map(|(_, dir)| dir.to_string())
614                    .unwrap_or_else(|| "asc".to_string())
615            }),
616            cursor: signal(cx, || match (&self.after, &self.before) {
617                (Some(token), _) => cursor_after(token),
618                (None, Some(token)) => cursor_before(token),
619                (None, None) => cursor_none(),
620            }),
621            group_by: signal(cx, || self.group_by.clone().unwrap_or_default()),
622            bulk: signal(cx, String::new),
623        }
624    }
625}

627/// The live seam's other direction: the signals the page owns,
628/// read back into the request state the shard loads and renders with.
629impl TableSignals {
630    /// Rebuild request state from the live signals.
631    ///
632    /// The one signal→state conversion: every value is client-owned by the time
633    /// the shard reads it back, so it goes through
634    /// [`TableState::from_live_args`] — the same `q` clamp and `filters` bound
635    /// the GET path applies — and the one cursor
636    /// wire is split into the `(after, before)` pair the loader consumes.
637    /// Pagination and the delete dialog always reset; a
638    /// token that does not decode fails loudly at load time.
639    pub fn to_state(&self) -> TableState {
640        let mut state = TableState::from_live_args(
641            &self.q.get(),
642            &self.filters.get(),
643            &self.sort.get(),
644            &self.dir.get(),
645            &self.group_by.get(),
646        );
647        (state.after, state.before) = split_cursor(&self.cursor.get());
648        state
649    }
650}

727/// Query-string pairs with the first occurrence winning.
728///
729/// A duplicate key keeps its first value rather than failing the decode:
730/// rejecting it would fail the whole parse, and answering empty state instead
731/// would silently drop filters and export's fail-closed guard. Unknown
732/// keys are ignored, like the typed decode.
733fn first_wins_query_params(query: &str) -> HashMap<String, String> {
734    let mut out = HashMap::new();
735    for (key, value) in form_urlencoded::parse(query.as_bytes()) {
736        if !out.contains_key(key.as_ref()) {
737            out.insert(key.into_owned(), value.into_owned());
738        }
739    }
740    out
741}

743/// Parse `filters` query param: `key:value,key2:value2` (trimmed, blank ignored).
744///
745/// `,`/`:`/`%` inside keys/values are `%`-escaped by [`TableState::filters_param`]
746/// decoding restores them. Duplicate keys keep the first occurrence
747/// instead of silent last-wins.
748///
749/// An over-long or over-full transport is refused *whole* — never partially
750/// applied, which would silently drop filters the caller did send — and the
751/// refusal rides the GH #148 malformed channel as `FILTERS_OVERFLOW_SEGMENT`
752/// so the list warns and the export 400s instead of running
753/// unfiltered. The bound lives here, where the value is parsed: the live shard
754/// hands this the client-owned `filters` signal, which the router buffers up to
755/// megabytes of.
756fn parse_filters_param(raw: &str) -> (HashMap<String, String>, Vec<String>) {
757    // The length test first: it is O(1) and short-circuits the segment scan
758    // for the oversized input this bound exists for.
759    if raw.len() > MAX_FILTERS_PARAM || raw.split(',').count() > MAX_FILTER_SEGMENTS {
760        return (HashMap::new(), vec![FILTERS_OVERFLOW_SEGMENT.to_string()]);
761    }
762    let mut map = HashMap::new();
763    let mut malformed = Vec::new();
764    for part in raw.split(',') {
765        let part = part.trim();
766        if part.is_empty() {
767            continue;
768        }
769        // Split on the first *unescaped* colon: `%3A` stays inside the key/value,
770        // so a plain `split_once(':')` is correct on the encoded form.
771        let Some((k_enc, v_enc)) = part.split_once(':') else {
772            malformed.push(part.to_string());
773            continue;
774        };
775        let k = decode_filter_component(k_enc.trim());
776        let v = decode_filter_component(v_enc.trim());
777        if k.is_empty() || v.is_empty() {
778            malformed.push(part.to_string());
779        } else {
780            map.entry(k).or_insert(v);
781        }
782    }
783    (map, malformed)
784}

786/// Escape `%`, `:`, `,` inside a filter key/value.
787fn encode_filter_component(s: &str) -> String {
788    s.replace('%', "%25")
789        .replace(':', "%3A")
790        .replace(',', "%2C")
791}
792
793/// Decode [`encode_filter_component`] (case-insensitive hex, single pass).
794fn decode_filter_component(s: &str) -> String {
795    s.replace("%2C", ",")
796        .replace("%2c", ",")
797        .replace("%3A", ":")
798        .replace("%3a", ":")
799        .replace("%25", "%")
800}

802/// Every byte outside the RFC 3986 `unreserved` set (`A-Z a-z 0-9 - _ . ~`)
803/// is percent-encoded in a query value or path segment.
804const NON_UNRESERVED: &AsciiSet = &NON_ALPHANUMERIC
805    .remove(b'-')
806    .remove(b'_')
807    .remove(b'.')
808    .remove(b'~');
809
810/// Percent-encode a query parameter value (`unreserved` RFC 3986 set passes).
811fn encode_query_value(value: &str) -> String {
812    utf8_percent_encode(value, NON_UNRESERVED).to_string()
813}
814
815/// Percent-encode a single path segment.
816///
817/// Row keys are `String` by contract, so `/`, `?`, `#`, `%`, `+` inside a key
818/// must not rewrite the action URL. Topcoat's `path_param_segment` returns
819/// the percent-decoded segment, so this round-trips; UUID keys pass through
820/// unchanged.
821pub(crate) fn encode_path_segment(value: &str) -> String {
822    encode_query_value(value)
823}

825/// FNV-1a (32-bit): stable across runs and Rust versions, no dependency.
826/// Used only to disambiguate DOM ids, never for anything security-relevant.
827fn fnv1a_32(s: &str) -> u32 {
828    let mut hash: u32 = 0x811c_9dc5;
829    for byte in s.bytes() {
830        hash ^= u32::from(byte);
831        hash = hash.wrapping_mul(0x0100_0193);
832    }
833    hash
834}

836/// Stable DOM id for a table row: Topcoat's morph (#392) follows
837/// elements by `id` across reruns, so reorderable row content needs one in
838/// addition to the keyed-diff `key:`. Derived from the row key (stable for
839/// the record, unlike a loop index), sanitized to an HTML-safe token plus a
840/// short hash: distinct keys (`Ada Lovelace`, `Ada-Lovelace`) can sanitize to
841/// the same token, and duplicate DOM ids would make the morph follow one row.
842pub(crate) fn row_dom_id(key: &str) -> String {
843    dom_id("row", key)
844}
845
846/// Stable DOM id for a page-local group header.
847///
848/// Same contract as [`row_dom_id`]: the header is injected, removed and moved
849/// as the page is re-sorted, so the in-place morph needs an id derived from the
850/// group label it belongs to rather than from its position in the page.
851pub(crate) fn group_header_dom_id(label: &str) -> String {
852    dom_id("group", label)
853}
854
855/// The one sanitizer behind both ids: `{prefix}-{token}-{hash}`.
856fn dom_id(prefix: &str, key: &str) -> String {
857    let mut out = String::with_capacity(prefix.len() + key.len() + 14);
858    out.push_str(prefix);
859    out.push('-');
860    for c in key.chars() {
861        if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | ':' | '.') {
862            out.push(c);
863        } else {
864            out.push('-');
865        }
866    }
867    out.push_str(&format!("-{:08x}", fnv1a_32(key)));
868    out
869}

871/// Build `path?k=v&…` from ordered optional parameters, skipping `None`.
872pub(crate) fn build_url(path: &str, params: &[(&str, Option<&str>)]) -> String {
873    let query = params
874        .iter()
875        .filter_map(|(k, v)| v.map(|v| format!("{k}={}", encode_query_value(v))))
876        .collect::<Vec<_>>()
877        .join("&");
878    if query.is_empty() {
879        path.to_string()
880    } else {
881        format!("{path}?{query}")
882    }
883}
