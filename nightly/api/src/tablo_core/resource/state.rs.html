18/// The live table's browser state: the list's query string and the bulk
19/// selection.
20///
21/// The page owns these signals and hands their handles to the `table_search`
22/// shard; each tracked read inside the shard becomes a `dep` marker the
23/// browser watches, so writing either re-renders the table in place, with no
24/// navigation and no scroll jump.
25///
26/// `query` is the URL query every control already spells in its `href`: a sort
27/// link or a pager link writes its own `href` query, and the search and filter
28/// scripts edit their own keys of the current value, clear links included, so
29/// the hoisted control a clear link names is cleared with the query. The
30/// shard parses it with [`TableState::from_query`], the GET path's parser, so
31/// the live table and the page cannot disagree about what a query means.
32/// `bulk` is the selection, which is not URL state and survives a rerun.
33///
34/// Both values are untrusted by the time the shard reads them back: the client
35/// owns the signals.
36#[derive(Clone)]
37pub(crate) struct TableSignals {
38    /// The list's URL query, without the leading `?`.
39    pub(crate) query: Signal<String>,
40    /// The bulk selection: comma-delimited record keys (`,a,b,`), empty when
41    /// nothing is selected. Row checkboxes carry no `checked` attribute:
42    /// `bulk.js` sets `checked` from the transport after every swap and
43    /// change, and the script writes the transport, whose bound `change`
44    /// handler writes this signal, so a rerun re-renders the boxes from the
45    /// selection instead of dropping it. The shard carries the handle without
46    /// reading it: a checkbox click must not reload rows.
47    pub(crate) bulk: Signal<String>,
48}
49
50/// Test helper: whether the comma-delimited selection wire names `key`.
51///
52/// The wire is `,a,b,`-delimited on both ends so membership is an exact
53/// segment match, not a substring test that would confuse `b` with `ab`.
54/// [`parse_bulk_ids`](crate::panel) already ignores the empty segments the
55/// delimiters produce, so the same wire is the form transport.
56#[cfg(test)]
57pub(crate) fn bulk_wire_contains(wire: &str, key: &str) -> bool {
58    wire.split(',').any(|segment| segment == key)
59}

71/// A pagination cursor: the page after, or the page before, an encoded row.
72///
73/// Toasty pages from one cursor at a time, so one value holds it: a URL that
74/// names both `?after=` and `?before=` parses as no cursor, the first page,
75/// which is where the cursor retry lands anyway.
76#[derive(Debug, Clone, PartialEq, Eq)]
77pub enum Cursor {
78    /// `?after=` — the page after the encoded row.
79    After(String),
80    /// `?before=` — the page before the encoded row.
81    Before(String),
82}

84/// Request-scoped table state, parsed from the list's URL query.
85///
86/// The single parse point shared by loaders (the search term, ordering via
87/// `Table::order_bys_for`) and render (active sort, toolbar values,
88/// pagination links), so the URL is the one truth for list state. A page that
89/// holds several tables — a record page with its relations — gives each a
90/// [`prefix`](Self::prefix) for its parameters (`comments.q=`), so the
91/// tables' states share one query without colliding.
92#[derive(Debug, Clone, Default, PartialEq, Eq)]
93pub struct TableState {
94    /// The prefix every parameter of this table carries (`comments` →
95    /// `comments.q=`), or `None` for a page's own list, whose parameters are
96    /// bare. Set by the parse ([`Self::from_query_prefixed`]); every link and
97    /// form input this state builds spells the same names.
98    pub prefix: Option<String>,
99    /// `?q=` — trimmed and clamped to `MAX_QUERY_TERM` chars; `None` when
100    /// absent or blank.
101    pub search: Option<String>,
102    /// `?sort=` + `?dir=` — `None` when absent or blank.
103    pub sort: Option<Sort>,
104    /// `?after=` or `?before=`.
105    pub cursor: Option<Cursor>,
106    /// `?f.<name>=<value>`, one parameter per active filter, by name. A blank
107    /// value is no filter.
108    pub filters: BTreeMap<String, String>,
109    /// Filter parameters arrived that the parse dropped: more than
110    /// `MAX_FILTERS`, a name or value over `MAX_FILTER_LEN` bytes, or the
111    /// retired `?filters=` spelling. `Table::unapplied_filters` reports it, so
112    /// the list warns and the export refuses instead of exporting an
113    /// over-broad CSV. No link carries it: a link rebuilds the query from the
114    /// filters that applied.
115    pub filters_dropped: bool,
116    /// `?group_by=` — field name to group by (in-memory, `count` summarizer).
117    pub group_by: Option<String>,
118    /// `?delete=` — the row key whose delete confirmation dialog opens on the
119    /// list page. The dialog's confirmed POST re-enters the delete
120    /// route; the parameter itself is never a write.
121    pub delete: Option<String>,
122    /// `?open=false` — set by `dialog.js` when Escape/backdrop dismisses the
123    /// delete dialog, so the next render stays closed. Absent (or `true`)
124    /// renders it open.
125    pub open: Option<bool>,
126}

128/// The URL parameters one table link projects, named so a call site reads
129/// which intent drops what.
130///
131/// `Default` is the projection that drops everything: a caller names only the
132/// parameters it keeps.
133#[derive(Default)]
134struct UrlProjection<'a> {
135    /// `?q=` search term.
136    search: Option<&'a str>,
137    /// `?sort=` column and `?dir=` value.
138    sort: Option<(&'a str, &'a str)>,
139    /// The `?f.<name>=` filters.
140    filters: bool,
141    /// `?group_by=` column.
142    group_by: Option<&'a str>,
143    /// `?after=` or `?before=`.
144    cursor: Option<&'a Cursor>,
145    /// `?delete=` row key for the confirmation dialog.
146    delete: Option<&'a str>,
147}
148
149/// Most filters one query applies. Every filter is a map entry that every
150/// rebuilt URL echoes, and the live shard parses a client-owned query, so the
151/// count is bounded where the query is parsed.
152pub(crate) const MAX_FILTERS: usize = 32;
153
154/// Longest filter name or value one query applies, in bytes: the bound on
155/// what every rebuilt link echoes per filter.
156pub(crate) const MAX_FILTER_LEN: usize = 256;
157
158/// The prefix that names a filter parameter: `?f.status=published`.
159const FILTER_PREFIX: &str = "f.";
160
161impl TableState {
162    /// Parse the state from the request in `cx`: [`Self::from_query`] over
163    /// the request URI's query. Renders without a request context (e.g. unit
164    /// tests) get neutral state.
165    pub fn from_cx(cx: &Cx) -> Self {
166        Self::from_query(&request_query(cx))
167    }
168
169    /// [`Self::from_cx`] for the table whose parameters carry `prefix`, on a
170    /// page of several.
171    pub fn from_cx_prefixed(cx: &Cx, prefix: &str) -> Self {
172        Self::from_query_prefixed(&request_query(cx), prefix)
173    }
174
175    /// [`Self::from_query`] for the table whose parameters carry `prefix`:
176    /// only the parameters spelled `{prefix}.{name}` are read, as `name`, and
177    /// the parsed state carries the prefix so its links spell the same names.
178    pub fn from_query_prefixed(query: &str, prefix: &str) -> Self {
179        let dotted = format!("{prefix}.");
180        let own = form_urlencoded::parse(query.as_bytes()).filter_map(|(name, value)| {
181            name.strip_prefix(dotted.as_str())
182                .map(|name| (Cow::Owned(name.to_string()), value))
183        });
184        Self {
185            prefix: Some(prefix.to_string()),
186            ..Self::from_pairs(own)
187        }
188    }
189
190    /// The URL parameter this table spells `name` as: `name` itself, or
191    /// `{prefix}.{name}` for a prefixed table.
192    pub(crate) fn param(&self, name: &str) -> String {
193        match &self.prefix {
194            Some(prefix) => format!("{prefix}.{name}"),
195            None => name.to_string(),
196        }
197    }
198
199    /// The URL parameter the filter `name` travels as: `f.{name}`, prefixed
200    /// like every other parameter.
201    pub(crate) fn filter_param(&self, name: &str) -> String {
202        self.param(&format!("{FILTER_PREFIX}{name}"))
203    }
204
205    /// Parse the state from a URL query (without the leading `?`): the one
206    /// parser behind the GET page and the live shard.
207    ///
208    /// A blank or unknown query parses as neutral state rather than failing
209    /// the request. A duplicate key keeps its first occurrence, so a repeated
210    /// filter never vanishes. A blank or dropped filter occurrence is no
211    /// filter, so a later occurrence of the same filter applies (a dropped one
212    /// still flags [`Self::filters_dropped`]). `q` is trimmed and clamped to
213    /// `MAX_QUERY_TERM`, `dir` is trimmed before comparing, and at most
214    /// `MAX_FILTERS` filters of at most `MAX_FILTER_LEN` bytes apply. A cursor
215    /// token is checked later, when it decodes.
216    ///
217    /// The live shard parses a client-owned query, so the parse is linear in
218    /// its length: only the known keys are remembered.
219    pub fn from_query(query: &str) -> Self {
220        Self::from_pairs(form_urlencoded::parse(query.as_bytes()))
221    }
222
223    /// The parse behind [`Self::from_query`] and [`Self::from_query_prefixed`],
224    /// over the query's decoded pairs with any table prefix already stripped.
225    fn from_pairs<'q>(pairs: impl Iterator<Item = (Cow<'q, str>, Cow<'q, str>)>) -> Self {
226        let mut state = Self::default();
227        let (mut sort, mut dir, mut after, mut before) = (None, None, None, None);
228        let mut seen = [false; 8];
229        for (key, value) in pairs {
230            if let Some(name) = key.strip_prefix(FILTER_PREFIX) {
231                let value = value.trim();
232                if name.is_empty() || value.is_empty() || state.filters.contains_key(name) {
233                    continue;
234                }
235                if state.filters.len() == MAX_FILTERS
236                    || name.len() > MAX_FILTER_LEN
237                    || value.len() > MAX_FILTER_LEN
238                {
239                    state.filters_dropped = true;
240                    continue;
241                }
242                state.filters.insert(name.to_string(), value.to_string());
243                continue;
244            }
245            let slot = match key.as_ref() {
246                "q" => 0,
247                "sort" => 1,
248                "dir" => 2,
249                "after" => 3,
250                "before" => 4,
251                "group_by" => 5,
252                "delete" => 6,
253                "open" => 7,
254                // The retired single-parameter filter spelling: a saved link
255                // must warn (and its export refuse), not list everything.
256                "filters" => {
257                    state.filters_dropped |= !value.trim().is_empty();
258                    continue;
259                }
260                _ => continue,
261            };
262            if std::mem::replace(&mut seen[slot], true) {
263                continue;
264            }
265            let non_empty = || Some(value.trim().to_string()).filter(|v| !v.is_empty());
266            match slot {
267                0 => state.search = Some(clamp_query_term(&value)).filter(|t| !t.is_empty()),
268                1 => sort = non_empty(),
269                2 => dir = Some(value.trim() == "desc"),
270                3 => after = non_empty(),
271                4 => before = non_empty(),
272                5 => state.group_by = non_empty(),
273                // The delete dialog is opt-in through `?delete=`;
274                // `?open=false` is the dismissal mirror `dialog.js` writes.
275                // Any other `open` value stays neutral (open).
276                6 => state.delete = non_empty(),
277                _ => {
278                    state.open = match value.as_ref() {
279                        "false" => Some(false),
280                        "true" => Some(true),
281                        _ => None,
282                    }
283                }
284            }
285        }
286        state.sort = sort.map(|column| Sort {
287            column,
288            descending: dir.unwrap_or(false),
289        });
290        state.cursor = match (after, before) {
291            (Some(token), None) => Some(Cursor::After(token)),
292            (None, Some(token)) => Some(Cursor::Before(token)),
293            _ => None,
294        };
295        state
296    }

298    /// The live page's signals: `query` seeded with the request's query as
299    /// written, the selection empty.
300    ///
301    /// The raw query, not a projection of the parsed state: the shard parses
302    /// it with [`Self::from_query`] and normalizes it exactly as the GET path
303    /// does, so an unknown `?group_by=` is dropped on the way back in and a
304    /// dropped filter still warns ([`Self::filters_dropped`]). Parameters the
305    /// list does not read ride along until a link replaces the query with its
306    /// own projection.
307    ///
308    /// Creates the signals, so it carries [`topcoat::runtime::signal`]'s
309    /// contract: call it while a view is collecting signal declarations — the
310    /// panel calls it from the live page's render, and the declarations ride
311    /// that page's hoisted parts.
312    pub(crate) fn signals_for(cx: &Cx, query: &str) -> TableSignals {
313        let query = query.to_string();
314        TableSignals {
315            query: signal(cx, move || query),
316            bulk: signal(cx, String::new),
317        }
318    }

321    /// without the path or the leading `?`.
322    pub(crate) fn query(&self) -> String {
323        self.project(self.full())

326    /// URL projection: `TableState` owns the table's URL vocabulary.
327    /// Callers ask for a user intent, never a parameter list, so adding a
328    /// parameter cannot silently drop it from half the links.
329    ///
330    /// One private encoder ([`Self::project`]) holds the vocabulary in
331    /// canonical order `q, sort, dir, f.*, group_by, after|before` (`delete`
332    /// appended by its intent). The parser is first-wins with unique keys, so
333    /// order is semantically irrelevant.
334    ///
335    /// Expects `group_by` pre-normalized: render seams normalize through
336    /// [`Table::normalize_state`](crate::resource::Table::normalize_state), so
337    /// the projection echoes `state.group_by` as-is. `open` is never emitted by
338    /// any link; `delete` only by [`Self::row_url_base`]'s dialog intent.
339    ///
340    /// Full state, including the cursor; never `delete`/`open`. The streamed
341    /// retry link for failures that keep their evidence.
342    pub(crate) fn list_url(&self, path: &str) -> String {
343        with_query(path, &self.query())
344    }

399    /// The shared parameters of every row-action URL on one page, encoded
400    /// once.
401    ///
402    /// A row's action URL is this base plus the row's record key, so a table
403    /// render pays for the projection once, however many rows the page holds.
404    /// Build it before the row loop and call [`RowUrlBase::delete_dialog`] per
405    /// row; that pair is the full-state-plus-`delete` projection, which keeps
406    /// the cursor and never emits `open`.
407    pub(crate) fn row_url_base(&self, path: &str) -> RowUrlBase {
408        RowUrlBase {
409            base: self.list_url(path),
410            delete_param: self.param("delete"),
411        }
412    }

414    /// `?sort=` column + `?dir=` value for the projection.
415    fn sort_pair(&self) -> Option<(&str, &str)> {
416        self.sort
417            .as_ref()
418            .map(|s| (s.column.as_str(), if s.descending { "desc" } else { "asc" }))
419    }

421    /// The projection that keeps every link parameter.
422    fn full(&self) -> UrlProjection<'_> {
423        UrlProjection {
424            search: self.search.as_deref(),
425            sort: self.sort_pair(),
426            filters: true,
427            group_by: self.group_by.as_deref(),
428            cursor: self.cursor.as_ref(),
429            delete: None,
430        }
431    }
432
433    /// The one encoder: every table link's parameter vocabulary lives here.
434    ///
435    /// The exhaustive destructure fails compilation when a field is added to
436    /// `TableState`, forcing the author to decide where it projects.
437    fn project(&self, projection: UrlProjection<'_>) -> String {
438        let TableState {
439            prefix: _,
440            search: _,
441            sort: _,
442            cursor: _,
443            filters: _,
444            filters_dropped: _,
445            group_by: _,
446            delete: _,
447            open: _,
448        } = self;
449        let filters = projection
450            .filters
451            .then_some(&self.filters)
452            .into_iter()
453            .flatten()
454            .map(|(name, value)| (self.filter_param(name), Some(value.as_str())));
455        let cursor = match projection.cursor {
456            Some(Cursor::After(token)) => ("after", Some(token.as_str())),
457            Some(Cursor::Before(token)) => ("before", Some(token.as_str())),
458            None => ("after", None),
459        };
460        let pairs: Vec<(String, &str)> = [
461            (self.param("q"), projection.search),
462            (
463                self.param("sort"),
464                projection.sort.map(|(column, _)| column),
465            ),
466            (self.param("dir"), projection.sort.map(|(_, dir)| dir)),
467        ]
468        .into_iter()
469        .chain(filters)
470        .chain([
471            (self.param("group_by"), projection.group_by),
472            (self.param(cursor.0), cursor.1),
473            (self.param("delete"), projection.delete),
474        ])
475        .filter_map(|(key, value)| value.map(|value| (key, value)))
476        .collect();
477        encode_query(&pairs)
478    }
479}

481/// One page's shared row-action URL parameters, encoded once.
482///
483/// The base is [`TableState::list_url`] — every parameter a row's action URL
484/// shares — so those parameters are encoded once per render, not once per
485/// row. Row-specific intents ([`Self::delete_dialog`]) append to it in the
486/// projection's own order.
487pub(crate) struct RowUrlBase {
488    base: String,
489    /// The table's `delete` parameter, prefixed like the rest.
490    delete_param: String,
491}
492
493impl RowUrlBase {
494    /// The `?delete=<key>` confirmation-dialog opener for one row.
495    ///
496    /// `base` is [`TableState::list_url`]'s output, which never carries
497    /// `delete`, and `delete` is the projection's last parameter — so this is
498    /// byte-for-byte what the one-pass projection builds, without re-encoding
499    /// the parameters it shares with the rest of the page.
500    pub(crate) fn delete_dialog(&self, key: &str) -> String {
501        let separator = if self.base.contains('?') { '&' } else { '?' };
502        format!(
503            "{}{separator}{}={}",
504            self.base,
505            encode_query_value(&self.delete_param),
506            encode_query_value(key)
507        )
508    }
509}

586/// The query parameter naming the page a write lands on after it commits,
587/// in place of the resource's list: `?return=/admin/posts/1`. The panel
588/// honours it only for a path under its own prefix.
589pub(crate) const RETURN_PARAM: &str = "return";
590
591/// `url`, which carries no query, with `?return={target}`.
592pub(crate) fn with_return(url: &str, target: &str) -> String {
593    format!("{url}?{RETURN_PARAM}={}", encode_query_value(target))
594}

596/// Every byte outside the RFC 3986 `unreserved` set (`A-Z a-z 0-9 - _ . ~`)
597/// is percent-encoded in a query value or path segment.
598const NON_UNRESERVED: &AsciiSet = &NON_ALPHANUMERIC
599    .remove(b'-')
600    .remove(b'_')
601    .remove(b'.')
602    .remove(b'~');
603
604/// Percent-encode a query parameter value (`unreserved` RFC 3986 set passes).
605fn encode_query_value(value: &str) -> String {
606    utf8_percent_encode(value, NON_UNRESERVED).to_string()
607}
608
609/// Percent-encode a single path segment.
610///
611/// Row keys are `String` by contract, so `/`, `?`, `#`, `%`, `+` inside a key
612/// must not rewrite the action URL. Topcoat's `path_param_segment` returns
613/// the percent-decoded segment, so this round-trips; UUID keys pass through
614/// unchanged.
615pub(crate) fn encode_path_segment(value: &str) -> String {
616    encode_query_value(value)
617}

619/// FNV-1a (32-bit): stable across runs and Rust versions, no dependency.
620/// Used only to disambiguate DOM ids, never for anything security-relevant.
621fn fnv1a_32(s: &str) -> u32 {
622    let mut hash: u32 = 0x811c_9dc5;
623    for byte in s.bytes() {
624        hash ^= u32::from(byte);
625        hash = hash.wrapping_mul(0x0100_0193);
626    }
627    hash
628}

630/// Stable DOM id for a table row: Topcoat's morph (#392) follows
631/// elements by `id` across reruns, so reorderable row content needs one in
632/// addition to the keyed-diff `key:`. Derived from the row key (stable for
633/// the record, unlike a loop index), sanitized to an HTML-safe token plus a
634/// short hash: distinct keys (`Ada Lovelace`, `Ada-Lovelace`) can sanitize to
635/// the same token, and duplicate DOM ids would make the morph follow one row.
636pub(crate) fn row_dom_id(key: &str) -> String {
637    dom_id("row", key)
638}
639
640/// Stable DOM id for a page-local group header.
641///
642/// Same contract as [`row_dom_id`]: the header is injected, removed and moved
643/// as the page is re-sorted, so the in-place morph needs an id derived from the
644/// group label it belongs to rather than from its position in the page.
645pub(crate) fn group_header_dom_id(label: &str) -> String {
646    dom_id("group", label)
647}
648
649/// The one sanitizer behind both ids: `{prefix}-{token}-{hash}`.
650fn dom_id(prefix: &str, key: &str) -> String {
651    let mut out = String::with_capacity(prefix.len() + key.len() + 14);
652    out.push_str(prefix);
653    out.push('-');
654    for c in key.chars() {
655        if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | ':' | '.') {
656            out.push(c);
657        } else {
658            out.push('-');
659        }
660    }
661    out.push_str(&format!("-{:08x}", fnv1a_32(key)));
662    out
663}

665/// Encode ordered `key=value` pairs as a URL query, without the leading `?`.
666fn encode_query(pairs: &[(String, &str)]) -> String {
667    pairs
668        .iter()
669        .map(|(key, value)| format!("{}={}", encode_query_value(key), encode_query_value(value)))
670        .collect::<Vec<_>>()
671        .join("&")
672}
673
674/// `path?query`, or `path` alone for an empty query.
675fn with_query(path: &str, query: &str) -> String {
676    if query.is_empty() {
677        path.to_string()
678    } else {
679        format!("{path}?{query}")
680    }
681}

683/// The request's URL query, without the leading `?`; empty without a
684/// request context.
685pub(crate) fn request_query(cx: &Cx) -> String {
686    topcoat::context::try_request_context::<http::request::Parts>(cx)

689}

691/// The query part of a URL this module built: what follows the `?`, or empty.
692pub(crate) fn query_of(url: &str) -> &str {
693    url.split_once('?').map_or("", |(_, query)| query)
694}
