43/// A pagination cursor: the page after, or the page before, an encoded row.

54/// Request-scoped table state, parsed from the list's URL query.
55#[derive(Debug, Clone, Default, PartialEq, Eq)]
56pub struct TableState {
57    /// The prefix every parameter of this table carries, or `None` for a page's
58    /// own list.
59    pub prefix: Option<String>,
60    /// `?q=` — trimmed and clamped to `MAX_QUERY_TERM` chars; `None` when
61    /// absent or blank.
62    pub search: Option<String>,
63    /// `?sort=` + `?dir=` — `None` when absent or blank.
64    pub sort: Option<Sort>,
65    /// `?after=` or `?before=`.
66    pub cursor: Option<Cursor>,
67    /// `?f.<name>=<value>`, one parameter per active filter, by name. A blank
68    /// value is no filter.
69    pub filters: BTreeMap<String, String>,
70    /// Filter parameters the parse dropped.
71    pub filters_dropped: bool,
72    /// `?group_by=` — field name to group by (in-memory, `count` summarizer).
73    pub group_by: Option<String>,
74    /// `?delete=` — the row key whose delete dialog opens; never a write.
75    pub delete: Option<String>,
76    /// `?open=false` — renders the delete dialog closed.
77    pub open: Option<bool>,
78}

80/// The URL parameters one table link projects.
81#[derive(Default)]
82struct UrlProjection<'a> {
83    /// `?q=` search term.
84    search: Option<&'a str>,
85    /// `?sort=` column and `?dir=` value.
86    sort: Option<(&'a str, &'a str)>,
87    /// The `?f.<name>=` filters.
88    filters: bool,
89    /// `?group_by=` column.
90    group_by: Option<&'a str>,
91    /// `?after=` or `?before=`.
92    cursor: Option<&'a Cursor>,
93    /// `?delete=` row key for the confirmation dialog.
94    delete: Option<&'a str>,
95}

97/// Caps applied filters per query.
98pub(crate) const MAX_FILTERS: usize = 32;
99
100/// Caps filter name and value length in bytes.
101pub(crate) const MAX_FILTER_LEN: usize = 256;
102
103/// The prefix that names a filter parameter: `?f.status=published`.
104const FILTER_PREFIX: &str = "f.";

106impl TableState {
107    /// Parse the state from the request in `cx`: [`Self::from_query`] over
108    /// the request URI's query. Renders without a request context (e.g. unit
109    /// tests) get neutral state.
110    pub fn from_cx(cx: &Cx) -> Self {
111        Self::from_query(&request_query(cx))
112    }
113
114    /// [`Self::from_cx`] for the table whose parameters carry `prefix`, on a
115    /// page of several.
116    pub fn from_cx_prefixed(cx: &Cx, prefix: &str) -> Self {
117        Self::from_query_prefixed(&request_query(cx), prefix)
118    }
119
120    /// [`Self::from_query`] for the table whose parameters carry `prefix`:
121    /// only the parameters spelled `{prefix}.{name}` are read, as `name`, and
122    /// the parsed state carries the prefix so its links spell the same names.
123    pub fn from_query_prefixed(query: &str, prefix: &str) -> Self {
124        let dotted = format!("{prefix}.");
125        let own = form_urlencoded::parse(query.as_bytes()).filter_map(|(name, value)| {
126            name.strip_prefix(dotted.as_str())
127                .map(|name| (Cow::Owned(name.to_string()), value))
128        });
129        Self {
130            prefix: Some(prefix.to_string()),
131            ..Self::from_pairs(own)
132        }
133    }
134
135    /// The URL parameter this table spells `name` as: `name` itself, or
136    /// `{prefix}.{name}` for a prefixed table.
137    pub(crate) fn param(&self, name: &str) -> String {
138        match &self.prefix {
139            Some(prefix) => format!("{prefix}.{name}"),
140            None => name.to_string(),
141        }
142    }
143
144    /// The URL parameter the filter `name` travels as: `f.{name}`, prefixed
145    /// like every other parameter.
146    pub(crate) fn filter_param(&self, name: &str) -> String {
147        self.param(&format!("{FILTER_PREFIX}{name}"))
148    }
149
150    /// Parse the state from a URL query (without the leading `?`): the one
151    /// parser behind the GET page and the live shard.
152    ///
153    /// A blank or unknown query parses as neutral state rather than failing
154    /// the request. A duplicate key keeps its first occurrence, so a repeated
155    /// filter never vanishes. A blank or dropped filter occurrence is no
156    /// filter, so a later occurrence of the same filter applies (a dropped one
157    /// still flags [`Self::filters_dropped`]). `q` is trimmed and clamped to
158    /// `MAX_QUERY_TERM`, `dir` is trimmed before comparing, and at most
159    /// `MAX_FILTERS` filters of at most `MAX_FILTER_LEN` bytes apply. A cursor
160    /// token is checked later, when it decodes.
161    ///
162    /// The live shard parses a client-owned query, so the parse is linear in
163    /// its length: only the known keys are remembered.
164    pub fn from_query(query: &str) -> Self {
165        Self::from_pairs(form_urlencoded::parse(query.as_bytes()))
166    }
167
168    /// The parse behind [`Self::from_query`] and [`Self::from_query_prefixed`],
169    /// over the query's decoded pairs with any table prefix already stripped.
170    fn from_pairs<'q>(pairs: impl Iterator<Item = (Cow<'q, str>, Cow<'q, str>)>) -> Self {
171        let mut state = Self::default();
172        let (mut sort, mut dir, mut after, mut before) = (None, None, None, None);
173        let mut seen = [false; 8];
174        for (key, value) in pairs {
175            if let Some(name) = key.strip_prefix(FILTER_PREFIX) {
176                let value = value.trim();
177                if name.is_empty() || value.is_empty() || state.filters.contains_key(name) {
178                    continue;
179                }
180                if state.filters.len() == MAX_FILTERS
181                    || name.len() > MAX_FILTER_LEN
182                    || value.len() > MAX_FILTER_LEN
183                {
184                    state.filters_dropped = true;
185                    continue;
186                }
187                state.filters.insert(name.to_string(), value.to_string());
188                continue;
189            }
190            let slot = match key.as_ref() {
191                "q" => 0,
192                "sort" => 1,
193                "dir" => 2,
194                "after" => 3,
195                "before" => 4,
196                "group_by" => 5,
197                "delete" => 6,
198                "open" => 7,
199                // The retired single-parameter filter spelling: a saved link
200                // must warn (and its export refuse), not list everything.
201                "filters" => {
202                    state.filters_dropped |= !value.trim().is_empty();
203                    continue;
204                }
205                _ => continue,
206            };
207            if std::mem::replace(&mut seen[slot], true) {
208                continue;
209            }
210            let non_empty = || Some(value.trim().to_string()).filter(|v| !v.is_empty());
211            match slot {
212                0 => state.search = Some(clamp_query_term(&value)).filter(|t| !t.is_empty()),
213                1 => sort = non_empty(),
214                2 => dir = Some(value.trim() == "desc"),
215                3 => after = non_empty(),
216                4 => before = non_empty(),
217                5 => state.group_by = non_empty(),
218                // The delete dialog is opt-in through `?delete=`;
219                // `?open=false` is the dismissal mirror `dialog.js` writes.
220                // Any other `open` value stays neutral (open).
221                6 => state.delete = non_empty(),
222                _ => {
223                    state.open = match value.as_ref() {
224                        "false" => Some(false),
225                        "true" => Some(true),
226                        _ => None,
227                    }
228                }
229            }
230        }
231        state.sort = sort.map(|column| Sort {
232            column,
233            descending: dir.unwrap_or(false),
234        });
235        state.cursor = match (after, before) {
236            (Some(token), None) => Some(Cursor::After(token)),
237            (None, Some(token)) => Some(Cursor::Before(token)),
238            _ => None,
239        };
240        state
241    }

243    /// The live page's signals: `query` seeded with the request's query as
244    /// written, the selection empty.
245    ///
246    /// The raw query, not a projection of the parsed state: the shard parses
247    /// it with [`Self::from_query`] and normalizes it exactly as the GET path
248    /// does, so an unknown `?group_by=` is dropped on the way back in and a
249    /// dropped filter still warns ([`Self::filters_dropped`]). Parameters the
250    /// list does not read ride along until a link replaces the query with its
251    /// own projection.
252    ///
253    /// Creates the signals, so it carries [`topcoat::runtime::signal`]'s
254    /// contract: call it while a view is collecting signal declarations — the
255    /// panel calls it from the live page's render, and the declarations ride
256    /// that page's hoisted parts.
257    pub(crate) fn signals_for(cx: &Cx, query: &str) -> TableSignals {
258        let query = query.to_string();
259        TableSignals {
260            query: signal(cx, move || query),
261            bulk: signal(cx, String::new),
262        }
263    }

266    /// without the path or the leading `?`.
267    pub(crate) fn query(&self) -> String {
268        self.project(self.full())

271    /// URL projection: `TableState` owns the table's URL vocabulary.
272    /// Callers ask for a user intent, never a parameter list, so adding a
273    /// parameter cannot silently drop it from half the links.
274    ///
275    /// One private encoder ([`Self::project`]) holds the vocabulary in
276    /// canonical order `q, sort, dir, f.*, group_by, after|before` (`delete`
277    /// appended by its intent). The parser is first-wins with unique keys, so
278    /// order is semantically irrelevant.
279    ///
280    /// Expects `group_by` pre-normalized: render seams normalize through
281    /// [`Table::normalize_state`](crate::resource::Table::normalize_state), so
282    /// the projection echoes `state.group_by` as-is. `open` is never emitted by
283    /// any link; `delete` only by [`Self::row_url_base`]'s dialog intent.
284    ///
285    /// Full state, including the cursor; never `delete`/`open`. The streamed
286    /// retry link for failures that keep their evidence.
287    pub(crate) fn list_url(&self, path: &str) -> String {
288        with_query(path, &self.query())
289    }

344    /// The shared parameters of every row-action URL on one page, encoded
345    /// once.
346    ///
347    /// A row's action URL is this base plus the row's record key, so a table
348    /// render pays for the projection once, however many rows the page holds.

359    /// `?sort=` column + `?dir=` value for the projection.
360    fn sort_pair(&self) -> Option<(&str, &str)> {
361        self.sort
362            .as_ref()
363            .map(|s| (s.column.as_str(), if s.descending { "desc" } else { "asc" }))
364    }

366    /// The projection that keeps every link parameter.
367    fn full(&self) -> UrlProjection<'_> {
368        UrlProjection {
369            search: self.search.as_deref(),
370            sort: self.sort_pair(),
371            filters: true,
372            group_by: self.group_by.as_deref(),
373            cursor: self.cursor.as_ref(),
374            delete: None,
375        }
376    }

378    /// The one encoder: every table link's parameter vocabulary lives here.
379    ///
380    /// The exhaustive destructure fails compilation when a field is added to
381    /// `TableState`, forcing the author to decide where it projects.
382    fn project(&self, projection: UrlProjection<'_>) -> String {
383        let TableState {
384            prefix: _,
385            search: _,
386            sort: _,
387            cursor: _,
388            filters: _,
389            filters_dropped: _,
390            group_by: _,
391            delete: _,
392            open: _,
393        } = self;
394        let filters = projection
395            .filters
396            .then_some(&self.filters)
397            .into_iter()
398            .flatten()
399            .map(|(name, value)| (self.filter_param(name), Some(value.as_str())));
400        let cursor = match projection.cursor {
401            Some(Cursor::After(token)) => ("after", Some(token.as_str())),
402            Some(Cursor::Before(token)) => ("before", Some(token.as_str())),
403            None => ("after", None),
404        };
405        let pairs: Vec<(String, &str)> = [
406            (self.param("q"), projection.search),
407            (
408                self.param("sort"),
409                projection.sort.map(|(column, _)| column),
410            ),
411            (self.param("dir"), projection.sort.map(|(_, dir)| dir)),
412        ]
413        .into_iter()
414        .chain(filters)
415        .chain([
416            (self.param("group_by"), projection.group_by),
417            (self.param(cursor.0), cursor.1),
418            (self.param("delete"), projection.delete),
419        ])
420        .filter_map(|(key, value)| value.map(|value| (key, value)))
421        .collect();
422        encode_query(&pairs)
423    }
424}

426/// One page's shared row-action URL parameters, encoded once.
427///
428/// The base is [`TableState::list_url`] — every parameter a row's action URL
429/// shares — so those parameters are encoded once per render, not once per
430/// row. Row-specific intents ([`Self::delete_dialog`]) append to it in the
431/// projection's own order.
432pub(crate) struct RowUrlBase {
433    base: String,
434    /// The table's `delete` parameter, prefixed like the rest.
435    delete_param: String,
436}
437
438impl RowUrlBase {
439    /// The `?delete=<key>` confirmation-dialog opener for one row.
440    ///
441    /// `base` is [`TableState::list_url`]'s output, which never carries
442    /// `delete`, and `delete` is the projection's last parameter — so this is
443    /// byte-for-byte what the one-pass projection builds, without re-encoding
444    /// the parameters it shares with the rest of the page.
445    pub(crate) fn delete_dialog(&self, key: &str) -> String {
446        let separator = if self.base.contains('?') { '&' } else { '?' };
447        format!(
448            "{}{separator}{}={}",
449            self.base,
450            encode_query_value(&self.delete_param),
451            encode_query_value(key)
452        )
453    }
454}

541/// Every byte outside the RFC 3986 `unreserved` set (`A-Z a-z 0-9 - _ . ~`)
542/// is percent-encoded in a query value or path segment.
543const NON_UNRESERVED: &AsciiSet = &NON_ALPHANUMERIC
544    .remove(b'-')
545    .remove(b'_')
546    .remove(b'.')
547    .remove(b'~');
548
549/// Percent-encode a query parameter value (`unreserved` RFC 3986 set passes).
550fn encode_query_value(value: &str) -> String {
551    utf8_percent_encode(value, NON_UNRESERVED).to_string()
552}

554/// Percent-encode a single path segment.
555///
556/// Row keys are `String` by contract, so `/`, `?`, `#`, `%`, `+` inside a key
557/// must not rewrite the action URL. Topcoat's `path_param_segment` returns
558/// the percent-decoded segment, so this round-trips; UUID keys pass through
559/// unchanged.
560pub(crate) fn encode_path_segment(value: &str) -> String {
561    encode_query_value(value)
562}

564/// FNV-1a (32-bit): stable across runs and Rust versions, no dependency.
565/// Used only to disambiguate DOM ids, never for anything security-relevant.
566fn fnv1a_32(s: &str) -> u32 {
567    let mut hash: u32 = 0x811c_9dc5;
568    for byte in s.bytes() {
569        hash ^= u32::from(byte);
570        hash = hash.wrapping_mul(0x0100_0193);
571    }
572    hash
573}

594/// The one sanitizer behind both ids: `{prefix}-{token}-{hash}`.
595fn dom_id(prefix: &str, key: &str) -> String {
596    let mut out = String::with_capacity(prefix.len() + key.len() + 14);
597    out.push_str(prefix);
598    out.push('-');
599    for c in key.chars() {
600        if c.is_ascii_alphanumeric() || matches!(c, '-' | '_' | ':' | '.') {
601            out.push(c);
602        } else {
603            out.push('-');
604        }
605    }
606    out.push_str(&format!("-{:08x}", fnv1a_32(key)));
607    out
608}
609
610/// Encode ordered `key=value` pairs as a URL query, without the leading `?`.
611fn encode_query(pairs: &[(String, &str)]) -> String {
612    pairs
613        .iter()
614        .map(|(key, value)| format!("{}={}", encode_query_value(key), encode_query_value(value)))
615        .collect::<Vec<_>>()
616        .join("&")
617}
618
619/// `path?query`, or `path` alone for an empty query.
620fn with_query(path: &str, query: &str) -> String {
621    if query.is_empty() {
622        path.to_string()
623    } else {
624        format!("{path}?{query}")
625    }
626}
627
628/// The request's URL query, without the leading `?`; empty without a
629/// request context.
630pub(crate) fn request_query(cx: &Cx) -> String {
631    topcoat::context::try_request_context::<http::request::Parts>(cx)
632        .and_then(|parts| parts.uri.query().map(str::to_string))
633        .unwrap_or_default()
634}

636/// The query part of a URL this module built: what follows the `?`, or empty.
637pub(crate) fn query_of(url: &str) -> &str {
638    url.split_once('?').map_or("", |(_, query)| query)
639}
