60/// Maps one Toasty `Model` to its admin UI.
61///
62/// # Contract
63///
64/// Requires [`Model`](Self::Model), [`Form`](Self::Form), and [`table`](Self::table);
65/// every other item has a default. Declarations build once at mount and serve
66/// every request; record fns run inside the handler transaction and fail
67/// without partial writes. Row chrome follows the declarations; the default
68/// policy denies all.
69///
70/// [`table`]: Self::table
71/// [`form`]: Self::form
72/// [`view`]: Self::view
73/// [`relations`]: Self::relations
74/// [`Schema::declaration_errors`]: crate::schema::Schema::declaration_errors
75pub trait Resource: Sized + Send + Sync + 'static {
76    /// The persisted model this resource administers.
77    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;
78
79    /// The typed value the create and edit forms parse into.
80    ///
81    /// A resource with create and edit pages names its
82    /// [`#[derive(RecordForm)]`](crate::RecordForm) struct; a list-only resource
83    /// names [`NoForm<Self::Model>`](crate::NoForm), and
84    /// [`Panel::resource`](crate::Panel::resource) registers no form route for
85    /// it ([`RecordForm::HAS_FORM`]).
86    type Form: RecordForm<Model = Self::Model>;
87
88    /// The columns an overriding [`Self::create_record`] sets itself, beyond
89    /// the form's fields.
90    const CREATE_COLUMNS: &'static [&'static str] = &[];
91
92    /// Gates every handler; defaults to [`Deny`]. Row scoping belongs in
93    /// [`query`](Self::query), not policy.
94    fn policy() -> impl Policy<Self::Model> {
95        Deny
96    }
97
98    /// Declares how rows belong to a tenant; defaults to unscoped.
99    fn tenancy() -> Tenancy<Self::Model> {
100        Tenancy::none()
101    }
102
103    /// Declares the read-only detail schema; empty disables the detail route.
104    fn view(_dx: &DeclCx) -> crate::schema::Schema {
105        crate::schema::Schema::empty()
106    }
107
108    /// Renders free-form content below [`view`](Self::view) and above
109    /// [`relations`](Self::relations).
110    fn view_content<'a>(_cx: &'a Cx, _record: &Self::Model) -> Option<topcoat::view::BoxView<'a>> {
111        None
112    }
113
114    /// Declares related resources rendered as narrowed tables on detail and
115    /// edit pages.
116    fn relations() -> Vec<Relation<Self::Model>> {
117        Vec::new()
118    }

120    /// The resource's custom [`Action`]s, in button order. Defaults to none.
121    ///
122    /// ```ignore
123    /// fn actions() -> Actions<Self> {
124    ///     Actions::new().add::<Publish>()
125    /// }
126    /// ```
127    fn actions() -> Actions<Self> {
128        Actions::new()
129    }

131    /// The record's label in the detail page's title, or `None` when
132    /// the record has no label to show.
133    ///
134    /// The detail page titles itself with this label when a resource returns
135    /// `Some`, and with [`label`](Self::label) plus the URL's record key when
136    /// it returns `None`, the default. The showcase's `PostResource` returns
137    /// the post's title, so its heading reads the title instead of
138    /// `Blog Post <record key>`.
139    ///
140    /// `cx` is the request's context — the same one [`view`](Self::view) and
141    /// [`view_values`](Self::view_values) receive — so a label
142    /// can read request state (a locale, a tenant). The default ignores both
143    /// arguments and returns `None`.
144    ///
145    /// A label is display text, not a key. Two records can share one (two
146    /// users named Ada), so it cannot replace the table key, whose projection
147    /// must stay injective within a page for keyed diffs and bulk selection,
148    /// or the record key, which the action routes resolve as the
149    /// model's typed PK.
150    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
151        None
152    }
153
154    /// A public URL for one record, rendered as a "View public post"-style
155    /// link on the detail and edit pages when `Some`.
156    ///
157    /// The default declares none, so no link renders. A resource whose records
158    /// have a public page overrides this with its URL — the showcase's posts
159    /// return their `/blog/{id}` page.
160    fn public_url(_cx: &Cx, _record: &Self::Model) -> Option<String> {
161        None
162    }
163
164    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
165    /// at `{panel prefix}/users`.
166    ///
167    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
168    /// take the resource type's name, strip a trailing `Resource`, pluralize
169    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
170    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
171    /// naming the rules cannot guess (`UsersResource` pluralizes to
172    /// `userses` — name resources singular, or override).
173    fn slug() -> String {
174        kebab_case(&pluralize(type_stem::<Self>("Resource")))
175    }
176
177    /// One record's name, e.g. `"User"`: the noun in the "Create {label}" and
178    /// "Edit {label}" titles.
179    ///
180    /// Defaults to the `Model` type name (Filament's model label). Override for
181    /// custom wording; [`navigation_label`](Self::navigation_label) pluralizes
182    /// it.
183    fn label() -> String {
184        type_short_name::<Self::Model>().to_string()
185    }
186
187    /// The sidebar label and list title, e.g. `"Users"`.
188    ///
189    /// Defaults to the pluralized [`label`](Self::label) (Filament's plural
190    /// model label): `User` → `Users`, `Category` → `Categories`, `Person` →
191    /// `People`. Override for a plural the rules cannot guess.
192    fn navigation_label() -> String {
193        pluralize(&Self::label())
194    }

196    /// Base query — the seam for a resource's **own** row scoping (ADR-0002):
197    /// soft deletes and row-level visibility. Every loader starts from it.
198    ///
199    /// **Relations are not this method's job.** The list and the export load
200    /// the relations the table's columns declare ([`TextColumn::include`]), and
201    /// the detail page loads [`Self::view_query`]. Include a relation here only
202    /// when a closure no column covers reads it on every loader's rows: one
203    /// [`policy`](Self::policy) reads, or one a table's `group_by` or row
204    /// key reads without a column including it.
205    ///
206    /// **Tenancy is not this method's job either.** For a resource whose
207    /// [`tenancy`](Self::tenancy) is scoped the framework ANDs the tenant filter
208    /// onto whatever this returns, at every loader through [`scoped_query`]. Do
209    /// not re-state it here. Code outside the framework's loaders starts from
210    /// [`scoped_query`].
211    ///
212    /// Returns the raw typed statement query, which composes generically —
213    /// `filter`, `order_by` and `Paginate::new` work on the raw form for any
214    /// `M: Model` — so one panel handler drives every resource's list page.
215    ///
216    /// # Keep unique constraints in step with this scope
217    ///
218    /// The app-side unique pre-check probes through the same tenant-scoped
219    /// query, so a `#[unique]` index *broader* than the scope is invisible: the
220    /// probe misses the colliding row and the user gets a 500 instead of the
221    /// inline "has already been taken". Scope the constraint to match —
222    /// `#[unique(tenant_id, email)]`. Not checkable at declaration time, since
223    /// a query's filters are not introspectable; upstream #117 is the fix.
224    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
225        toasty::stmt::Query::<List<Self::Model>>::all()
226    }
227
228    /// The detail page's query: [`Self::query`] plus the relations the page
229    /// reads off the loaded row — in [`view_values`](Self::view_values) or
230    /// [`view_content`](Self::view_content) — so include them here. A
231    /// [`relation`](Self::relations) table runs its own query and needs none.
232    ///
233    /// ```ignore
234    /// fn view_query(cx: &Cx) -> Query<List<Post>> {
235    ///     let author: Include<Post, Author> = Post::fields().author().into();
236    ///     Self::query(cx).include(author)
237    /// }
238    /// ```
239    ///
240    /// The default is [`Self::query`] unchanged. The framework ANDs the tenant
241    /// scope onto it, as it does onto [`Self::query`].
242    fn view_query(cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
243        Self::query(cx)
244    }

246    /// Description of the list view.
247    ///
248    /// Every resource declares its table with [`Table::new`]: columns and the
249    /// row key the list renders (see [`Table::render`]). It takes no context:
250    /// the panel calls it once, at build, and serves that table to every
251    /// request. What depends on the request — which rows a user may act on —
252    /// is the policies' business, which the panel wires per request.
253    fn table() -> Table<Self::Model>;
254
255    /// The schema the create and edit forms render.
256    ///
257    /// Defaults to the record form's derived schema
258    /// ([`RecordForm::schema`]): one control per field, in declaration order.
259    /// Override it to arrange the controls into a layout — the derive's
260    /// `controls(dx)` hands each one over, ready for its modifiers:
261    ///
262    /// ```ignore
263    /// fn form(dx: &DeclCx) -> Schema {
264    ///     let c = PostForm::controls(dx);
265    ///     Schema::new(Section::new("Content").schema((c.title, c.body.multiline(6))))
266    /// }
267    /// ```
268    ///
269    /// `dx` carries the app schema and nothing from a request: the panel calls
270    /// this once, when it is mounted, and refuses a
271    /// record form field this schema does not declare, and a schema on a
272    /// resource whose [`Form`](Self::Form) is [`NoForm`](crate::NoForm).
273    fn form(dx: &DeclCx) -> crate::schema::Schema {
274        <Self::Form as RecordForm>::schema(dx)
275    }
276
277    /// App-level rules on the parsed form. The errors render inline with a
278    /// 200 and nothing is written; a record fn error keeps its own mapping, so
279    /// a range or cross-field rule belongs here.
280    ///
281    /// Each error names the key it renders under: a control's own key, or a
282    /// [`Repeater`](crate::Repeater) group's label. A key the submitted form
283    /// renders nowhere fails the submit as a declaration error instead of
284    /// writing past the rule.
285    fn validate_record(_cx: &Cx, _form: &Self::Form) -> FieldErrors {
286        FieldErrors::new()
287    }

289    /// Create a record from the parsed form, inside the handler's transaction.
290    ///
291    /// Defaults to the derived write, [`write_create`]; override to check
292    /// something inside the transaction, then delegate. An override on a
293    /// [`Tenancy::column`](crate::Tenancy::column) resource stamps the tenant
294    /// only by delegating to [`write_create`]. `ex` is the open
295    /// transaction: run every statement through it. Return the created row; it
296    /// is what [`Self::after_commit`] receives.
297    fn create_record(
298        cx: &Cx,
299        form: Self::Form,
300        ex: &mut dyn Executor,
301    ) -> impl Future<Output = Result<Self::Model>> + Send {
302        write_create::<Self>(cx, form, ex)
303    }

305    /// Update the already-authorized `record` from the posted form, inside the
306    /// handler's transaction.
307    ///
308    /// Defaults to the derived write, [`write_update`]. `record` is the
309    /// snapshot the handler loaded and policy-checked inside the transaction:
310    /// use it, never re-query. Return the row as it now stands.
311    fn update_record(
312        cx: &Cx,
313        record: Self::Model,
314        posted: Posted<Self::Form>,
315        ex: &mut dyn Executor,
316    ) -> impl Future<Output = Result<Self::Model>> + Send {
317        write_update::<Self>(cx, record, posted, ex)
318    }

321    ///
322    /// The default declares a label ([`Self::navigation_label`]) and no URL:
323    /// the Panel that owns the resource resolves where it is mounted, so this

326    /// Override to curate this resource's sidebar entry: `Panel::resource`

339    /// Delete the already-authorized `record` (#86).
340    ///
341    /// The handler loads `record` through the tenancy-scoped query inside the
342    /// framework transaction and checks the policy on that snapshot, then
343    /// calls this with the same transaction as `ex`. The after-commit hook
344    /// receives the snapshot once the delete commits.
345    ///
346    /// The default deletes the row through [`scoped_query`], filtered to the
347    /// record's key: the table's record key, which is the model's primary key
348    /// (see [`Table::new_split`]). Override to delete another way, such as a
349    /// soft delete.
350    fn delete_record(
351        cx: &Cx,
352        record: &Self::Model,
353        ex: &mut dyn toasty::Executor,
354    ) -> impl std::future::Future<Output = Result<()>> + Send
355    where
356        Self: Sized,
357    {
358        let key = declared::<Self>(cx).table.record_key_of(record);
359        let query = scoped_query::<Self>(cx).and_then(|query| {
360            let filter = crate::schema::pk_eq_expr::<Self::Model>(&key).ok_or_else(|| {
361                // The handler found the row by this key, so a key that does
362                // not parse back is a table whose record key is not the
363                // primary key (`Table::new_split`). The detail stays in the
364                // log; the page gets the delete-failure toast.
365                tracing::error!(
366                    resource = Self::slug(),
367                    key,
368                    "default delete_record: the table's record key does not parse as the primary \
369                     key; declare the primary key as the record key or override delete_record"
370                );
371                TabloError::Declaration("delete failed".to_string())
372            })?;
373            Ok(query.filter(filter))
374        });
375        async move {
376            query?
377                .delete()
378                .exec(ex)
379                .await
380                .map_err(|error| -> topcoat::Error { error.into() })?;
381            Ok(())
382        }
383    }
384
385    /// Bulk-delete the already-authorized `records`: the handler
386    /// fetches through the tenancy-scoped `IN` query inside the framework
387    /// transaction and checks the policy on every row before calling this.
388    /// The default deletes each record through [`Self::delete_record`] in
389    /// order, through the same `ex` — any error rolls the whole batch back,
390    /// so mid-loop failures delete zero rows. An override of `delete_record`,
391    /// such as a soft delete, therefore covers bulk delete too. Override this
392    /// for a single-statement batch.
393    fn bulk_delete_records(
394        cx: &Cx,
395        records: &[Self::Model],
396        ex: &mut dyn toasty::Executor,
397    ) -> impl std::future::Future<Output = Result<()>> + Send
398    where
399        Self: Sized,
400    {
401        async move {
402            for record in records {
403                Self::delete_record(cx, record, &mut *ex).await?;
404            }
405            Ok(())
406        }
407    }

409    /// Post-commit work for a mutation this resource committed.
410    ///
411    /// The place for a side effect that must not survive a rollback: an email, a
412    /// webhook, an audit row, cache invalidation. Called once per successful
413    /// write, after `tx.commit()` and before the response — running it inside a
414    /// record fn would leak the effect on rollback, and the write handlers' pool
415    /// discipline forbids a second handle while the transaction is open.
416    ///
417    /// [`Committed`] names the mutation and the rows it wrote: the committed row
418    /// a create or update returned, the rows a delete or bulk delete removed
419    /// (one call with every row). It is never called when nothing committed: a
420    /// validation error, a policy denial, a failed record fn, or a failed commit
421    /// all leave the hook untouched, so a rollback cannot produce the effect. A
422    /// hook that returns `Err` is logged and ignored — the write is committed,
423    /// and retries are the app's to build; a panic surfaces as Topcoat's
424    /// panic-isolated 500.
425    ///
426    /// ```ignore
427    /// async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
428    ///     let mut db = db(cx); // a fresh handle is allowed here
429    ///     for post in committed.records() { notify(post).await?; }
430    ///     Ok(())
431    /// }
432    /// ```
433    fn after_commit(
434        _cx: &Cx,
435        _committed: Committed<Self::Model>,
436    ) -> impl std::future::Future<Output = Result<()>> + Send
437    where
438        Self: Sized,
439    {
440        async move { Ok(()) }
441    }

460/// The tenant-scoped base query.
461///
462/// [`Resource::query`] with the [`Resource::tenancy`] filter ANDed onto it, so a
463/// resource that overrides `query` for soft deletes cannot drop the tenant
464/// scope by forgetting to re-state it. Every framework loader and app code
465/// start here.
466///
467/// # Errors
468///
469/// A tenant-scoped resource and no tenant in `cx`: 403, the same answer the
470/// handler gate gives.
471///
472/// App code that loads rows itself must call this: on a scoped resource
473/// [`Resource::query`] is the *tenant-unscoped* base by design.
474pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
475    apply_tenant_scope::<R>(cx, R::query(cx))
476}
477
478/// [`Resource::view_query`] under the same tenant gate and filter as
479/// [`scoped_query`]: the detail page's loader, and the entry point for a page
480/// that owns its own detail view.
481///
482/// # Errors
483///
484/// The same as [`scoped_query`].
485pub fn scoped_view_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
486    apply_tenant_scope::<R>(cx, R::view_query(cx))
487}
488
489/// AND the resource's tenant filter onto `query`: the body of
490/// [`scoped_query`] and [`scoped_view_query`].
491fn apply_tenant_scope<R: Resource>(
492    cx: &Cx,
493    query: Query<List<R::Model>>,
494) -> Result<Query<List<R::Model>>> {
495    let tenancy = R::tenancy();
496    if !tenancy.is_scoped() {
497        return Ok(query);
498    }
499    if let Some(Err(error)) = tenancy.column_field() {
500        return Err(TabloError::Declaration(format!(
501            "resource `{}`'s `Tenancy::column` lens binds no column: {error}",
502            std::any::type_name::<R>(),
503        ))
504        .into());
505    }
506    let tenant = crate::tenancy::require_tenant(cx)?;
507    Ok(match tenancy.filter(tenant) {
508        Some(filter) => query.filter(filter),
509        None => query,
510    })
511}

513/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource): the
514/// bridge that lets the relationship option loaders be generic over the source
515/// surface instead of over `Resource`, so `schema` does not depend on
516/// `resource`.
517///
518/// [`scoped_query`](crate::schema::OptionSource::scoped_query) forwards to
519/// [`scoped_query`], so an option load inherits the tenant gate and filter
520/// exactly as every other loader does. The policy and the tenancy forward
521/// unchanged, and the search expression and default ordering come from the
522/// resource's declared [`table`](Resource::table), which is where "the option
523/// search searches the related resource's searchable columns" lives.
524impl<R: Resource> crate::schema::OptionSource for R {
525    type Model = R::Model;
526
527    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
528        scoped_query::<R>(cx)
529    }
530
531    fn policy() -> impl Policy<R::Model> {
532        <R as Resource>::policy()
533    }
534
535    fn requires_tenant() -> bool {
536        <R as Resource>::tenancy().is_scoped()
537    }
538
539    fn slug() -> String {
540        <R as Resource>::slug()
541    }
542
543    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
544        declared::<R>(cx).table.search_expr(term)
545    }
546
547    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
548        declared::<R>(cx).table.order_by(false)
549    }
550}
