52/// Maps one Toasty `Model` to its admin UI.
53///
54/// # Contract
55///
56/// **Every item but [`Model`](Self::Model), [`Form`](Self::Form) and
57/// [`table`](Self::table) has a default**, so a resource compiles as soon as it
58/// declares its model, its record form (or [`NoForm`](crate::NoForm)) and its
59/// list view — and an omission must fail loudly rather than silently:
60///
61/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**: the declared table must serve a
62///   list, and [`form`](Self::form) must agree with [`Form`](Self::Form): a record form's fields
63///   are the schema's controls, and a [`NoForm`](crate::NoForm) resource declares no schema. A
64///   resource with no form must not allow [`can_create`](Self::can_create). These are declarations,
65///   checked with a Db-only context.
66/// - **Loud at request time**: [`delete_record`](Self::delete_record) defaults to an error naming
67///   the type, so a resource that never implemented delete says so instead of writing nothing
68///   quietly. [`bulk_delete_records`](Self::bulk_delete_records) loops `delete_record` by default,
69///   so it stays loud through the same stub.
70/// - **Chrome follows the declarations**: the row Delete control and the bulk column render when
71///   [`can_delete_any`](Self::can_delete_any) allows, the Edit link when the resource has a record
72///   form, and the View link when it declares [`view`](Self::view); the row predicates then gate
73///   each row.
74/// - **Default-deny is untouched**: every `can_*` defaults to `false`, except `can_delete`, which
75///   defaults to `can_delete_any`; an unconfigured resource exposes no data and no mutation.
76pub trait Resource: Sized + Send + Sync + 'static {
77    /// The persisted model this resource administers.
78    ///
79    /// `Send + Sync` holds for every data-only model struct and is required
80    /// for concurrent rendering of the resource's pages.
81    ///
82    /// `Clone` is part of the contract because a committed mutation names its
83    /// rows: a handler keeps a copy of the rows it loaded while the
84    /// record fn consumes them, so the hook can be handed what was written.
85    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;
86
87    /// The typed value the create and edit forms parse into.
88    ///
89    /// A resource with create and edit pages names its
90    /// [`#[derive(RecordForm)]`](crate::RecordForm) struct; a list-only resource
91    /// names [`NoForm<Self::Model>`](crate::NoForm), and
92    /// [`Panel::resource`](crate::Panel::resource) registers no form route for
93    /// it ([`RecordForm::HAS_FORM`]).
94    type Form: RecordForm<Model = Self::Model>;
95
96    /// The columns an overriding [`Self::create_record`] sets itself, beyond
97    /// the form's fields.
98    ///
99    /// [`Panel::build`](crate::Panel::build) refuses a resource that allows
100    /// create when a non-nullable column is neither a form field, nor filled by
101    /// toasty (`#[auto]`, `#[default(..)]`), nor the stamped tenant column: the
102    /// create would fail at the driver on every submit. A record fn that sets
103    /// such a column by hand names it here.
104    const CREATE_COLUMNS: &'static [&'static str] = &[];
105
106    /// Whether the current user may view the list page.
107    ///
108    /// Also gates relationship option loads: a related resource
109    /// that denies this cannot offer its records as options at all.
110    fn can_view_any(_cx: &Cx) -> bool {
111        false
112    }
113
114    /// Whether the current user may view the given record.
115    ///
116    /// Checked on the edit page (GET), the edit POST (which requires both
117    /// `can_view` and `can_update`), per row in CSV export, on each
118    /// record behind a relationship `Select`'s options, on each row
119    /// of a detail page's relation table, and on the list page as
120    /// the per-row gate of every action link (GH #235: the View link, and the
121    /// `can_view` half of Edit and Delete). Note both hooks default-deny: a
122    /// resource used as a relationship target for option loads must allow
123    /// `can_view_any` **and** `can_view` (overriding one does not imply the
124    /// other), while a relation table consults `can_view` alone. The
125    /// list page deliberately checks only
126    /// `can_view_any` for *membership*: `can_view` is an in-memory Rust
127    /// predicate that cannot run in SQL, and filtering rows after cursor
128    /// pagination would mislabel pages. Row-level visibility that must hold on
129    /// the list belongs in [`Self::query`].
130    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
131        false
132    }
133
134    /// Whether the current user may create a new record.
135    ///
136    /// `Panel::build` calls this with a Db-only context to decide which
137    /// declaration checks apply, so a predicate that reads the request (a
138    /// tenant, a user) answers as it would for an anonymous request there. The
139    /// list page links to the create page only for a resource with a record
140    /// form ([`RecordForm::HAS_FORM`]), whatever this answers.
141    fn can_create(_cx: &Cx) -> bool {
142        false
143    }
144
145    /// Whether the current user may update the given record.
146    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
147        false
148    }
149
150    /// Whether the current user may delete records of this resource at all.
151    ///
152    /// Decides whether the list renders the row Delete control, the bulk
153    /// column, and the bulk bar, and gates the single-delete and bulk-delete
154    /// POSTs before any record loads. The column decision takes no record, so
155    /// the streamed skeleton and the table agree on their columns. The default
156    /// [`can_delete`](Self::can_delete) also calls it once per row.
157    fn can_delete_any(_cx: &Cx) -> bool {
158        false
159    }
160
161    /// Whether the current user may delete the given record.
162    ///
163    /// Defaults to [`can_delete_any`](Self::can_delete_any); override to
164    /// refuse some rows. Checked on the single-delete and bulk-delete POSTs
165    /// after `can_delete_any` and together with `can_view` — the edit
166    /// contract: a record that cannot be viewed cannot be deleted by
167    /// UUID-guessing the route. A row it refuses renders no Delete control and
168    /// no bulk checkbox.
169    fn can_delete(cx: &Cx, _record: &Self::Model) -> bool {
170        Self::can_delete_any(cx)
171    }
172
173    /// How one record is displayed on the detail page, read-only.
174    ///
175    /// The same [`Schema`](crate::schema::Schema) a form uses, rendered for reading: a `TextInput`
176    /// shows its stored value instead of an `<input>`, a `Select` shows the
177    /// option label the form offered, and a layout block keeps the structure it
178    /// declares. Declaring a view is what turns the detail page on — the default
179    /// declares nothing, so the route 404s and no `View` row action renders.
180    ///
181    /// Values come from [`view_values`](Self::view_values). A
182    /// relation is not one of these fields — it is a list of records, not a
183    /// string — and renders through [`view_relations`](Self::view_relations).
184    ///
185    /// Read-only is a promise, not a disabled form: nothing here validates or
186    /// submits, and no field renders a required marker or an error slot.
187    fn view(_cx: &Cx) -> crate::schema::Schema {
188        crate::schema::Schema::empty()
189    }
190
191    /// The related records on this resource's detail page.
192    ///
193    /// [`view`](Self::view) renders from the record's *string projection* — one
194    /// `HashMap<String, String>` — because that is what every field binds. A
195    /// relation is not a string and may be a list of records, so it cannot ride
196    /// that map; this hook renders it instead, with the loaded record in hand.
197    ///
198    /// This is the half that makes [`Self::query`]'s `include` pay: the related
199    /// rows are already loaded on the record, so a hook that reads
200    /// `record.comments.get()` issues no query at all. Touching an un-included
201    /// relation panics ([`Deferred::get`](toasty::Deferred)), so check
202    /// [`Deferred::is_unloaded`](toasty::Deferred::is_unloaded) — the same marker
203    /// the list columns check.
204    ///
205    /// Returns `None` (the default) for a resource with no related records to
206    /// show, which renders nothing. The two lifetimes are deliberately separate:
207    /// the returned view may borrow the request context, never the record — a
208    /// view holding the record would pin the handler's local binding for as long
209    /// as the page, which does not compile, and the projections return owned
210    /// strings ([`render_relation`]), so nothing needs to.
211    fn view_relations<'a>(
212        _cx: &'a Cx,
213        _record: &Self::Model,
214    ) -> Option<topcoat::view::BoxView<'a>> {
215        None
216    }
217
218    /// Whether this resource declares a detail page.
219    ///
220    /// Derived from [`view`](Self::view) rather than declared twice, so the
221    /// route and the row link cannot disagree with the schema that renders
222    /// them. The detail handler uses it to 404 a resource that declares
223    /// nothing, and the row chrome uses it to leave the link off.
224    fn viewed(cx: &Cx) -> bool {
225        !Self::view(cx).is_empty()
226    }

228    /// The record's label in the detail page's title, or `None` when
229    /// the record has no label to show.
230    ///
231    /// The detail page titles itself with this label when a resource returns
232    /// `Some`, and with [`navigation_label`](Self::navigation_label) plus the
233    /// URL's record key when it returns `None` — the default, so a resource
234    /// that declares nothing keeps the title it has. The showcase's
235    /// `PostResource` is the worked example: it returns the post's title, so
236    /// its heading reads the title instead of `Blog Posts <record key>`.
237    ///
238    /// `cx` is the request's context — the same one [`view`](Self::view) and
239    /// [`view_values`](Self::view_values) receive — so a label
240    /// can read request state (a locale, a tenant). The default ignores both
241    /// arguments and returns `None`.
242    ///
243    /// A label is display text, not a key. Two records can share one (two
244    /// users named Ada), so it cannot replace the table key, whose projection
245    /// must stay injective within a page for keyed diffs and bulk selection,
246    /// or the record key, which the action routes resolve as the
247    /// model's typed PK.
248    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
249        None
250    }
251
252    /// A public URL for one record, rendered as a "View public post"-style
253    /// link on the detail and edit pages when `Some`.
254    ///
255    /// The default declares none, so no link renders. A resource whose records
256    /// have a public page overrides this with its URL — the showcase's posts
257    /// return their `/blog/{id}` page.
258    fn public_url(_cx: &Cx, _record: &Self::Model) -> Option<String> {
259        None
260    }
261
262    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
263    /// at `{panel prefix}/users`.
264    ///
265    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
266    /// take the resource type's name, strip a trailing `Resource`, pluralize
267    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
268    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
269    /// naming the rules cannot guess (`UsersResource` pluralizes to
270    /// `userses` — name resources singular, or override).
271    fn slug() -> String {
272        let name = type_short_name::<Self>();
273        let singular = name.strip_suffix("Resource").unwrap_or(name);
274        kebab_case(&pluralize(singular))
275    }
276
277    /// The sidebar label, e.g. `"Users"`.
278    ///
279    /// Defaults to the pluralized `Model` type name (Filament's plural model
280    /// label): `User` → `Users`, `Category` → `Categories`, `Person` →
281    /// `People`. Override for custom wording.
282    fn navigation_label() -> String {
283        pluralize(type_short_name::<Self::Model>())
284    }

286    /// Base query — the seam for a resource's **own** row scoping (ADR-0002):
287    /// soft deletes, row-level visibility, and the relations a page loads.
288    ///
289    /// **Tenancy is not this method's job.** When
290    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
291    /// the tenant filter, derived from the model's `tenant_id` column, onto
292    /// whatever this returns, at every loader through [`scoped_query`]. Do not
293    /// re-state `tenant_id().eq(tenant_id(cx))`: the copy is redundant, and one
294    /// that disagreed with the derived column would hide rows rather than widen
295    /// access. Code outside the framework's loaders starts from
296    /// [`scoped_query`].
297    ///
298    /// Returns the raw typed statement query, which composes generically —
299    /// `filter`, `order_by` and `Paginate::new` work on the raw form for any
300    /// `M: Model` — so one panel handler drives every resource's list page.
301    ///
302    /// # Keep unique constraints in step with this scope
303    ///
304    /// The app-side unique pre-check probes through the same tenant-scoped
305    /// query, so a `#[unique]` index *broader* than the scope is invisible: the
306    /// probe misses the colliding row and the user gets a 500 instead of the
307    /// inline "has already been taken". Scope the constraint to match —
308    /// `#[unique(tenant_id, email)]`. Not checkable at declaration time, since
309    /// a query's filters are not introspectable; upstream #117 is the fix.
310    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
311        toasty::stmt::Query::<List<Self::Model>>::all()
312    }
313
314    /// [`Self::query`] narrowed to the relations `needs` asks for.
315    ///
316    /// A loader that reads only part of what [`Self::query`] loads states the
317    /// includes it reads here, and the resource answers with the matching branch
318    /// of its base query. The names are the opaque vocabulary [`IncludeNeeds`]
319    /// documents; the resource maps them onto its typed `include(..)` calls.
320    ///
321    /// **The default ignores `needs` and returns [`Self::query`] unchanged**, so
322    /// narrowing is opt-in per resource. An override must keep the non-tenant
323    /// scope [`Self::query`] carries and any relation its
324    /// [`can_view`](Self::can_view) reads, because the loaders run that
325    /// predicate over the loaded rows; the tenant half is the framework's to AND
326    /// on. The list and the export ask for their table's
327    /// [`include_needs`](Table::include_needs). The detail page reads
328    /// [`view_relations`](Self::view_relations), an opaque hook with no
329    /// declaration, so it loads [`Self::query`] unchanged; edit, delete, bulk
330    /// delete and the option and pagination probes read no relation, so they ask
331    /// for an empty set.
332    fn query_with(cx: &Cx, _needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
333        Self::query(cx)
334    }
335
336    /// The CSV export's base query: [`Self::query`], narrowed to the relations
337    /// the rendered columns declared.
338    ///
339    /// The export writes a cell per column, so it asks the table which relations
340    /// those columns' projections read ([`Table::include_needs`] — each column
341    /// declares them with [`TextColumn::needs`]) and hands the answer here.
342    /// `docs/guide/src/resources.md` states the override contract: one branch per
343    /// declared name, so an include `query` carries for the detail page or the
344    /// live list rides along on an export only when a rendered column declared
345    /// it.
346    ///
347    /// **The default delegates to [`Self::query_with`]**, which returns
348    /// [`Self::query`] unchanged unless the resource overrides it. Over-fetching
349    /// costs a join; dropping a relation a column reads breaks the render, so the
350    /// default over-fetches.
351    ///
352    /// # What an override must keep
353    ///
354    /// - **The non-tenant scope of [`Self::query`]** (soft deletes, row-level visibility): the
355    ///   export is a reader like any other, and an override that drops that half exports other
356    ///   rows. The tenant half is the framework's.
357    /// - **Whatever the policy path reads.** The visibility scan calls
358    ///   [`Self::can_view`](Self::can_view) on every row of both passes before any cell is written,
359    ///   so a `can_view` that reads a relation needs it included even though no column declared it.
360    /// - **Every name a column declared.** A declared name with no matching include renders an
361    ///   unloaded relation, which the column's `is_unloaded` guard (ADR-0011) reports in test
362    ///   builds instead of a silent `"-"`.
363    fn export_query(cx: &Cx, needs: &IncludeNeeds) -> toasty::stmt::Query<List<Self::Model>> {
364        Self::query_with(cx, needs)
365    }

367    /// Whether this resource requires a tenant in every handler.
368    ///
369    /// Opt-in and default-open: `false` preserves [`Self::query`] exactly as
370    /// written. `true` means two things:
371    ///
372    /// 1. **The gate.** Every handler 403s when the request carries no tenant, instead of leaking
373    ///    unscoped rows or minting nil-tenant orphans.
374    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto the base query,
375    ///    deriving the column from the model's own schema or the resource's [`Self::tenant_scope`].
376    ///    A gated resource that declares neither a `tenant_id` UUID column nor an override is
377    ///    refused by [`Panel::build`](crate::Panel::build) at boot rather than served unscoped or
378    ///    failing per request.
379    ///
380    /// A resource that must genuinely serve more than the request tenant
381    /// declares `false` and scopes in [`Self::query`] by hand, giving up the
382    /// gate above along with the derived filter.
383    fn requires_tenant() -> bool {
384        false
385    }
386
387    /// The predicate the framework ANDs onto this resource's base query to
388    /// scope it to `tenant`, or `None` when there is nothing to AND.
389    ///
390    /// The default derives it from the model: `tenant_id = tenant`, on the
391    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
392    /// Override it when the resource's tenancy is not a column on its own
393    /// model — a row that inherits its parent's tenant states the relation
394    /// path here instead, and the framework applies it exactly as it applies
395    /// the derived one. The showcase's comments are the worked example.
396    ///
397    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
398    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
399    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
400    /// and every loader keeps answering an error naming the resource
401    /// rather than running its query without a tenant predicate — the backstop
402    /// for a predicate that is only `None` for some tenants. There is
403    /// deliberately no override that *removes* the scope — a resource that
404    /// must serve more than one tenant declares `requires_tenant() = false`
405    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
406    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
407        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
408    }

479    /// Delete the already-authorized `record` (#86).
480    ///
481    /// The handler loads `record` through the tenancy-scoped query inside the
482    /// framework transaction and checks `can_delete` on that snapshot: use it,
483    /// never re-query by id, and write through `ex`.
484    fn delete_record(
485        _cx: &Cx,
486        _record: Self::Model,
487        _ex: &mut dyn toasty::Executor,

489    where

492        async move {
493            Err(std::io::Error::other(format!(
494                "delete not implemented for {}",
495                std::any::type_name::<Self>()
496            ))
497            .into())
498        }
499    }

501    /// Bulk-delete the already-authorized `records`: the handler
502    /// fetches through the tenancy-scoped `IN` query inside the framework
503    /// transaction and checks `can_delete` on every row before calling this.
504    /// The default deletes each record through [`Self::delete_record`] in
505    /// order, through the same `ex` — any error rolls the whole batch back,
506    /// so mid-loop failures delete zero rows. Override for a single-statement
507    /// batch.
508    fn bulk_delete_records(
509        cx: &Cx,
510        records: Vec<Self::Model>,
511        ex: &mut dyn toasty::Executor,
512    ) -> impl std::future::Future<Output = Result<()>> + Send
513    where
514        Self: Sized,
515    {
516        async move {
517            for record in records {
518                Self::delete_record(cx, record, &mut *ex).await?;
519            }
520            Ok(())
521        }
522    }

524    /// Post-commit work for a mutation this resource committed.
525    ///
526    /// The place for a side effect that must not survive a rollback: an email, a
527    /// webhook, an audit row, cache invalidation. Called once per successful
528    /// write, after `tx.commit()` and before the response — running it inside a
529    /// record fn would leak the effect on rollback, and the write handlers' pool
530    /// discipline forbids a second handle while the transaction is open.
531    ///
532    /// [`Committed`] names the mutation and the rows it wrote: the committed row
533    /// a create or update returned, the rows a delete or bulk delete removed
534    /// (one call with every row). It is never called when nothing committed: a
535    /// validation error, a policy denial, a failed record fn, or a failed commit
536    /// all leave the hook untouched, so a rollback cannot produce the effect. A
537    /// hook that returns `Err` is logged and ignored — the write is committed,
538    /// and retries are the app's to build; a panic surfaces as Topcoat's
539    /// panic-isolated 500.
540    ///
541    /// ```ignore
542    /// async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
543    ///     let mut db = db(cx); // a fresh handle is allowed here
544    ///     for post in committed.records() { notify(post).await?; }
545    ///     Ok(())
546    /// }
547    /// ```
548    fn after_commit(
549        _cx: &Cx,
550        _committed: Committed<Self::Model>,
551    ) -> impl std::future::Future<Output = Result<()>> + Send
552    where
553        Self: Sized,
554    {
555        async move { Ok(()) }
556    }

558    /// The record's values for the detail page, keyed by the name each
559    /// [`view`](Self::view) field binds.
560    ///
561    /// The detail page reads the form's keys from
562    /// [`RecordForm::hydrate`], and this adds any
563    /// key only the view shows; the form's keys win on a collision. A
564    /// resource with no form ([`NoForm`](crate::NoForm)) supplies every key its
565    /// view shows here.
566    ///
567    /// `cx` carries the app schema, which an embedded value's keys need
568    /// ([`write_embedded`](crate::schema::write_embedded)). The default is
569    /// empty.
570    fn view_values(_cx: &Cx, _record: &Self::Model) -> HashMap<String, String> {
571        HashMap::new()
572    }

575/// The tenant-scoped full base query.
576///
577/// [`Resource::query`] with the predicate from [`Resource::tenant_scope`]
578/// ANDed onto it, so a resource that overrides `query` for includes or soft
579/// deletes cannot drop the tenant scope by forgetting to re-state it. This is
580/// the entry point for a reader that wants the full base query — the detail
581/// page and app code. A framework loader that reads only part of it uses
582/// `scoped_query_with`.
583///
584/// # Errors
585///
586/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer the handler gate
587///   gives.
588/// - A gated resource that supplies no tenant predicate — no discoverable `tenant_id` UUID column,
589///   no [`Resource::tenant_scope`] override → an error naming the resource and the model.
590///   [`Panel::build`](crate::Panel::build) refuses that declaration at boot, so this is the
591///   backstop for a predicate that is `None` for the request's tenant, and for app code outside a
592///   panel. It is deliberately **not** a fallback to the unscoped query: a silent miss would be the
593///   leak [`Resource::requires_tenant`] exists to prevent.
594///
595/// App code that loads rows itself must call this — on a gated resource
596/// [`Resource::query`] is the *tenant-unscoped* base by design, so calling it
597/// directly is safe only for rows whose tenant membership is already settled (a
598/// write by id against a record the framework loaded and authorized).
599pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
600    apply_tenant_scope::<R>(cx, R::query(cx))
601}
602
603/// [`scoped_query`] over a loader's declared includes: the same tenant gate
604/// and derived predicate, seeded from [`Resource::query_with`] instead of
605/// [`Resource::query`]. A loader that reads no relation passes
606/// `IncludeNeeds::default()`.
607pub(crate) fn scoped_query_with<R: Resource>(
608    cx: &Cx,
609    needs: &IncludeNeeds,
610) -> Result<Query<List<R::Model>>> {
611    apply_tenant_scope::<R>(cx, R::query_with(cx, needs))
612}
613
614/// AND the framework's tenant predicate onto `query`.
615///
616/// The body of [`scoped_query`] and [`scoped_query_with`], split out so every
617/// seed — the base query, a loader's narrowed branch, the export's
618/// [`Resource::export_query`](Resource::export_query) — shares one gate, one
619/// predicate, and one fail-closed error, and so the seeds cannot drift apart.
620/// It is crate-internal because a caller outside the crate always has a
621/// `Resource`, and so always wants one of the `scoped_query*` entry points.
622pub(crate) fn apply_tenant_scope<R: Resource>(
623    cx: &Cx,
624    query: Query<List<R::Model>>,
625) -> Result<Query<List<R::Model>>> {
626    if !R::requires_tenant() {
627        return Ok(query);
628    }
629    let tenant = crate::tenancy::require_tenant(cx)?;
630    let Some(filter) = R::tenant_scope(tenant) else {
631        // Fail closed and loudly: the resource declared a gate whose scope the
632        // framework cannot derive and the resource did not state, and running
633        // the query unscoped is the one outcome that declaration exists to
634        // prevent. `Panel::build` already refused the resource if *no* tenant
635        // could scope it; this is the backstop for a `tenant_scope`
636        // that answers `None` only for this tenant, and for callers outside a
637        // panel.
638        tracing::error!(
639            resource = R::slug(),
640            model = std::any::type_name::<R::Model>(),
641            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
642             UUID column on the model and no `tenant_scope` override (GH #223)"
643        );
644        return Err(std::io::Error::other(format!(
645            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
646             `tenant_id` UUID column to derive the filter from, and the resource does not override \
647             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
648             `requires_tenant` and scope in `query`",
649            R::slug(),
650            std::any::type_name::<R::Model>(),
651        ))
652        .into());
653    };
654    Ok(query.filter(filter))
655}

657/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
658/// bridge that lets the relationship option loaders be generic over the source
659/// surface instead of over `Resource`, so `schema` does not depend on
660/// `resource`.
661///
662/// A resource answers the loaders here, so the loaders never name `Resource`.
663/// [`scoped_query`](crate::schema::OptionSource::scoped_query), the one required
664/// method, forwards to [`scoped_query`], so an option load inherits the tenant
665/// gate and the derived tenant predicate exactly as every other loader does; it
666/// is deliberately not [`Resource::query`], which on a gated resource is the
667/// *tenant-unscoped* base.
668///
669/// [`options_query`](crate::schema::OptionSource::options_query) forwards to
670/// `scoped_query_with` with an empty [`IncludeNeeds`]: an option load projects a
671/// value and a label off each row and reads no relation of its own, so a
672/// resource that narrows its loaders narrows option loads too. A source that
673/// overrides nothing keeps the full [`scoped_query`]. The policy predicates and
674/// the tenant declaration forward unchanged, and the search expression and
675/// default ordering come from the resource's declared [`table`](Resource::table),
676/// which is where "the option search searches the related resource's searchable
677/// columns" lives.
678impl<R: Resource> crate::schema::OptionSource for R {
679    type Model = R::Model;
680
681    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
682        scoped_query::<R>(cx)
683    }
684
685    fn options_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
686        scoped_query_with::<R>(cx, &IncludeNeeds::default())
687    }
688
689    fn can_view_any(cx: &Cx) -> bool {
690        <R as Resource>::can_view_any(cx)
691    }
692
693    fn can_view(cx: &Cx, record: &R::Model) -> bool {
694        <R as Resource>::can_view(cx, record)
695    }
696
697    fn requires_tenant() -> bool {
698        <R as Resource>::requires_tenant()
699    }
700
701    fn slug() -> String {
702        <R as Resource>::slug()
703    }
704
705    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
706        <R as Resource>::table(cx).search_expr(term)
707    }
708
709    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
710        <R as Resource>::table(cx).order_by(false)
711    }
712}
