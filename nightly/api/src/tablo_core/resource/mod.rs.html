62/// Maps one Toasty `Model` to its admin UI.
63///
64/// # Contract
65///
66/// **Every item but [`Model`](Self::Model), [`Form`](Self::Form) and
67/// [`table`](Self::table) has a default**, so a resource compiles as soon as it
68/// declares its model, its record form (or [`NoForm`](crate::NoForm)) and its
69/// list view — and an omission must fail loudly rather than silently:
70///
71/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**: the declared table must serve a
72///   list, and [`form`](Self::form) must agree with [`Form`](Self::Form): a record form's fields
73///   are the schema's controls, and a [`NoForm`](crate::NoForm) resource declares no schema. A
74///   resource with no form must not allow [`can_create`](Self::can_create). These are declarations,
75///   checked with a Db-only context.
76/// - **Loud at request time**: a record fn's error fails the write and rolls its transaction back,
77///   never a partial write. [`delete_record`](Self::delete_record) defaults to deleting the row
78///   through [`scoped_query`], filtered to the record's key, and refuses a table whose record key
79///   does not parse back as the primary key — naming the override that fixes it — rather than
80///   deleting the wrong row or nothing. [`bulk_delete_records`](Self::bulk_delete_records) loops
81///   `delete_record` by default, so an override covers bulk delete too.
82/// - **Chrome follows the declarations**: the row Delete control and the bulk column render when
83///   [`can_delete_any`](Self::can_delete_any) allows, the Edit link when the resource has a record
84///   form, and the View link when it declares [`view`](Self::view); the row predicates then gate
85///   each row.
86/// - **Default-deny is untouched**: every `can_*` defaults to `false`, except `can_delete`, which
87///   defaults to `can_delete_any`; an unconfigured resource exposes no data and no mutation.
88pub trait Resource: Sized + Send + Sync + 'static {
89    /// The persisted model this resource administers.
90    ///
91    /// `Send + Sync` holds for every data-only model struct and is required
92    /// for concurrent rendering of the resource's pages.
93    ///
94    /// `Clone` is part of the contract because a committed mutation names its
95    /// rows: a handler keeps a copy of the rows it loaded while the
96    /// record fn consumes them, so the hook can be handed what was written.
97    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;
98
99    /// The typed value the create and edit forms parse into.
100    ///
101    /// A resource with create and edit pages names its
102    /// [`#[derive(RecordForm)]`](crate::RecordForm) struct; a list-only resource
103    /// names [`NoForm<Self::Model>`](crate::NoForm), and
104    /// [`Panel::resource`](crate::Panel::resource) registers no form route for
105    /// it ([`RecordForm::HAS_FORM`]).
106    type Form: RecordForm<Model = Self::Model>;
107
108    /// The columns an overriding [`Self::create_record`] sets itself, beyond
109    /// the form's fields.
110    ///
111    /// [`Panel::build`](crate::Panel::build) refuses a resource that allows
112    /// create when a non-nullable column is neither a form field, nor filled by
113    /// toasty (`#[auto]`, `#[default(..)]`), nor the stamped tenant column: the
114    /// create would fail at the driver on every submit. A record fn that sets
115    /// such a column by hand names it here.
116    const CREATE_COLUMNS: &'static [&'static str] = &[];
117
118    /// Whether the current user may view the list page.
119    ///
120    /// Also gates relationship option loads: a related resource
121    /// that denies this cannot offer its records as options at all.
122    fn can_view_any(_cx: &Cx) -> bool {
123        false
124    }
125
126    /// Whether the current user may view the given record.
127    ///
128    /// Checked on the edit page (GET), the edit POST (which requires both
129    /// `can_view` and `can_update`), per row in CSV export, on each
130    /// record behind a relationship choice's options, on each row
131    /// of a detail page's relation table, and on the list page as
132    /// the per-row gate of every action link (GH #235: the View link, and the
133    /// `can_view` half of Edit and Delete). Note both hooks default-deny: a
134    /// resource used as a relationship target for option loads must allow
135    /// `can_view_any` **and** `can_view` (overriding one does not imply the
136    /// other), while a relation table consults `can_view` alone. The
137    /// list page deliberately checks only
138    /// `can_view_any` for *membership*: `can_view` is an in-memory Rust
139    /// predicate that cannot run in SQL, and filtering rows after cursor
140    /// pagination would mislabel pages. Row-level visibility that must hold on
141    /// the list belongs in [`Self::query`].
142    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
143        false
144    }
145
146    /// Whether the current user may create a new record.
147    ///
148    /// `Panel::build` calls this with a Db-only context to decide which
149    /// declaration checks apply, so a predicate that reads the request (a
150    /// tenant, a user) answers as it would for an anonymous request there. The
151    /// list page links to the create page only for a resource with a record
152    /// form ([`RecordForm::HAS_FORM`]), whatever this answers.
153    fn can_create(_cx: &Cx) -> bool {
154        false
155    }
156
157    /// Whether the current user may update the given record.
158    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
159        false
160    }
161
162    /// Whether the current user may delete records of this resource at all.
163    ///
164    /// Decides whether the list renders the row Delete control, the bulk
165    /// column, and the bulk bar, and gates the single-delete and bulk-delete
166    /// POSTs before any record loads. The column decision takes no record, so
167    /// the streamed skeleton and the table agree on their columns. The default
168    /// [`can_delete`](Self::can_delete) also calls it once per row.
169    fn can_delete_any(_cx: &Cx) -> bool {
170        false
171    }
172
173    /// Whether the current user may delete the given record.
174    ///
175    /// Defaults to [`can_delete_any`](Self::can_delete_any); override to
176    /// refuse some rows. Checked on the single-delete and bulk-delete POSTs
177    /// after `can_delete_any` and together with `can_view` — the edit
178    /// contract: a record that cannot be viewed cannot be deleted by
179    /// UUID-guessing the route. A row it refuses renders no Delete control and
180    /// no bulk checkbox.
181    fn can_delete(cx: &Cx, _record: &Self::Model) -> bool {
182        Self::can_delete_any(cx)
183    }
184
185    /// How one record is displayed on the detail page, read-only.
186    ///
187    /// The same [`Schema`](crate::schema::Schema) vocabulary a form uses,
188    /// rendered for reading: a text field shows its stored value instead of an
189    /// `<input>`, a choice shows the option label the form offered, and a
190    /// layout block keeps the structure it declares. Like Filament's infolist,
191    /// it is its own declaration: it may show keys the form does not, and a
192    /// resource with no form declares one too. Declaring a view is what turns
193    /// the detail page on — the default declares nothing, so the route 404s
194    /// and no `View` row action renders.
195    ///
196    /// Values come from [`view_values`](Self::view_values) and the record
197    /// form's `hydrate`. A field whose key neither supplies renders
198    /// `(missing)` and fails a `debug_assert!`. A relation is not one of these
199    /// fields — it is a list of records, not a string — and renders through
200    /// [`relations`](Self::relations).
201    ///
202    /// Read-only is a promise, not a disabled form: nothing here validates or
203    /// submits, and no field renders a required marker or an error slot.
204    fn view(_cx: &Cx) -> crate::schema::Schema {
205        crate::schema::Schema::empty()
206    }
207
208    /// Free-form content on this resource's detail page, below the
209    /// [`view`](Self::view) schema and above the [`relations`](Self::relations):
210    /// anything computed from the loaded record that is not one of its fields,
211    /// such as a word count.
212    ///
213    /// Returns `None` (the default) to render nothing. The two lifetimes are
214    /// deliberately separate: the returned view may borrow the request context,
215    /// never the record — a view holding the record would pin the handler's
216    /// local binding for as long as the page, which does not compile.
217    fn view_content<'a>(_cx: &'a Cx, _record: &Self::Model) -> Option<topcoat::view::BoxView<'a>> {
218        None
219    }

221    /// The related resources whose rows belong to a record: each renders on
222    /// this resource's detail and edit pages as the related resource's own
223    /// list table, narrowed to the record (Filament's relation managers) —
224    /// read-only on the detail page, and on the edit page with its write
225    /// actions and a create link that opens its form with the record already
226    /// chosen.
227    ///
228    /// Takes no `Cx`, like [`navigation`](Self::navigation): the relations are
229    /// a declaration, and each related resource's policies decide per request
230    /// what its table shows. The default declares none.
231    fn relations() -> Vec<Relation<Self::Model>> {
232        Vec::new()
233    }

235    /// The resource's custom [`Action`]s, in button order. Defaults to none.
236    ///
237    /// ```ignore
238    /// fn actions() -> Actions<Self> {
239    ///     Actions::new().add::<Publish>()
240    /// }
241    /// ```
242    fn actions() -> Actions<Self> {
243        Actions::new()
244    }

246    /// Whether this resource declares a detail page.
247    ///
248    /// Derived from [`view`](Self::view) rather than declared twice, so the
249    /// route and the row link cannot disagree with the schema that renders
250    /// them. The detail handler uses it to 404 a resource that declares
251    /// nothing, and the row chrome uses it to leave the link off.
252    fn viewed(cx: &Cx) -> bool {
253        !Self::view(cx).is_empty()
254    }

256    /// The record's label in the detail page's title, or `None` when
257    /// the record has no label to show.
258    ///
259    /// The detail page titles itself with this label when a resource returns
260    /// `Some`, and with [`label`](Self::label) plus the URL's record key when
261    /// it returns `None`, the default. The showcase's `PostResource` returns
262    /// the post's title, so its heading reads the title instead of
263    /// `Blog Post <record key>`.
264    ///
265    /// `cx` is the request's context — the same one [`view`](Self::view) and
266    /// [`view_values`](Self::view_values) receive — so a label
267    /// can read request state (a locale, a tenant). The default ignores both
268    /// arguments and returns `None`.
269    ///
270    /// A label is display text, not a key. Two records can share one (two
271    /// users named Ada), so it cannot replace the table key, whose projection
272    /// must stay injective within a page for keyed diffs and bulk selection,
273    /// or the record key, which the action routes resolve as the
274    /// model's typed PK.
275    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
276        None
277    }

279    /// A public URL for one record, rendered as a "View public post"-style
280    /// link on the detail and edit pages when `Some`.
281    ///
282    /// The default declares none, so no link renders. A resource whose records
283    /// have a public page overrides this with its URL — the showcase's posts
284    /// return their `/blog/{id}` page.
285    fn public_url(_cx: &Cx, _record: &Self::Model) -> Option<String> {
286        None
287    }

289    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
290    /// at `{panel prefix}/users`.
291    ///
292    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
293    /// take the resource type's name, strip a trailing `Resource`, pluralize
294    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
295    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
296    /// naming the rules cannot guess (`UsersResource` pluralizes to
297    /// `userses` — name resources singular, or override).
298    fn slug() -> String {
299        kebab_case(&pluralize(type_stem::<Self>("Resource")))
300    }

302    /// One record's name, e.g. `"User"`: the noun in the "Create {label}" and
303    /// "Edit {label}" titles.
304    ///
305    /// Defaults to the `Model` type name (Filament's model label). Override for
306    /// custom wording; [`navigation_label`](Self::navigation_label) pluralizes
307    /// it.
308    fn label() -> String {
309        type_short_name::<Self::Model>().to_string()
310    }

312    /// The sidebar label and list title, e.g. `"Users"`.
313    ///
314    /// Defaults to the pluralized [`label`](Self::label) (Filament's plural
315    /// model label): `User` → `Users`, `Category` → `Categories`, `Person` →
316    /// `People`. Override for a plural the rules cannot guess.

318        pluralize(&Self::label())
319    }

321    /// Base query — the seam for a resource's **own** row scoping (ADR-0002):
322    /// soft deletes and row-level visibility. Every loader starts from it.
323    ///
324    /// **Relations are not this method's job.** The list and the export load
325    /// the relations the table's columns declare ([`TextColumn::include`]), and
326    /// the detail page loads [`Self::view_query`]. Include a relation here only
327    /// when a closure no column covers reads it on every loader's rows: one
328    /// [`can_view`](Self::can_view) reads, or one a table's `group_by` or row
329    /// key reads without a column including it.
330    ///
331    /// **Tenancy is not this method's job either.** When
332    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
333    /// the tenant filter, derived from the model's `tenant_id` column, onto
334    /// whatever this returns, at every loader through [`scoped_query`]. Do not
335    /// re-state `tenant_id().eq(tenant_id(cx))`: the copy is redundant, and one
336    /// that disagreed with the derived column would hide rows rather than widen
337    /// access. Code outside the framework's loaders starts from
338    /// [`scoped_query`].
339    ///
340    /// Returns the raw typed statement query, which composes generically —
341    /// `filter`, `order_by` and `Paginate::new` work on the raw form for any
342    /// `M: Model` — so one panel handler drives every resource's list page.
343    ///
344    /// # Keep unique constraints in step with this scope
345    ///
346    /// The app-side unique pre-check probes through the same tenant-scoped
347    /// query, so a `#[unique]` index *broader* than the scope is invisible: the
348    /// probe misses the colliding row and the user gets a 500 instead of the
349    /// inline "has already been taken". Scope the constraint to match —
350    /// `#[unique(tenant_id, email)]`. Not checkable at declaration time, since
351    /// a query's filters are not introspectable; upstream #117 is the fix.
352    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
353        toasty::stmt::Query::<List<Self::Model>>::all()
354    }
355
356    /// The detail page's query: [`Self::query`] plus the relations the page
357    /// reads off the loaded row — in [`view_values`](Self::view_values) or
358    /// [`view_content`](Self::view_content) — so include them here. A
359    /// [`relation`](Self::relations) table runs its own query and needs none.
360    ///
361    /// ```ignore
362    /// fn view_query(cx: &Cx) -> Query<List<Post>> {
363    ///     let author: Include<Post, Author> = Post::fields().author().into();
364    ///     Self::query(cx).include(author)
365    /// }
366    /// ```
367    ///
368    /// The default is [`Self::query`] unchanged. The framework ANDs the tenant
369    /// scope onto it, as it does onto [`Self::query`].
370    fn view_query(cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
371        Self::query(cx)
372    }

374    /// Whether this resource requires a tenant in every handler.
375    ///
376    /// Opt-in and default-open: `false` preserves [`Self::query`] exactly as
377    /// written. `true` means two things:
378    ///
379    /// 1. **The gate.** Every handler 403s when the request carries no tenant, instead of leaking
380    ///    unscoped rows or minting nil-tenant orphans.
381    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto the base query,
382    ///    deriving the column from the model's own schema or the resource's [`Self::tenant_scope`].
383    ///    A gated resource that declares neither a `tenant_id` UUID column nor an override is
384    ///    refused by [`Panel::build`](crate::Panel::build) at boot rather than served unscoped or
385    ///    failing per request.
386    ///
387    /// A resource that must genuinely serve more than the request tenant
388    /// declares `false` and scopes in [`Self::query`] by hand, giving up the
389    /// gate above along with the derived filter.
390    fn requires_tenant() -> bool {
391        false
392    }
393
394    /// The predicate the framework ANDs onto this resource's base query to
395    /// scope it to `tenant`, or `None` when there is nothing to AND.
396    ///
397    /// The default derives it from the model: `tenant_id = tenant`, on the
398    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
399    /// Override it when the resource's tenancy is not a column on its own
400    /// model — a row that inherits its parent's tenant states the relation
401    /// path here instead, and the framework applies it exactly as it applies
402    /// the derived one. The showcase's comments are the worked example.
403    ///
404    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
405    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
406    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
407    /// and every loader keeps answering an error naming the resource
408    /// rather than running its query without a tenant predicate — the backstop
409    /// for a predicate that is only `None` for some tenants. There is
410    /// deliberately no override that *removes* the scope — a resource that
411    /// must serve more than one tenant declares `requires_tenant() = false`
412    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
413    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
414        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
415    }

492    /// Delete the already-authorized `record` (#86).
493    ///
494    /// The handler loads `record` through the tenancy-scoped query inside the
495    /// framework transaction and checks `can_delete` on that snapshot, then
496    /// calls this with the same transaction as `ex`. The after-commit hook
497    /// receives the snapshot once the delete commits.
498    ///
499    /// The default deletes the row through [`scoped_query`], filtered to the
500    /// record's key: the table's record key, which is the model's primary key
501    /// (see [`Table::new_split`]). Override to delete another way, such as a
502    /// soft delete.
503    fn delete_record(
504        cx: &Cx,
505        record: &Self::Model,
506        ex: &mut dyn toasty::Executor,
507    ) -> impl std::future::Future<Output = Result<()>> + Send
508    where
509        Self: Sized,
510    {
511        let key = Self::table(cx).record_key_of(record);
512        let query = scoped_query::<Self>(cx).and_then(|query| {
513            let filter = crate::schema::pk_eq_expr::<Self::Model>(&key).ok_or_else(|| {
514                // The handler found the row by this key, so a key that does
515                // not parse back is a table whose record key is not the
516                // primary key (`Table::new_split`). The detail stays in the
517                // log; the page gets the delete-failure toast.
518                tracing::error!(
519                    resource = Self::slug(),
520                    key,
521                    "default delete_record: the table's record key does not parse as the primary \
522                     key; declare the primary key as the record key or override delete_record"
523                );
524                TabloError::Declaration("delete failed".to_string())
525            })?;
526            Ok(query.filter(filter))
527        });
528        async move {
529            query?
530                .delete()
531                .exec(ex)
532                .await
533                .map_err(|error| -> topcoat::Error { error.into() })?;
534            Ok(())
535        }
536    }

538    /// Bulk-delete the already-authorized `records`: the handler
539    /// fetches through the tenancy-scoped `IN` query inside the framework
540    /// transaction and checks `can_delete` on every row before calling this.
541    /// The default deletes each record through [`Self::delete_record`] in
542    /// order, through the same `ex` — any error rolls the whole batch back,
543    /// so mid-loop failures delete zero rows. An override of `delete_record`,
544    /// such as a soft delete, therefore covers bulk delete too. Override this
545    /// for a single-statement batch.
546    fn bulk_delete_records(
547        cx: &Cx,
548        records: &[Self::Model],
549        ex: &mut dyn toasty::Executor,
550    ) -> impl std::future::Future<Output = Result<()>> + Send
551    where
552        Self: Sized,
553    {
554        async move {
555            for record in records {
556                Self::delete_record(cx, record, &mut *ex).await?;
557            }
558            Ok(())
559        }
560    }

562    /// Post-commit work for a mutation this resource committed.
563    ///
564    /// The place for a side effect that must not survive a rollback: an email, a
565    /// webhook, an audit row, cache invalidation. Called once per successful
566    /// write, after `tx.commit()` and before the response — running it inside a
567    /// record fn would leak the effect on rollback, and the write handlers' pool
568    /// discipline forbids a second handle while the transaction is open.
569    ///
570    /// [`Committed`] names the mutation and the rows it wrote: the committed row
571    /// a create or update returned, the rows a delete or bulk delete removed
572    /// (one call with every row). It is never called when nothing committed: a
573    /// validation error, a policy denial, a failed record fn, or a failed commit
574    /// all leave the hook untouched, so a rollback cannot produce the effect. A
575    /// hook that returns `Err` is logged and ignored — the write is committed,
576    /// and retries are the app's to build; a panic surfaces as Topcoat's
577    /// panic-isolated 500.
578    ///
579    /// ```ignore
580    /// async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
581    ///     let mut db = db(cx); // a fresh handle is allowed here
582    ///     for post in committed.records() { notify(post).await?; }
583    ///     Ok(())
584    /// }
585    /// ```
586    fn after_commit(
587        _cx: &Cx,
588        _committed: Committed<Self::Model>,
589    ) -> impl std::future::Future<Output = Result<()>> + Send
590    where
591        Self: Sized,
592    {
593        async move { Ok(()) }
594    }

596    /// The record's values for the detail page, keyed by the name each
597    /// [`view`](Self::view) field binds.
598    ///
599    /// The detail page reads the form's keys from
600    /// [`RecordForm::hydrate`], and this adds any
601    /// key only the view shows; the form's keys win on a collision. A
602    /// resource with no form ([`NoForm`](crate::NoForm)) supplies every key its

605    /// `cx` carries the app schema, which an embedded value's keys need

607    /// The default is empty.
608    fn view_values(_cx: &Cx, _record: &Self::Model) -> HashMap<String, String> {
609        HashMap::new()
610    }

613/// The tenant-scoped base query.
614///
615/// [`Resource::query`] with the predicate from [`Resource::tenant_scope`]
616/// ANDed onto it, so a resource that overrides `query` for soft deletes cannot
617/// drop the tenant scope by forgetting to re-state it. Every framework loader
618/// and app code start here.
619///
620/// # Errors
621///
622/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer the handler gate
623///   gives.
624/// - A gated resource that supplies no tenant predicate — no discoverable `tenant_id` UUID column,
625///   no [`Resource::tenant_scope`] override → an error naming the resource and the model.
626///   [`Panel::build`](crate::Panel::build) refuses that declaration at boot, so this is the
627///   backstop for a predicate that is `None` for the request's tenant, and for app code outside a
628///   panel. It is deliberately **not** a fallback to the unscoped query: a silent miss would be the
629///   leak [`Resource::requires_tenant`] exists to prevent.
630///
631/// App code that loads rows itself must call this — on a gated resource
632/// [`Resource::query`] is the *tenant-unscoped* base by design, so calling it
633/// directly is safe only for rows whose tenant membership is already settled (a
634/// write by id against a record the framework loaded and authorized).
635pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
636    apply_tenant_scope::<R>(cx, R::query(cx))
637}
638
639/// [`Resource::view_query`] under the same tenant gate and predicate as
640/// [`scoped_query`]: the detail page's loader, and the entry point for a page
641/// that owns its own detail view.
642///
643/// # Errors
644///
645/// The same as [`scoped_query`].
646pub fn scoped_view_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
647    apply_tenant_scope::<R>(cx, R::view_query(cx))
648}
649
650/// AND the framework's tenant predicate onto `query`.
651///
652/// The body of [`scoped_query`] and [`scoped_view_query`], split out so both
653/// seeds share one gate, one predicate, and one fail-closed error.
654/// It is crate-internal because a caller outside the crate always has a
655/// `Resource`, and so always wants [`scoped_query`] or [`scoped_view_query`].
656pub(crate) fn apply_tenant_scope<R: Resource>(
657    cx: &Cx,
658    query: Query<List<R::Model>>,
659) -> Result<Query<List<R::Model>>> {
660    if !R::requires_tenant() {
661        return Ok(query);
662    }
663    let tenant = crate::tenancy::require_tenant(cx)?;
664    let Some(filter) = R::tenant_scope(tenant) else {
665        // Fail closed and loudly: the resource declared a gate whose scope the
666        // framework cannot derive and the resource did not state, and running
667        // the query unscoped is the one outcome that declaration exists to
668        // prevent. `Panel::build` already refused the resource if *no* tenant
669        // could scope it; this is the backstop for a `tenant_scope`
670        // that answers `None` only for this tenant, and for callers outside a
671        // panel.
672        tracing::error!(
673            resource = R::slug(),
674            model = std::any::type_name::<R::Model>(),
675            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
676             UUID column on the model and no `tenant_scope` override (GH #223)"
677        );
678        return Err(TabloError::Declaration(format!(
679            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
680             `tenant_id` UUID column to derive the filter from, and the resource does not override \
681             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
682             `requires_tenant` and scope in `query`",
683            R::slug(),
684            std::any::type_name::<R::Model>(),
685        ))
686        .into());
687    };
688    Ok(query.filter(filter))
689}

691/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
692/// bridge that lets the relationship option loaders be generic over the source
693/// surface instead of over `Resource`, so `schema` does not depend on
694/// `resource`.
695///
696/// A resource answers the loaders here, so the loaders never name `Resource`.
697/// [`scoped_query`](crate::schema::OptionSource::scoped_query), the one required
698/// method, forwards to [`scoped_query`], so an option load inherits the tenant
699/// gate and the derived tenant predicate exactly as every other loader does; it
700/// is deliberately not [`Resource::query`], which on a gated resource is the
701/// *tenant-unscoped* base.
702///
703/// The policy predicates and the tenant declaration forward unchanged, and the
704/// search expression and default ordering come from the resource's declared
705/// [`table`](Resource::table), which is where "the option search searches the
706/// related resource's searchable columns" lives.
707impl<R: Resource> crate::schema::OptionSource for R {
708    type Model = R::Model;
709
710    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
711        scoped_query::<R>(cx)
712    }
713
714    fn can_view_any(cx: &Cx) -> bool {
715        <R as Resource>::can_view_any(cx)
716    }
717
718    fn can_view(cx: &Cx, record: &R::Model) -> bool {
719        <R as Resource>::can_view(cx, record)
720    }
721
722    fn requires_tenant() -> bool {
723        <R as Resource>::requires_tenant()
724    }
725
726    fn slug() -> String {
727        <R as Resource>::slug()
728    }
729
730    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
731        <R as Resource>::table(cx).search_expr(term)
732    }
733
734    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
735        <R as Resource>::table(cx).order_by(false)
736    }
737}
