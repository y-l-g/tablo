57/// Maps one Toasty `Model` to its admin UI.
58///
59/// # Contract
60///
61/// **Every item but [`Model`](Self::Model), [`Form`](Self::Form) and
62/// [`table`](Self::table) has a default**, so a resource compiles as soon as it
63/// declares its model, its record form (or [`NoForm`](crate::NoForm)) and its
64/// list view — and an omission must fail loudly rather than silently:
65///
66/// - **Checked at [`Panel::build`](crate::panel::Panel::build)**: the declared table must serve a
67///   list, and [`form`](Self::form) must agree with [`Form`](Self::Form): a record form's fields
68///   are the schema's controls, and a [`NoForm`](crate::NoForm) resource declares no schema. A
69///   resource with no form must not allow [`can_create`](Self::can_create). These are declarations,
70///   checked with a Db-only context.
71/// - **Loud at request time**: a record fn's error fails the write and rolls its transaction back,
72///   never a partial write. [`delete_record`](Self::delete_record) defaults to deleting the row
73///   through [`scoped_query`], filtered to the record's key, and refuses a table whose record key
74///   does not parse back as the primary key — naming the override that fixes it — rather than
75///   deleting the wrong row or nothing. [`bulk_delete_records`](Self::bulk_delete_records) loops
76///   `delete_record` by default, so an override covers bulk delete too.
77/// - **Chrome follows the declarations**: the row Delete control and the bulk column render when
78///   [`can_delete_any`](Self::can_delete_any) allows, the Edit link when the resource has a record
79///   form, and the View link when it declares [`view`](Self::view); the row predicates then gate
80///   each row.
81/// - **Default-deny is untouched**: every `can_*` defaults to `false`, except `can_delete`, which
82///   defaults to `can_delete_any`; an unconfigured resource exposes no data and no mutation.
83pub trait Resource: Sized + Send + Sync + 'static {
84    /// The persisted model this resource administers.
85    ///
86    /// `Send + Sync` holds for every data-only model struct and is required
87    /// for concurrent rendering of the resource's pages.
88    ///
89    /// `Clone` is part of the contract because a committed mutation names its
90    /// rows: a handler keeps a copy of the rows it loaded while the
91    /// record fn consumes them, so the hook can be handed what was written.
92    type Model: toasty::schema::Model + Send + Sync + Clone + 'static;
93
94    /// The typed value the create and edit forms parse into.
95    ///
96    /// A resource with create and edit pages names its
97    /// [`#[derive(RecordForm)]`](crate::RecordForm) struct; a list-only resource
98    /// names [`NoForm<Self::Model>`](crate::NoForm), and
99    /// [`Panel::resource`](crate::Panel::resource) registers no form route for
100    /// it ([`RecordForm::HAS_FORM`]).
101    type Form: RecordForm<Model = Self::Model>;
102
103    /// The columns an overriding [`Self::create_record`] sets itself, beyond
104    /// the form's fields.
105    ///
106    /// [`Panel::build`](crate::Panel::build) refuses a resource that allows
107    /// create when a non-nullable column is neither a form field, nor filled by
108    /// toasty (`#[auto]`, `#[default(..)]`), nor the stamped tenant column: the
109    /// create would fail at the driver on every submit. A record fn that sets
110    /// such a column by hand names it here.
111    const CREATE_COLUMNS: &'static [&'static str] = &[];
112
113    /// Whether the current user may view the list page.
114    ///
115    /// Also gates relationship option loads: a related resource
116    /// that denies this cannot offer its records as options at all.
117    fn can_view_any(_cx: &Cx) -> bool {
118        false
119    }
120
121    /// Whether the current user may view the given record.
122    ///
123    /// Checked on the edit page (GET), the edit POST (which requires both
124    /// `can_view` and `can_update`), per row in CSV export, on each
125    /// record behind a relationship choice's options, on each row
126    /// of a detail page's relation table, and on the list page as
127    /// the per-row gate of every action link (GH #235: the View link, and the
128    /// `can_view` half of Edit and Delete). Note both hooks default-deny: a
129    /// resource used as a relationship target for option loads must allow
130    /// `can_view_any` **and** `can_view` (overriding one does not imply the
131    /// other), while a relation table consults `can_view` alone. The
132    /// list page deliberately checks only
133    /// `can_view_any` for *membership*: `can_view` is an in-memory Rust
134    /// predicate that cannot run in SQL, and filtering rows after cursor
135    /// pagination would mislabel pages. Row-level visibility that must hold on
136    /// the list belongs in [`Self::query`].
137    fn can_view(_cx: &Cx, _record: &Self::Model) -> bool {
138        false
139    }
140
141    /// Whether the current user may create a new record.
142    ///
143    /// `Panel::build` calls this with a Db-only context to decide which
144    /// declaration checks apply, so a predicate that reads the request (a
145    /// tenant, a user) answers as it would for an anonymous request there. The
146    /// list page links to the create page only for a resource with a record
147    /// form ([`RecordForm::HAS_FORM`]), whatever this answers.
148    fn can_create(_cx: &Cx) -> bool {
149        false
150    }
151
152    /// Whether the current user may update the given record.
153    fn can_update(_cx: &Cx, _record: &Self::Model) -> bool {
154        false
155    }
156
157    /// Whether the current user may delete records of this resource at all.
158    ///
159    /// Decides whether the list renders the row Delete control, the bulk
160    /// column, and the bulk bar, and gates the single-delete and bulk-delete
161    /// POSTs before any record loads. The column decision takes no record, so
162    /// the streamed skeleton and the table agree on their columns. The default
163    /// [`can_delete`](Self::can_delete) also calls it once per row.
164    fn can_delete_any(_cx: &Cx) -> bool {
165        false
166    }
167
168    /// Whether the current user may delete the given record.
169    ///
170    /// Defaults to [`can_delete_any`](Self::can_delete_any); override to
171    /// refuse some rows. Checked on the single-delete and bulk-delete POSTs
172    /// after `can_delete_any` and together with `can_view` — the edit
173    /// contract: a record that cannot be viewed cannot be deleted by
174    /// UUID-guessing the route. A row it refuses renders no Delete control and
175    /// no bulk checkbox.
176    fn can_delete(cx: &Cx, _record: &Self::Model) -> bool {
177        Self::can_delete_any(cx)
178    }
179
180    /// How one record is displayed on the detail page, read-only.
181    ///
182    /// The same [`Schema`](crate::schema::Schema) vocabulary a form uses,
183    /// rendered for reading: a text field shows its stored value instead of an
184    /// `<input>`, a choice shows the option label the form offered, and a
185    /// layout block keeps the structure it declares. Like Filament's infolist,
186    /// it is its own declaration: it may show keys the form does not, and a
187    /// resource with no form declares one too. Declaring a view is what turns
188    /// the detail page on — the default declares nothing, so the route 404s
189    /// and no `View` row action renders.
190    ///
191    /// Values come from [`view_values`](Self::view_values) and the record
192    /// form's `hydrate`. A field whose key neither supplies renders
193    /// `(missing)` and fails a `debug_assert!`. A relation is not one of these
194    /// fields — it is a list of records, not a string — and renders through
195    /// [`relations`](Self::relations).
196    ///
197    /// Read-only is a promise, not a disabled form: nothing here validates or
198    /// submits, and no field renders a required marker or an error slot.
199    fn view(_cx: &Cx) -> crate::schema::Schema {
200        crate::schema::Schema::empty()
201    }
202
203    /// Free-form content on this resource's detail page, below the
204    /// [`view`](Self::view) schema and above the [`relations`](Self::relations):
205    /// anything computed from the loaded record that is not one of its fields,
206    /// such as a word count.
207    ///
208    /// Returns `None` (the default) to render nothing. The two lifetimes are
209    /// deliberately separate: the returned view may borrow the request context,
210    /// never the record — a view holding the record would pin the handler's
211    /// local binding for as long as the page, which does not compile.
212    fn view_content<'a>(_cx: &'a Cx, _record: &Self::Model) -> Option<topcoat::view::BoxView<'a>> {
213        None
214    }
215
216    /// The related resources whose rows belong to a record: each renders on
217    /// this resource's detail and edit pages as the related resource's own
218    /// list table, narrowed to the record (Filament's relation managers) —
219    /// read-only on the detail page, and on the edit page with its write
220    /// actions and a create link that opens its form with the record already
221    /// chosen.
222    ///
223    /// Takes no `Cx`, like [`navigation`](Self::navigation): the relations are
224    /// a declaration, and each related resource's policies decide per request
225    /// what its table shows. The default declares none.
226    fn relations() -> Vec<Relation<Self::Model>> {
227        Vec::new()
228    }
229
230    /// Whether this resource declares a detail page.
231    ///
232    /// Derived from [`view`](Self::view) rather than declared twice, so the
233    /// route and the row link cannot disagree with the schema that renders
234    /// them. The detail handler uses it to 404 a resource that declares
235    /// nothing, and the row chrome uses it to leave the link off.
236    fn viewed(cx: &Cx) -> bool {
237        !Self::view(cx).is_empty()
238    }

240    /// The record's label in the detail page's title, or `None` when
241    /// the record has no label to show.
242    ///
243    /// The detail page titles itself with this label when a resource returns
244    /// `Some`, and with [`label`](Self::label) plus the URL's record key when
245    /// it returns `None`, the default. The showcase's `PostResource` returns
246    /// the post's title, so its heading reads the title instead of
247    /// `Blog Post <record key>`.
248    ///
249    /// `cx` is the request's context — the same one [`view`](Self::view) and
250    /// [`view_values`](Self::view_values) receive — so a label
251    /// can read request state (a locale, a tenant). The default ignores both
252    /// arguments and returns `None`.
253    ///
254    /// A label is display text, not a key. Two records can share one (two
255    /// users named Ada), so it cannot replace the table key, whose projection
256    /// must stay injective within a page for keyed diffs and bulk selection,
257    /// or the record key, which the action routes resolve as the
258    /// model's typed PK.
259    fn record_label(_cx: &Cx, _record: &Self::Model) -> Option<String> {
260        None
261    }
262
263    /// A public URL for one record, rendered as a "View public post"-style
264    /// link on the detail and edit pages when `Some`.
265    ///
266    /// The default declares none, so no link renders. A resource whose records
267    /// have a public page overrides this with its URL — the showcase's posts
268    /// return their `/blog/{id}` page.
269    fn public_url(_cx: &Cx, _record: &Self::Model) -> Option<String> {
270        None
271    }
272
273    /// The URL slug for this resource's pages, e.g. `"users"` mounts the list
274    /// at `{panel prefix}/users`.
275    ///
276    /// Defaults to the Filament convention (`HasRoutes::resolveDefaultSlug`):
277    /// take the resource type's name, strip a trailing `Resource`, pluralize
278    /// (`UserResource` → `Users`, `CategoryResource` → `Categories`), then
279    /// kebab-case (`BlogPostResource` → `blog-posts`). Override for irregular
280    /// naming the rules cannot guess (`UsersResource` pluralizes to
281    /// `userses` — name resources singular, or override).
282    fn slug() -> String {
283        kebab_case(&pluralize(type_stem::<Self>("Resource")))
284    }
285
286    /// One record's name, e.g. `"User"`: the noun in the "Create {label}" and
287    /// "Edit {label}" titles.
288    ///
289    /// Defaults to the `Model` type name (Filament's model label). Override for
290    /// custom wording; [`navigation_label`](Self::navigation_label) pluralizes
291    /// it.
292    fn label() -> String {
293        type_short_name::<Self::Model>().to_string()
294    }
295
296    /// The sidebar label and list title, e.g. `"Users"`.
297    ///
298    /// Defaults to the pluralized [`label`](Self::label) (Filament's plural
299    /// model label): `User` → `Users`, `Category` → `Categories`, `Person` →
300    /// `People`. Override for a plural the rules cannot guess.
301    fn navigation_label() -> String {
302        pluralize(&Self::label())
303    }

305    /// Base query — the seam for a resource's **own** row scoping (ADR-0002):
306    /// soft deletes and row-level visibility. Every loader starts from it.
307    ///
308    /// **Relations are not this method's job.** The list and the export load
309    /// the relations the table's columns declare ([`TextColumn::include`]), and
310    /// the detail page loads [`Self::view_query`]. Include a relation here only
311    /// when a closure no column covers reads it on every loader's rows: one
312    /// [`can_view`](Self::can_view) reads, or one a table's `group_by` or row
313    /// key reads without a column including it.
314    ///
315    /// **Tenancy is not this method's job either.** When
316    /// [`requires_tenant`](Self::requires_tenant) is `true` the framework ANDs
317    /// the tenant filter, derived from the model's `tenant_id` column, onto
318    /// whatever this returns, at every loader through [`scoped_query`]. Do not
319    /// re-state `tenant_id().eq(tenant_id(cx))`: the copy is redundant, and one
320    /// that disagreed with the derived column would hide rows rather than widen
321    /// access. Code outside the framework's loaders starts from
322    /// [`scoped_query`].
323    ///
324    /// Returns the raw typed statement query, which composes generically —
325    /// `filter`, `order_by` and `Paginate::new` work on the raw form for any
326    /// `M: Model` — so one panel handler drives every resource's list page.
327    ///
328    /// # Keep unique constraints in step with this scope
329    ///
330    /// The app-side unique pre-check probes through the same tenant-scoped
331    /// query, so a `#[unique]` index *broader* than the scope is invisible: the
332    /// probe misses the colliding row and the user gets a 500 instead of the
333    /// inline "has already been taken". Scope the constraint to match —
334    /// `#[unique(tenant_id, email)]`. Not checkable at declaration time, since
335    /// a query's filters are not introspectable; upstream #117 is the fix.
336    fn query(_cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
337        toasty::stmt::Query::<List<Self::Model>>::all()
338    }
339
340    /// The detail page's query: [`Self::query`] plus the relations the page
341    /// reads off the loaded row — in [`view_values`](Self::view_values) or
342    /// [`view_content`](Self::view_content) — so include them here. A
343    /// [`relation`](Self::relations) table runs its own query and needs none.
344    ///
345    /// ```ignore
346    /// fn view_query(cx: &Cx) -> Query<List<Post>> {
347    ///     let author: Include<Post, Author> = Post::fields().author().into();
348    ///     Self::query(cx).include(author)
349    /// }
350    /// ```
351    ///
352    /// The default is [`Self::query`] unchanged. The framework ANDs the tenant
353    /// scope onto it, as it does onto [`Self::query`].
354    fn view_query(cx: &Cx) -> toasty::stmt::Query<List<Self::Model>> {
355        Self::query(cx)
356    }

358    /// Whether this resource requires a tenant in every handler.
359    ///
360    /// Opt-in and default-open: `false` preserves [`Self::query`] exactly as
361    /// written. `true` means two things:
362    ///
363    /// 1. **The gate.** Every handler 403s when the request carries no tenant, instead of leaking
364    ///    unscoped rows or minting nil-tenant orphans.
365    /// 2. **The scope.** Every loader ANDs `tenant_id = <request tenant>` onto the base query,
366    ///    deriving the column from the model's own schema or the resource's [`Self::tenant_scope`].
367    ///    A gated resource that declares neither a `tenant_id` UUID column nor an override is
368    ///    refused by [`Panel::build`](crate::Panel::build) at boot rather than served unscoped or
369    ///    failing per request.
370    ///
371    /// A resource that must genuinely serve more than the request tenant
372    /// declares `false` and scopes in [`Self::query`] by hand, giving up the
373    /// gate above along with the derived filter.
374    fn requires_tenant() -> bool {
375        false
376    }
377
378    /// The predicate the framework ANDs onto this resource's base query to
379    /// scope it to `tenant`, or `None` when there is nothing to AND.
380    ///
381    /// The default derives it from the model: `tenant_id = tenant`, on the
382    /// field named `tenant_id` whose type is a UUID (see [`crate::tenancy`]).
383    /// Override it when the resource's tenancy is not a column on its own
384    /// model — a row that inherits its parent's tenant states the relation
385    /// path here instead, and the framework applies it exactly as it applies
386    /// the derived one. The showcase's comments are the worked example.
387    ///
388    /// Only consulted when [`requires_tenant`](Self::requires_tenant) is
389    /// `true`. `None` from a gated resource is a **misdeclaration**, not a way
390    /// to be unscoped: [`Panel::build`](crate::Panel::build) refuses it at boot
391    /// and every loader keeps answering an error naming the resource
392    /// rather than running its query without a tenant predicate — the backstop
393    /// for a predicate that is only `None` for some tenants. There is
394    /// deliberately no override that *removes* the scope — a resource that
395    /// must serve more than one tenant declares `requires_tenant() = false`
396    /// and owns the scope in [`Self::query`], visibly, with the gate given up.
397    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
398        crate::tenancy::derived_tenant_filter::<Self::Model>(tenant)
399    }

476    /// Delete the already-authorized `record` (#86).
477    ///
478    /// The handler loads `record` through the tenancy-scoped query inside the
479    /// framework transaction and checks `can_delete` on that snapshot, then
480    /// calls this with the same transaction as `ex`. The after-commit hook
481    /// receives the snapshot once the delete commits.
482    ///
483    /// The default deletes the row through [`scoped_query`], filtered to the
484    /// record's key: the table's record key, which is the model's primary key
485    /// (see [`Table::new_split`]). Override to delete another way, such as a
486    /// soft delete.
487    fn delete_record(
488        cx: &Cx,
489        record: &Self::Model,
490        ex: &mut dyn toasty::Executor,
491    ) -> impl std::future::Future<Output = Result<()>> + Send
492    where
493        Self: Sized,
494    {
495        let key = Self::table(cx).record_key_of(record);
496        let query = scoped_query::<Self>(cx).and_then(|query| {
497            let filter = crate::schema::pk_eq_expr::<Self::Model>(&key).ok_or_else(|| {
498                // The handler found the row by this key, so a key that does
499                // not parse back is a table whose record key is not the
500                // primary key (`Table::new_split`). The detail stays in the
501                // log; the page gets the delete-failure toast.
502                tracing::error!(
503                    resource = Self::slug(),
504                    key,
505                    "default delete_record: the table's record key does not parse as the primary \
506                     key; declare the primary key as the record key or override delete_record"
507                );
508                TabloError::Declaration("delete failed".to_string())
509            })?;
510            Ok(query.filter(filter))
511        });
512        async move {
513            query?
514                .delete()
515                .exec(ex)
516                .await
517                .map_err(|error| -> topcoat::Error { error.into() })?;
518            Ok(())
519        }
520    }

522    /// Bulk-delete the already-authorized `records`: the handler
523    /// fetches through the tenancy-scoped `IN` query inside the framework
524    /// transaction and checks `can_delete` on every row before calling this.
525    /// The default deletes each record through [`Self::delete_record`] in
526    /// order, through the same `ex` — any error rolls the whole batch back,
527    /// so mid-loop failures delete zero rows. An override of `delete_record`,
528    /// such as a soft delete, therefore covers bulk delete too. Override this
529    /// for a single-statement batch.
530    fn bulk_delete_records(
531        cx: &Cx,
532        records: &[Self::Model],
533        ex: &mut dyn toasty::Executor,
534    ) -> impl std::future::Future<Output = Result<()>> + Send
535    where
536        Self: Sized,
537    {
538        async move {
539            for record in records {
540                Self::delete_record(cx, record, &mut *ex).await?;
541            }
542            Ok(())
543        }
544    }
545
546    /// Post-commit work for a mutation this resource committed.
547    ///
548    /// The place for a side effect that must not survive a rollback: an email, a
549    /// webhook, an audit row, cache invalidation. Called once per successful
550    /// write, after `tx.commit()` and before the response — running it inside a
551    /// record fn would leak the effect on rollback, and the write handlers' pool
552    /// discipline forbids a second handle while the transaction is open.
553    ///
554    /// [`Committed`] names the mutation and the rows it wrote: the committed row
555    /// a create or update returned, the rows a delete or bulk delete removed
556    /// (one call with every row). It is never called when nothing committed: a
557    /// validation error, a policy denial, a failed record fn, or a failed commit
558    /// all leave the hook untouched, so a rollback cannot produce the effect. A
559    /// hook that returns `Err` is logged and ignored — the write is committed,
560    /// and retries are the app's to build; a panic surfaces as Topcoat's
561    /// panic-isolated 500.
562    ///
563    /// ```ignore
564    /// async fn after_commit(cx: &Cx, committed: Committed<Post>) -> Result<()> {
565    ///     let mut db = db(cx); // a fresh handle is allowed here
566    ///     for post in committed.records() { notify(post).await?; }
567    ///     Ok(())
568    /// }
569    /// ```
570    fn after_commit(
571        _cx: &Cx,
572        _committed: Committed<Self::Model>,
573    ) -> impl std::future::Future<Output = Result<()>> + Send
574    where
575        Self: Sized,
576    {
577        async move { Ok(()) }
578    }
579
580    /// The record's values for the detail page, keyed by the name each
581    /// [`view`](Self::view) field binds.
582    ///
583    /// The detail page reads the form's keys from
584    /// [`RecordForm::hydrate`], and this adds any
585    /// key only the view shows; the form's keys win on a collision. A
586    /// resource with no form ([`NoForm`](crate::NoForm)) supplies every key its
587    /// view shows here.
588    ///
589    /// `cx` carries the app schema, which an embedded value's keys need
590    /// ([`EmbeddedForm::write_form`](crate::schema::EmbeddedForm::write_form)).
591    /// The default is empty.
592    fn view_values(_cx: &Cx, _record: &Self::Model) -> HashMap<String, String> {
593        HashMap::new()
594    }
595}

597/// The tenant-scoped base query.
598///
599/// [`Resource::query`] with the predicate from [`Resource::tenant_scope`]
600/// ANDed onto it, so a resource that overrides `query` for soft deletes cannot
601/// drop the tenant scope by forgetting to re-state it. Every framework loader
602/// and app code start here.
603///
604/// # Errors
605///
606/// - A gated resource and no tenant in `cx` → 403, the same fail-closed answer the handler gate
607///   gives.
608/// - A gated resource that supplies no tenant predicate — no discoverable `tenant_id` UUID column,
609///   no [`Resource::tenant_scope`] override → an error naming the resource and the model.
610///   [`Panel::build`](crate::Panel::build) refuses that declaration at boot, so this is the
611///   backstop for a predicate that is `None` for the request's tenant, and for app code outside a
612///   panel. It is deliberately **not** a fallback to the unscoped query: a silent miss would be the
613///   leak [`Resource::requires_tenant`] exists to prevent.
614///
615/// App code that loads rows itself must call this — on a gated resource
616/// [`Resource::query`] is the *tenant-unscoped* base by design, so calling it
617/// directly is safe only for rows whose tenant membership is already settled (a
618/// write by id against a record the framework loaded and authorized).
619pub fn scoped_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
620    apply_tenant_scope::<R>(cx, R::query(cx))
621}
622
623/// [`Resource::view_query`] under the same tenant gate and predicate as
624/// [`scoped_query`]: the detail page's loader, and the entry point for a page
625/// that owns its own detail view.
626///
627/// # Errors
628///
629/// The same as [`scoped_query`].
630pub fn scoped_view_query<R: Resource>(cx: &Cx) -> Result<Query<List<R::Model>>> {
631    apply_tenant_scope::<R>(cx, R::view_query(cx))
632}
633
634/// AND the framework's tenant predicate onto `query`.
635///
636/// The body of [`scoped_query`] and [`scoped_view_query`], split out so both
637/// seeds share one gate, one predicate, and one fail-closed error.
638/// It is crate-internal because a caller outside the crate always has a
639/// `Resource`, and so always wants [`scoped_query`] or [`scoped_view_query`].
640pub(crate) fn apply_tenant_scope<R: Resource>(
641    cx: &Cx,
642    query: Query<List<R::Model>>,
643) -> Result<Query<List<R::Model>>> {
644    if !R::requires_tenant() {
645        return Ok(query);
646    }
647    let tenant = crate::tenancy::require_tenant(cx)?;
648    let Some(filter) = R::tenant_scope(tenant) else {
649        // Fail closed and loudly: the resource declared a gate whose scope the
650        // framework cannot derive and the resource did not state, and running
651        // the query unscoped is the one outcome that declaration exists to
652        // prevent. `Panel::build` already refused the resource if *no* tenant
653        // could scope it; this is the backstop for a `tenant_scope`
654        // that answers `None` only for this tenant, and for callers outside a
655        // panel.
656        tracing::error!(
657            resource = R::slug(),
658            model = std::any::type_name::<R::Model>(),
659            "requires_tenant is true but the resource supplies no tenant predicate: no `tenant_id` \
660             UUID column on the model and no `tenant_scope` override (GH #223)"
661        );
662        return Err(TabloError::Declaration(format!(
663            "resource '{}' requires a tenant, but the framework cannot scope it: {} declares no \
664             `tenant_id` UUID column to derive the filter from, and the resource does not override \
665             `tenant_scope` (GH #223); declare the column, override `tenant_scope`, or drop \
666             `requires_tenant` and scope in `query`",
667            R::slug(),
668            std::any::type_name::<R::Model>(),
669        ))
670        .into());
671    };
672    Ok(query.filter(filter))
673}

675/// Every `Resource` is an [`OptionSource`](crate::schema::OptionSource) — the
676/// bridge that lets the relationship option loaders be generic over the source
677/// surface instead of over `Resource`, so `schema` does not depend on
678/// `resource`.
679///
680/// A resource answers the loaders here, so the loaders never name `Resource`.
681/// [`scoped_query`](crate::schema::OptionSource::scoped_query), the one required
682/// method, forwards to [`scoped_query`], so an option load inherits the tenant
683/// gate and the derived tenant predicate exactly as every other loader does; it
684/// is deliberately not [`Resource::query`], which on a gated resource is the
685/// *tenant-unscoped* base.
686///
687/// The policy predicates and the tenant declaration forward unchanged, and the
688/// search expression and default ordering come from the resource's declared
689/// [`table`](Resource::table), which is where "the option search searches the
690/// related resource's searchable columns" lives.
691impl<R: Resource> crate::schema::OptionSource for R {
692    type Model = R::Model;
693
694    fn scoped_query(cx: &Cx) -> Result<Query<List<R::Model>>> {
695        scoped_query::<R>(cx)
696    }
697
698    fn can_view_any(cx: &Cx) -> bool {
699        <R as Resource>::can_view_any(cx)
700    }
701
702    fn can_view(cx: &Cx, record: &R::Model) -> bool {
703        <R as Resource>::can_view(cx, record)
704    }
705
706    fn requires_tenant() -> bool {
707        <R as Resource>::requires_tenant()
708    }
709
710    fn slug() -> String {
711        <R as Resource>::slug()
712    }
713
714    fn search_expr(cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
715        <R as Resource>::table(cx).search_expr(term)
716    }
717
718    fn order_by(cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
719        <R as Resource>::table(cx).order_by(false)
720    }
721}
