9/// The share of the table a [`ColumnWidth::Narrow`] column claims, in whole
10/// percent.
11pub(crate) const NARROW_DEFAULT_PERCENT: u8 = 10;
12
13/// The width a [`TextColumn`] claims in the table's fixed layout.
14///
15/// Widths are **shares of the table**, so what a table declares is a fraction
16/// of its container rather than a length that can outgrow it: the columns that
17/// declare none take what the declared ones leave. A length
18/// ([`Rem`](Self::Rem)) is the exception — lengths do not shrink with the
19/// table, and a table whose lengths exceed its width gives the columns that
20/// declare none no space at all, header text included.
21///
22/// The renderer writes the width into the column's `th` and every row's `td`
23/// as an inline `style` attribute — data, never a generated Tailwind class.
24/// Tailwind generates only the class literals it finds in source, so a width
25/// assembled at render (`w-[{n}%]`) would emit no CSS at all (ADR-0006); a
26/// declared width is read by the layout directly.
27///
28/// A column's **kind** picks the default: `TextColumn::r#for` binds a `String`
29/// field, so its cells hold the row's own text — a title, a name, a body — and
30/// it defaults to [`Wide`](Self::Wide), taking a share of what the declared
31/// columns leave; [`TextColumn::computed`] derives its cell (a status, a
32/// boolean, a date, a count) and defaults to [`Narrow`](Self::Narrow), a share
33/// of the table. [`TextColumn::width`] overrides either, which is the seam for
34/// a column whose content disagrees with its kind.
35#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
36pub enum ColumnWidth {
37    /// Take a share of whatever the declared columns leave: the column
38    /// declares no width, and `table-fixed` splits the remainder between the
39    /// wide columns instead of measuring the rows currently rendered.
40    #[default]
41    Wide,
42    /// A share of the table for a status, boolean, date or count cell: the
43    /// default for a [`TextColumn::computed`] column. The renderer resolves
44    /// the share (10% nominally) against the table's other kind defaults.
45    Narrow,
46    /// An explicit length in whole rem: `Rem(14)` declares `14rem`. A length
47    /// does not shrink with the table, so a table narrower than the lengths it
48    /// declares gives the columns that declare none no space at all.
49    Rem(u8),
50    /// An explicit share of the table in whole percent: `Percent(30)`
51    /// declares `30%`.
52    Percent(u8),
53}
54
55impl ColumnWidth {
56    /// The share of the table this column claims as a **kind default**, in
57    /// whole percent, or `None` for a column that declares an explicit width
58    /// or none at all.
59    ///
60    /// A nominal: the renderer scales the kind defaults down together when
61    /// their total would leave the wide columns less than their share of the
62    /// table.
63    pub(crate) fn default_percent(self) -> Option<u8> {
64        match self {
65            Self::Narrow => Some(NARROW_DEFAULT_PERCENT),
66            Self::Wide | Self::Rem(_) | Self::Percent(_) => None,
67        }
68    }
69
70    /// The `style` attribute value an **explicit** declaration emits, or
71    /// `None` for [`Wide`](Self::Wide), which declares nothing, and for
72    /// [`Narrow`](Self::Narrow), whose share the renderer resolves against the
73    /// rest of the table.
74    pub(crate) fn explicit_css(self) -> Option<Cow<'static, str>> {
75        match self {
76            Self::Rem(rem) => Some(Cow::Owned(format!("width: {rem}rem"))),
77            Self::Percent(percent) => Some(Cow::Owned(format!("width: {percent}%"))),
78            Self::Wide | Self::Narrow => None,
79        }
80    }
81}

83/// Text column bound to a typed lens **and** a typed projection.
84///
85/// The lens (`FieldLens<M, String>`) is the query side: it names the column
86/// and produces search/sort predicates — `TextColumn::for(User::fields().name(), ..)`
87/// fails to compile if the field does not exist (ADR-0001).
88///
89/// The projection closure is the render side: it reads the value off a model
90/// instance for the cell (`|u| u.name.clone()`). Toasty models are plain
91/// structs and expose no instance→field reflection, so the closure is the
92/// only way to read a field generically (upstream gap #119: instance →
93/// field-value extraction). A typo in the closure body fails at compile
94/// time — there is no string dispatch and no panic at render.
95///
96/// A projection that reads a **relation** declares it with [`Self::include`],
97/// because the closure is opaque to the framework: the list and the export
98/// load every relation the table's columns declared, and nothing else.
99#[derive(Clone)]
100pub struct TextColumn<M> {
101    /// The query-side lens; `None` for [`Self::computed`] columns, which
102    /// render a value but declare no predicates.
103    path: Option<FieldLens<M, String>>,
104    name: String,
105    label: String,
106    project: Arc<dyn Fn(&M) -> String + Send + Sync>,
107    searchable: bool,
108    sortable: bool,
109    /// The width this column claims in the table's fixed layout.
110    width: ColumnWidth,
111    /// Relations this column's projection reads, as includes on the model.
112    includes: Vec<toasty_core::stmt::Include>,
113}
114
115/// The escape character the search pattern declares to `LIKE`:
116/// backslash, escaped in the pattern by [`escape_like_pattern`].
117pub(crate) const LIKE_ESCAPE: char = '\\';
118
119/// Wrap `term` as a `LIKE` pattern matching it anywhere in the column, with
120/// `%`, `_` and the escape character itself escaped so the term stays literal.
121///
122/// Toasty ships the SQL half (`like_with_escape`) but not this one: escaping is
123/// app-side because only the app knows whether it is building a literal or a
124/// pattern.
125pub(crate) fn escape_like_pattern(term: &str) -> String {
126    let mut pattern = String::with_capacity(term.len() + 2);
127    pattern.push('%');
128    for c in term.chars() {
129        if c == LIKE_ESCAPE || c == '%' || c == '_' {
130            pattern.push(LIKE_ESCAPE);
131        }
132        pattern.push(c);
133    }
134    pattern.push('%');
135    pattern
136}

138impl<M> TextColumn<M>
139where
140    M: toasty::schema::Model,
141{
142    /// Bind a column to a `String` field lens plus a projection closure:
143    /// `TextColumn::for(User::fields().name(), |u| u.name.clone())`.
144    ///
145    /// The closure receives each rendered row and returns the cell text, so
146    /// computed cells (`|u| u.active.then(|| "Active".into()).unwrap_or_default()`)
147    /// are as natural as plain field reads.
148    pub fn r#for(
149        path: FieldLens<M, String>,
150        project: impl Fn(&M) -> String + Send + Sync + 'static,
151    ) -> Self {
152        let field = lens_field(path.clone(), &M::schema());
153        Self {
154            path: Some(path),
155            name: field.name.app_unwrap().to_string(),
156            label: lens_label(&field),
157            project: Arc::new(project),
158            searchable: false,
159            sortable: false,
160            width: ColumnWidth::Wide,
161            includes: Vec::new(),
162        }
163    }
164
165    /// A computed, display-only column (CONTEXT.md Column: "a computed value").
166    ///
167    /// No field lens — so it cannot be searchable or sortable (it maps to no
168    /// query predicate) — but any cell projection compiles: booleans,
169    /// timestamps, joined values. Calling `.searchable()` / `.sortable()` on
170    /// a computed column panics: a lying sort link / search promise
171    /// is worse than a loud build error.
172    pub fn computed(
173        label: impl Into<String>,
174        project: impl Fn(&M) -> String + Send + Sync + 'static,
175    ) -> Self {
176        let label = label.into();
177        let name = label.to_lowercase();
178        Self {
179            path: None,
180            name,
181            label,
182            project: Arc::new(project),
183            searchable: false,
184            sortable: false,
185            width: ColumnWidth::Narrow,
186            includes: Vec::new(),
187        }
188    }
189
190    /// Declare a relation this column's projection reads:
191    /// `.include(Post::fields().author())` for `|p| p.author.get().name.clone()`.
192    /// The path is typed on the table's model, so a relation the model does not
193    /// have is a compile error.
194    ///
195    /// The list and the export load every relation the table's columns declared,
196    /// so **declare every relation the closure reads**. One it does not declare
197    /// arrives unloaded, and the closure's `is_unloaded` guard (the
198    /// unloaded-relation contract of ADR-0011, `"(unloaded)"` plus a
199    /// `debug_assert!`) turns that into a loud failure instead of a silent
200    /// `"-"`.
201    ///
202    /// Repeat calls accumulate:
203    /// `.include(Post::fields().author()).include(Post::fields().comments())`.
204    ///
205    /// One query loads a relation once for the whole table: two includes of the
206    /// same relation merge, and an unfiltered one wins over a filtered one
207    /// (Toasty ORs their filters). A column that counts a filtered subset should
208    /// filter in its closure rather than rely on a filtered include.
209    pub fn include<T>(mut self, relation: impl Into<toasty::stmt::Include<M, T>>) -> Self {
210        let include: toasty_core::stmt::Include = relation.into().into();
211        if !self.includes.contains(&include) {
212            self.includes.push(include);
213        }
214        self
215    }
216
217    /// The relations this column declared, in declaration order.
218    pub(crate) fn includes(&self) -> &[toasty_core::stmt::Include] {
219        &self.includes
220    }
221
222    pub fn searchable(mut self) -> Self {
223        assert!(
224            self.path.is_some(),
225            "searchable() on computed column '{}': computed columns map to no query predicate",
226            self.label
227        );
228        self.searchable = true;
229        self
230    }
231
232    pub fn sortable(mut self) -> Self {
233        assert!(
234            self.path.is_some(),
235            "sortable() on computed column '{}': computed columns map to no query predicate",
236            self.label
237        );
238        self.sortable = true;
239        self
240    }

242    pub fn is_searchable(&self) -> bool {
243        self.searchable
244    }
245
246    pub fn is_sortable(&self) -> bool {
247        self.sortable
248    }

250    /// Declare this column's width in the table's fixed layout:
251    /// `.width(ColumnWidth::Percent(20))` for a column that knows its own
252    /// measure.
253    ///
254    /// The default follows the column's kind — see [`ColumnWidth`]. Override
255    /// it when the content disagrees with the kind: a `computed` column that
256    /// holds a name or a title is [`Wide`](ColumnWidth::Wide), a `String` field
257    /// that holds a status is [`Narrow`](ColumnWidth::Narrow).
258    pub fn width(mut self, width: ColumnWidth) -> Self {
259        self.width = width;
260        self
261    }

263    /// The width this column declares, which the renderer emits on its `th`
264    /// and on every `td` of its column.
265    pub fn column_width(&self) -> ColumnWidth {
266        self.width
267    }

269    pub fn label(&self) -> &str {
270        &self.label
271    }

273    /// App-level field name (from the lens). Identifies the column in the
274    /// `?sort=` URL parameter.

276        &self.name
277    }

279    /// Render the cell for one row via the typed projection.
280    pub fn render_cell(&self, row: &M) -> String {
281        (self.project)(row)
282    }
283
284    /// The search predicate for this column: a portable, escaped
285    /// **substring** match.
286    ///
287    /// `like_with_escape` keeps the pattern parameterised and lowers to the
288    /// same `LIKE … ESCAPE '\\'` on every driver, and
289    /// `escape_like_pattern` makes the term literal — a `%` or `_` the user
290    /// typed matches that character, it does not act as a wildcard. Note the
291    /// driver difference `LIKE` brings: SQLite compares ASCII
292    /// case-insensitively, PostgreSQL case-sensitively.
293    pub fn to_search_expr(&self, term: &str) -> Option<Expr<bool>> {
294        let t = term.trim();
295        if !self.searchable || t.is_empty() {
296            return None;
297        }
298        Some(
299            self.path
300                .clone()?
301                .like_with_escape(escape_like_pattern(t), LIKE_ESCAPE),
302        )
303    }
304
305    pub fn to_order_by(&self, descending: bool) -> Option<OrderByExpr> {
306        if self.sortable {
307            // Cursor determinism is the engine's job: toasty's
308            // `normalize_cursor_order` appends the physical PK columns to
309            // ambiguous cursor orderings internally.
310            let path = self.path.clone()?;
311            Some(if descending { path.desc() } else { path.asc() })
312        } else {
313            None
314        }
315    }
316}

331/// Convert a single column or tuple of columns into `Vec<TextColumn<M>>`.
332///
333/// Tuple members are `TextColumn<M>` themselves, so nothing sits between the
334/// column types. A single column converts on its own, with no
335/// one-element tuple.
336///
337/// Tuple arities stop at eight, the ceiling every tuple-collection trait
338/// shares: `IntoFilters` in `resource/filter.rs`, `IntoSchema` in
339/// `schema/tree.rs`, and `IntoRelationColumns` in `resource/relation.rs`.
340/// Without variadic generics the idiom is one `macro_rules!` invocation per
341/// arity, and eight covers the widest tuple a Resource declares. Extend every
342/// list together when a real Resource needs more.
343pub trait IntoColumns<M> {
344    fn into_columns(self) -> Vec<TextColumn<M>>;
345}

347impl<M> IntoColumns<M> for TextColumn<M> {
348    fn into_columns(self) -> Vec<TextColumn<M>> {
349        vec![self]
350    }
351}

353/// Generate the tuple impls of a column-list trait, arities two to eight:
354/// `$trait::$method` collects a tuple of `$col<T>` into a `Vec` and hands it to
355/// `$wrap`. [`IntoColumns`] and
356/// [`IntoRelationColumns`](super::IntoRelationColumns) share it, so the two
357/// column lists accept the same tuple shapes. Every element is one column type,
358/// so `(a, (b, c))` is not a column list: a table's columns sit in one flat
359/// tuple.
360macro_rules! column_tuples {
361    ($trait:ident, $method:ident, $col:ident, $out:ty, $wrap:expr) => {
362        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b);
363        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c);
364        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c, d);
365        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c, d, e);
366        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c, d, e, f);
367        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c, d, e, f, g);
368        column_tuples!(@arity $trait, $method, $col, $out, $wrap; a, b, c, d, e, f, g, h);
369    };
370    (@arity $trait:ident, $method:ident, $col:ident, $out:ty, $wrap:expr; $($v:ident),+) => {
371        impl<T> $trait<T> for ($(column_tuples!(@element $col $v)),+) {
372            fn $method(self) -> $out {
373                let ($($v,)+) = self;
374                ($wrap)(vec![$($v,)+])
375            }
376        }
377    };
378    (@element $col:ident $v:ident) => { $col<T> };
379}
380pub(crate) use column_tuples;

387    |columns| columns

390/// The table-level floor a wide column contributes to the table's
391/// `min-width`, in whole rem.
392///
393/// A wide column declares no width, so a sum of declared widths alone would
394/// let it crush to zero on a narrow viewport. Six rem keeps body text readable
395/// and, summed across the wide columns, trips the wrapper's horizontal scroll
396/// before the fixed layout crushes them.
397pub(crate) const WIDE_COLUMN_MIN_REM: u8 = 6;
398
399/// The most of the table the kind defaults claim together.
400///
401/// The defaults are shares of the table, and the columns that declare none
402/// take what they leave: a total over 100% gives those columns no space at
403/// all, and `table-fixed` renders a column with no space at zero width, header
404/// text included. The budget keeps the rest of the table for them whatever the
405/// column set.
406pub(crate) const DEFAULT_WIDTH_BUDGET_PERCENT: u8 = 60;
407
408/// The share a kind default claims, scaled down when the table's defaults
409/// together (`total`) exceed [`DEFAULT_WIDTH_BUDGET_PERCENT`].
410pub(crate) fn scaled_default_percent(nominal: u8, total: u32) -> u8 {
411    if total <= u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) {
412        return nominal;
413    }
414    let scaled = u32::from(nominal) * u32::from(DEFAULT_WIDTH_BUDGET_PERCENT) / total;
415    // `scaled` is at most the budget, so the conversion cannot fail.
416    u8::try_from(scaled).unwrap_or(DEFAULT_WIDTH_BUDGET_PERCENT)
417}
418
419/// The `style` value a kind default emits.
420pub(crate) fn default_width_style(percent: u8) -> Cow<'static, str> {
421    Cow::Owned(format!("width: {percent}%"))
422}
423
424/// The `style` a data column's cells carry: an explicit `Rem`/`Percent`
425/// verbatim, a kind default scaled against the table's defaults (`total`), and
426/// nothing for a wide column, which takes a share of what the declared ones
427/// leave.
428pub(crate) fn column_width_style(width: ColumnWidth, total: u32) -> Option<Cow<'static, str>> {
429    width.explicit_css().or_else(|| {
430        width
431            .default_percent()
432            .map(|nominal| default_width_style(scaled_default_percent(nominal, total)))
433    })
434}

436/// The terms of a fixed-layout table's `min-width`: every share as emitted and
437/// the lengths as one rem total.
438///
439/// With `w-full` the table never exceeds its container on its own, so without
440/// the floor the wrapper's `overflow-x-auto` never scrolls; with it the table
441/// keeps its measure on a narrow viewport and the wrapper scrolls.
442#[derive(Default)]
443pub(crate) struct MinWidth {
444    percent: Vec<u8>,
445    rem: u32,
446}
447
448impl MinWidth {
449    /// A share of the table, as the column emits it.
450    pub(crate) fn share(&mut self, percent: u8) {
451        self.percent.push(percent);
452    }
453
454    /// A length, in whole rem.
455    pub(crate) fn rem(&mut self, rem: u8) {
456        self.rem += u32::from(rem);
457    }
458
459    /// A data column's term: its scaled share or its length, and
460    /// [`WIDE_COLUMN_MIN_REM`] for a wide column, which declares nothing.
461    pub(crate) fn column(&mut self, width: ColumnWidth, total: u32) {
462        match width {
463            ColumnWidth::Wide => self.rem(WIDE_COLUMN_MIN_REM),
464            ColumnWidth::Narrow => {
465                self.share(scaled_default_percent(NARROW_DEFAULT_PERCENT, total))
466            }
467            ColumnWidth::Rem(rem) => self.rem(rem),
468            ColumnWidth::Percent(share) => self.share(share),
469        }
470    }
471
472    /// The `min-width` style, emitted only when the sum carries a length:
473    /// shares alone are a fraction of the container and can never overflow it.
474    pub(crate) fn style(&self) -> Option<Cow<'static, str>> {
475        (self.rem > 0).then(|| {
476            let mut parts: Vec<String> = self
477                .percent
478                .iter()
479                .map(|share| format!("{share}%"))
480                .collect();
481            parts.push(format!("{}rem", self.rem));
482            if parts.len() == 1 {
483                Cow::Owned(format!("min-width: {}", parts[0]))
484            } else {
485                Cow::Owned(format!("min-width: calc({})", parts.join(" + ")))
486            }
487        })
488    }
489}
