11/// One table column: what its header says, what each row's cell shows, and
12/// which query predicates it contributes.
13///
14/// The built-in [`TextColumn`] and [`BooleanColumn`] implement this trait
15/// and nothing more, so an app column has the same reach: implement it, and
16/// pass the value to [`Table::new`](super::Table::new) in a tuple or to
17/// [`Table::column`](super::Table::column).
18///
19/// ```ignore
20/// struct Initials;
21///
22/// impl Column<User> for Initials {
23///     fn name(&self) -> &str { "initials" }
24///     fn label(&self) -> &str { "Initials" }
25///     fn text(&self, row: &User) -> String {
26///         row.name.split_whitespace().filter_map(|w| w.chars().next()).collect()
27///     }
28/// }
29/// ```
30///
31/// Only [`name`](Self::name), [`label`](Self::label) and
32/// [`text`](Self::text) are required. The cell defaults to the text, and a
33/// column contributes no search, sort or relation until it says so.
34pub trait Column<M>: Send + Sync {
35    /// The column's identifier, distinct within its table: the `?sort=`
36    /// value that names it.
37    fn name(&self) -> &str;
38
39    /// The header text, which the CSV export also writes.
40    fn label(&self) -> &str;
41
42    /// The row's value as plain text: the CSV export's cell, and the table
43    /// cell unless [`cell`](Self::cell) renders something else.
44    fn text(&self, row: &M) -> String;
45
46    /// The row's table cell. Defaults to [`text`](Self::text).
47    ///
48    /// The cell sits in a `td` that truncates, so a view wider than its
49    /// column clips to an ellipsis.
50    fn cell<'a>(&self, cx: &'a Cx, row: &M) -> BoxView<'a> {
51        let text = self.text(row);
52        view! { cx => (text) }.boxed()
53    }
54
55    /// The width the column claims in the table's fixed layout. Defaults to
56    /// [`ColumnWidth::Narrow`].
57    fn column_width(&self) -> ColumnWidth {
58        ColumnWidth::Narrow
59    }
60
61    /// Whether the column joins the table's search: the search toolbar
62    /// renders when any column does. Defaults to `false`.
63    fn is_searchable(&self) -> bool {
64        false
65    }
66
67    /// The predicate a search for `term` adds, OR-ed with the other
68    /// searchable columns'. `term` is trimmed and not empty. Defaults to
69    /// none.
70    fn search_expr(&self, _term: &str) -> Option<Expr<bool>> {
71        None
72    }
73
74    /// Whether the header links to a sort on this column. Defaults to
75    /// `false`.
76    fn is_sortable(&self) -> bool {
77        false
78    }
79
80    /// The ordering a sort on this column applies. Defaults to none.
81    fn order_by(&self, _descending: bool) -> Option<OrderByExpr> {
82        None
83    }
84
85    /// The relations [`text`](Self::text) and [`cell`](Self::cell) read,
86    /// which the list and the export load. Defaults to none.
87    fn includes(&self) -> Includes<M> {
88        Includes::new()
89    }
90
91    /// What is wrong with this column's declaration, which
92    /// [`Panel::build`](crate::Panel::build) reports. The built-in columns
93    /// record a lens that binds no single field here, and a search or sort
94    /// asked of a computed column.
95    #[doc(hidden)]
96    fn misdeclared(&self) -> Option<String> {
97        None
98    }
99}

129impl<M> Includes<M> {
130    /// No relation.
131    pub fn new() -> Self {
132        Self::default()
133    }
134
135    /// Add `relation`: `Includes::new().with(Post::fields().author())`. A
136    /// relation already present is not added twice.
137    pub fn with<T>(mut self, relation: impl Into<toasty::stmt::Include<M, T>>) -> Self {
138        let include: toasty_core::stmt::Include = relation.into().into();
139        if !self.0.contains(&include) {
140            self.0.push(include);
141        }
142        self
143    }
144
145    /// The relations, in the order they were added.
146    pub(crate) fn into_vec(self) -> Vec<toasty_core::stmt::Include> {
147        self.0
148    }
149
150    /// How many relations there are.
151    pub fn len(&self) -> usize {
152        self.0.len()
153    }
154
155    /// Whether there are none.
156    pub fn is_empty(&self) -> bool {
157        self.0.is_empty()
158    }
159}

161/// The share of the table a [`ColumnWidth::Narrow`] column claims, in whole
162/// percent.
163pub(crate) const NARROW_DEFAULT_PERCENT: u8 = 10;
164
165/// The width a [`Column`] claims in the table's fixed layout.
166///
167/// Widths are **shares of the table**, so what a table declares is a fraction
168/// of its container rather than a length that can outgrow it: the columns that
169/// declare none take what the declared ones leave. A length
170/// ([`Rem`](Self::Rem)) is the exception — lengths do not shrink with the
171/// table, and a table whose lengths exceed its width gives the columns that
172/// declare none no space at all, header text included.
173///
174/// The renderer writes the width into the column's `th` and every row's `td`
175/// as an inline `style` attribute — data, never a generated Tailwind class.
176/// Tailwind generates only the class literals it finds in source, so a width
177/// assembled at render (`w-[{n}%]`) would emit no CSS at all (ADR-0006); a
178/// declared width is read by the layout directly.
179///
180/// A column's **kind** picks the default: `TextColumn::r#for` binds a `String`
181/// field, so its cells hold the row's own text — a title, a name, a body — and
182/// it defaults to [`Wide`](Self::Wide), taking a share of what the declared
183/// columns leave; [`TextColumn::computed`] derives its cell (a status, a
184/// boolean, a date, a count) and defaults to [`Narrow`](Self::Narrow), a share
185/// of the table, as does a [`BooleanColumn`] and any [`Column`] that does not
186/// override [`Column::column_width`]. [`TextColumn::width`] overrides either,
187/// which is the seam for a column whose content disagrees with its kind.
188#[derive(Clone, Copy, Debug, Default, PartialEq, Eq)]
189pub enum ColumnWidth {
190    /// Take a share of whatever the declared columns leave: the column
191    /// declares no width, and `table-fixed` splits the remainder between the
192    /// wide columns instead of measuring the rows currently rendered.
193    #[default]
194    Wide,
195    /// A share of the table for a status, boolean, date or count cell: the
196    /// default for a [`TextColumn::computed`] column. The renderer resolves
197    /// the share (10% nominally) against the table's other kind defaults.
198    Narrow,
199    /// An explicit length in whole rem: `Rem(14)` declares `14rem`. A length
200    /// does not shrink with the table, so a table narrower than the lengths it
201    /// declares gives the columns that declare none no space at all.
202    Rem(u8),
203    /// An explicit share of the table in whole percent: `Percent(30)`
204    /// declares `30%`.
205    Percent(u8),
206}
207
208impl ColumnWidth {
209    /// The share of the table this column claims as a **kind default**, in
210    /// whole percent, or `None` for a column that declares an explicit width
211    /// or none at all.
212    ///
213    /// A nominal: the renderer scales the kind defaults down together when
214    /// their total would leave the wide columns less than their share of the
215    /// table.
216    pub(crate) fn default_percent(self) -> Option<u8> {
217        match self {
218            Self::Narrow => Some(NARROW_DEFAULT_PERCENT),
219            Self::Wide | Self::Rem(_) | Self::Percent(_) => None,
220        }
221    }
222
223    /// The `style` attribute value an **explicit** declaration emits, or
224    /// `None` for [`Wide`](Self::Wide), which declares nothing, and for
225    /// [`Narrow`](Self::Narrow), whose share the renderer resolves against the
226    /// rest of the table.
227    pub(crate) fn explicit_css(self) -> Option<Cow<'static, str>> {
228        match self {
229            Self::Rem(rem) => Some(Cow::Owned(format!("width: {rem}rem"))),
230            Self::Percent(percent) => Some(Cow::Owned(format!("width: {percent}%"))),
231            Self::Wide | Self::Narrow => None,
232        }
233    }
234}

236/// Text column bound to a typed lens **and** a typed projection.
237///
238/// The lens (`FieldLens<M, String>`) is the query side: it names the column
239/// and produces search/sort predicates — `TextColumn::for(User::fields().name(), ..)`
240/// fails to compile if the field does not exist (ADR-0001).
241///
242/// The projection closure is the render side: it reads the value off a model
243/// instance for the cell (`|u| u.name.clone()`). Toasty models are plain
244/// structs and expose no instance→field reflection, so the closure is the
245/// only way to read a field generically (upstream gap #119: instance →
246/// field-value extraction). A typo in the closure body fails at compile
247/// time — there is no string dispatch and no panic at render.
248///
249/// A projection that reads a **relation** declares it with [`Self::include`],
250/// because the closure is opaque to the framework: the list and the export
251/// load every relation the table's columns declared, and nothing else.
252#[derive(Clone)]
253pub struct TextColumn<M> {
254    /// The query-side lens; `None` for [`Self::computed`] columns, which
255    /// render a value but declare no predicates.
256    path: Option<FieldLens<M, String>>,
257    name: String,
258    label: String,
259    project: Arc<dyn Fn(&M) -> String + Send + Sync>,
260    searchable: bool,
261    sortable: bool,
262    /// The width this column claims in the table's fixed layout.
263    width: ColumnWidth,
264    /// Relations this column's projection reads.
265    includes: Includes<M>,
266    /// What is wrong with the declaration ([`Column::misdeclared`]).
267    misdeclared: Option<String>,
268}

270/// The escape character the search pattern declares to `LIKE`:
271/// backslash, escaped in the pattern by [`escape_like_pattern`].
272pub(crate) const LIKE_ESCAPE: char = '\\';
273
274/// Wrap `term` as a `LIKE` pattern matching it anywhere in the column, with
275/// `%`, `_` and the escape character itself escaped so the term stays literal.
276///
277/// Toasty ships the SQL half (`like_with_escape`) but not this one: escaping is
278/// app-side because only the app knows whether it is building a literal or a
279/// pattern.
280pub(crate) fn escape_like_pattern(term: &str) -> String {
281    let mut pattern = String::with_capacity(term.len() + 2);
282    pattern.push('%');
283    for c in term.chars() {
284        if c == LIKE_ESCAPE || c == '%' || c == '_' {
285            pattern.push(LIKE_ESCAPE);
286        }
287        pattern.push(c);
288    }
289    pattern.push('%');
290    pattern
291}

293impl<M> TextColumn<M>
294where
295    M: toasty::schema::Model,
296{
297    /// Bind a column to a `String` field lens plus a projection closure:
298    /// `TextColumn::for(User::fields().name(), |u| u.name.clone())`.
299    ///
300    /// The closure receives each rendered row and returns the cell text, so
301    /// computed cells (`|u| u.active.then(|| "Active".into()).unwrap_or_default()`)
302    /// are as natural as plain field reads.
303    pub fn r#for(
304        path: FieldLens<M, String>,
305        project: impl Fn(&M) -> String + Send + Sync + 'static,
306    ) -> Self {
307        let binding = LensBinding::of(path.clone());
308        Self {
309            path: Some(path),
310            name: binding.name,
311            label: binding.label,
312            project: Arc::new(project),
313            searchable: false,
314            sortable: false,
315            width: ColumnWidth::Wide,
316            includes: Includes::new(),
317            misdeclared: binding.misdeclared,
318        }
319    }
320
321    /// A computed, display-only column (CONTEXT.md Column: "a computed value").
322    ///
323    /// No field lens — so it cannot be searchable or sortable (it maps to no
324    /// query predicate) — but any cell projection compiles: booleans,
325    /// timestamps, joined values. Calling `.searchable()` / `.sortable()` on
326    /// a computed column is a misdeclaration
327    /// [`Panel::build`](crate::Panel::build) refuses: a lying sort link or
328    /// search promise is worse than a loud build error.
329    pub fn computed(
330        label: impl Into<String>,
331        project: impl Fn(&M) -> String + Send + Sync + 'static,
332    ) -> Self {
333        let label = label.into();
334        let name = label.to_lowercase();
335        Self {
336            path: None,
337            name,
338            label,
339            project: Arc::new(project),
340            searchable: false,
341            sortable: false,
342            width: ColumnWidth::Narrow,
343            includes: Includes::new(),
344            misdeclared: None,
345        }
346    }
347
348    /// Declare a relation this column's projection reads:
349    /// `.include(Post::fields().author())` for `|p| p.author.get().name.clone()`.
350    /// The path is typed on the table's model, so a relation the model does not
351    /// have is a compile error.
352    ///
353    /// The list and the export load every relation the table's columns declared,
354    /// so **declare every relation the closure reads**. One it does not declare
355    /// arrives unloaded, and the closure's `is_unloaded` guard (the
356    /// unloaded-relation contract of ADR-0011, `"(unloaded)"` plus a
357    /// `debug_assert!`) turns that into a loud failure instead of a silent
358    /// `"-"`.
359    ///
360    /// Repeat calls accumulate:
361    /// `.include(Post::fields().author()).include(Post::fields().comments())`.
362    ///
363    /// One query loads a relation once for the whole table: two includes of the
364    /// same relation merge, and an unfiltered one wins over a filtered one
365    /// (Toasty ORs their filters). A column that counts a filtered subset should
366    /// filter in its closure rather than rely on a filtered include.
367    pub fn include<T>(mut self, relation: impl Into<toasty::stmt::Include<M, T>>) -> Self {
368        self.includes = self.includes.with(relation);
369        self
370    }
371
372    pub fn searchable(mut self) -> Self {
373        self.refuse_computed("searchable");
374        self.searchable = true;
375        self
376    }
377
378    pub fn sortable(mut self) -> Self {
379        self.refuse_computed("sortable");
380        self.sortable = true;
381        self
382    }
383
384    /// Record `modifier` on a computed column as a misdeclaration: it maps to
385    /// no query predicate.
386    fn refuse_computed(&mut self, modifier: &str) {
387        if self.path.is_none() && self.misdeclared.is_none() {
388            self.misdeclared = Some(format!(
389                "{modifier}() on computed column '{}': computed columns map to no query predicate",
390                self.label
391            ));
392        }
393    }
394
395    /// Declare this column's width in the table's fixed layout:
396    /// `.width(ColumnWidth::Percent(20))` for a column that knows its own
397    /// measure.
398    ///
399    /// The default follows the column's kind — see [`ColumnWidth`]. Override
400    /// it when the content disagrees with the kind: a `computed` column that
401    /// holds a name or a title is [`Wide`](ColumnWidth::Wide), a `String` field
402    /// that holds a status is [`Narrow`](ColumnWidth::Narrow).
403    pub fn width(mut self, width: ColumnWidth) -> Self {
404        self.width = width;
405        self
406    }
407}

409impl<M> Column<M> for TextColumn<M>
410where
411    M: toasty::schema::Model + Send + Sync + 'static,
412{
413    fn name(&self) -> &str {
414        &self.name
415    }
416
417    fn label(&self) -> &str {
418        &self.label
419    }
420
421    /// The typed projection's output.
422    fn text(&self, row: &M) -> String {
423        (self.project)(row)
424    }
425
426    fn column_width(&self) -> ColumnWidth {
427        self.width
428    }
429
430    fn is_searchable(&self) -> bool {
431        self.searchable
432    }
433
434    /// A portable, escaped **substring** match.
435    ///
436    /// `like_with_escape` keeps the pattern parameterised and lowers to the
437    /// same `LIKE … ESCAPE '\\'` on every driver, and
438    /// `escape_like_pattern` makes the term literal — a `%` or `_` the user
439    /// typed matches that character, it does not act as a wildcard. Note the
440    /// driver difference `LIKE` brings: SQLite compares ASCII
441    /// case-insensitively, PostgreSQL case-sensitively.
442    fn search_expr(&self, term: &str) -> Option<Expr<bool>> {
443        let t = term.trim();
444        if !self.searchable || t.is_empty() {
445            return None;
446        }
447        Some(
448            self.path
449                .clone()?
450                .like_with_escape(escape_like_pattern(t), LIKE_ESCAPE),
451        )
452    }
453
454    fn is_sortable(&self) -> bool {
455        self.sortable
456    }
457
458    fn order_by(&self, descending: bool) -> Option<OrderByExpr> {
459        if self.sortable {
460            // Cursor determinism is the engine's job: toasty's
461            // `normalize_cursor_order` appends the physical PK columns to
462            // ambiguous cursor orderings internally.
463            let path = self.path.clone()?;
464            Some(if descending { path.desc() } else { path.asc() })
465        } else {
466            None
467        }
468    }
469
470    fn includes(&self) -> Includes<M> {
471        self.includes.clone()
472    }
473
474    fn misdeclared(&self) -> Option<String> {
475        self.misdeclared.clone()
476    }
477}

575        };

577            cx =>
578            <span class="inline-flex items-center" data-boolean=(value.to_string())>
579                icon(
580                    data: data,
581                    attrs: attributes! { class=(class) aria-hidden="true" }

614/// A table's columns, as the table stores them: shared, so the list, the
615/// export and the live-search handler read one declaration.

618/// Convert a single built-in column, or a tuple of any [`Column`]s, into a
619/// table's column list.
620///
621/// A tuple takes columns of any type, an app's own among them. A single
622/// column converts on its own when it is a built-in; a single app column is
623/// a one-element tuple, `(MyColumn,)`, or goes through
624/// [`Table::column`](super::Table::column).
625///
626/// Tuple arities stop at eight, the ceiling every tuple-collection trait
627/// shares: `IntoFilters` in `resource/filter.rs` and `IntoSchema` in
628/// `schema/tree.rs`. Without variadic generics the idiom is one
629/// `macro_rules!` invocation per arity; [`Table::column`](super::Table::column)
630/// appends past it.
631pub trait IntoColumns<M> {
632    #[doc(hidden)]
633    fn into_columns(self) -> Vec<BoxColumn<M>>;
634}

649    fn into_columns(self) -> Vec<BoxColumn<M>> {
650        vec![Arc::new(self)]
651    }
652}
653
654/// The tuple impls of [`IntoColumns`], one arity per invocation. Every
655/// element is a [`Column`], so `(a, (b, c))` is not a column list: a
656/// table's columns sit in one flat tuple.
657macro_rules! into_columns_tuples {
658    ($($T:ident => $v:ident),+ $(,)?) => {
659        impl<M, $($T),+> IntoColumns<M> for ($($T,)+)
660        where
661            $($T: Column<M> + 'static,)+
662        {
663            fn into_columns(self) -> Vec<BoxColumn<M>> {
664                let ($($v,)+) = self;
665                vec![$(Arc::new($v) as BoxColumn<M>,)+]
666            }
667        }
668    };
669}
670
671into_columns_tuples!(A => a);
672into_columns_tuples!(A => a, B => b);
673into_columns_tuples!(A => a, B => b, C => c);
674into_columns_tuples!(A => a, B => b, C => c, D => d);
675into_columns_tuples!(A => a, B => b, C => c, D => d, E => e);
676into_columns_tuples!(A => a, B => b, C => c, D => d, E => e, F => f);
677into_columns_tuples!(A => a, B => b, C => c, D => d, E => e, F => f, G => g);
678into_columns_tuples!(
679    A => a, B => b, C => c, D => d, E => e, F => f, G => g, H => h
680);
