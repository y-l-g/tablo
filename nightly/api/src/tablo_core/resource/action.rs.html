10/// A mutation beyond create, update and delete: "publish", "archive",
11/// "resend the invite".
12///
13/// An action runs on one record, from a button in its row, or on the
14/// selection, from the bulk bar, or both ([`ROW`](Self::ROW),
15/// [`BULK`](Self::BULK)). [`Resource::actions`] declares it:
16///
17/// ```ignore
18/// struct Publish;
19///
20/// impl Action<PostResource> for Publish {
21///     const NAME: &'static str = "publish";
22///
23///     fn label() -> String {
24///         "Publish".to_string()
25///     }
26///
27///     fn can_run(_cx: &Cx, post: &Post) -> bool {
28///         post.status != "published"
29///     }
30///
31///     async fn run(_cx: &Cx, posts: &[Post], ex: &mut dyn Executor) -> Result<()> {
32///         for post in posts {
33///             Post::filter(Post::fields().id().eq(post.id))
34///                 .update()
35///                 .status("published".to_string())
36///                 .exec(&mut *ex)
37///                 .await?;
38///         }
39///         Ok(())
40///     }
41/// }
42/// ```
43///
44/// The framework owns everything around [`run`](Self::run), as it does for
45/// a delete:
46///
47/// - the route, `{list}/{key}/actions/{NAME}` for a row and `{list}/actions/{NAME}` for the
48///   selection, and its CSRF check;
49/// - the transaction: the records are loaded through [`scoped_query`](super::scoped_query) inside
50///   it, `run` writes through the same executor, and an error rolls everything back;
51/// - the policy: every record must pass [`Ability::View`](crate::policy::Ability::View) and
52///   [`can_run`](Self::can_run), checked on the loaded rows before `run`;
53/// - [`Resource::after_commit`] with [`Mutation::Action`](super::Mutation::Action) once the
54///   transaction commits, and the success notification.
55///
56/// A row whose record fails `can_run` renders no button for the action, and
57/// a row that no bulk action and no delete allows renders no checkbox.
58pub trait Action<R: Resource>: 'static {
59    /// The action's URL segment, distinct among the resource's actions.
60    /// [`RouterBuilderPanelExt::panel`](crate::RouterBuilderPanelExt::panel) refuses one that is
61    /// not a single path segment, or that another action of the resource shares.
62    const NAME: &'static str;
63
64    /// Whether a row renders the action's button. Defaults to `true`.
65    const ROW: bool = true;
66
67    /// Whether the bulk bar renders the action for the selection. Defaults
68    /// to `true`.
69    const BULK: bool = true;
70
71    /// The button text.
72    fn label() -> String;
73
74    /// Whether the action may run on `record`.
75    fn can_run(cx: &Cx, record: &R::Model) -> bool;
76
77    /// Perform the action on `records`, all of which passed
78    /// [`can_run`](Self::can_run), through the framework's transaction `ex`.
79    fn run(
80        cx: &Cx,
81        records: &[R::Model],
82        ex: &mut dyn toasty::Executor,
83    ) -> impl Future<Output = Result<()>> + Send;
84
85    /// The success notification after a commit. Defaults to the label and
86    /// the record count: `"Publish: 3 records"`.
87    fn success(count: usize) -> String {
88        let noun = if count == 1 { "record" } else { "records" };
89        format!("{}: {count} {noun}", Self::label())
90    }
91}

93/// What an erased action's `run` returns.
94pub(crate) type ActionFuture<'a> = Pin<Box<dyn Future<Output = Result<()>> + Send + 'a>>;
95
96/// The actions a [`Resource`] declares, in button order:
97/// `Actions::new().add::<Publish>().add::<Archive>()`.
98pub struct Actions<R: Resource> {
99    entries: Vec<ActionEntry<R>>,
100}
101
102impl<R: Resource> Default for Actions<R> {
103    fn default() -> Self {
104        Self {
105            entries: Vec::new(),
106        }
107    }
108}
109
110impl<R: Resource> std::fmt::Debug for Actions<R> {
111    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
112        f.debug_list()
113            .entries(self.entries.iter().map(|e| e.name))
114            .finish()
115    }
116}
117
118impl<R: Resource> Actions<R> {
119    /// No actions.
120    pub fn new() -> Self {
121        Self::default()
122    }
123
124    /// Append the action `A`.
125    pub fn add<A: Action<R>>(mut self) -> Self {
126        self.entries.push(ActionEntry {
127            name: A::NAME,
128            label: A::label,
129            row: A::ROW,
130            bulk: A::BULK,
131            can_run: A::can_run,
132            run: run_erased::<R, A>,
133            success: A::success,
134        });
135        self
136    }
137
138    /// The declared actions, in order.
139    pub(crate) fn entries(&self) -> &[ActionEntry<R>] {
140        &self.entries
141    }
142
143    /// The action named `name`.
144    pub(crate) fn find(&self, name: &str) -> Option<&ActionEntry<R>> {
145        self.entries.iter().find(|e| e.name == name)
146    }
147}

149/// One declared action with its type erased, so a resource's actions sit
150/// in one list.
151pub(crate) struct ActionEntry<R: Resource> {
152    pub(crate) name: &'static str,
153    pub(crate) label: fn() -> String,
154    pub(crate) row: bool,
155    pub(crate) bulk: bool,
156    pub(crate) can_run: fn(&Cx, &R::Model) -> bool,
157    pub(crate) run:
158        for<'a> fn(&'a Cx, &'a [R::Model], &'a mut dyn toasty::Executor) -> ActionFuture<'a>,
159    pub(crate) success: fn(usize) -> String,
160}
161
162/// [`Action::run`] behind a function pointer.
163fn run_erased<'a, R: Resource, A: Action<R>>(
164    cx: &'a Cx,
165    records: &'a [R::Model],
166    ex: &'a mut dyn toasty::Executor,
167) -> ActionFuture<'a> {
168    Box::pin(A::run(cx, records, ex))
169}
