12/// Derive `EmbeddedForm` for an embedded struct or enum.
13///
14/// The generated impl builds the value's schema node from the columns the app
15/// schema resolves for the parent path — one text field per leaf, a nested
16/// node per `#[form(embed)]` value, and for an enum the variant control plus
17/// one group per variant — and converts the value to and from the panel's flat
18/// form map through that node's keys. It also generates `form(cx, parent)`,
19/// the value's schema. The framework supplies the storage names, the derive
20/// supplies the Rust shape.
21///
22/// ```ignore
23/// #[derive(Debug, Clone, toasty::Embed, tablo_core::EmbeddedForm)]
24/// pub enum Publication {
25///     #[column(variant = 1)]
26///     Scheduled {
27///         #[shared(timestamp)]
28///         #[form(label = "Publication timestamp")]
29///         scheduled_at: String,
30///         scheduled_for: String,
31///     },
32///     #[column(variant = 2)]
33///     Published { #[shared(timestamp)] published_at: String, canonical_url: String },
34/// }
35///
36/// // form declaration — no field bindings written by hand
37/// Section::new("Publication").schema(Publication::form(cx, Post::fields().publication()))
38/// ```
39///
40/// # How a field is classified
41///
42/// A field marked `#[form(embed)]` is another **embedded value**, delegated to
43/// its own `EmbeddedForm`. Every other field is a **scalar**: one column, read
44/// and written through `FormScalar` (`String`, a `TypedValue` type, or an
45/// `Option` of one). A scalar of another type fails to compile at the field,
46/// naming the trait. An empty scalar is its declared `#[form(blank = ..)]`,
47/// else its `FormScalar::blank()`; with neither, the parse refuses its key.
48///
49/// # Which variant an enum reads
50///
51/// A named discriminant always wins, and an undeclared one is refused;
52/// otherwise the first variant, in declaration order, with a **payload of its
53/// own** submitted — a `#[shared(..)]` column belongs to several variants and
54/// never selects one; otherwise the first variant.
55///
56/// # Per-field attributes
57///
58/// - `#[form(embed)]` — a nested `EmbeddedForm` value.
59/// - `#[form(label = "Canonical URL")]` — the control's label (default: the field name, humanized).
60/// - `#[form(multiline = 3)]` — a `<textarea>` of 3 rows.
61/// - `#[form(blank = ..)]` — what an empty submission reads as, overriding the leaf type's own
62///   answer.
63///
64/// Anything else in `#[form(..)]` is a compile error, as are `label`, `multiline`, and `blank` on
65/// an embedded value.
66#[proc_macro_derive(EmbeddedForm, attributes(form))]
67pub fn embedded_form(input: TokenStream) -> TokenStream {
68    let input = syn::parse_macro_input!(input as DeriveInput);
69    embedded::expand(input)
70}

72/// Derive `RecordForm` for the typed value a resource's form writes.
73///
74/// One field per model column the form writes, named and typed like the
75/// model's field. A scalar (`String`, a `TypedValue` type, or an `Option` of
76/// one) binds the key its control posts; a `#[form(embed)]` field binds every
77/// key of an `EmbeddedForm` value and is written whole.
78///
79/// ```ignore
80/// #[derive(tablo_core::RecordForm)]
81/// #[form(model = User)]
82/// pub struct UserForm {
83///     pub name: String,
84///     #[form(blank = "member")]
85///     pub role: String,
86///     #[form(blank = 0)]
87///     pub age: i64,
88/// }
89/// ```
90///
91/// The derive also emits `UserFormField`, one variant per field, which
92/// `Posted` keys on and `RecordForm::fields` answers with each variant's keys.
93/// It emits `UserFormControls`, one control per field chosen from the field —
94/// a `bool` is a toggle, `#[form(options = T)]` a choice over `T`'s options,
95/// `#[form(choice)]` a bare choice, `#[form(file)]` a file field,
96/// `#[form(embed)]` the embedded value's schema, and any other field a text
97/// field — with `controls(dx)` handing them over and `RecordForm::schema`
98/// arranging one per field in declaration order. `Resource::form` defaults to
99/// that schema; an override arranges the controls into a layout instead.
100///
101/// # Attributes
102///
103/// - `#[form(model = User)]` on the struct: the model the form writes.
104/// - `#[form(blank = <expr>)]` on a scalar: the value an empty submission reads as, overriding the
105///   default (`String` answers `""` and `Option<T>` answers `None` through the type's own blank,
106///   and `bool` answers `false` through the derive's default).
107/// - `#[form(options = Status)]`: a choice over `Status::options()`.
108/// - `#[form(choice)]`: a bare choice, whose options or relationship the resource's `form` may add.
109/// - `#[form(file)]` on a `String`: a file field.
110/// - `#[form(embed)]` on an `EmbeddedForm` value.
111///
112/// A generic struct, a tuple struct, an empty struct, a `Deferred<_>` field,
113/// `blank` on an `Option` or an embedded value, and an unknown key are compile
114/// errors. So are a field the model lacks, a type the model's field does not
115/// have, and a scalar that is not a `FormScalar`.
116#[proc_macro_derive(RecordForm, attributes(form))]
117pub fn record_form(input: TokenStream) -> TokenStream {
118    let input = syn::parse_macro_input!(input as DeriveInput);
119    record_form::expand_tokens(input).into()
120}

122/// Derive `Options` for a unit-variant enum: the `(value, label)` list a
123/// choice field, a select filter and a column share.
124///
125/// ```ignore
126/// #[derive(tablo::Options)]
127/// pub enum Status {
128///     Draft,
129///     #[option(label = "Live")]
130///     Published,
131/// }
132///
133/// assert_eq!(Status::Published.value(), "published");
134/// assert_eq!(Status::Published.label(), "Live");
135/// assert_eq!(Status::from_value("draft"), Some(Status::Draft));
136/// ```
137///
138/// Each variant stores its `snake_case` name and reads as that name in
139/// sentence case. `#[option(value = "..")]` and `#[option(label = "..")]`
140/// override either. A generic enum, a variant with fields, two variants
141/// storing one value, and an unknown key are compile errors.
142#[proc_macro_derive(Options, attributes(option))]
143pub fn options(input: TokenStream) -> TokenStream {
144    let input = syn::parse_macro_input!(input as DeriveInput);
145    options::expand_tokens(input).into()
146}

148/// The path the generated code names `tablo-core` by: the `tablo` facade, which
149/// re-exports it at its root, else `::tablo_core`, either under the consumer's
150/// rename. The facade comes first: it is what an app depends on, and it
151/// reaches every path the generated code names.
152fn tablo_core_path(span: &syn::Ident, derive: &str) -> syn::Result<proc_macro2::TokenStream> {
153    let found = proc_macro_crate::crate_name("tablo")
154        .map(|found| (found, "tablo"))
155        .or_else(|_| proc_macro_crate::crate_name("tablo-core").map(|found| (found, "tablo_core")));
156    match found {
157        Ok((found, own_name)) => {
158            let name = match found {
159                proc_macro_crate::FoundCrate::Itself => own_name.to_string(),
160                proc_macro_crate::FoundCrate::Name(n) => n,
161            };
162            let ident = syn::Ident::new(&name.replace('-', "_"), proc_macro2::Span::call_site());
163            Ok(quote::quote! { ::#ident })
164        }
165        Err(_) => Err(syn::Error::new_spanned(
166            span,
167            format!("`tablo` (or `tablo-core`) must be a dependency to #[derive({derive})]"),
168        )),
169    }
