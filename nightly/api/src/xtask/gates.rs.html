1//! Gate orchestration: `fmt`, `bump-upstream`, `verify-locks`, `check`.
2//!
3//! Each CI job calls one subcommand instead of inlining its own bash loop, so
4//! the loops are typed, tested, and runnable locally. Jobs stay parallel —
5//! `check` is a local fail-fast convenience runner only, never a CI job.

51/// The repo root (xtask lives at `<root>/xtask`).
52pub fn repo_root() -> PathBuf {
53    let manifest_dir = PathBuf::from(env!("CARGO_MANIFEST_DIR"));
54    manifest_dir
55        .parent()
56        .unwrap_or(Path::new("."))
57        .to_path_buf()
58}

60/// How a gate command runs: `prog` with `args` in `dir`, plus extra `env`.
61/// Commands inherit stdio and propagate their exit code — never through a
62/// pipe, so no status gets masked (AGENTS.md rule 5).
63pub trait Runner {
64    fn run(
65        &self,
66        prog: &str,
67        args: &[&str],
68        dir: Option<&Path>,
69        env: &[(&str, &str)],
70    ) -> anyhow::Result<()>;
71}
72
73/// The real runner: inherit stdio, fail on a non-zero exit.
74pub struct RealRunner;
75
76impl Runner for RealRunner {
77    fn run(
78        &self,
79        prog: &str,
80        args: &[&str],
81        dir: Option<&Path>,
82        env: &[(&str, &str)],
83    ) -> anyhow::Result<()> {
84        let mut command = Command::new(prog);
85        command.args(args);
86        if let Some(dir) = dir {
87            command.current_dir(dir);
88        }
89        for (key, value) in env {
90            command.env(key, value);
91        }
92        let status = command
93            .status()
94            .map_err(|error| anyhow::anyhow!("failed to run {prog}: {error}"))?;
95        if status.success() {
96            Ok(())
97        } else {
98            anyhow::bail!("`{prog} {}` exited with {status}", args.join(" "))
99        }
100    }
101}

103/// The pinned-nightly workspace fmt check (CONTRIBUTING gate 3).
104pub fn nightly_fmt(run: &dyn Runner, root: &Path) -> anyhow::Result<()> {
105    let toolchain = format!("+{NIGHTLY_FMT}");
106    run.run(
107        "cargo",
108        &[&toolchain, "fmt", "--all", "--", "--check"],
109        Some(root),
110        &[],
111    )
112}

114/// The detached-bench fmt checks: `cargo fmt` never sees workspace-excluded members.
115pub fn detached_fmt(run: &dyn Runner, root: &Path) -> anyhow::Result<()> {
116    for bench in DETACHED_BENCHES {
117        run.run(
118            "cargo",
119            &["fmt", "--", "--check"],
120            Some(&root.join(bench)),
121            &[],
122        )?;
123    }
124    Ok(())
125}

130/// The locked-rev `topcoat fmt` check plus diff guard (CONTRIBUTING gate 4).
131/// Check-half only: installing the CLI (~4 min build) stays in CI's cache-keyed
132/// step, so a missing or wrong-rev CLI fails here with the install command.
133pub fn topcoat_fmt(run: &dyn Runner, root: &Path) -> anyhow::Result<()> {
134    run.run("topcoat", &["fmt"], Some(root), &[])
135        .map_err(|error| {
136            anyhow::anyhow!(
137                "{error}\n`topcoat fmt` needs the CLI built from the locked rev: {TOPCOAT_INSTALL}"
138            )
139        })?;
140    run.run("git", &["diff", "--exit-code"], Some(root), &[])
141        .map_err(|error| {
142            anyhow::anyhow!(
143                "{error}\nA diff that only reflows `view!` markup means the CLI is the wrong rev, not a hand-fix: {TOPCOAT_INSTALL}"
144            )
145        })
146}

148/// The formatting subset: nightly fmt, detached-bench fmt, locked-rev topcoat fmt.
149pub fn fmt_check(run: &dyn Runner) -> anyhow::Result<()> {
150    let root = repo_root();
151    nightly_fmt(run, &root)?;
152    detached_fmt(run, &root)?;
153    topcoat_fmt(run, &root)
154}

156/// The `source` field's resolved git sha (`git+https://…?rev=<sha>#<sha>`).
157fn git_pin(source: &str) -> Option<&str> {
158    source
159        .rsplit_once('#')
160        .map(|(_, sha)| sha)
161        .filter(|sha| !sha.is_empty())
162}

164/// The lockstep pins parsed from one `cargo metadata` document (serde_json is
165/// already an xtask dep, so no new `toml` dep risks the MSRV floor).
166pub fn pins_from_metadata(metadata: &serde_json::Value) -> BTreeMap<String, String> {
167    let mut pins = BTreeMap::new();
168    for package in metadata["packages"].as_array().into_iter().flatten() {
169        let name = package["name"].as_str().unwrap_or("");
170        if !LOCKSTEP_CRATES.contains(&name) {
171            continue;
172        }
173        if let Some(sha) = package["source"].as_str().and_then(git_pin) {
174            pins.insert(name.to_string(), sha.to_string());
175        }
176    }
177    pins
178}

180/// Fail unless every lockstep crate pins the same rev in both lockfiles.
181pub fn check_lockstep(
182    workspace: &BTreeMap<String, String>,
183    bench: &BTreeMap<String, String>,
184) -> anyhow::Result<()> {
185    let mut drift = Vec::new();
186    for name in LOCKSTEP_CRATES {
187        match (workspace.get(*name), bench.get(*name)) {
188            (Some(workspace_rev), Some(bench_rev)) if workspace_rev == bench_rev => {
189                println!("{name}: {workspace_rev} (in sync)");
190            }
191            (Some(workspace_rev), Some(bench_rev)) => drift.push(format!(
192                "{name} rev drift: workspace={workspace_rev} bench={bench_rev}"
193            )),
194            _ => drift.push(format!(
195                "{name} rev missing: workspace={} bench={}",
196                workspace.get(*name).map_or("none", String::as_str),
197                bench.get(*name).map_or("none", String::as_str),
198            )),
199        }
200    }
201    if drift.is_empty() {
202        Ok(())
203    } else {
204        anyhow::bail!(
205            "lockstep pin drift — bump both manifests in one commit:\n{}",
206            drift.join("\n")
207        )
208    }
209}

211/// Fail unless both manifests pin the same single rev per upstream repo —
212/// the companion pins (`toasty-core`, `topcoat-ui*`) `bump-upstream` manages,
213/// which the lock comparison never sees.
214pub fn check_manifest_lockstep(
215    workspace: &BTreeMap<String, BTreeSet<String>>,
216    bench: &BTreeMap<String, BTreeSet<String>>,
217) -> anyhow::Result<()> {
218    let show = |revs: Option<&BTreeSet<String>>| {
219        revs.map_or("none".to_string(), |set| {
220            set.iter()
221                .map(String::as_str)
222                .collect::<Vec<_>>()
223                .join(", ")
224        })
225    };
226    let mut drift = Vec::new();
227    for (name, _) in UPSTREAM_REPOS {
228        match (workspace.get(*name), bench.get(*name)) {
229            (Some(workspace_revs), Some(bench_revs))
230                if workspace_revs == bench_revs && workspace_revs.len() == 1 =>
231            {
232                println!(
233                    "{name} manifest pins: {} (in sync)",
234                    show(Some(workspace_revs))
235                );
236            }
237            (Some(_), Some(_)) => drift.push(format!(
238                "{name} manifest rev drift: workspace={} bench={}",
239                show(workspace.get(*name)),
240                show(bench.get(*name)),
241            )),
242            _ => drift.push(format!(
243                "{name} manifest rev missing: workspace={} bench={}",
244                show(workspace.get(*name)),
245                show(bench.get(*name)),
246            )),
247        }
248    }
249    if drift.is_empty() {
250        Ok(())
251    } else {
252        anyhow::bail!(
253            "manifest pin drift — bump both manifests in one commit:\n{}",
254            drift.join("\n")
255        )
256    }
257}

259/// `cargo metadata` for one manifest. `--locked` (not `--frozen`): a stale
260/// tree must fail instead of healing the lockfile, but fresh CI caches
261/// legitimately need the network to resolve — `--offline` would mask the
262/// drift signal with download errors.
263fn fetch_metadata(manifest: &Path) -> anyhow::Result<serde_json::Value> {
264    let output = Command::new("cargo")
265        .args([
266            "metadata",
267            "--format-version",
268            "1",
269            "--locked",
270            "--manifest-path",
271        ])
272        .arg(manifest)
273        .output()
274        .map_err(|error| anyhow::anyhow!("failed to run cargo metadata: {error}"))?;
275    if !output.status.success() {
276        anyhow::bail!(
277            "cargo metadata failed for {}: {}",
278            manifest.display(),
279            String::from_utf8_lossy(&output.stderr).trim()
280        );
281    }
282    serde_json::from_slice(&output.stdout)
283        .map_err(|error| anyhow::anyhow!("could not parse cargo metadata: {error}"))
284}

286/// Workspace vs bench rev equality, local and CI (the `bench-check` lockstep):
287/// the `rev =` manifest pins `bump-upstream` manages, grouped by repo, then
288/// the resolved lock pins. Manifests are plain file reads first — no
289/// subprocess, no network, nothing to heal — so a hand-edited manifest
290/// reports drift here instead of failing inside `cargo metadata --locked`
291/// with a resolution error.
292pub fn verify_locks() -> anyhow::Result<()> {
293    let root = repo_root();
294    let workspace_manifest = std::fs::read_to_string(root.join("Cargo.toml"))
295        .map_err(|error| anyhow::anyhow!("cannot read Cargo.toml: {error}"))?;
296    let bench_manifest = std::fs::read_to_string(root.join("benchmarks/tablo/Cargo.toml"))
297        .map_err(|error| anyhow::anyhow!("cannot read benchmarks/tablo/Cargo.toml: {error}"))?;
298    check_manifest_lockstep(
299        &manifest_revs(&workspace_manifest),
300        &manifest_revs(&bench_manifest),
301    )?;
302    let workspace_meta = fetch_metadata(&root.join("Cargo.toml"))?;
303    let bench_meta = fetch_metadata(&root.join("benchmarks/tablo/Cargo.toml"))?;
304    check_lockstep(
305        &pins_from_metadata(&workspace_meta),
306        &pins_from_metadata(&bench_meta),
307    )
308}

310fn is_rev(rev: &str) -> bool {
311    rev.len() == 40 && rev.bytes().all(|byte| byte.is_ascii_hexdigit())
312}
313
314/// The byte span of the `rev = "…"` value on one manifest line, if present.
315fn rev_span(line: &str) -> Option<(usize, usize)> {
316    let start = line.find("rev = \"")? + "rev = \"".len();
317    let end = line[start..].find('"')? + start;
318    Some((start, end))
319}
320
321/// The `rev = "…"` value on one manifest line, if present.
322fn read_rev(line: &str) -> Option<&str> {
323    let (start, end) = rev_span(line)?;
324    Some(&line[start..end])
325}
326
327/// Replace the `rev = "…"` value on one manifest line.
328fn replace_rev(line: &str, rev: &str) -> Option<String> {
329    let (start, end) = rev_span(line)?;
330    Some(format!("{}{rev}{}", &line[..start], &line[end..]))
331}

333/// The `rev =` pins in one manifest, grouped by repo short name: each repo
334/// maps to its distinct pinned revs (one element when the manifest agrees
335/// with itself). Reads through [`UPSTREAM_REPOS`], the same table
336/// `set_upstream_revs` writes through.
337pub fn manifest_revs(manifest: &str) -> BTreeMap<String, BTreeSet<String>> {
338    let mut pins: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
339    for line in manifest.split('\n') {
340        let Some((name, _)) = UPSTREAM_REPOS.iter().find(|(_, url)| line.contains(url)) else {
341            continue;
342        };
343        if let Some(rev) = read_rev(line) {
344            pins.entry(name.to_string())
345                .or_default()
346                .insert(rev.to_string());
347        }
348    }
349    pins
350}

352/// Rewrite the `rev =` pins for both upstream repos in one manifest.
353/// Matches through [`UPSTREAM_REPOS`] so `toasty-core` and `topcoat-ui*`
354/// follow their repo.
355pub fn set_upstream_revs(
356    manifest: &str,
357    topcoat_rev: &str,
358    toasty_rev: &str,
359) -> anyhow::Result<String> {
360    let revs = [topcoat_rev, toasty_rev];
361    let mut hits = [0, 0];
362    let mut out = Vec::new();
363    for line in manifest.split('\n') {
364        let mut rewritten = None;
365        for (index, (_, url)) in UPSTREAM_REPOS.iter().enumerate() {
366            if !line.contains(url) {
367                continue;
368            }
369            if let Some(new_line) = replace_rev(line, revs[index]) {
370                hits[index] += 1;
371                rewritten = Some(new_line);
372            }
373            break;
374        }
375        out.push(rewritten.unwrap_or_else(|| line.to_string()));
376    }
377    if hits[0] == 0 || hits[1] == 0 {
378        anyhow::bail!(
379            "expected topcoat and toasty git pins, found {} and {}",
380            hits[0],
381            hits[1]
382        );
383    }
384    Ok(out.join("\n"))
385}

387/// Bump both upstream revs in both manifests, re-resolve both lockfiles,
388/// prove the revs resolve from the local git cache, and assert lockstep.
389pub fn bump_upstream(run: &dyn Runner, topcoat_rev: &str, toasty_rev: &str) -> anyhow::Result<()> {
390    for (name, rev) in [("topcoat", topcoat_rev), ("toasty", toasty_rev)] {
391        if !is_rev(rev) {
392            anyhow::bail!("{name} rev must be a 40-char hex sha, got `{rev}`");
393        }
394    }
395    let root = repo_root();
396    for manifest in ["Cargo.toml", "benchmarks/tablo/Cargo.toml"] {
397        let path = root.join(manifest);
398        let text = std::fs::read_to_string(&path)
399            .map_err(|error| anyhow::anyhow!("cannot read {}: {error}", path.display()))?;
400        let updated = set_upstream_revs(&text, topcoat_rev, toasty_rev)?;
401        std::fs::write(&path, updated)
402            .map_err(|error| anyhow::anyhow!("cannot write {}: {error}", path.display()))?;
403        println!("updated {manifest}");
404    }
405    run.run(
406        "cargo",
407        &["update", "-p", "topcoat", "-p", "toasty"],
408        Some(&root),
409        &[],
410    )?;
411    run.run(
412        "cargo",
413        &[
414            "update",
415            "--manifest-path",
416            "benchmarks/tablo/Cargo.toml",
417            "-p",
418            "topcoat",
419            "-p",
420            "toasty",
421        ],
422        Some(&root),
423        &[],
424    )?;
425    run.run("cargo", &["check", "--offline"], Some(&root), &[])?;
426    run.run(
427        "cargo",
428        &[
429            "check",
430            "--offline",
431            "--manifest-path",
432            "benchmarks/tablo/Cargo.toml",
433        ],
434        Some(&root),
435        &[],
436    )?;
437    verify_locks()
438}

440/// The gate set as a local fail-fast convenience runner: the eight CONTRIBUTING
441/// gates in order, then docs, detached-bench fmt, and the lockstep check.
442pub fn check(run: &dyn Runner) -> anyhow::Result<()> {
443    check_with(run, &verify_locks)
444}

446fn check_with(run: &dyn Runner, verify: &dyn Fn() -> anyhow::Result<()>) -> anyhow::Result<()> {
447    let root = repo_root();
448    run.run(
449        "cargo",
450        &["test", "--workspace", "--locked"],
451        Some(&root),
452        &[],
453    )?;
454    run.run(
455        "cargo",
456        &[
457            "clippy",
458            "--workspace",
459            "--all-targets",
460            "--locked",
461            "--",
462            "-D",
463            "warnings",
464        ],
465        Some(&root),
466        &[],
467    )?;
468    nightly_fmt(run, &root)?;
469    topcoat_fmt(run, &root)?;
470    run.run(
471        "cargo",
472        &[
473            "clippy",
474            "--locked",
475            "--manifest-path",
476            "benchmarks/tablo/Cargo.toml",
477            "--all-targets",
478            "--",
479            "-D",
480            "warnings",
481        ],
482        Some(&root),
483        &[],
484    )?;
485    let msrv = format!("+{MSRV}");
486    run.run(
487        "cargo",
488        &[&msrv, "check", "--workspace", "--locked"],
489        Some(&root),
490        &[],
491    )?;
492    let mut assets: Vec<&str> = vec!["--test"];
493    assets.extend_from_slice(ASSET_SUITES);
494    run.run("node", &assets, Some(&root), &[])?;
495    run.run(
496        "cargo",
497        &["+nightly", "install", "cargo-udeps", "--locked"],
498        Some(&root),
499        &[],
500    )?;
501    run.run(
502        "cargo",
503        &[
504            "+nightly",
505            "udeps",
506            "--workspace",
507            "--all-targets",
508            "--locked",
509        ],
510        Some(&root),
511        &[],
512    )?;
513    run.run(
514        "cargo",
515        &["doc", "--workspace", "--no-deps", "--locked"],
516        Some(&root),
517        &[("RUSTDOCFLAGS", "-D warnings")],
518    )?;
519    run.run("mdbook", &["build", "docs/guide"], Some(&root), &[])?;
520    detached_fmt(run, &root)?;
521    verify()
522}
