8/// Pre-paint theme application — the blocking head script that reconciles the
9/// `dark` class on `<html>` before the first pixel (shadcn `theme-provider`
10/// parity: `attribute="class"` + pre-paint apply).
11///
12/// `theme.js` wires the toggle buttons and persists the choice to
13/// `localStorage` + a `theme` cookie, but it runs on `DOMContentLoaded`, i.e.
14/// after first paint. Render this in the `<head>` — before any body content —
15/// so the class is right before the first pixel.
16///
17/// The stored preference is **authoritative in both directions**: a
18/// stored `light` *removes* the server-rendered `dark` class rather than
19/// leaving it in place.
20///
21/// The script deliberately avoids `&&`: topcoat has no DOM-safe inline-script
22/// primitive, so the body goes through the normal HTML escape path, which
23/// rewrites `&&` to `&amp;&amp;` — and character references are **not** decoded
24/// inside `<script>` (HTML's script-data state), so the emitted script would
25/// be a syntax error. Nested `if`s keep the body free of escapable characters.
26fn theme_init_script_body(default_dark: bool) -> String {
27    let mut script = String::from(
28        "(function(){var t=null;try{t=localStorage.getItem('theme')}catch(e){}\
29if(!t){var m=document.cookie.match(/theme=([^;]+)/);if(m)t=m[1]}\
30if(t!=='light'){if(t!=='dark')t=",
31    );
32    let _ = write!(script, "'{}'", if default_dark { "dark" } else { "light" });
33    script.push_str(
34        ";}if(t==='dark')document.documentElement.classList.add('dark');\
35else document.documentElement.classList.remove('dark')})();",
36    );
37    script
38}

40/// Render in the `<head>` of every layout, before the stylesheet link.
41///
42/// `default_dark` is the server's build-time preference (`Panel::dark_mode`),
43/// used only when the visitor has no stored choice.
44///
45/// The body goes through the normal escaping path rather than `Unescaped`:
46/// topcoat has no DOM-safe script primitive, and the body is ours — a fixed
47/// ASCII program with one `true`/`false` interpolated — so escaping is
48/// harmless as long as the program stays free of characters HTML would
49/// rewrite (see `theme_init_script_body`).
50///
51/// ```ignore
52/// <head>
53///     theme_init_script(default_dark)
54///     <link rel="stylesheet" href=(tailwind::stylesheet!())>
55/// </head>
56/// ```
57#[component]
58pub async fn theme_init_script(default_dark: bool) -> Result<impl View> {
59    let body = theme_init_script_body(default_dark);
60    Ok(view! { <script>(body)</script> })
61}
