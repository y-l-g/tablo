1//! Build-script helpers: scans Tablo's sources for Tailwind classes the app's
2//! stylesheet cannot name, published via Cargo `links` metadata.
3//!
4//! In the app's `build.rs` `main`:
5//!
6//! ```no_run
7//! tablo_build::tailwind().expect("the Tailwind build runs");
8//! ```
9//!
10//! The package must depend on `tablo` (or on `tablo-core` and `tablo-ui`) as a
11//! normal dependency, and name this crate under `[build-dependencies]`.
12
13use std::{
14    env,
15    ffi::OsString,
16    fmt, fs, io,
17    path::{Path, PathBuf},
18};
19
20/// Holds one published source directory each, as Cargo names them for a
21/// dependent's build script.
22const SOURCE_VARS: &[&str] = &[
23    "DEP_TABLO_CORE",
24    "DEP_TABLO_UI",
25    "DEP_TABLO_CORE_SRC",
26    "DEP_TABLO_UI_SRC",
27];
28
29/// The stylesheet [`tailwind`] reads, relative to the package root.
30pub const STYLESHEET: &str = "styles.css";
31
32/// Why [`tailwind`] failed.
33#[derive(Debug)]
34pub enum Error {
35    /// No Tablo crate published its sources to this build script: the package
36    /// does not depend on `tablo`, `tablo-core` or `tablo-ui` directly.
37    NoSources,
38    /// A Cargo variable a build script always has is unset: the function ran
39    /// outside a build script.
40    NotABuildScript(&'static str),
41    /// Reading the stylesheet or writing the generated input failed.
42    Io { path: PathBuf, source: io::Error },
43    /// The Tailwind CLI could not be fetched or failed.
44    Tailwind(topcoat::tailwind::BuildError),
45}
46
47impl fmt::Display for Error {
48    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
49        match self {
50            Self::NoSources => write!(
51                f,
52                "no Tablo sources to scan: none of {} is set; depend on `tablo` (or `tablo-core` \
53                 and `tablo-ui`) under [dependencies]",
54                SOURCE_VARS.join(", ")
55            ),
56            Self::NotABuildScript(var) => {
57                write!(f, "`{var}` is not set; call this from a build script")
58            }
59            Self::Io { path, source } => write!(f, "{}: {source}", path.display()),
60            Self::Tailwind(error) => write!(f, "tailwind: {error}"),
61        }
62    }
63}
64
65impl std::error::Error for Error {
66    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
67        match self {
68            Self::Io { source, .. } => Some(source),
69            Self::Tailwind(error) => Some(error),
70            Self::NoSources | Self::NotABuildScript(_) => None,
71        }
72    }
73}
74
75impl From<topcoat::tailwind::BuildError> for Error {
76    fn from(error: topcoat::tailwind::BuildError) -> Self {
77        Self::Tailwind(error)
78    }
79}

81/// Builds the app's stylesheet with Tablo's sources in scope, writing
82/// `$OUT_DIR/tailwind.css` and returning that path.
83///
84/// Prints `cargo::rerun-if-changed` for the stylesheet and each source directory.
85///
86/// # Errors
87///
88/// [`Error::NotABuildScript`] outside a build script, [`Error::NoSources`]
89/// when no Tablo crate published its sources to this build script,
90/// [`Error::Io`] when the stylesheet is missing or the input cannot be
91/// written, and [`Error::Tailwind`] when the CLI fails.
92pub fn tailwind() -> Result<PathBuf, Error> {
93    let manifest_dir = var("CARGO_MANIFEST_DIR")?;
94    let out_dir = PathBuf::from(var("OUT_DIR")?);
95    let stylesheet = PathBuf::from(manifest_dir).join(STYLESHEET);
96    if !stylesheet.is_file() {
97        return Err(Error::Io {
98            path: stylesheet,
99            source: io::Error::new(io::ErrorKind::NotFound, "the app's stylesheet is missing"),
100        });
101    }
102    let sources = source_dirs(|name| env::var_os(name));
103    if sources.is_empty() {
104        return Err(Error::NoSources);
105    }
106    println!("cargo::rerun-if-changed={}", stylesheet.display());
107    for dir in &sources {
108        println!("cargo::rerun-if-changed={}", dir.display());
109    }
110    let input = out_dir.join("tablo-tailwind-input.css");
111    write_if_changed(&input, &input_css(&stylesheet, &sources))?;
112    Ok(topcoat::tailwind::BuildConfig::new()
113        .input(input)
114        .render()?)
115}
116
117fn var(name: &'static str) -> Result<OsString, Error> {
118    env::var_os(name).ok_or(Error::NotABuildScript(name))
119}
120
121/// Collects the published directories, deduplicated in first-seen order.
122fn source_dirs(lookup: impl Fn(&str) -> Option<OsString>) -> Vec<PathBuf> {
123    let mut dirs: Vec<PathBuf> = Vec::new();
124    for name in SOURCE_VARS {
125        let Some(value) = lookup(name) else {
126            continue;
127        };
128        let dir = PathBuf::from(value);
129        if !dir.as_os_str().is_empty() && !dirs.contains(&dir) {
130            dirs.push(dir);
131        }
132    }
133    dirs
134}
135
136fn input_css(stylesheet: &Path, sources: &[PathBuf]) -> String {
137    let mut css = format!("@import {};\n", css_string(stylesheet));
138    for dir in sources {
139        css.push_str(&format!("@source {};\n", css_string(&dir.join("**/*.rs"))));
140    }
141    css
142}
143
144fn css_string(path: &Path) -> String {
145    let path = path.to_string_lossy().replace('\\', "/");
146    format!("\"{}\"", path.replace('"', "\\\""))
147}
148
149fn write_if_changed(path: &Path, contents: &str) -> Result<(), Error> {
150    if fs::read_to_string(path).is_ok_and(|existing| existing == contents) {
151        return Ok(());
152    }
153    fs::write(path, contents).map_err(|source| Error::Io {
154        path: path.to_path_buf(),
155        source,
156    })
157}
