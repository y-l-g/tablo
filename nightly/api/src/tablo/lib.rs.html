1//! Tablo — an admin toolkit for Rust, server-rendered on Topcoat and persisted
2//! with Toasty.
3//!
4//! This crate is the one Tablo dependency an app names, beside `topcoat` and
5//! `toasty` at the revisions Tablo pins:
6//!
7//! ```toml
8//! [dependencies]
9//! tablo = { git = "https://github.com/y-l-g/tablo", features = ["sqlite"] }
10//!
11//! [build-dependencies]
12//! tablo-build = { git = "https://github.com/y-l-g/tablo" }
13//! ```
14//!
15//! It re-exports [`tablo_core`] at its root, the UI components as [`ui`], and
16//! the in-memory HTTP client as `testing` with the `testing` feature; the driver
17//! features (`sqlite`, `postgresql`, `mysql`) turn on Toasty's driver of the same
18//! name, and the toolkit itself enables none.
19//!
20//! A minimal app is one model, one [`Resource`], and the [`Panel`] that serves
21//! it — the full version lives in `examples/quickstart`:
22//!
23//! ```ignore
24//! use tablo::prelude::*;
25//! use toasty::Db;
26//!
27//! #[derive(Debug, Clone, toasty::Model)]
28//! pub struct Book {
29//!     #[key]
30//!     #[auto]
31//!     pub id: uuid::Uuid,
32//!     pub title: String,
33//! }
34//!
35//! #[derive(tablo::RecordForm)]
36//! #[form(model = Book)]
37//! pub struct BookForm {
38//!     pub title: String,
39//! }
40//!
41//! pub struct BookResource;
42//!
43//! impl Resource for BookResource {
44//!     type Model = Book;
45//!     type Form = BookForm;
46//!
47//!     fn policy() -> impl Policy<Book> {
48//!         Allow
49//!     }
50//!
51//!     fn table() -> Table<Book> {
52//!         Table::new(
53//!             |b: &Book| b.id.to_string(),
54//!             TextColumn::r#for(Book::fields().title(), |b: &Book| b.title.clone())
55//!                 .searchable()
56//!                 .sortable(),
57//!         )
58//!     }
59//!
60//!     fn form(_dx: &tablo::DeclCx) -> Schema {
61//!         Schema::new(Field::text(Book::fields().title()))
62//!     }
63//! }
64//!
65//! let router = Router::builder()
66//!     .discover()
67//!     .app_context(db)
68//!     .panel(Panel::new("admin").resource::<BookResource>())?
69//!     .build();
70//! ```
71//!
72//! The user guide starts at "Your first panel".
