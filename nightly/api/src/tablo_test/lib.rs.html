1//! In-memory HTTP test client for Tablo panels.
2//!
3//! [`TestClient`] drives a built `Router` without a socket: it carries
4//! cookies, a CSRF token, and a tenant, and the helpers below write form and
5//! multipart bodies and read values back out of the rendered HTML. An app
6//! reaches it as `tablo::testing` with the facade's `testing` feature; the
7//! showcase and `tablo-core` suites use the same client. Seed and database
8//! fixtures stay with each suite: they name its own models.
9
10use http::header::{CONTENT_TYPE, COOKIE};
11use http_body_util::BodyExt;
12use topcoat::router::{Body, Router};
13
14/// One request client for every suite.
15///
16/// Every request is built here, so a suite attaches cookies, a tenant, or a
17/// session in one place. Builder methods clone the client, leaving the base
18/// reusable: `client.tenant(t).csrf(&token).post_form(uri, body)`.
19#[derive(Clone)]
20pub struct TestClient<'a> {
21    router: &'a Router,
22    cookies: Vec<(String, String)>,
23    tenant: Option<uuid::Uuid>,
24}
25
26impl<'a> TestClient<'a> {
27    pub fn new(router: &'a Router) -> Self {
28        Self {
29            router,
30            cookies: Vec::new(),
31            tenant: None,
32        }
33    }
34
35    /// Attach a cookie to every request this client sends. A later value for
36    /// the same name replaces the earlier one, like a browser jar.
37    pub fn cookie(&self, name: &str, value: &str) -> Self {
38        let mut client = self.clone();
39        match client.cookies.iter_mut().find(|(kept, _)| kept == name) {
40            Some(existing) => existing.1 = value.to_string(),
41            None => client.cookies.push((name.to_string(), value.to_string())),
42        }
43        client
44    }
45
46    /// Attach every `(name, value)` pair, e.g. the cookies a response set.
47    /// Like a browser jar: a later value for the same name replaces the
48    /// earlier one instead of appending a duplicate `Cookie` entry.
49    pub fn cookies(&self, cookies: &[(String, String)]) -> Self {
50        let mut client = self.clone();
51        for (name, value) in cookies {
52            match client.cookies.iter_mut().find(|(kept, _)| kept == name) {
53                Some(existing) => existing.1 = value.clone(),
54                None => client.cookies.push((name.clone(), value.clone())),
55            }
56        }
57        client
58    }
59
60    /// Attach the CSRF cookie the form's `csrf_token` field must match.
61    pub fn csrf(&self, token: &str) -> Self {
62        self.cookie(tablo_core::csrf::COOKIE_NAME, token)
63    }
64
65    /// Carry a tenant as a `Tenant` request extension — the server-set
66    /// override seam. It takes precedence over the logged-in user's
67    /// tenant, letting a suite scope one request to another tenant.
68    pub fn tenant(&self, tenant: uuid::Uuid) -> Self {
69        let mut client = self.clone();
70        client.tenant = Some(tenant);
71        client
72    }
73
74    pub async fn get(&self, uri: &str) -> http::Response<Body> {
75        self.router
76            .handle(self.request(http::Method::GET, uri))
77            .await
78    }
79
80    /// POST an urlencoded form.
81    pub async fn post_form(&self, uri: &str, body: String) -> http::Response<Body> {
82        let mut request = self.request(http::Method::POST, uri);
83        request.headers_mut().insert(
84            CONTENT_TYPE,
85            http::HeaderValue::from_static("application/x-www-form-urlencoded"),
86        );
87        *request.body_mut() = Body::from(body);
88        self.router.handle(request).await
89    }
90
91    /// POST a multipart body (file uploads).
92    pub async fn post_multipart(
93        &self,
94        uri: &str,
95        boundary: &str,
96        body: String,
97    ) -> http::Response<Body> {
98        let mut request = self.request(http::Method::POST, uri);
99        request.headers_mut().insert(
100            CONTENT_TYPE,
101            format!("multipart/form-data; boundary={boundary}")
102                .parse()
103                .expect("multipart content type"),
104        );
105        *request.body_mut() = Body::from(body);
106        self.router.handle(request).await
107    }
108
109    /// POST a JSON body to a runtime endpoint (a shard or procedure), with the
110    /// page identity header the browser runtime sends.
111    pub async fn post_json(&self, uri: &str, body: String, identity: &str) -> http::Response<Body> {
112        let mut request = self.request(http::Method::POST, uri);
113        request.headers_mut().insert(
114            CONTENT_TYPE,
115            http::HeaderValue::from_static("application/json"),
116        );
117        request.headers_mut().insert(
118            topcoat::router::request::IDENTITY_HEADER,
119            http::HeaderValue::from_str(identity).expect("identity header"),
120        );
121        *request.body_mut() = Body::from(body);
122        self.router.handle(request).await
123    }
124
125    /// Build a request carrying this client's cookies and tenant.
126    fn request(&self, method: http::Method, uri: &str) -> http::Request<Body> {
127        let mut builder = http::Request::builder().method(method).uri(uri);
128        if let Some(jar) = cookie_header(
129            self.cookies
130                .iter()
131                .map(|(name, value)| (name.as_str(), value.as_str())),
132        ) {
133            builder = builder.header(COOKIE, jar);
134        }
135        let (mut parts, body) = builder.body(Body::empty()).unwrap().into_parts();
136        if let Some(tenant) = self.tenant {
137            parts.extensions.insert(tablo_core::Tenant(tenant));
138        }
139        http::Request::from_parts(parts, body)
140    }
141}

143/// One `Cookie` header value from `(name, value)` pairs, or `None` when there
144/// are no pairs to send.
145pub fn cookie_header<'a>(cookies: impl IntoIterator<Item = (&'a str, &'a str)>) -> Option<String> {
146    let mut jar = String::new();
147    for (name, value) in cookies {
148        if !jar.is_empty() {
149            jar.push_str("; ");
150        }
151        jar.push_str(name);
152        jar.push('=');
153        jar.push_str(value);
154    }
155    (!jar.is_empty()).then_some(jar)
156}

158/// Collect a response body as bytes.
159pub async fn body_bytes(response: http::Response<Body>) -> Vec<u8> {
160    response
161        .into_body()
162        .collect()
163        .await
164        .expect("collect body")
165        .to_bytes()
166        .to_vec()
167}

169/// Collect a response body as a lossy UTF-8 string.
170pub async fn body_string(response: http::Response<Body>) -> String {
171    String::from_utf8_lossy(&body_bytes(response).await).into_owned()
172}

174/// URL-encode `(key, value)` pairs into an urlencoded form body.
175pub fn form_body(pairs: &[(&str, &str)]) -> String {
176    let mut serializer = form_urlencoded::Serializer::new(String::new());
177    for (key, value) in pairs {
178        serializer.append_pair(key, value);
179    }
180    serializer.finish()
181}

183/// A multipart body, one part per entry: `None` is a text part, `Some("")` the
184/// browser's "no file chosen" file part, `Some(name)` a chosen file.
185///
186/// Parts carry no per-part `Content-Type`: neither server parser reads one.
187/// The framework parser (`tablo_core` multipart values, over Topcoat's
188/// multer-based extractor) tells file parts from text parts by the
189/// `filename` parameter alone, and the showcase media upload reads the part's
190/// content type only as a display-kind hint defaulting to file. A suite that
191/// needs a part content type (the media kind cases) builds that body ad-hoc.
192pub fn multipart_body(boundary: &str, parts: &[(&str, Option<&str>, &str)]) -> String {
193    let mut body = String::new();
194    for (name, filename, content) in parts {
195        body.push_str(&format!(
196            "--{boundary}\r\nContent-Disposition: form-data; name=\"{name}\""
197        ));
198        if let Some(filename) = filename {
199            body.push_str(&format!("; filename=\"{filename}\""));
200        }
201        body.push_str("\r\n\r\n");
202        body.push_str(content);
203        body.push_str("\r\n");
204    }
205    body.push_str(&format!("--{boundary}--\r\n"));
206    body
207}

209/// The `value` attribute of the named `<input>` in rendered HTML, in either
210/// attribute order.
211///
212/// Handles both quote styles (`value="…"` and `value='…'`); the controlled
213/// UUID markup only emits double quotes today, but an encoder change must
214/// not silently turn every lookup into `None` (harness hardening).
215pub fn input_value(html: &str, name: &str) -> Option<String> {
216    let double = format!("name=\"{name}\"");
217    let single = format!("name='{name}'");
218    for tag in html.split('<').skip(1) {
219        if !tag.contains(&double) && !tag.contains(&single) {
220            continue;
221        }
222        let attrs = &tag[..tag.find('>')?];
223        for (prefix, term) in [("value=\"", '"'), ("value='", '\'')] {
224            if let Some(start) = attrs.find(prefix) {
225                let rest = &attrs[start + prefix.len()..];
226                if let Some(end) = rest.find(term) {
227                    // Unescape the matching quote entity the encoder may emit.
228                    let raw = &rest[..end];
229                    return Some(raw.replace("&quot;", "\"").replace("&#x27;", "'"));
230                }
231            }
232        }
233    }
234    None
235}

237/// The session cookie name Topcoat's default token store writes (`__Host-`
238/// prefix plus the `session` name, per its hardened cookie contract).
239pub const SESSION_COOKIE: &str = "__Host-session";
240
241/// The `(name, value)` pairs a response's `Set-Cookie` headers carry.
242pub fn response_cookies(response: &http::Response<Body>) -> Vec<(String, String)> {
243    response
244        .headers()
245        .get_all(http::header::SET_COOKIE)
246        .iter()
247        .filter_map(|value| value.to_str().ok())
248        .filter_map(|value| value.split(';').next())
249        .filter_map(|pair| pair.split_once('='))
250        .map(|(name, value)| (name.trim().to_string(), value.trim().to_string()))
251        .collect()
252}
253
254/// The full `Set-Cookie` header for `name`, so tests can assert attributes.
255pub fn set_cookie_header(response: &http::Response<Body>, name: &str) -> Option<String> {
256    response
257        .headers()
258        .get_all(http::header::SET_COOKIE)
259        .iter()
260        .filter_map(|value| value.to_str().ok())
261        .find(|value| value.starts_with(&format!("{name}=")))
262        .map(str::to_string)
263}
264
265/// The session cookie value a response set, if any.
266pub fn session_cookie_value(response: &http::Response<Body>) -> Option<String> {
267    response_cookies(response)
268        .into_iter()
269        .find(|(name, _)| name == SESSION_COOKIE)
270        .map(|(_, value)| value)
271}
