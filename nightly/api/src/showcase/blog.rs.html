74/// The list: published posts only, newest first, each linking to its page.
75#[page("/blog")]
76async fn page(cx: &Cx) -> Result<impl View> {
77    let mut db = db(cx);
78    // N+1 touch: the author is included.
79    let include_author: Include<Post, Author> = Post::fields().author().into();
80    let posts = Post::filter(Post::fields().status().eq(PUBLISHED.to_string()))
81        .order_by(Post::fields().created_at().desc())
82        .include(include_author)
83        .exec(&mut db)
84        .await?;
85
86    Ok(view! {
87        <h1 class="text-3xl font-bold tracking-tight">"Blog"</h1>
88
89        if posts.is_empty() {
90            <p class="mt-6 text-muted-foreground">
91                "No posts have been published yet."
92            </p>
93        } else {
94            <ul class="mt-8 flex flex-col gap-8">
95                for post in &posts {
96                    <li>
97                        <article>
98                            <h2 class="text-xl font-semibold tracking-tight">
99                                <a
100                                    href=(href!(post_page, Id(post.id)))
101                                    class="hover:underline"
102                                >
103                                    (&post.title)
104                                </a>
105                            </h2>
106                            <p class="mt-1 text-sm text-muted-foreground">
107                                (author_name(post))
108                                " · "
109                                (post.created_at.strftime("%Y-%m-%d").to_string())
110                            </p>
111                            if !post.seo.description.is_empty() {
112                                <p class="mt-3 text-muted-foreground">
113                                    (&post.seo.description)
114                                </p>
115                            }
116                        </article>
117                    </li>
118                }
119            </ul>
120        }
121    })
122}

124/// Renders one published post; drafts answer not-found.
125#[page("/blog/{id}")]
126async fn post_page(cx: &Cx) -> Result<impl View> {
127    // A malformed id answers not-found.
128    let id = path_param::<Id>(cx).map_err(|_| not_found())?.to_owned();
129
130    let mut db = db(cx);
131    let include_author: Include<Post, Author> = Post::fields().author().into();
132    let post = Post::filter(
133        Post::fields()
134            .id()
135            .eq(id)
136            .and(Post::fields().status().eq(PUBLISHED.to_string())),
137    )
138    .include(include_author)
139    .first()
140    .exec(&mut db)
141    .await?
142    .ok_or_not_found()?;
143
144    let cover = match post.cover_id {
145        Some(cover_id) => {
146            MediaAsset::filter(MediaAsset::fields().id().eq(cover_id))
147                .first()
148                .exec(&mut db)
149                .await?
150        }
151        None => None,
152    };
153
154    Ok(view! {
155        <a
156            href=(href!(page))
157            class="text-sm text-muted-foreground hover:text-foreground"
158        >
159            "← All posts"
160        </a>
161
162        <article class="mt-4">
163            <h1 class="text-3xl font-bold tracking-tight">(&post.title)</h1>
164            <p class="mt-2 text-sm text-muted-foreground">
165                (author_name(&post))
166                " · "
167                (post.created_at.strftime("%Y-%m-%d").to_string())
168            </p>
169
170            if !post.seo.description.is_empty() {
171                <p class="mt-6 text-lg text-muted-foreground">
172                    (&post.seo.description)
173                </p>
174            }
175
176            if let Some(asset) = cover.as_ref().and_then(|asset| cover_url(asset)) {
177                <img
178                    src=(asset.to_string())
179                    alt=(post.title.clone())
180                    class="mt-8 w-full rounded-lg border border-border"
181                >
182            }
183
184            <div class="mt-8 leading-7">(&post.body)</div>
185        </article>
186    })
187}

189/// Reads the author's display name.
190fn author_name(post: &Post) -> String {
191    if post.author.is_unloaded() {
192        "(unknown author)".to_string()
193    } else {
194        post.author.get().name.clone()
195    }
196}

198/// Returns the cover URL when the row names a servable path.
199fn cover_url(asset: &MediaAsset) -> Option<&str> {
200    let path = asset.path.trim();
201    path.starts_with('/').then_some(path)
