1//! The media library: the `medias` table and the page that fills it.
2//!
3//! A WordPress-style library: one row per stored file with the tenant that
4//! uploaded it, the `path` the `Uploader` returned, the client's `filename`,
5//! a `kind`, and a timestamp. Rows carry no owner: a post shows one row as its
6//! cover through its own `cover_id`, and the library lists one tenant's rows
7//! (ADR-0021). The page uploads through the app's own [`Uploader`] — the
8//! `DirUploader` the panel installs for its file fields — writes the row, and
9//! lists what the library holds: a thumbnail for an image, a link for anything
10//! else.
11

36/// The upload route's path. A `#[route]` path is a literal, so it spells the
37/// panel prefix; the page's form and the redirect after an upload take the
38/// page's URL from `tablo_core::url::page` instead.
39pub const MEDIA_PATH: &str = "/admin/media";

41/// The widget script the page emits.
42///
43/// An app asset: ADR-0014's nine scripts are the shell's, and this one belongs
44/// to the page that renders the widget. The page links it `defer`red, and only
45/// when the router carries an asset bundle — the test router has none, like the
46/// public blog's document.
47pub const MEDIA_JS: topcoat::asset::Asset = topcoat::asset::asset!("../assets/media.js");

49/// The `kind` of a row whose bytes are an image: the row renders a thumbnail.
50pub const KIND_IMAGE: &str = "image";
51
52/// The `kind` of every other row: it renders a link.
53pub const KIND_FILE: &str = "file";

55/// The upload form's file field.
56const FILE_FIELD: &str = "file";

58/// The media library as a relationship source: one tenant's rows.
59///
60/// A post's cover picker loads its options through this source, so the tenant
61/// gate and the tenant filter apply to the choice exactly as they apply to the
62/// page that lists the same rows.
63pub struct MediaLibrary;
64
65impl OptionSource for MediaLibrary {
66    type Model = MediaAsset;
67
68    fn scoped_query(cx: &Cx) -> Result<toasty::stmt::Query<toasty::stmt::List<MediaAsset>>> {
69        let tenant = require_tenant(cx)?;
70        Ok(toasty::stmt::Query::<toasty::stmt::List<MediaAsset>>::all()
71            .filter(MediaAsset::fields().tenant_id().eq(tenant)))
72    }
73
74    fn policy() -> impl Policy<MediaAsset> {
75        ReadOnly
76    }
77
78    fn requires_tenant() -> bool {
79        true
80    }
81
82    fn slug() -> String {
83        "media".to_string()
84    }
85
86    fn search_expr(_cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
87        let term = term.trim();
88        if term.is_empty() {
89            return None;
90        }
91        Some(
92            MediaAsset::fields()
93                .filename()
94                .like_with_escape(format!("%{term}%"), '\\'),
95        )
96    }
97
98    fn order_by(_cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
99        Some(MediaAsset::fields().filename().asc())
100    }
101}

103/// One media row's file: a thumbnail for an image, a link for anything else.
104///
105/// The framework's file field links every stored path the same way;
106/// telling an image from the rest is the media library's job, and `kind` is
107/// what the row recorded when the upload was stored. The public blog renders a
108/// post's cover through this too, so one row looks the same wherever it is
109/// shown.
110pub fn media_file_view<'a>(cx: &'a Cx, asset: &MediaAsset) -> BoxView<'a> {
111    if asset.kind == KIND_IMAGE {
112        let src = asset.path.clone();
113        let alt = asset.filename.clone();
114        view! {
115            cx =>
116            <img
117                src=(src)
118                alt=(alt)
119                data-media-thumbnail=""
120                class="size-12 shrink-0 rounded-md border border-border object-cover"
121            >
122        }
123        .boxed()
124    } else {
125        let href = asset.path.clone();
126        let name = asset.filename.clone();
127        view! {
128            cx =>
129            <a
130                href=(href)
131                data-media-link=""
132                class="min-w-0 truncate text-sm underline"
133            >
134                (name)
135            </a>
136        }
137        .boxed()
138    }
139}

141/// The media library: every row this tenant holds, and the form that adds one.
142///
143/// The list is one query, not a paginated `Table`: a `Table` renders text
144/// columns, and a thumbnail is not text. A library that outgrows one page wants
145/// its own loader and pager, which is a different seam from this demo's.
146pub struct MediaLibraryPage;
147
148impl Page for MediaLibraryPage {
149    fn navigation() -> NavigationItem {
150        NavigationItem::for_page::<Self>().icon(tablo_ui::icons::IMAGE)
151    }
152
153    /// The library's segment under the panel prefix; the upload route at
154    /// [`MEDIA_PATH`] shares the URL.
155    fn slug() -> String {
156        "media".to_string()
157    }
158
159    async fn render(cx: &Cx) -> Result<impl View> {
160        // One tenant's library: a tenantless request is refused rather
161        // than served every tenant's rows.
162        let tenant = require_tenant(cx)?;
163        let mut db = db(cx);
164        // No resource owns `MediaAsset`, so its tenant filter is this page's —
165        // written once, on the column the model declares.
166        let media = MediaAsset::filter(MediaAsset::fields().tenant_id().eq(tenant))
167            .order_by(MediaAsset::fields().created_at().desc())
168            .exec(&mut db)
169            .await?;
170
171        // The form is the app's, so the token is the app's to embed.
172        let csrf_token = csrf::ensure_token(cx);
173        let has_assets = try_app_context::<AssetConfig>(cx).is_some();
174
175        Ok(view! {
176            cx =>
177            tablo_ui::page(
178                tablo_ui::page_header(
179                    tablo_ui::page_title("Media library")
180                    tablo_ui::page_description(
181                        "Files stored through the app's uploader, picked as post covers."
182                    )
183                )
184                tablo_ui::page_content(
185                    tablo_ui::card(
186                        tablo_ui::card_header(tablo_ui::card_title("Upload"))
187                        tablo_ui::card_content(
188                            <form
189                                method="post"
190                                action=(tablo_core::url::page::<MediaLibraryPage>(cx))
191                                enctype="multipart/form-data"
192                                class="flex flex-col gap-4"
193                            >
194                                (csrf::field(cx, &csrf_token))
195                                tablo_ui::field(
196                                    tablo_ui::field_label(
197                                        attrs: attributes! { for="media-file" },
198                                        "File"
199                                    )
200                                    <div class="flex items-center gap-2">
201                                        tablo_ui::input(
202                                            attrs: attributes! {
203                                                id="media-file"
204                                                type="file"
205                                                name=(FILE_FIELD)
206                                                required=""
207                                                data-media-file=""
208                                            }
209                                        )
210                                        // The × is a reset control: with no script
211                                        // the browser resets the form and the file
212                                        // input empties; `media.js` empties the input
213                                        // and the preview itself and cancels that
214                                        // reset, so a file clear keeps the form
215                                        // usable (ADR-0021).
216                                        tablo_ui::button(
217                                            variant: tablo_ui::ButtonVariant::Outline,
218                                            size: tablo_ui::ButtonSize::Icon,
219                                            attrs: attributes! {
220                                                type="reset"
221                                                data-media-clear=""
222                                                aria-label="Clear the selected file"
223                                                title="Clear the selected file"
224                                            },
225                                            icon(data: tablo_ui::icons::X)
226                                        )
227                                    </div>
228                                    <div
229                                        data-media-preview=""
230                                        hidden=""
231                                        class="flex items-center gap-3 text-xs text-muted-foreground"
232                                    ></div>
233                                )
234                                <div>
235                                    tablo_ui::button(
236                                        variant: tablo_ui::ButtonVariant::Primary,
237                                        attrs: attributes! { type="submit" },
238                                        "Upload"
239                                    )
240                                </div>
241                            </form>
242                        )
243                    )
244                    tablo_ui::card(
245                        tablo_ui::card_header(tablo_ui::card_title("Stored media"))
246                        tablo_ui::card_content(
247                            <div class="flex flex-col gap-3">
248                                if media.is_empty() {
249                                    tablo_ui::empty_state(
250                                        title: "No media has been uploaded yet.",
251                                        detail: "Uploaded files are listed here.",
252                                        attrs: attributes! { data-media-empty="" }
253                                    )
254                                } else {
255                                    <ul
256                                        data-media-list=""
257                                        class="flex flex-col divide-y divide-border"
258                                    >
259                                        for asset in &media {
260                                            <li
261                                                data-media-row=(asset.id.to_string())
262                                                class="flex items-center gap-4 py-3"
263                                            >
264                                                (media_file_view(cx, asset))
265                                                <div class="flex min-w-0 flex-col">
266                                                    if asset.kind == KIND_IMAGE {
267                                                        <span class="truncate text-sm font-medium">
268                                                            (asset.filename.clone())
269                                                        </span>
270                                                    }
271                                                    <span class="text-xs text-muted-foreground">
272                                                        (asset.created_at.strftime("%Y-%m-%d %H:%M").to_string())
273                                                    </span>
274                                                </div>
275                                            </li>
276                                        }
277                                    </ul>
278                                }
279                            </div>
280                        )
281                    )
282                    if has_assets {
283                        <script src=(MEDIA_JS) defer=""></script>
284                    }
285                )
286            )
287        })
288    }

291/// `POST /admin/media` — store one uploaded file and write the row for it.
292///
293/// The page renders its own form, so it parses its own multipart body: the
294/// framework's parser serves the fields a `Schema` declares, and this form is
295/// not one. The bytes go through the app's own [`Uploader`] — the same
296/// `DirUploader` the app gives `Panel::uploads` — outside any
297/// transaction, like every upload (ADR-0017).
298#[route(POST "/admin/media")]
299async fn upload(cx: &Cx, mut multipart: Multipart) -> Result<SeeOther> {
300    let tenant = require_tenant(cx)?;
301    let mut values = HashMap::new();
302    let mut file: Option<UploadedPart> = None;
303    while let Some(field) = multipart.next_field().await? {
304        let Some(name) = field.name().map(str::to_string) else {
305            continue;
306        };
307        if name == FILE_FIELD {
308            let filename = field.file_name().unwrap_or_default().to_string();
309            let content_type = field.content_type().unwrap_or_default().to_string();
310            let bytes = field.bytes().await?;
311            file = Some(UploadedPart {
312                filename,
313                content_type,
314                bytes: bytes.to_vec(),
315            });
316        } else {
317            values.insert(name, field.text().await?);
318        }
319    }
320    // The framework verifies the forms it renders; this one is the
321    // app's, so the check is the app's too.
322    csrf::verify(cx, &values)?;
323    let part = file.ok_or_else(|| bad_request("Choose a file before uploading."))?;
324    // The name the row records and the store writes: one rule, so the row's
325    // `filename` and the file on disk cannot disagree.
326    let filename = basename(&part.filename);
327    if filename.is_empty() || part.bytes.is_empty() {
328        return Err(bad_request("Choose a file before uploading.").into());
329    }
330    let mut db = db(cx);
331    // The app's own store, pointed at the directory the panel serves: the
332    // `Uploader` `Panel::uploads` installs lives on the panel for the
333    // framework's form parser and is not readable from a page, so the page
334    // builds the same store from the same configuration. What it returns is
335    // the row's `path` verbatim — a URL that resolves back to these bytes.
336    let path = DirUploader::new(upload_dir())
337        .store(&filename, &part.bytes)
338        .await
339        .map_err(bad_request)?;
340    toasty::create!(MediaAsset {
341        tenant_id: tenant,
342        path: path,
343        filename: filename,
344        kind: kind_of(&part.content_type).to_string(),
345        created_at: jiff::Timestamp::now(),
346    })
347    .exec(&mut db)
348    .await?;
349    set_notification(cx, Notification::success("Media uploaded"));
350    let library = tablo_core::url::page::<MediaLibraryPage>(cx)
351        .expect("media page is registered on this panel");
352    Ok(see_other(library))
353}

355/// The file part an upload form submitted, before anything is stored.
356struct UploadedPart {
357    filename: String,
358    content_type: String,
359    bytes: Vec<u8>,
360}
361
362/// Whether an uploaded part is an image, from the content type the browser sent
363/// with it.
364///
365/// The framework's file field reads no extension and renders every stored path
366/// the same way; deciding that a thumbnail suits *these* bytes is the
367/// media library's, and the part's `Content-Type` is what the browser says they
368/// are. It is a claim, not a sniff: a library that served those bytes to other
369/// people would read their magic numbers instead.
370fn kind_of(content_type: &str) -> &'static str {
371    if content_type
372        .trim()
373        .to_ascii_lowercase()
374        .starts_with("image/")
375    {
376        KIND_IMAGE
377    } else {
378        KIND_FILE
379    }
380}
