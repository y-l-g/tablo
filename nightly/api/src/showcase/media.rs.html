1//! The media library: the `medias` table and the page that fills it.
2//!
3//! A WordPress-style library: one row per stored file with the tenant that
4//! uploaded it, the `path` the `Uploader` returned, the client's `filename`,
5//! a `kind`, and a timestamp. Rows carry no owner: a post shows one row as its
6//! cover through its own `cover_id`, and the library lists one tenant's rows
7//! (ADR-0021). The page uploads through the app's own [`Uploader`] — the
8//! `DirUploader` the panel installs for its file fields — writes the row, and
9//! lists what the library holds: a thumbnail for an image, a link for anything
10//! else.
11

35/// Where the media library lives: the page, the upload route, and the form's
36/// own action.
37pub const MEDIA_PATH: &str = "/admin/media";

39/// The widget script the page emits.
40///
41/// An app asset: ADR-0014's nine scripts are the shell's, and this one belongs
42/// to the page that renders the widget. The page links it `defer`red, and only
43/// when the router carries an asset bundle — the test router has none, like the
44/// public blog's document.
45pub const MEDIA_JS: topcoat::asset::Asset = topcoat::asset::asset!("../assets/media.js");

47/// The `kind` of a row whose bytes are an image: the row renders a thumbnail.
48pub const KIND_IMAGE: &str = "image";
49
50/// The `kind` of every other row: it renders a link.
51pub const KIND_FILE: &str = "file";

53/// The upload form's file field.
54const FILE_FIELD: &str = "file";

56/// The media library as a relationship source: one tenant's rows.
57///
58/// A post's cover picker loads its options through this source, so the tenant
59/// gate and the tenant filter apply to the choice exactly as they apply to the
60/// page that lists the same rows.
61pub struct MediaLibrary;
62
63impl OptionSource for MediaLibrary {
64    type Model = MediaAsset;
65
66    fn scoped_query(cx: &Cx) -> Result<toasty::stmt::Query<toasty::stmt::List<MediaAsset>>> {
67        let tenant = require_tenant(cx)?;
68        Ok(toasty::stmt::Query::<toasty::stmt::List<MediaAsset>>::all()
69            .filter(MediaAsset::fields().tenant_id().eq(tenant)))
70    }
71
72    fn can_view_any(_cx: &Cx) -> bool {
73        true
74    }
75
76    fn can_view(_cx: &Cx, _record: &MediaAsset) -> bool {
77        true
78    }
79
80    fn requires_tenant() -> bool {
81        true
82    }
83
84    fn slug() -> String {
85        "media".to_string()
86    }
87
88    fn search_expr(_cx: &Cx, term: &str) -> Option<toasty::stmt::Expr<bool>> {
89        let term = term.trim();
90        if term.is_empty() {
91            return None;
92        }
93        Some(
94            MediaAsset::fields()
95                .filename()
96                .like_with_escape(format!("%{term}%"), '\\'),
97        )
98    }
99
100    fn order_by(_cx: &Cx) -> Option<toasty::stmt::OrderByExpr> {
101        Some(MediaAsset::fields().filename().asc())
102    }
103}

105/// One media row's file: a thumbnail for an image, a link for anything else.
106///
107/// The framework's file field links every stored path the same way;
108/// telling an image from the rest is the media library's job, and `kind` is
109/// what the row recorded when the upload was stored. The public blog renders a
110/// post's cover through this too, so one row looks the same wherever it is
111/// shown.
112pub fn media_file_view<'a>(cx: &'a Cx, asset: &MediaAsset) -> BoxView<'a> {
113    if asset.kind == KIND_IMAGE {
114        let src = asset.path.clone();
115        let alt = asset.filename.clone();
116        view! {
117            cx =>
118            <img
119                src=(src)
120                alt=(alt)
121                data-media-thumbnail=""
122                class="size-12 shrink-0 rounded-md border border-border object-cover"
123            >
124        }
125        .boxed()
126    } else {
127        let href = asset.path.clone();
128        let name = asset.filename.clone();
129        view! {
130            cx =>
131            <a
132                href=(href)
133                data-media-link=""
134                class="min-w-0 truncate text-sm underline"
135            >
136                (name)
137            </a>
138        }
139        .boxed()
140    }
141}

143/// The media library: every row this tenant holds, and the form that adds one.
144///
145/// The list is one query, not a paginated `Table`: a `Table` renders text
146/// columns, and a thumbnail is not text. A library that outgrows one page wants
147/// its own loader and pager, which is a different seam from this demo's.
148pub struct MediaLibraryPage;
149
150impl Page for MediaLibraryPage {
151    /// The library keeps the short URL [`MEDIA_PATH`] names; the upload route
152    /// and the form's action share it.
153    fn slug() -> String {
154        "media".to_string()
155    }
156
157    async fn render(cx: &Cx) -> Result<impl View> {
158        // One tenant's library: a tenantless request is refused rather
159        // than served every tenant's rows.
160        let tenant = require_tenant(cx)?;
161        let mut db = db(cx);
162        // No resource owns `MediaAsset`, so its tenant filter is this page's —
163        // written once, on the column the model declares.
164        let media = MediaAsset::filter(MediaAsset::fields().tenant_id().eq(tenant))
165            .order_by(MediaAsset::fields().created_at().desc())
166            .exec(&mut db)
167            .await?;
168
169        // The form is the app's, so the token is the app's to embed.
170        let csrf_token = csrf::ensure_token(cx);
171        let has_assets = try_app_context::<AssetConfig>(cx).is_some();
172
173        Ok(view! {
174            cx =>
175            tablo_ui::page(
176                tablo_ui::page_header(
177                    tablo_ui::page_title("Media library")
178                    tablo_ui::page_description(
179                        "Files stored through the app's uploader, picked as post covers."
180                    )
181                )
182                tablo_ui::page_content(
183                    tablo_ui::card(
184                        tablo_ui::card_header(tablo_ui::card_title("Upload"))
185                        tablo_ui::card_content(
186                            <form
187                                method="post"
188                                action=(MEDIA_PATH)
189                                enctype="multipart/form-data"
190                                class="flex flex-col gap-4"
191                            >
192                                (csrf::field(cx, &csrf_token))
193                                <div class="flex flex-col gap-2">
194                                    <label class="text-sm font-medium" for="media-file">
195                                        "File"
196                                    </label>
197                                    <div class="flex items-center gap-2">
198                                        tablo_ui::input(
199                                            attrs: attributes! {
200                                                id="media-file"
201                                                type="file"
202                                                name=(FILE_FIELD)
203                                                required=""
204                                                data-media-file=""
205                                            }
206                                        )
207                                        // The × is a reset control: with no script
208                                        // the browser resets the form and the file
209                                        // input empties; `media.js` empties the input
210                                        // and the preview itself and cancels that
211                                        // reset, so a file clear keeps the form
212                                        // usable (ADR-0021).
213                                        tablo_ui::button(
214                                            variant: tablo_ui::ButtonVariant::Outline,
215                                            size: tablo_ui::ButtonSize::Sm,
216                                            attrs: attributes! {
217                                                type="reset"
218                                                data-media-clear=""
219                                                aria-label="Clear the selected file"
220                                            },
221                                            "×"
222                                        )
223                                    </div>
224                                    <div
225                                        data-media-preview=""
226                                        hidden=""
227                                        class="flex items-center gap-3 text-xs text-muted-foreground"
228                                    ></div>
229                                </div>
230                                tablo_ui::button(
231                                    variant: tablo_ui::ButtonVariant::Primary,
232                                    attrs: attributes! { type="submit" },
233                                    "Upload"
234                                )
235                            </form>
236                        )
237                    )
238                    tablo_ui::card(
239                        tablo_ui::card_header(tablo_ui::card_title("Stored media"))
240                        tablo_ui::card_content(
241                            <div class="flex flex-col gap-3">
242                                if media.is_empty() {
243                                    <p
244                                        data-media-empty=""
245                                        class="text-sm text-muted-foreground"
246                                    >
247                                        "No media has been uploaded yet."
248                                    </p>
249                                } else {
250                                    <ul
251                                        data-media-list=""
252                                        class="flex flex-col divide-y divide-border"
253                                    >
254                                        for asset in &media {
255                                            <li
256                                                data-media-row=(asset.id.to_string())
257                                                class="flex items-center gap-4 py-3"
258                                            >
259                                                (media_file_view(cx, asset))
260                                                <div class="flex min-w-0 flex-col">
261                                                    if asset.kind == KIND_IMAGE {
262                                                        <span class="truncate text-sm font-medium">
263                                                            (asset.filename.clone())
264                                                        </span>
265                                                    }
266                                                    <span class="text-xs text-muted-foreground">
267                                                        (asset.created_at.strftime("%Y-%m-%d %H:%M").to_string())
268                                                    </span>
269                                                </div>
270                                            </li>
271                                        }
272                                    </ul>
273                                }
274                            </div>
275                        )
276                    )
277                    if has_assets {
278                        <script src=(MEDIA_JS) defer=""></script>
279                    }
280                )
281            )
282        })
283    }

286/// `POST /admin/media` — store one uploaded file and write the row for it.
287///
288/// The page renders its own form, so it parses its own multipart body: the
289/// framework's parser serves the fields a `Schema` declares, and this form is
290/// not one. The bytes go through the app's own [`Uploader`] — the same
291/// `DirUploader` the app gives `Panel::uploads` — outside any
292/// transaction, like every upload (ADR-0017).
293#[route(POST "/admin/media")]
294async fn upload(cx: &Cx, mut multipart: Multipart) -> Result<SeeOther> {
295    let tenant = require_tenant(cx)?;
296    let mut values = HashMap::new();
297    let mut file: Option<UploadedPart> = None;
298    while let Some(field) = multipart.next_field().await? {
299        let Some(name) = field.name().map(str::to_string) else {
300            continue;
301        };
302        if name == FILE_FIELD {
303            let filename = field.file_name().unwrap_or_default().to_string();
304            let content_type = field.content_type().unwrap_or_default().to_string();
305            let bytes = field.bytes().await?;
306            file = Some(UploadedPart {
307                filename,
308                content_type,
309                bytes: bytes.to_vec(),
310            });
311        } else {
312            values.insert(name, field.text().await?);
313        }
314    }
315    // The framework verifies the forms it renders; this one is the
316    // app's, so the check is the app's too.
317    csrf::verify(cx, &values)?;
318    let part = file.ok_or_else(|| bad_request("Choose a file before uploading."))?;
319    // The name the row records and the store writes: one rule, so the row's
320    // `filename` and the file on disk cannot disagree.
321    let filename = basename(&part.filename);
322    if filename.is_empty() || part.bytes.is_empty() {
323        return Err(bad_request("Choose a file before uploading.").into());
324    }
325    let mut db = db(cx);
326    // The app's own store, pointed at the directory the panel serves: the
327    // `Uploader` `Panel::uploads` installs lives on the app context for the
328    // framework's form parser and is not readable from a page, so the page
329    // builds the same store from the same configuration. What it returns is
330    // the row's `path` verbatim — a URL that resolves back to these bytes.
331    let path = DirUploader::new(upload_dir())
332        .store(&filename, &part.bytes)
333        .await
334        .map_err(bad_request)?;
335    toasty::create!(MediaAsset {
336        tenant_id: tenant,
337        path: path,
338        filename: filename,
339        kind: kind_of(&part.content_type).to_string(),
340        created_at: jiff::Timestamp::now(),
341    })
342    .exec(&mut db)
343    .await?;
344    set_notification(cx, Notification::success("Media uploaded"));
345    Ok(see_other(MEDIA_PATH))
346}

348/// The file part an upload form submitted, before anything is stored.
349struct UploadedPart {
350    filename: String,
351    content_type: String,
352    bytes: Vec<u8>,
353}
354
355/// Whether an uploaded part is an image, from the content type the browser sent
356/// with it.
357///
358/// The framework's file field reads no extension and renders every stored path
359/// the same way; deciding that a thumbnail suits *these* bytes is the
360/// media library's, and the part's `Content-Type` is what the browser says they
361/// are. It is a claim, not a sniff: a library that served those bytes to other
362/// people would read their magic numbers instead.
363fn kind_of(content_type: &str) -> &'static str {
364    if content_type
365        .trim()
366        .to_ascii_lowercase()
367        .starts_with("image/")
368    {
369        KIND_IMAGE
370    } else {
371        KIND_FILE
372    }
373}
