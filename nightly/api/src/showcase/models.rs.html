12/// User shown in the admin list — the realistic spec model (US16):
13/// role/active/created_at plus `#[index]` on the searchable `name` column.
14/// `email` keeps only `#[unique]` — a unique constraint already implies an
15/// index, and stacking `#[index]` on top would double it.
16#[derive(Debug, Clone, toasty::Model)]
17pub struct User {
18    #[key]
19    #[auto]
20    pub id: uuid::Uuid,
21    #[index]
22    pub name: String,
23    #[unique]
24    pub email: String,
25    /// "admin" or "member" — the form renders them as a static-options choice.
26    pub role: String,
27    pub active: bool,
28    /// A stored integer the form binds as a typed text field: optional, zero
29    /// or more.
30    pub age: i64,
31    #[default(jiff::Timestamp::now())]
32    pub created_at: Timestamp,
33}

36// Scoped, not global: the form's unique probe runs through the
37// tenant-scoped query (`scoped_query`), so a *global* unique index
38// on `email` would be rejected by the database for an email another tenant
39// already owns — after the probe passed — and surface as a 500. Constraining
40// `(tenant_id, email)` makes the constraint say what the probe enforces, so two
41// tenants may share an email.
42#[unique(tenant_id, email)]
43pub struct Author {
44    #[key]
45    #[auto]
46    pub id: uuid::Uuid,
47    #[index]
48    pub tenant_id: uuid::Uuid,
49    /// Display name.
50    pub name: String,
51    pub email: String,
52    #[has_many]
53    pub posts: Deferred<Vec<Post>>,
54}

56/// SEO metadata for a post — an embedded struct.
57///
58/// Flattens into the parent table as `seo_title` / `seo_description`: the same
59/// row, no join, but two more columns the form binds like any other.
60#[derive(Debug, Clone, toasty::Embed, tablo_core::EmbeddedForm)]
61pub struct Seo {
62    pub title: String,
63    /// A multi-line control: the derive renders one text field per leaf, and
64    /// this is the one leaf the panel wants as a `<textarea>`.
65    #[form(multiline = 3)]
66    pub description: String,
67}

69/// A post's lifecycle — an embedded enum whose **timestamps are shared**.
70///
71/// Every variant declares a timestamp under the same `#[shared(timestamp)]`
72/// identifier, so the three coalesce into one `publication_timestamp` column
73/// instead of one column per variant. The rest of each variant is its own
74/// nullable column (`publication_scheduled_for`, `publication_canonical_url`,
75/// `publication_reason`).
76///
77/// The timestamps are `Option<jiff::Timestamp>`: the form binds them through
78/// a typed text field, which renders `type="datetime-local"`, reads the
79/// submission back as UTC, and stores no time for an empty control.
80#[derive(Debug, Clone, PartialEq, toasty::Embed, tablo_core::EmbeddedForm)]
81pub enum Publication {
82    #[column(variant = 1)]
83    Scheduled {
84        /// The shared column's one control renders from the first variant that
85        /// declares it, so its label is written there.
86        #[shared(timestamp)]
87        #[form(label = "Publication timestamp")]
88        scheduled_at: Option<Timestamp>,
89        scheduled_for: Option<Timestamp>,
90    },
91    #[column(variant = 2)]
92    Published {
93        #[shared(timestamp)]
94        published_at: Option<Timestamp>,
95        #[form(label = "Canonical URL")]
96        canonical_url: String,
97    },
98    #[column(variant = 3)]
99    Archived {
100        #[shared(timestamp)]
101        archived_at: Option<Timestamp>,
102        #[form(label = "Archive reason")]
103        reason: String,
104    },
105}

152/// One stored file in the media library — the `medias` table.
153///
154/// A WordPress-style library: one row per stored file with the tenant that
155/// uploaded it, the `path` the `Uploader` returned, the client's `filename`,
156/// a `kind`, and a timestamp. Rows carry no owner: a post shows one row as
157/// its cover through its own `cover_id`, and the library lists one tenant's
158/// rows. ADR-0021 records the shape.
159#[derive(Debug, Clone, toasty::Model)]
160#[table = "medias"]
161pub struct MediaAsset {
162    #[key]
163    #[auto]
164    pub id: uuid::Uuid,
165    /// The tenant that uploaded the file, like every other showcase row
166    /// the library lists one tenant's media.
167    #[index]
168    pub tenant_id: uuid::Uuid,
169    /// What the app's [`Uploader`](tablo_core::Uploader) returned, stored
170    /// verbatim and rendered as the URL the file is served at.
171    pub path: String,
172    /// The client's filename, a basename, for display.
173    pub filename: String,
174    /// `"image"` or `"file"` — the app's own kind, from the uploaded part's
175    /// content type. It decides whether a row renders a thumbnail or a link.
176    pub kind: String,
177    pub created_at: Timestamp,
178}
