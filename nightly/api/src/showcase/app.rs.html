38/// How many words `body` holds: whitespace-separated tokens.
39///
40/// Empty bodies hold none.
41pub fn word_count(body: &str) -> usize {
42    body.split_whitespace().count()
43}
44
45/// How many minutes `words` take to read at 200 words per minute, rounded up.
46///
47/// Empty bodies read in no minutes.
48pub fn read_minutes(words: usize) -> i64 {
49    if words == 0 {
50        0
51    } else {
52        words.div_ceil(200) as i64
53    }
54}

514            ),

573/// The author must exist *in this tenant* at write time: `validate_async`
574/// checked the option set before the transaction opened, but the author may
575/// be cross-tenant or deleted since, so the check re-runs through the
576/// tenant-scoped query inside the transaction. A miss is a 500.
577async fn ensure_author_in_tenant(
578    cx: &Cx,
579    author_id: uuid::Uuid,
580    ex: &mut dyn toasty::Executor,
581) -> Result<()> {
582    let author_exists = scoped_query::<AuthorResource>(cx)?
583        .filter(Author::fields().id().eq(author_id))
584        .first()
585        .exec(&mut *ex)
586        .await
587        .map_err(|e| -> topcoat::Error { e.into() })?
588        .is_some();
589    if !author_exists {
590        return Err(topcoat::Error::from(std::io::Error::other(
591            "author not found",
592        )));
593    }
594    Ok(())
595}

597/// Comments resource over `Comment`: the moderation queue.
598///
599/// Comments carry no tenant of their own — they inherit visibility from their
600/// post — so the request tenant is required like any other gated
601/// resource, and the scope is the parent post's tenant, declared in
602/// [`tenant_scope`](Resource::tenant_scope).
603///
604/// That declaration is what the framework's default cannot supply: the default
605/// derives the filter from a `tenant_id` column on the model, and `Comment` has
606/// none. Leaving `requires_tenant` false and writing the filter inside `query`
607/// was the GH #223 review's leak — a tenantless request silently *skipped* the
608/// filter instead of being refused, and with `can_view_any`/`can_view` true the
609/// caller could read and moderate every tenant's comments. Gated + declared is
610/// the shape that fails closed: no tenant is a 403 everywhere, and the
611/// predicate is the framework's to apply.
612///
613/// The queue moderates: `can_delete_any` enables row and bulk delete, and
614/// `delete_record` writes them — bulk delete rides
615/// the framework default that loops `delete_record`. A resource that wants a
616/// read-only queue leaves [`can_delete_any`](Resource::can_delete_any) at its default instead.
617pub struct CommentResource;
618
619/// Re-resolve a comment's parent post through the tenant-scoped
620/// [`scoped_query::<PostResource>`] inside the caller's open transaction.
621/// `Schema::validate_async` already rejects a `post_id` outside the
622/// tenant-scoped option set before the tx opens, but that
623/// is a pre-write check in a different window: a policy or tenant change between
624/// the two would slip through, and a direct `create_record` / `update_record`
625/// caller never ran it at all. Mirroring `PostResource`'s author double-check,
626/// this is the defense-in-depth half — one query on the seam that owns tenancy.
627///
628/// The miss is a 404, not the 500 `PostResource` uses for a missing author: a
629/// parent in another tenant is an authorization boundary, and "wrong tenant
630/// looks exactly like unknown id" is this panel's contract everywhere else
631/// (#169).
632async fn ensure_post_in_tenant(
633    cx: &Cx,
634    post_id: uuid::Uuid,
635    ex: &mut dyn toasty::Executor,
636) -> Result<()> {
637    let in_tenant = scoped_query::<PostResource>(cx)?
638        .filter(Post::fields().id().eq(post_id))
639        .first()
640        .exec(&mut *ex)
641        .await
642        .map_err(|e| -> topcoat::Error { e.into() })?
643        .is_some();
644    if !in_tenant {
645        return Err(topcoat::router::error::not_found().into());
646    }
647    Ok(())
648}

676        cx: &Cx,
677        form: CommentForm,
678        ex: &mut dyn toasty::Executor,
679    ) -> Result<Comment> {

681        // validation is not a write-time guarantee.
682        ensure_post_in_tenant(cx, form.post_id, ex).await?;
683        write_create::<Self>(cx, form, ex).await
684    }

687        cx: &Cx,
688        record: Comment,
689        posted: Posted<CommentForm>,
690        ex: &mut dyn toasty::Executor,
691    ) -> Result<Comment> {
692        // An update can re-point the comment at another post, which is exactly
693        // the move the pre-tx check cannot be trusted to catch.

696    }

699        // "Comments", not "Discussion": the entity is a comment, the
700        // route and model say so, and a discussion — if it means anything here
701        // — would be the set of comments on one post, which is not a record the
702        // panel can list or moderate.
703        "Comments".to_string()
704    }

706    /// Tenant-scoped from the parent post, and gated like every other
707    /// tenant-owned resource.

709    /// `true` is what makes a tenantless request a 403 here instead of a read
710    /// that quietly dropped the filter; the predicate below replaces the
711    /// framework's name-based derivation, which finds no `tenant_id` on
712    /// `Comment`.
713    fn requires_tenant() -> bool {
714        true
715    }

717    /// Inherit-through-the-relation: scope through the parent post's
718    /// tenant. Toasty rewrites the relation-path comparison into a foreign-key
719    /// subquery, and the framework ANDs the result onto `query`/`view_query`

721    fn tenant_scope(tenant: uuid::Uuid) -> Option<toasty::stmt::Expr<bool>> {
722        Some(Comment::fields().post().tenant_id().eq(tenant))
723    }

726        true
727    }

731    /// refuses, which is the same answer — a placeholder has nothing

733    fn can_view(_cx: &Cx, record: &Comment) -> bool {
734        record.body != REMOVED_COMMENT_BODY
735    }

817/// The URL prefix uploads are served under.
818///
819/// It matches the `serve_dir` route below, and the app owns both ends: the
820/// store decides the path it returns, so the framework never has to guess a
821/// URL convention.
822pub const UPLOAD_URL_PREFIX: &str = "/uploads";
823
824/// The showcase's own uploader: write the bytes into the served directory and
825/// return the URL they are served at.
826///
827/// A demo, not a framework default — the trait is the seam and drivers are the
828/// app's business. The UUID prefix keeps two uploads of `cover.png` apart, and
829/// the framework hands this store a name already sanitized to a basename.
830/// Writing the file is this app's job; swapping in an object store
831/// means replacing this type and nothing else.
832///
833/// The media library's page builds this store too: it parses its own
834/// multipart body, so the sanitized name is not the framework's to guarantee
835/// there — the store applies [`basename`] itself rather than trusting every
836/// caller to have done it.
837pub(crate) struct DirUploader {
838    dir: PathBuf,
839}
840
841impl DirUploader {
842    pub(crate) fn new(dir: impl Into<PathBuf>) -> Self {
843        Self { dir: dir.into() }
844    }
845}
846
847impl Uploader for DirUploader {
848    async fn store(&self, filename: &str, bytes: &[u8]) -> Result<String, String> {
849        let name = format!("{}-{}", uuid::Uuid::new_v4(), basename(filename));
850        // Failure reasons are rendered to the user, so they say what the user
851        // can act on and never leak the path that failed.
852        tokio::fs::create_dir_all(&self.dir)
853            .await
854            .map_err(|_| "the upload directory is not writable".to_string())?;
855        tokio::fs::write(self.dir.join(&name), bytes)
856            .await
857            .map_err(|_| "the upload could not be written".to_string())?;
858        // The caller stores this string and renders it verbatim as the file's
859        // URL, so it has to be one: the segment is percent-encoded,
860        // or `cover #1.png` would be served as `cover ` plus a fragment, and a
861        // `%22` the browser sent in the filename would decode to a quote the
862        // file on disk does not carry.
863        Ok(format!("{UPLOAD_URL_PREFIX}/{}", url_segment(&name)))
864    }
865
866    /// Whether the served directory still holds the file a URL names.
867    ///
868    /// The candidate comes from a form the panel re-rendered, so it is a URL
869    /// this store itself produced; the check is a lookup rather than a path
870    /// join — every entry in the store's own root is encoded back to its URL
871    /// form and compared, so a `%2F`, a `..` or an absolute path a client
872    /// submits never reaches the filesystem.
873    async fn holds(&self, path: &str) -> bool {
874        let Some(segment) = path.strip_prefix(UPLOAD_URL_PREFIX) else {
875            return false;
876        };
877        let Some(segment) = segment.strip_prefix('/') else {
878            return false;
879        };
880        if segment.is_empty() {
881            return false;
882        }
883        let Ok(mut entries) = tokio::fs::read_dir(&self.dir).await else {
884            return false;
885        };
886        while let Ok(Some(entry)) = entries.next_entry().await {
887            if url_segment(&entry.file_name().to_string_lossy()) == segment {
888                return true;
889            }
890        }
891        false
892    }
893}
894
895/// The longest client filename the showcase keeps, in bytes.
896///
897/// 255 bytes is the common per-component limit on Linux filesystems, and the
898/// store's `{uuid}-` prefix takes 37 of them, so the basename is capped to what
899/// is left. The media library records the name it stores, so the row's
900/// `filename` and the file on disk cannot disagree.
901const MAX_BASENAME_BYTES: usize = 218;
902
903/// Reduce a client-supplied filename to the basename this app stores and shows.
904/// Strips directory components (`../../etc/passwd` → `passwd`), drops control
905/// characters, trims the ends, and caps the byte length preserving the tail, so
906/// the extension survives. The framework sanitizes the names its own form
907/// parser hands an [`Uploader`]; the media library's page parses its own
908/// multipart body, and this is the one rule both callers apply.
909pub(crate) fn basename(raw: &str) -> String {
910    let base = raw.rsplit(['/', '\\']).next().unwrap_or(raw);
911    let clean: String = base.chars().filter(|c| !c.is_control()).collect();
912    let trimmed = clean.trim();
913    if trimmed.len() <= MAX_BASENAME_BYTES {
914        return trimmed.to_string();
915    }
916    // Walk the cut point forward to a char boundary: slicing a multibyte
917    // character would panic on an attacker-controlled filename.
918    let mut start = trimmed.len() - MAX_BASENAME_BYTES;
919    while !trimmed.is_char_boundary(start) {
920        start += 1;
921    }
922    trimmed[start..].to_string()
923}
924
925/// `name` as one URL path segment: everything outside the unreserved set is
926/// percent-encoded (RFC 3986 §2.3).
927///
928/// The store's contract is that the string it returns is fetchable verbatim
929/// and a client filename is arbitrary: a space must not become the
930/// end of the URL, a `#` must not start a fragment, and a `%` must not decode
931/// to something else.
932fn url_segment(name: &str) -> String {
933    use std::fmt::Write as _;
934
935    let mut out = String::with_capacity(name.len());
936    for byte in name.bytes() {
937        match byte {
938            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~' => {
939                out.push(byte as char);
940            }
941            _ => {
942                let _ = write!(out, "%{byte:02X}");
943            }
944        }
945    }
946    out
947}

949fn build_router(db: Db, bundle: Option<AssetBundle>, uploads: Option<PathBuf>) -> Router {
950    let mut panel = Panel::new("admin")
951        .brand(
952            Brand::new("Tablo Blog").logo(
953                "data:image/svg+xml,%3Csvg%20xmlns='http://www.w3.org/2000/svg'%20viewBox='0%200%2024%2024'%3E%3Ccircle%20cx='12'%20cy='12'%20r='10'%20fill='%236366f1'/%3E%3Ctext%20x='12'%20y='16'%20text-anchor='middle'%20font-size='12'%20fill='white'%20font-family='sans-serif'%3EA%3C/text%3E%3C/svg%3E",
954            ),
955        )
956        // Light by default: the header toggle is the only thing that
957        // turns dark on. `Panel::dark_mode` stays available for an app that
958        // wants a dark-first panel.
959        .home::<Dashboard>()
960        .resource::<UserResource>()
961        .resource::<AuthorResource>()
962        .resource::<PostResource>()
963        .resource::<CommentResource>()
964        .page::<MediaLibraryPage>()
965        .page::<LiveActivityPage>();
966    // No "Published" saved-view entry: it would point at
967    // `/admin/posts?f.status=published`, i.e. the Blog Posts table with a
968    // filter — the same page twice in the sidebar, and the one arrangement the
969    // shell's path matching highlights twice at once.
970    // Demo credentials stay available for local development via
971    // SHOWCASE_LOGIN_HINT, but the default login page is shippable with no
972    // hint. Empty values install nothing (no empty hint paragraph).
973    if let Ok(hint) = std::env::var("SHOWCASE_LOGIN_HINT")
974        && !hint.trim().is_empty()
975    {
976        panel = panel.login_hint(hint);
977    }
978    if let Some(dir) = uploads {
979        // Both ends of the demo: the store writes into `dir` and
980        // returns `{UPLOAD_URL_PREFIX}/…`, and the panel serves exactly that
981        // prefix from the same directory — which is why the stored path is
982        // fetchable without the framework inventing a URL convention.
983        panel = panel
984            .serve_dir(format!("{UPLOAD_URL_PREFIX}/{{*file}}"), dir.clone())
985            .uploads(DirUploader::new(dir));
986    }
987    // The app owns the router: the public blog's pages and the panel's are
988    // discovered and mounted side by side.
989    let mut builder = Router::builder().discover().app_context(db);
990    if let Some(bundle) = bundle {
991        builder = builder.assets(bundle);
992        panel = panel.shell_assets(tailwind::stylesheet!(), GEIST);
993    }
994    builder.panel(panel).expect("showcase panel mounts").build()
995}

997fn load_assets() -> AssetBundle {
998    match AssetBundle::load() {
999        Ok(bundle) => bundle,
1000        Err(near_executable) => {
1001            // Cargo places test executables in `target/*/deps`, while the
1002            // bundle remains beside the package binary in `target/*`.
1003            let test_bundle = std::env::current_exe()
1004                .ok()
1005                .and_then(|exe| {
1006                    let dir = exe.parent()?;
1007                    if dir.file_name().is_some_and(|name| name == "deps") {
1008                        Some(dir.parent()?.to_path_buf())
1009                    } else {
1010                        None
1011                    }
1012                })
1013                .map(|dir| dir.join("assets"));
1014            match test_bundle {
1015                Some(dir) => AssetBundle::load_dir(&dir).unwrap_or_else(|test_error| {
1016                    panic!(
1017                        "showcase asset bundle is unavailable: executable lookup failed ({near_executable}); tried {} ({test_error})",
1018                        dir.display()
1019                    )
1020                }),
1021                None => panic!(
1022                    "showcase asset bundle is unavailable: executable lookup failed ({near_executable})"
1023                ),
1024            }
1025        }
1026    }
1027}
