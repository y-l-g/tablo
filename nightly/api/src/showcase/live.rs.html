16/// Where the live feed lives.

19/// How many of the newest users the feed shows.
20const FEED_ROWS: usize = 20;

22/// The process-wide wake channel behind every open feed.
23static BOARD: LazyLock<Board> = LazyLock::new(Board::default);
24
25/// Wakes every connected feed after a committed write.
26struct Board {
27    changed: tokio::sync::broadcast::Sender<()>,
28}
29
30impl Default for Board {
31    fn default() -> Self {
32        Self {
33            changed: tokio::sync::broadcast::channel(16).0,
34        }
35    }
36}
37
38impl Board {
39    fn notify(&self) {
40        let _ = self.changed.send(());
41    }
42
43    fn subscribe(&self) -> tokio::sync::broadcast::Receiver<()> {
44        self.changed.subscribe()
45    }
46}
47
48/// Wakes every connected feed the process holds.
49pub(crate) fn notify() {
50    BOARD.notify();
51}
52
53fn subscribe() -> tokio::sync::broadcast::Receiver<()> {
54    BOARD.subscribe()
55}

90/// Streams the newest users to every open page.
91#[shard]
92async fn live_feed(cx: &Cx) -> Result<impl View> {
93    // The shard runs the panel's guard itself.
94    tablo_core::auth::guard(cx)?;
95    Ok(live! {
96        let mut changed = subscribe();
97        loop {
98            let users = newest_users(cx).await?;
99            let empty = users.is_empty();
100            let token = emit! {
101                if empty {
102                    <p class="text-sm text-muted-foreground" data-live-feed="">
103                        "No users yet."
104                    </p>
105                } else {
106                    <ul class="flex flex-col divide-y divide-border" data-live-feed="">
107                        for user in users {
108                            <li
109                                class="flex items-center justify-between gap-4 py-2.5 text-sm"
110                            >
111                                <span class="font-medium text-foreground">
112                                    (user.name)
113                                </span>
114                                <span class="truncate text-muted-foreground">
115                                    (user.email)
116                                </span>
117                            </li>
118                        }
119                    </ul>
120                }
121            }?;
122            if !connected(cx) {
123                break Ok(token);
124            }
125            match changed.recv().await {
126                Err(tokio::sync::broadcast::error::RecvError::Lagged(_)) => {}
127                Err(tokio::sync::broadcast::error::RecvError::Closed) => break Ok(token),
128                Ok(()) => {}
129            }
130        }
131    })
132}

134/// The newest users the panel serves, newest first.
135async fn newest_users(cx: &Cx) -> Result<Vec<User>> {
136    let mut db = db(cx);
137    let users = UserResource::query(cx)
138        .order_by(User::fields().created_at().desc())
139        .limit(FEED_ROWS)
140        .exec(&mut db)
141        .await?;
142    Ok(users)
143}
