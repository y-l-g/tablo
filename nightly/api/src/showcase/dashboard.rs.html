1//! The panel's home page, served at the panel prefix by `Panel::home`.

3use tablo_core::{NavigationItem, Page, Resource, can_list, db::db, scoped_query};
4use topcoat::{
5    Result,
6    context::Cx,
7    icon::{IconData, icon},
8    view::{BoxView, View, ViewExt, view},
9};
10
11use crate::app::{AuthorResource, CommentResource, PostResource, UserResource};
12
13/// The page the panel serves at its prefix.
14pub struct Dashboard;
15
16/// One resource's tile: its label, its list, and how many records the
17/// caller can see there.
18struct Stat {
19    label: String,
20    url: String,
21    icon: IconData,
22    /// `None` when the count could not be read: the tile shows a dash rather
23    /// than a wrong zero.
24    count: Option<u64>,
25}
26
27/// `R`'s tile, or `None` when the caller may not open `R`'s list, so the
28/// dashboard links to no list that would answer 403, or when the panel does
29/// not register `R`.
30async fn stat<R: Resource>(cx: &Cx, glyph: IconData) -> Option<Stat> {
31    if !can_list::<R>(cx) {
32        return None;
33    }
34    let url = tablo_core::url::resource::<R>(cx)?;
35    // The list's own scoped query, so the tile counts exactly the rows the
36    // list would page through.
37    let count = match scoped_query::<R>(cx) {
38        Ok(query) => query.count().exec(&mut db(cx)).await.ok(),
39        Err(_) => None,
40    };
41    Some(Stat {
42        label: R::navigation_label(),
43        url,
44        icon: glyph,
45        count,
46    })
47}
48
49fn stat_card(cx: &Cx, stat: Stat) -> BoxView<'_> {
50    let count = stat
51        .count
52        .map_or_else(|| "—".to_string(), |count| count.to_string());
53    view! {
54        cx =>
55        <a
56            href=(stat.url)
57            class="group rounded-xl focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
58        >
59            tablo_ui::card(
60                attrs: topcoat::view::attributes! { class="transition-colors group-hover:bg-muted/40" },
61                tablo_ui::card_content(
62                    <div class="flex items-center justify-between gap-2">
63                        <p class="text-sm font-medium text-muted-foreground">
64                            (stat.label)
65                        </p>
66                        <span
67                            class="text-muted-foreground [&>svg]:size-4"
68                            aria-hidden="true"
69                        >
70                            icon(data: stat.icon)
71                        </span>
72                    </div>
73                    <p
74                        class="mt-2 text-3xl font-semibold tracking-tight text-foreground"
75                    >
76                        (count)
77                    </p>
78                )
79            )
80        </a>
81    }
82    .boxed()
83}
84
85impl Page for Dashboard {
86    fn navigation() -> NavigationItem {
87        NavigationItem::for_page::<Self>().icon(tablo_ui::icons::LAYOUT_DASHBOARD)
88    }
89
90    async fn render(cx: &Cx) -> Result<impl View> {
91        let stats = [
92            stat::<UserResource>(cx, tablo_ui::icons::USERS).await,
93            stat::<AuthorResource>(cx, tablo_ui::icons::PEN_LINE).await,
94            stat::<PostResource>(cx, tablo_ui::icons::FILE_TEXT).await,
95            stat::<CommentResource>(cx, tablo_ui::icons::MESSAGE_SQUARE).await,
96        ];
97        let cards: Vec<BoxView<'_>> = stats
98            .into_iter()
99            .flatten()
100            .map(|stat| stat_card(cx, stat))
101            .collect();
102        Ok(view! {
103            cx =>
104            tablo_ui::page(
105                tablo_ui::page_header(
106                    tablo_ui::page_title("Dashboard")
107                    tablo_ui::page_description(
108                        "The blog's admin: its users, authors, posts, comments and media."
109                    )
110                )
111                tablo_ui::page_content(
112                    <div class="grid gap-4 sm:grid-cols-2 xl:grid-cols-4">
113                        for card in cards {
114                            (card)
115                        }
116                    </div>
117                )
118            )
119        })
120    }
